51class LFRicCollection():
52 '''
53 Base class for managing the declaration and initialisation of a
54 group of related entities within an Invoke or Kernel stub
55
56 :param node: the Kernel or Invoke for which to manage variable \
57 declarations and initialisation.
58 :type node: :py:class:`psyclone.domain.lfric.LFRicInvoke` or \
59 :py:class:`psyclone.domain.lfric.LFRicKern`
60
61 :raises InternalError: if the supplied node is not an LFRicInvoke or an \
62 LFRicKern.
63
64 '''
65 def __init__(self, node):
66 if isinstance(node, LFRicInvoke):
67 # We are handling declarations/initialisations for an Invoke
68 self._invoke = node
69 self._kernel = None
70 self._symbol_table = self._invoke.schedule.symbol_table
71 # The list of Kernel calls we are responsible for
72 self._calls = node.schedule.kernels()
73 elif isinstance(node, LFRicKern):
74 # We are handling declarations for a Kernel stub
75 self._invoke = None
76 self._kernel = node
77 # TODO #719 The symbol table is not connected to other parts of
78 # the Stub generation.
79 self._symbol_table = LFRicSymbolTable()
80 # We only have a single Kernel call in this case
81 self._calls = [node]
82 else:
83 raise InternalError(f"LFRicCollection takes only an LFRicInvoke "
84 f"or an LFRicKern but got: {type(node)}")
85
86 # Whether or not the associated Invoke contains only Kernels that
87 # operate on DoFs.
88 if self._invoke:
89 self._dofs_only = self._invoke.operates_on_dofs_only
90 else:
91 self._dofs_only = False
92
93 def declarations(self, parent):
94 '''
95 Insert declarations for all necessary variables into the AST of
96 the generated code. Simply calls either '_invoke_declarations()' or
97 '_stub_declarations()' depending on whether we're handling an Invoke
98 or a Kernel stub.
99
100 :param parent: the node in the f2pygen AST representing the routine \
101 in which to insert the declarations.
102 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
103
104 :raises InternalError: if neither 'self._invoke' nor 'self._kernel' \
105 are set.
106
107 '''
108 if self._invoke:
109 self._invoke_declarations(parent)
110 elif self._kernel:
111 self._stub_declarations(parent)
112 else:
113 raise InternalError("LFRicCollection has neither a Kernel "
114 "nor an Invoke - should be impossible.")
