45 '''Symbol identifying a generic interface that maps to a number of

107 @routines.setter
108 def routines(self, values):
109 '''
110 Setter for the list of routines to which this interface provides
111 access.
112
113 :param values: the RoutineSymbols and whether or not each of them
114 is a module procedure.
115 :type values: list[
116 tuple[:py:class:`psyclone.psyir.symbols.RoutineSymbol`, bool]]
117
118 :raises ValueError: if no (or an empty) list of values is provided.
119 :raises TypeError: if `symbols` is not a list that consists only of
120 (RoutineSymbol, bool) tuples.
121 '''
122 if not values:
123 raise ValueError("A GenericInterfaceSymbol requires a list of "
124 "RoutineSymbols but none were provided.")
125 if not isinstance(values, list):
126 raise TypeError(f"A GenericInterfaceSymbol requires a list of "
127 f"tuples describing its member routines but got: "
128 f"'{values}'")
129 self._routines = []
130 for item in values:
131 if not isinstance(item, tuple):
132 raise TypeError(
133 f"A GenericInterfaceSymbol ('{self.name}') requires a "
134 f"list of tuples but got: '{type(item).__name__}'")
135 if (len(item) != 2 or not isinstance(item[0], RoutineSymbol) or
136 not isinstance(item[1], bool)):
137 raise TypeError(
138 f"Each tuple used to define a routine within the "
139 f"GenericInterfaceSymbol '{self.name}' must consist of a "
140 f"RoutineSymbol and a bool but got: "
141 f"{[type(rt).__name__ for rt in item]}")
142 self._routines.append(self.RoutineInfo(item[0], item[1]))

190 def replace_symbols_using(self, table):
191 '''
192 Replace any Symbols referred to by this object with those in the
193 supplied SymbolTable with matching names. If there
194 is no match for a given Symbol then it is left unchanged.
195
196 :param table: the symbol table from which to get replacement symbols.
197 :type table: :py:class:`psyclone.psyir.symbols.SymbolTable`

200 # Construct a new list of RoutineSymbols.
201 new_routines = []
