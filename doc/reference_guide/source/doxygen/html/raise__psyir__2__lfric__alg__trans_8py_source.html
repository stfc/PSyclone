36'''Specialise generic PSyIR to LFRic algorithm-layer PSyIR. Currently
37we transform PSyIR invoke calls to LFRic algorithm-layer-specific
38invoke calls which uses specialised classes.
39
40'''
41from psyclone.domain.common.transformations import RaisePSyIR2AlgTrans
42from psyclone.domain.lfric.algorithm.psyir import (
43 LFRicBuiltinFunctorFactory, LFRicKernelFunctor, LFRicAlgorithmInvokeCall)
44from psyclone.psyir.nodes import ArrayReference
45
46
47class RaisePSyIR2LFRicAlgTrans(RaisePSyIR2AlgTrans):
48 '''Transform a generic PSyIR representation of an Algorithm-layer
49 invoke call to an LFRic version with specialised domain-specific
50 nodes.
51
52 '''
53 def apply(self, call, index, options=None):
54 ''' Apply the transformation to the supplied node.
55
56 :param call: a PSyIR call node capturing an invoke call in \
57 generic PSyIR.
58 :type call: :py:class:`psyclone.psyir.nodes.Call`
59 :param int index: the position of this invoke call relative to \
60 other invokes in the algorithm layer.
61 :param options: a dictionary with options for transformations.
62 :type options: Optional[Dict[str, Any]]
63
64 '''
65 self.validatevalidate(call, options=options)
66
67 call_name = None
68 calls = []
69 table = call.scope.symbol_table
70
71 factory = LFRicBuiltinFunctorFactory.get()
72
73 for idx, call_arg in enumerate(call.arguments):
74
75 if call.argument_names[idx]:
76 call_name = f"{call_arg.value}"
77 elif isinstance(call_arg, ArrayReference):
78 # kernel or builtin misrepresented as ArrayReference
79 args = call_arg.pop_all_children()
80 try:
81 calls.append(factory.create(call_arg.name, table, args))
82 except KeyError:
83 # No match for a builtin so create a user-defined kernel.
84 self._specialise_symbol(call_arg.symbol)
85 calls.append(LFRicKernelFunctor.create(call_arg.symbol,
86 args))
87 else:
88 for fp2_node in call_arg.get_ast_nodes:
89 # This child is a kernel or builtin
90 name = fp2_node.children[0].string
91 args = RaisePSyIR2AlgTrans._parse_args(call_arg,
92 fp2_node)
93 name = fp2_node.children[0].string
94 try:
95 calls.append(factory.create(name, table, args))
96 except KeyError:
97 # No match for a builtin so create a user-defined
98 # kernel.
99 type_symbol = RaisePSyIR2AlgTrans._get_symbol(
100 call, fp2_node)
101 self._specialise_symbol(type_symbol)
102 calls.append(LFRicKernelFunctor.create(type_symbol,
103 args))
104
105 invoke_call = LFRicAlgorithmInvokeCall.create(
106 call.routine.symbol, calls, index, name=call_name)
107 call.replace_with(invoke_call)
