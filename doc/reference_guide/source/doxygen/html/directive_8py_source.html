79 def create_data_movement_deep_copy_refs(self):
80 '''
81 Creates the References required to perform a deep copy (in e.g.
82 OpenACC or OpenMP) of all of the quantities accessed in Nodes below
83 this one in the tree. It distringuishes between those quantities that
84 are only read, only written or are both read and written. The necessary
85 References are added to the returned OrderedDicts in the order in which
86 they must be copied.
87
88 :returns: a 3-tuple containing dicts describing the quantities that
89 are read-only, write-only and readwrite. Each dict contains
90 References indexed by Signatures.
91 :rtype: Tuple[OrderedDict[:py:class:`psyclone.core.Signature`,
92 :py:class:`psyclone.psyir.nodes.Reference`]]
93
94 '''
95 readwrites = OrderedDict()
96 read_only = OrderedDict()
97 write_only = OrderedDict()
98 table = self.scope.symbol_table
99
100 var_info = VariablesAccessInfo()
101 self.reference_accesses(var_info)
102
103 for sig in var_info.all_signatures:
104 vinfo = var_info[sig]
105 node = vinfo.all_accesses[0].node
106 sym = table.lookup(sig.var_name)
107
108 if not vinfo.has_data_access():
109 # Ignore references that don't correspond to data accesses.
110 continue
111
112 if isinstance(sym.datatype, ScalarType):
113 # We ignore scalars as these are typically copied by value.
114 continue
115
116 if var_info.has_read_write(sig):
117 access_dict = readwrites
118 else:
119 if var_info.is_read(sig):
120 if var_info.is_written(sig):
121 if vinfo.is_written_first():
122 access_dict = write_only
123 else:
124 access_dict = readwrites
125 else:
126 access_dict = read_only
127 else:
128 access_dict = write_only
129
130 if not sig.is_structure:
131 # This must be an array.
132 # TODO #2304 - in languages such as C++ it will be necessary to
133 # supply the extent of an array that is being accessed. For now
134 # we only supply a Reference (which is sufficient in Fortran).
135 if sig not in access_dict:
136 access_dict[sig] = Reference(node.symbol)
137 continue
138
139 # We have a structure access and so we need the list of
140 # references required to do a 'deep copy'. This means that if
141 # we have an access `a%b%c(i)` then we need references to `a`,
142 # `a%b` and then `a%b%c`.
143
144 # A Signature does not contain indexing information so we use
145 # a PSyIR node that corresponds to this access.
146 _, index_lists = node.get_signature_and_indices()
147
148 # First add the root access (`a` in the above example).
149 if Signature(node.symbol.name) not in access_dict:
150 access_dict[Signature(node.symbol.name)] = Reference(
151 node.symbol)
152
153 # Then work our way down the various members.
154 for depth in range(1, len(sig)):
155 if sig[:depth+1] not in access_dict:
156 if node.is_array:
157 base_cls = ArrayOfStructuresReference
158 # Copy the indices so as not to modify the original
159 # node.
160 base_args = [node.symbol,
161 [idx.copy() for idx in node.indices]]
162 else:
163 base_cls = StructureReference
164 base_args = [node.symbol]
165 # Create the new lists of indices, one list for each
166 # member of the structure access apart from the last
167 # one where we assume the whole array (if it is an
168 # array) is accessed. Hence the loop is 1:depth and
169 # then we set the last one separately.
170 new_lists = []
171 for idx_list in index_lists[1:depth]:
172 new_lists.append([idx.copy() for idx in idx_list])
173 members = list(zip(sig[1:depth], new_lists))
174 # The last member has no array indexing.
175 members.append(sig[depth])
176 access_dict[sig[:depth+1]] = base_cls.create(
177 *base_args, members)
178 return read_only, write_only, readwrites

181class RegionDirective(Directive):
182 '''
183 Base class for all Directive nodes that have an associated
184 region of code with them.
185
186 All classes that generate RegionDirective statements (e.g. OpenMP,
187 OpenACC, compiler-specific) inherit from this class.
188
189 :param ast: the entry in the fparser2 parse tree representing the code
190 contained within this directive or None.
191 :type ast: Optional[:py:class:`fparser.two.Fortran2003.Base`]
192 :param children: the nodes that will be children of this
193 Directive node or None.
194 :type children: Optional[List[:py:class:`psyclone.psyir.nodes.Node`]]
195 :param parent: PSyIR node that is the parent of this Directive or None.
196 :type parent: Optional[:py:class:`psyclone.psyir.nodes.Node`]
197
198 '''
199 # Textual description of the node.
200 _children_valid_format = "Schedule"
201
202 def __init__(self, ast=None, children=None, parent=None):
203 # A Directive always contains a Schedule
204 super().__init__(ast, parent=parent)
205 self.addchild(Schedule(children=children))

247 def gen_post_region_code(self, parent):
248 '''
249 Generates any code that must be executed immediately after the end of
250 the region defined by this directive.
251
252 TODO #1648 this method is only used by the gen_code() code-generation
253 path and should be replaced by functionality in a
254 'lower_to_language_level' method in an LFRic-specific subclass
255 of the appropriate directive.
256
257 :param parent: where to add new f2pygen nodes.
258 :type parent: :py:class:`psyclone.f2pygen.BaseGen`
259
260 '''
261 if not Config.get().distributed_memory or self.ancestor(Loop):
262 return
263 # Have to import PSyLoop here to avoid a circular dependence.
264 # pylint: disable=import-outside-toplevel
265 from psyclone.domain.common.psylayer import PSyLoop
266
267 commented = False
268 for loop in self.walk(PSyLoop):
269 if not isinstance(loop.parent, Loop):
270 if not commented and loop.unique_modified_args("gh_field"):
271 commented = True
272 parent.add(CommentGen(parent, ""))
273 parent.add(CommentGen(parent,
274 " Set halos dirty/clean for fields "
275 "modified in the above loop(s)"))
276 parent.add(CommentGen(parent, ""))
277 loop.gen_mark_halos_clean_dirty(parent)
278
279 if commented:
280 parent.add(CommentGen(parent, ""))
281 parent.add(CommentGen(parent,
282 " End of set dirty/clean section for "
283 "above loop(s)"))
