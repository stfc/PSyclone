56class AssignmentTrans(AdjointTransformation):
57 '''Implements a transformation to translate a Tangent-Linear
58 assignment to its Adjoint form.
59
60 '''
61 def apply(self, node, options=None):
62 '''Apply the Assignment transformation to the specified node. The node
63 must be a valid tangent-linear assignment. The assignment is
64 replaced with its adjoint version.
65
66 :param node: an Assignment node.
67 :type node: :py:class:`psyclone.psyir.nodes.Assignment`
68 :param options: a dictionary with options for transformations.
69 :type options: Optional[Dict[str, Any]]
70
71 '''
72 self.validatevalidate(node, options)
73
74 # Split the RHS of the assignment into [-]<term> +- <term> +- ...
75 rhs_terms = self._split_nodes(
76 node.rhs, [BinaryOperation.Operator.ADD,
77 BinaryOperation.Operator.SUB])
78
79 deferred_inc = []
80 sym_maths = SymbolicMaths.get()
81 for rhs_term in rhs_terms:
82
83 # Find the active var in rhs_term if one exists (we may
84 # find 0.0), storing it in 'active_var' and if so replace
85 # it with lhs_active_var storing the modified term in
86 # 'new_rhs_term'. Also determine whether this is an
87 # increment, storing the result in 'increment'.
88 increment = False
89 active_var = None
90 new_rhs_term = rhs_term.copy()
91 for ref in new_rhs_term.walk(Reference):
92 if ref.symbol not in self._active_variables_active_variables:
93 continue
94 active_var = ref
95 # Identify whether this reference on the RHS matches the
96 # one on the LHS - if so we have an increment.
97 if node.is_array_assignment and isinstance(ref, ArrayMixin):
98 # TODO #1537 - we can't just do `sym_maths.equal` if we
99 # have an array range because the SymbolicMaths class does
100 # not currently support them.
101 # Since we have already checked (in validate) that any
102 # references to the same symbol on the RHS have the same
103 # range, this is an increment if the symbols match.
104 if node.lhs.symbol is ref.symbol:
105 increment = True
106 else:
107 if sym_maths.equal(ref, node.lhs):
108 increment = True
109 if ref.parent:
110 ref.replace_with(node.lhs.copy())
111 else:
112 new_rhs_term = node.lhs.copy()
113 break
114
115 # Work out whether the binary operation for this term is a
116 # '+' or a '-' and return it in 'rhs_operator'.
117 rhs_operator = BinaryOperation.Operator.ADD
118 previous = rhs_term
119 candidate = rhs_term.parent
120 while not isinstance(candidate, Assignment):
121 if (isinstance(candidate, BinaryOperation) and
122 candidate.operator == BinaryOperation.Operator.SUB and
123 candidate.children[1] is previous):
124 # Rules: + + -> +; - - -> +; + - -> -; - + -> -
125 # If the higher level op is + then there is no
126 # change to the existing op. If it is - then
127 # we flip the op i.e. - => + and + => -.
128 if rhs_operator == BinaryOperation.Operator.SUB:
129 rhs_operator = BinaryOperation.Operator.ADD
130 else:
131 rhs_operator = BinaryOperation.Operator.SUB
132 previous = candidate
133 candidate = candidate.parent
134
135 if not active_var:
136 # This is an expression without an active variable
137 # (which must be 0.0, otherwise validation will have
138 # rejected it). There is therefore nothing to output.
139 continue
140
141 if increment:
142 # The output of an increment needs to be deferred as
143 # other terms must be completed before the LHS TL
144 # active variable is modified. Save the rhs term
145 # and its associated operator.
146 deferred_inc.append((new_rhs_term, rhs_operator))
147 else:
148 # Output the adjoint for this term
149 rhs = BinaryOperation.create(
150 rhs_operator, active_var.copy(), new_rhs_term)
151 assignment = Assignment.create(active_var.copy(), rhs)
152 node.parent.children.insert(node.position, assignment)
153
154 if (len(deferred_inc) == 1 and
155 isinstance(deferred_inc[0][0], Reference)):
156 # No need to output anything as the adjoint is A = A.
157 pass
158 elif deferred_inc:
159 # Output the adjoint for all increment terms in a single line.
160 rhs, _ = deferred_inc.pop(0)
161 for term, operator in deferred_inc:
162 rhs = BinaryOperation.create(operator, rhs, term)
163 assignment = Assignment.create(node.lhs.copy(), rhs)
164 node.parent.children.insert(node.position, assignment)
165 else:
166 # The assignment is not an increment. The LHS active
167 # variable needs to be zero'ed.
168 assignment = Assignment.create(
169 node.lhs.copy(), Literal("0.0", REAL_TYPE))
170 node.parent.children.insert(node.position, assignment)
171
172 # Remove the original node
173 node.detach()

175 def _array_ranges_match(self, assign, active_variable):
176 '''
177 If the supplied assignment is to an array range and the supplied
178 active variable is the entity being assigned to then this routine
179 checks that the array ranges of the LHS and the supplied reference
180 match. If they do not then an exception is raised.
181
182 :param assign: the assignment that we are checking.
183 :type assign: :py:class:`psyclone.psyir.nodes.Assignment`
184 :param active_variable: an active variable that appears on the \
185 LHS and RHS of the supplied assignment.
186 :type active_variable: :py:class:`psyclone.psyir.nodes.Reference`
187
188 :raises TangentLinearError: if the supplied assignment is to a \
189 symbol with an array range but the same symbol occurs on the \
190 RHS without an array range.
191 :raises NotImplementedError: if the array ranges on the LHS and \
192 RHS of the assignment for the supplied variable do not match.
193
194 '''
195 # This check only needs to proceed if the assignment is to an array
196 # range and the supplied active variable is the one being assigned to.
197 if not (assign.is_array_assignment and active_variable.symbol is
198 assign.lhs.symbol):
199 return
200
201 if not isinstance(active_variable, ArrayMixin):
202 raise TangentLinearError(
203 f"Assignment is to an array range but found a "
204 f"reference to the LHS variable "
205 f"'{assign.lhs.symbol.name}' without array notation"
206 f" on the RHS: '{assign.debug_string()}'")
207
208 for pos, idx in enumerate(active_variable.indices):
209 lhs_idx = assign.lhs.indices[pos]
210 # pylint: disable=unidiomatic-typecheck
211 if not (type(idx) is type(lhs_idx) and
212 lhs_idx.parent.same_range(lhs_idx.position,
213 idx.parent, pos)):
214 raise NotImplementedError(
215 f"Different sections of the same active array "
216 f"'{assign.lhs.symbol.name}' are "
217 f"accessed on the LHS and RHS of an assignment: "
218 f"'{assign.debug_string()}'. This is not supported.")

220 def validate(self, node, options=None):
221 '''Perform various checks to ensure that it is valid to apply the
222 AssignmentTrans transformation to the supplied PSyIR Node.
223
224 :param node: the node that is being checked.
225 :type node: :py:class:`psyclone.psyir.nodes.Assignment`
226 :param options: a dictionary with options for transformations.
227 :type options: Optional[Dict[str, Any]]
228
229 :raises TransformationError: if the node argument is not an \
230 Assignment.
231 :raises TangentLinearError: if the assignment does not conform \
232 to the required tangent-linear structure.
233
234 '''
235 # Check node argument is an assignment node
236 if not isinstance(node, Assignment):
237 raise TransformationError(
238 f"Node argument in assignment transformation should be a "
239 f"PSyIR Assignment, but found '{type(node).__name__}'.")
240 assign = node
241
242 # If there are no active variables then return
243 assignment_active_var_names = [
244 var.name for var in assign.walk(Reference)
245 if var.symbol in self._active_variables_active_variables]
246 if not assignment_active_var_names:
247 # No active variables in this assignment so the assignment
248 # remains unchanged.
249 return
250
251 # The lhs of the assignment node should be an active variable
252 if assign.lhs.symbol not in self._active_variables_active_variables:
253 # There are active vars on RHS but not on LHS
254 raise TangentLinearError(
255 f"Assignment node '{assign.debug_string()}' has the following "
256 f"active variables on its RHS '{assignment_active_var_names}' "
257 f"but its LHS '{assign.lhs.name}' is not an active variable.")
258
259 # Split the RHS of the assignment into <expr> +- <expr> +- <expr>
260 rhs_terms = self._split_nodes(
261 assign.rhs, [BinaryOperation.Operator.ADD,
262 BinaryOperation.Operator.SUB])
263
264 # Check for the special case where RHS=0.0. This is really a
265 # representation of multiplying an active variable by zero but
266 # this is obviously not visible in the code. Use 'float' to
267 # normalise different representations of 0.
268 if (len(rhs_terms) == 1 and isinstance(rhs_terms[0], Literal) and
269 float(rhs_terms[0].value) == 0.0):
270 return
271
272 # Check each expression term. It must be in the form
273 # A */ <expr> where A is an active variable.
274 for rhs_term in rhs_terms:
275
276 # When searching for references to an active variable we must
277 # take care to exclude those cases where they are present as
278 # arguments to the L/UBOUND intrinsics (as they will be when
279 # array notation is used).
280 active_vars = []
281 lu_bound_ops = [IntrinsicCall.Intrinsic.LBOUND,
282 IntrinsicCall.Intrinsic.UBOUND]
283 for ref in rhs_term.walk(Reference):
284 if (ref.symbol in self._active_variables_active_variables and
285 not (isinstance(ref.parent, IntrinsicCall) and
286 ref.parent.intrinsic in lu_bound_ops)):
287 active_vars.append(ref)
288
289 if not active_vars:
290 # This term must contain an active variable
291 raise TangentLinearError(
292 f"Each non-zero term on the RHS of the assignment "
293 f"'{assign.debug_string()}' must have an active variable "
294 f"but '{rhs_term.debug_string()}' does not.")
295
296 if len(active_vars) > 1:
297 # This term can only contain one active variable
298 raise TangentLinearError(
299 f"Each term on the RHS of the assignment "
300 f"'{assign.debug_string()}' must not have more than one "
301 f"active variable but '{rhs_term.debug_string()}' has "
302 f"{len(active_vars)}.")
303
304 if (isinstance(rhs_term, Reference) and rhs_term.symbol
305 in self._active_variables_active_variables):
306 self._array_ranges_match(assign, rhs_term)
307 # This term consists of a single active variable (with
308 # a multiplier of unity) and is therefore valid.
309 continue
310
311 # Ignore unary minus if it is the parent. unary minus does
312 # not cause a problem when applying the transformation but
313 # does cause a problem here in the validation when
314 # splitting the term into expressions.
315 if (isinstance(rhs_term, UnaryOperation) and
316 rhs_term.operator ==
317 UnaryOperation.Operator.MINUS):
318 rhs_term = rhs_term.children[0]
319
320 # Split the term into <expr> */ <expr> */ <expr>
321 expr_terms = self._split_nodes(
322 rhs_term, [BinaryOperation.Operator.MUL,
323 BinaryOperation.Operator.DIV])
324
325 # One of the expression terms must be an active variable
326 # or an active variable with a preceding + or -.
327 for expr_term in expr_terms:
328 check_term = expr_term
329 if (isinstance(expr_term, UnaryOperation) and
330 expr_term.operator in [UnaryOperation.Operator.PLUS,
331 UnaryOperation.Operator.MINUS]):
332 check_term = expr_term.children[0]
333 if (isinstance(check_term, Reference) and
334 check_term.symbol in self._active_variables_active_variables):
335 active_variable = check_term
336 break
337 else:
338 raise TangentLinearError(
339 f"Each term on the RHS of the assignment "
340 f"'{assign.debug_string()}' must be linear with respect "
341 f"to the active variable, but found "
342 f"'{rhs_term.debug_string()}'.")
343
344 # The term must be a product of an active variable with an
345 # inactive expression. Check that the active variable does
346 # not appear in a denominator.
347 candidate = active_variable
348 parent = candidate.parent
349 while not isinstance(parent, Assignment):
350 # Starting with the active variable reference, look up
351 # the tree for an ancestor divide operation until
352 # reaching the assignment node.
353 if (isinstance(parent, BinaryOperation) and
354 parent.operator == BinaryOperation.Operator.DIV and
355 parent.children[1] is candidate):
356 # Found a divide and the active variable is on its RHS
357 raise TangentLinearError(
358 f"In tangent-linear code an active variable cannot "
359 f"appear as a denominator but "
360 f"'{rhs_term.debug_string()}' was found in "
361 f"'{assign.debug_string()}'.")
362 # Continue up the PSyIR tree
363 candidate = parent
364 parent = candidate.parent
365
366 # If the LHS of the assignment is an array range then we only
367 # support accesses of the same variable on the RHS if they have
368 # the same range.
369 self._array_ranges_match(assign, active_variable)

371 @staticmethod
372 def _split_nodes(node, binary_operator_list):
373 '''Utility to split an expression into a series of sub-expressions
374 separated by one of the binary operators specified in
375 binary_operator_list.
376
377 :param node: the node containing the expression to split.
378 :type node: :py:class:`psyclone.psyir.nodes.DataNode`
379 :param binary_operator_list: list of binary operators.
380 :type binary_operator_list: list of
381 :py:class:`psyclone.psyir.nodes.BinaryOperations.Operator`
382
383 :returns: a list of sub-expressions.
384 :rtype: list of :py:class:`psyclone.psyir.nodes.DataNode`
385
386 '''
387 if (isinstance(node, BinaryOperation)) and \
388 (node.operator in binary_operator_list):
389 lhs_node_list = AssignmentTrans._split_nodes(
390 node.children[0], binary_operator_list)
391 rhs_node_list = AssignmentTrans._split_nodes(
392 node.children[1], binary_operator_list)
393 return lhs_node_list + rhs_node_list
394 return [node]
