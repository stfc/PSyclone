63class Symbol(CommentableMixin):
64 '''Generic Symbol item for the Symbol Table and PSyIR References.
65 It has an immutable name label because it must always match with the
66 key in the SymbolTable. If the symbol is private then it is only visible
67 to those nodes that are descendants of the Node to which its containing
68 Symbol Table belongs.
69
70 :param str name: name of the symbol.
71 :param visibility: the visibility of the symbol.
72 :type visibility: :py:class:`psyclone.psyir.symbols.Symbol.Visibility`
73 :param interface: optional object describing the interface to this \
74 symbol (i.e. whether it is passed as a routine argument or \
75 accessed in some other way). Defaults to \
76 :py:class:`psyclone.psyir.symbols.AutomaticInterface`
77 :type interface: Optional[ \
78 :py:class:`psyclone.psyir.symbols.symbol.SymbolInterface`]
79
80 :raises TypeError: if the name is not a str.
81
82 '''
83
84 class Visibility(Enum):
85 ''' Enumeration of the different visibility attributes supported in
86 the PSyIR. If no visibility information is supplied for a Symbol then
87 it is given the DEFAULT_VISIBILITY value.
88
89 PUBLIC: the symbol is visible in any scoping region that has access to
90 the SymbolTable containing it.
91 PRIVATE: the symbol is only visibile inside the scoping region that
92 contains the SymbolTable to which it belongs.
93 '''
94 PUBLIC = 1
95 PRIVATE = 2
96
97 # The visibility given to any PSyIR symbols that are created without being
98 # given an explicit visibility.
99 DEFAULT_VISIBILITY = Visibility.PUBLIC

171 def specialise(self, subclass, **kwargs):
172 '''Specialise this symbol so that it becomes an instance of the class
173 provided in the subclass argument. This allows this instance
174 to become a subclass without any references to it becoming
175 invalid.
176
177 :param subclass: the class that this symbol will become.
178 :type subclass: type of sub-class of \
179 :py:class:`psyclone.psyir.symbols.Symbol`
180
181 :raises TypeError: if subclass is not a sub-class of Symbol.
182
183 '''
184 try:
185 is_subclass = issubclass(subclass, self.__class__)
186 except TypeError as info:
187 raise TypeError(
188 f"The specialise method in '{self.name}' expects the "
189 f"subclass argument to be a class.") from info
190 # pylint: disable = unidiomatic-typecheck
191 if not is_subclass or type(self) is subclass:
192 raise TypeError(
193 f"The specialise method in '{self.name}', an instance of "
194 f"'{type(self).__name__}', expects the subclass argument to "
195 f"be a subclass of '{type(self).__name__}', but found "
196 f"'{subclass.__name__}'.")
197 self.__class__ = subclass
198 self._process_arguments(**kwargs)

201 def get_external_symbol(self):
202 '''
203 Looks-up and returns the Symbol referred to by this Symbol's
204 Import Interface.
205
206 :raises SymbolError: if the module pointed to by the symbol interface
207 does not contain the symbol (or the symbol is
208 not public).
209 :raises NotImplementedError: if this symbol does not have an
210 ImportInterface.
211 '''
212 if not self.is_importis_import:
213 raise NotImplementedError(
214 f"Error trying to resolve symbol '{self.name}' properties, "
215 f"the lazy evaluation of '{self.interface}' interfaces is "
216 f"not supported.")
217
218 csym = self.interfaceinterfaceinterface.container_symbol
219
220 try:
221 container = csym.find_container_psyir()
222 if not container:
223 raise SymbolError(
224 f"Error trying to resolve the properties of symbol "
225 f"'{self.name}'. The interface points to module "
226 f"'{csym.name}' but could not obtain its PSyIR.")
227 return container.symbol_table.lookup(
228 self.namename, visibility=self.Visibility.PUBLIC)
229 except KeyError as kerr:
230 raise SymbolError(
231 f"Error trying to resolve the properties of symbol "
232 f"'{self.name}'. The interface points to module "
233 f"'{csym.name}' but could not find the definition of "
234 f"'{self.name}' in that module.") from kerr
235 except SymbolError as err:
236 raise SymbolError(
237 f"Error trying to resolve the properties of symbol "
238 f"'{self.name}' in module '{csym.name}': {err.value}") from err

240 def resolve_type(self):
241 '''
242 Update the properties of this Symbol by using the definition imported
243 from the external Container. If this symbol does not have an
244 ImportInterface then there is no lookup needed and we just return this
245 symbol.
246
247 :returns: a symbol object with the class and type determined by
248 examining the Container from which it is imported.
249 :rtype: subclass of :py:class:`psyclone.psyir.symbols.Symbol`
250
251 '''
252 if self.is_importis_import:
253 extern_symbol = self.get_external_symbol()
254 from psyclone.psyir.symbols import RoutineSymbol
255 if isinstance(extern_symbol, RoutineSymbol):
256 # Specialise the existing Symbol in-place so that all
257 # References to it remain valid.
258 self.specialise(type(extern_symbol),
259 datatype=extern_symbol.datatype)
260 else:
261 init_value = None
262 if extern_symbol.initial_value:
263 init_value = extern_symbol.initial_value.copy()
264 # Specialise the existing Symbol in-place so that all
265 # References to it remain valid.
266 self.specialise(type(extern_symbol),
267 datatype=extern_symbol.datatype,
268 is_constant=extern_symbol.is_constant,
269 initial_value=init_value)
270 return self

402 def find_symbol_table(self, node):
403 '''
404 Searches back up the PSyIR tree for the SymbolTable that contains
405 this Symbol.
406
407 :param node: the PSyIR node from which to search.
408 :type node: :py:class:`psyclone.psyir.nodes.Node`
409
410 :returns: the SymbolTable containing this Symbol or None.
411 :rtype: :py:class:`psyclone.psyir.symbols.SymbolTable` or NoneType
412
413 :raises TypeError: if the supplied `node` argument is not a PSyIR Node.
414
415 '''
416 # This import has to be local to this method to avoid circular
417 # dependencies.
418 # pylint: disable=import-outside-toplevel
419 from psyclone.psyir.nodes import Node
420 if not isinstance(node, Node):
421 raise TypeError(
422 f"find_symbol_table: expected to be passed an instance of "
423 f"psyir.nodes.Node but got '{type(node).__name__}'")
424
425 try:
426 current = node.scope.symbol_table
427 while current:
428 if self.namename in current:
429 return current
430 if current.node.parent:
431 current = current.node.parent.scope.symbol_table
432 else:
433 # We can't go any further up the hierarchy and we haven't
434 # found a SymbolTable that contains this symbol.
435 return None
436 except SymbolError:
437 # Failed to find any enclosing symbol table
438 return None

453 def is_array_access(self, index_variable=None, access_info=None):
454 '''This method detects if a variable is used as an array or not.
455 If available, it will use the access information, i.e. how the
456 variable is used (e.g. if it has indices somewhere, like `a(i)%b`).
457 This can be incorrect in case of implicit loops (e.g. `a=b+1`,
458 where `a` and `b` are arrays) where the variable usage information
459 will not have information about indices. In this case this function
460 will fallback to querying the symbol itself.
461
462 This can cause significant slowdown if this symbol is imported
463 from a module, since then these modules need to be parsed.
464 # TODO #1213: Parsing modules is not yet supported.
465
466 If a `loop_variable` is specified, a variable access will only be
467 considered an array access if the specified variable is used in
468 at least one of the indices. For example:
469 >>> do i=1, n
470 >>> a(j) = 2
471
472 the access to `a` is not considered an array if `loop_variable` is
473 set to `i`. If `loop_variable` is specified, `access_information`
474 must be specified.
475
476 :param str index_variable: optional loop variable that is used to
477 to determine if an access is an array access using this variable.
478 :param access_info: variable access information, optional.
479 :type access_info:
480 :py:class:`psyclone.core.SingleVariableAccessInfo`
481
482 :returns: whether or not the variable is an array.
483 :rtype bool:
484
485 :raises InternalError: if a loop_variable is specified, but no
486 access information is given.
487
488 '''
489 # TODO #1270: this function might either be better off elsewhere,
490 # or even do not implement one function that uses both access
491 # information and symbol table - if required, the user can
492 # query both in two simple statements anyway.
493 if index_variable and not access_info:
494 raise InternalError(f"In Symbol.is_array_access: index variable "
495 f"'{index_variable}' specified, but no access "
496 f"information given.")
497
498 # TODO #1244: If as a result of 1244 we have more reliable
499 # information in the symbol table, the implementation here might
500 # be changed.
501
502 # Prioritise access information, since this can take the index
503 # variable into account, and avoids potentially parsing other
504 # modules to find type information.
505 if access_info:
506 # Access Info might not have information if a variable is used
507 # as array (e.g. in case of an array expression). In this case
508 # we still need to check the type information in the symbol table.
509 is_array = access_info.is_array(index_variable=index_variable)
510
511 # Access information might indicate that a variable is not an
512 # array if it is used in array expressions only. In order to
513 # verify, we need to check the symbol type information. But
514 # if an index variable is specified, an array used in an array
515 # expression without the index variable is not considered to
516 # be an array access, since it is independent of the loop
517 # variable. In this case also return the value from `is_array`.
518 if is_array or index_variable is not None:
519 return is_array
520
521 # Either we don't have access information, or the access information
522 # does not indicate an array. In the latter case we still need to
523 # test the symbol table, since the variable might be used in array
524 # expressions only. Note that we cannot check for index variable usage
525 # in this case.
526 # TODO #1213: check for wildcard imports
527 return self.is_array
