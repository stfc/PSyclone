39'''
40This module contains the LFRicFields class which manages the
41declarations for field arguments required by an Invoke or
42Kernel stub.
43'''

45# Imports
46from collections import OrderedDict
47
48from psyclone import psyGen
49from psyclone.domain.lfric import LFRicCollection, LFRicConstants
50from psyclone.errors import InternalError
51from psyclone.f2pygen import DeclGen, TypeDeclGen

54class LFRicFields(LFRicCollection):
55 '''
56 Manages the declarations for all field arguments required by an Invoke
57 or Kernel stub.
58
59 '''
60 def _invoke_declarations(self, parent):
61 '''
62 Add field-related declarations to the PSy-layer routine.
63 Note: PSy layer in LFRic does not modify the field objects. Hence,
64 their Fortran intents are always in (the data updated in the kernels
65 is only pointed to from the field object and is thus not a part of
66 the object).
67
68 :param parent: the node in the f2pygen AST representing the PSy-layer
69 routine to which to add declarations.
70 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
71
72 :raises InternalError: for unsupported intrinsic types of field
73 argument data.
74
75 '''
76 # Create dict of all field arguments for checks
77 const = LFRicConstants()
78 fld_args = self._invoke.unique_declarations(
79 argument_types=const.VALID_FIELD_NAMES)
80 # Filter field arguments by intent and intrinsic type
81 real_field_args = self._invoke.unique_declarations(
82 argument_types=const.VALID_FIELD_NAMES,
83 intrinsic_type=const.MAPPING_DATA_TYPES["gh_real"])
84 int_field_args = self._invoke.unique_declarations(
85 argument_types=const.VALID_FIELD_NAMES,
86 intrinsic_type=const.MAPPING_DATA_TYPES["gh_integer"])
87
88 # Create lists of field names for real- and integer-valued fields
89 fld_arg_list = [arg.declaration_name for arg in fld_args]
90 real_field_arg_list = [arg.declaration_name for arg in real_field_args]
91 int_field_arg_list = [arg.declaration_name for arg in int_field_args]
92 # Check for unsupported intrinsic types
93 fld_inv = (set(fld_arg_list) -
94 set(real_field_arg_list).union(set(int_field_arg_list)))
95 if fld_inv:
96 raise InternalError(
97 f"Found unsupported intrinsic types for the field arguments "
98 f"{list(fld_inv)} to Invoke '{self._invoke.name}'. Supported "
99 f"types are {const.VALID_FIELD_INTRINSIC_TYPES}.")
100
101 # Create a field argument map that splits the (real and
102 # integer) fields into their different datatypes.
103 field_datatype_map = OrderedDict()
104 for arg in real_field_args + int_field_args:
105 try:
106 field_datatype_map[
107 (arg.data_type, arg.module_name)].append(arg)
108 except KeyError:
109 # This datatype has not been seen before so create a
110 # new entry
111 field_datatype_map[(arg.data_type, arg.module_name)] = [arg]
112
113 # Add the Invoke subroutine argument declarations for the
114 # different fields types. They are declared as intent "in" as
115 # they contain a pointer to the data that is modified.
116 for fld_type, fld_mod in field_datatype_map:
117 args = field_datatype_map[(fld_type, fld_mod)]
118 arg_list = [arg.declaration_name for arg in args]
119 parent.add(TypeDeclGen(parent, datatype=fld_type,
120 entity_decls=arg_list,
121 intent="in"))
122 (self._invoke.invokes.psy.
123 infrastructure_modules[fld_mod].add(fld_type))
124
125 def _stub_declarations(self, parent):
126 '''
127 Add field-related declarations to a Kernel stub.
128
129 :param parent: the node in the f2pygen AST representing the Kernel
130 stub to which to add declarations.
131 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
132
133 :raises InternalError: for an unsupported data type of field
134 argument data.
135
136 '''
137 const = LFRicConstants()
138
139 fld_args = psyGen.args_filter(
140 self._kernel.args, arg_types=const.VALID_FIELD_NAMES)
141 for fld in fld_args:
142 undf_name = fld.function_space.undf_name
143 fld_dtype = fld.intrinsic_type
144 fld_kind = fld.precision
145
146 # Check for invalid descriptor data type
147 fld_ad_dtype = fld.descriptor.data_type
148 if fld_ad_dtype not in const.VALID_FIELD_DATA_TYPES:
149 raise InternalError(
150 f"Found an unsupported data type '{fld_ad_dtype}' in "
151 f"kernel stub declarations for the field argument "
152 f"'{fld.declaration_name}'. Supported types are "
153 f"{const.VALID_FIELD_DATA_TYPES}.")
154
155 if fld.vector_size > 1:
156 for idx in range(1, fld.vector_size+1):
157 text = (fld.name + "_" +
158 fld.function_space.mangled_name +
159 "_v" + str(idx))
160 parent.add(
161 DeclGen(parent, datatype=fld_dtype, kind=fld_kind,
162 dimension=undf_name,
163 intent=fld.intent, entity_decls=[text]))
164 else:
165 parent.add(
166 DeclGen(parent, datatype=fld_dtype, kind=fld_kind,
167 intent=fld.intent,
168 dimension=undf_name,
169 entity_decls=[fld.name + "_" +
170 fld.function_space.mangled_name]))
