49class GOceanExtractTrans(ExtractTrans):
50 ''' GOcean1.0 API application of ExtractTrans transformation \
51 to extract code into a stand-alone program. For example:
52
53 >>> from psyclone.parse.algorithm import parse
54 >>> from psyclone.psyGen import PSyFactory
55 >>>
56 >>> API = "gocean"
57 >>> FILENAME = "shallow_alg.f90"
58 >>> ast, invokeInfo = parse(FILENAME, api=API)
59 >>> psy = PSyFactory(API, distributed_memory=False).create(invoke_info)
60 >>> schedule = psy.invokes.get('invoke_0').schedule
61 >>>
62 >>> from psyclone.domain.gocean.transformations import GOceanExtractTrans
63 >>> etrans = GOceanExtractTrans()
64 >>>
65 >>> # Apply GOceanExtractTrans transformation to selected Nodes
66 >>> etrans.apply(schedule.children[0])
67 >>> print(schedule.view())
68 '''

70 def __init__(self):
71 super().__init__(ExtractNode)
72 # Set the integer and real types to use. If required, the constructor
73 # could take a parameter to change these.
74
75 self._driver_creator = ExtractDriverCreator(INTEGER_TYPE, REAL8_TYPE)

77 # ------------------------------------------------------------------------
78 def validate(self, node_list, options=None):
79 ''' Perform GOcean1.0 API specific validation checks before applying
80 the transformation.
81
82 :param node_list: the list of Node(s) we are checking.
83 :type node_list: list of :py:class:`psyclone.psyir.nodes.Node`
84 :param options: a dictionary with options for transformations.
85 :type options: Optional[Dict[str, Any]]
86 :param bool options["create_driver"]: whether or not to create a \
87 driver program at code-generation time. If set, the driver will \
88 be created in the current working directory with the name \
89 "driver-MODULE-REGION.f90" where MODULE and REGION will be the \
90 corresponding values for this region. This flag is forwarded to \
91 the ExtractNode. Its default value is False.
92 :param (str,str) options["region_name"]: an optional name to \
93 use for this data-extraction region, provided as a 2-tuple \
94 containing a module name followed by a local name. The pair of \
95 strings should uniquely identify a region unless aggregate \
96 information is required (and is supported by the runtime \
97 library). This option is forwarded to the PSyDataNode (where it \
98 changes the region names) and to the ExtractNode (where it \
99 changes the name of the created output files and the name of the \
100 driver program).
101
102 :raises TransformationError: if transformation is applied to an \
103 inner Loop without its parent outer Loop.
104 '''
105
106 # First check constraints on Nodes in the node_list inherited from
107 # the parent classes (ExtractTrans and RegionTrans)
108 super().validate(node_list, options)
109
110 # Check GOceanExtractTrans specific constraints
111 for node in node_list:
112
113 # Check that ExtractNode is not inserted between an inner
114 # and an outer Loop.
115 ancestor = node.ancestor(GOLoop)
116 if ancestor and ancestor.loop_type == 'outer':
117 raise TransformationError(
118 f"Error in {self.name}: Application to an "
119 f"inner Loop without its ancestor outer Loop is not "
120 f"allowed.")

123 def apply(self, nodes, options=None):
124 # pylint: disable=arguments-differ
125 '''Apply this transformation to a subset of the nodes within a
126 schedule - i.e. enclose the specified Nodes in the schedule within
127 a single PSyData region. Note that this implementation just calls
128 the base class, it is only added here to provide the documentation
129 for this function, since it accepts different options
130 to the base class (e.g. create_driver, which is passed to the
131 ExtractNode instance that will be inserted.).
132
133 :param nodes: can be a single node or a list of nodes.
134 :type nodes: :py:class:`psyclone.psyir.nodes.Node` or list of \
135 :py:class:`psyclone.psyir.nodes.Node`
136 :param options: a dictionary with options for transformations.
137 :type options: Optional[Dict[str, Any]]
138 :param str options["prefix"]: a prefix to use for the PSyData module \
139 name (``prefix_psy_data_mod``) and the PSyDataType \
140 (``prefix_PSyDataType``) - a "_" will be added automatically. \
141 It defaults to "extract", resulting in e.g. \
142 ``extract_psy_data_mod``.
143 :param bool options["create_driver"]: whether or not to create a \
144 driver program at code-generation time. If set, the driver will \
145 be created in the current working directory with the name \
146 "driver-MODULE-REGION.f90" where MODULE and REGION will be the \
147 corresponding values for this region. Defaults to False.
148 :param (str,str) options["region_name"]: an optional name to \
149 use for this PSyData area, provided as a 2-tuple containing a \
150 location name followed by a local name. The pair of strings \
151 should uniquely identify a region unless aggregate information \
152 is required (and is supported by the runtime library).
153
154 '''
155 if options is None:
156 my_options = {}
157 else:
158 # We will add a default prefix, so create a copy to avoid
159 # changing the user's options:
160 my_options = options.copy()
161
162 ctu = CallTreeUtils()
163 nodes = self.get_node_list(nodes)
164 region_name = self.get_unique_region_name(nodes, my_options)
165 my_options["region_name"] = region_name
166 my_options["prefix"] = my_options.get("prefix", "extract")
167
168 read_write_info = ctu.get_in_out_parameters(
169 nodes, include_non_data_accesses=True)
170
171 # Even variables that are output-only need to be written with their
172 # values at the time the kernel is called: many kernels will only
173 # write to part of a field (e.g. in case of MPI the halo region
174 # will not be written). Since the comparison in the driver uses
175 # the whole field (including values not updated), we need to write
176 # the current value of an output-only field as well. This is
177 # achieved by adding any written-only field to the list of fields
178 # read. This will trigger to write the values in the extraction,
179 # and the driver code created will read in their values.
180 for sig in read_write_info.write_list:
181 if sig not in read_write_info.read_list:
182 read_write_info.read_list.append(sig)
183
184 # Determine a unique postfix to be used for output variables
185 # that avoid any name clashes
186 postfix = ExtractTrans.determine_postfix(read_write_info,
187 postfix="_post")
188 my_options["post_var_postfix"] = postfix
189
190 if my_options.get("create_driver", False):
191 # We need to create the driver before inserting the ExtractNode
192 # (since some of the visitors used in driver creation do not
193 # handle an ExtractNode in the tree)
194 self._driver_creator.write_driver(nodes, read_write_info,
195 postfix=postfix,
196 prefix=my_options["prefix"],
197 region_name=region_name)
198
199 my_options["read_write_info"] = read_write_info
200 super().apply(nodes, options=my_options)
