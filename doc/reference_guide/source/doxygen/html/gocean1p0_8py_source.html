128class GOInvokes(Invokes):
129 '''
130 The GOcean specific invokes class. This passes the GOcean specific
131 invoke class to the base class so it creates the one we require.
132
133 :param alg_calls: The Invoke calls discovered in the Algorithm layer.
134 :type alg_calls: OrderedDict of :py:class:`psyclone.parse.InvokeCall` \
135 objects.
136 :param psy: the PSy object containing this GOInvokes object.
137 :type psy: :py:class:`psyclone.gocean1p0.GOPSy`
138
139 '''
140 def __init__(self, alg_calls, psy):
141 self._0_to_n = GOInvoke(None, None, None) # for pyreverse
142 Invokes.__init__(self, alg_calls, GOInvoke, psy)
143
144 index_offsets = []
145 # Loop over all of the kernels in all of the invoke() calls
146 # and check that they work on compatible grid-index offsets.
147 # Strictly speaking this check should be done in the parsing
148 # code since it is a check on the correctness of the meta-data.
149 # However, that would require a fundamental change to the parsing
150 # code since it requires information on all of the invokes and
151 # kernels in an application. Therefore it is much simpler to
152 # do it here where we have easy access to that information.
153 for invoke in self.invoke_listinvoke_list:
154 for kern_call in invoke.schedule.coded_kernels():
155 # We only care if the index offset is not offset_any (since
156 # that is compatible with any other offset)
157 if kern_call.index_offset != "go_offset_any":
158 # Loop over the offsets we've seen so far
159 for offset in index_offsets:
160 if offset != kern_call.index_offset:
161 raise GenerationError(
162 f"Meta-data error in kernel {kern_call.name}: "
163 f"INDEX_OFFSET of '{kern_call.index_offset}' "
164 f"does not match that ({offset}) of other "
165 f"kernels. This is not supported.")
166 # Append the index-offset of this kernel to the list of
167 # those seen so far
168 index_offsets.append(kern_call.index_offset)

170 def gen_code(self, parent):
171 '''
172 GOcean redefines the Invokes.gen_code() to start using the PSyIR
173 backend when possible. In cases where the backend can not be used yet
174 (e.g. OpenCL and PSyDataNodes) the parent class will be called. This
175 is a temporary workaround to avoid modifying the generator file while
176 other APIs still use the f2pygen module for code generation.
177 Once the PSyIR backend has generated an output, this is added into a
178 f2pygen PSyIRGen block in the f2pygen AST for each Invoke in the
179 PSy layer.
180
181 :param parent: the parent node in the f2pygen AST to which to add \
182 content.
183 :type parent: `psyclone.f2pygen.ModuleGen`
184 '''
185 if self.invoke_listinvoke_list:
186 # We just need one invoke as they all have a common root.
187 invoke = self.invoke_listinvoke_list[0]
188
189 # Lower the GOcean PSyIR to language level so it can be visited
190 # by the backends
191 invoke.schedule.parent.lower_to_language_level()
192 # Then insert it into a f2pygen AST as a PSyIRGen node.
193 # Note that other routines besides the Invoke could have been
194 # inserted during the lowering (e.g. module-inlined kernels),
195 # so have to iterate over all current children of parent.
196 for child in invoke.schedule.parent.children:
197 parent.add(PSyIRGen(parent, child))

703 @staticmethod
704 def add_bounds(bound_info):
705 # pylint: disable=too-many-locals
706 '''
707 Adds a new iteration space to PSyclone. An iteration space in the
708 gocean API is for a certain offset type and field type. It defines
709 the loop boundaries for the outer and inner loop. The format is a
710 ":" separated tuple:
711
712 >>> bound_info = offset-type:field-type:iteration-space:outer-start:
713 outer-stop:inner-start:inner-stop
714
715 Example:
716
717 >>> bound_info = go_offset_ne:go_ct:go_all_pts:
718 {start}-1:{stop}+1:{start}:{stop}
719
720 The expressions {start} and {stop} will be replaced with the loop
721 indices that correspond to the inner points (i.e. non-halo or
722 boundary points) of the field. So the index {start}-1 is actually
723 on the halo / boundary.
724
725 :param str bound_info: A string that contains a ":" separated \
726 tuple with the iteration space definition.
727
728 :raises ValueError: if bound_info is not a string.
729 :raises ConfigurationError: if bound_info is not formatted correctly.
730
731 '''
732 if not isinstance(bound_info, str):
733 raise InternalError(f"The parameter 'bound_info' must be a "
734 f"string, got '{bound_info}' "
735 f"(type {type(bound_info)})")
736
737 data = bound_info.split(":")
738 if len(data) != 7:
739 raise ConfigurationError(f"An iteration space must be in the form "
740 f"\"offset-type:field-type:"
741 f"iteration-space:outer-start:"
742 f"outer-stop:inner-start:inner-stop\"\n"
743 f"But got \"{bound_info}\"")
744
745 if not GOLoop._bounds_lookup:
746 GOLoop.setup_bounds()
747
748 # Check that all bound specifications (min and max index) are valid.
749 # ------------------------------------------------------------------
750 # Regular expression that finds stings surrounded by {}
751 bracket_regex = re.compile("{[^}]+}")
752 for bound in data[3:7]:
753 all_expr = bracket_regex.findall(bound)
754 for bracket_expr in all_expr:
755 if bracket_expr not in ["{start}", "{stop}"]:
756 raise ConfigurationError(f"Only '{{start}}' and "
757 f"'{{stop}}' are allowed as "
758 f"bracketed expression in an "
759 f"iteration space. But got "
760 f"{bracket_expr}")
761
762 # We need to make sure the fparser is properly initialised, which
763 # typically has not yet happened when the config file is read.
764 # Otherwise the Nonlabel_Do_Stmt cannot parse valid expressions.
765 ParserFactory().create(std="f2008")
766
767 # Test both the outer loop indices (index 3 and 4) and inner
768 # indices (index 5 and 6):
769 for bound in data[3:7]:
770 do_string = f"do i=1, {bound}"
771 # Now replace any {start}/{stop} expression in the loop
772 # with a valid integer value:
773 do_string = do_string.format(start='15', stop='25')
774 # Check if the do loop can be parsed as a nonlabel do loop
775 try:
776 _ = Nonlabel_Do_Stmt(do_string)
777 except NoMatchError as err:
778 raise ConfigurationError(f"Expression '{bound}' is not a "
779 f"valid do loop boundary. Error "
780 f"message: '{err}'.") from err
781
782 # All tests successful, so add the new bounds:
783 # --------------------------------------------
784 current_bounds = GOLoop._bounds_lookup # Shortcut
785 # Check offset-type exists
786 if not data[0] in current_bounds:
787 current_bounds[data[0]] = {}
788
789 # Check field-type exists
790 if not data[1] in current_bounds[data[0]]:
791 current_bounds[data[0]][data[1]] = {}
792
793 const = GOceanConstants()
794 # Check iteration space exists:
795 if not data[2] in current_bounds[data[0]][data[1]]:
796 current_bounds[data[0]][data[1]][data[2]] = {}
797 const.VALID_ITERATES_OVER.append(data[2])
798
799 current_bounds[data[0]][data[1]][data[2]] = \
800 {'outer': {'start': data[3], 'stop': data[4]},
801 'inner': {'start': data[5], 'stop': data[6]}}

1133 def _record_stencil_accesses(self, signature, arg, var_accesses):
1134 '''This function adds accesses to a field depending on the
1135 meta-data declaration for this argument (i.e. accounting for
1136 any stencil accesses).
1137
1138 :param signature: signature of the variable.
1139 :type signature: :py:class:`psyclone.core.Signature`

1156 # depth will be 1, so one access to (i,j) is then added.

1166 current_depth)

1168 # can only ever write to (i,j), so any other references
1169 # must be read:
1170 if i == 0 and j == 0:
1171 acc = arg.access
1172 else:
1173 acc = AccessType.READ

1175 var_accesses.add_access(signature, acc, self,
1176 [i_expr, j_expr])

1707class GOStencil():
1708 '''GOcean 1.0 stencil information for a kernel argument as obtained by
1709 parsing the kernel meta-data. The expected structure of the
1710 metadata and its meaning is provided in the description of the
1711 load method
1712
1713 '''
1714 def __init__(self):
1715 ''' Set up any internal variables. '''
1716 self._has_stencil = None
1717 self._stencil = [[0 for _ in range(3)] for _ in range(3)]
1718 self._name = None
1719 self._initialised = False
1720
1721 # pylint: disable=too-many-branches
1722 def load(self, stencil_info, kernel_name):
1723 '''Take parsed stencil metadata information, check it is valid and
1724 store it in a convenient form. The kernel_name argument is
1725 only used to provide the location if there is an error.
1726
1727 The stencil information should either be a name which
1728 indicates a particular type of stencil or in the form
1729 stencil(xxx,yyy,zzz) which explicitly specifies a stencil
1730 shape where xxx, yyy and zzz are triplets of integers
1731 indicating whether there is a stencil access in a particular
1732 direction and the depth of that access. For example:
1733
1734 go_stencil(010, ! N
1735 212, ! W E
1736 010) ! S
1737
1738 indicates that there is a stencil access of depth 1 in the
1739 "North" and "South" directions and stencil access of depth 2
1740 in the "East" and "West" directions. The value at the centre
1741 of the stencil will not be used by PSyclone but can be 0 or 1
1742 and indicates whether the local field value is accessed.
1743
1744 The convention is for the associated arrays to be
1745 2-dimensional. If we denote the first dimension as "i" and the
1746 second dimension as "j" then the following directions are
1747 assumed:
1748
1749
1750 > j
1751 > ^
1752 > |
1753 > |
1754 > ---->i
1755
1756 For example a stencil access like:
1757
1758 a(i,j) + a(i+1,j) + a(i,j-1)
1759
1760 would be stored as:
1761
1762 go_stencil(000,
1763 011,
1764 010)
1765
1766 :param stencil_info: contains the appropriate part of the parser AST
1767 :type stencil_info: :py:class:`psyclone.expression.FunctionVar`
1768 :param string kernel_name: the name of the kernel from where this \
1769 stencil information came from.
1770
1771 :raises ParseError: if the supplied stencil information is invalid.
1772
1773 '''
1774 self._initialised = True
1775
1776 if not isinstance(stencil_info, expr.FunctionVar):
1777 # the stencil information is not in the expected format
1778 raise ParseError(
1779 f"Meta-data error in kernel '{kernel_name}': 3rd descriptor "
1780 f"(stencil) of field argument is '{stencil_info}' but "
1781 f"expected either a name or the format 'go_stencil(...)'")
1782
1783 # Get the name
1784 name = stencil_info.name.lower()
1785 const = GOceanConstants()
1786
1787 if stencil_info.args:
1788 # The stencil info is of the form 'name(a,b,...), so the
1789 # name should be 'stencil' and there should be 3
1790 # arguments'
1791 self._has_stencil = True
1792 args = stencil_info.args
1793 if name != "go_stencil":
1794 raise ParseError(
1795 f"Meta-data error in kernel '{kernel_name}': 3rd "
1796 f"descriptor (stencil) of field argument is '{name}' but "
1797 f"must be 'go_stencil(...)")
1798 if len(args) != 3:
1799 raise ParseError(
1800 f"Meta-data error in kernel '{kernel_name}': 3rd "
1801 f"descriptor (stencil) of field argument with format "
1802 f"'go_stencil(...)', has {len(args)} arguments but should "
1803 f"have 3")
1804 # Each of the 3 args should be of length 3 and each
1805 # character should be a digit from 0-9. Whilst we are
1806 # expecting numbers, the parser represents these numbers
1807 # as strings so we have to perform string manipulation to
1808 # check and that extract them
1809 for arg_idx in range(3):
1810 arg = args[arg_idx]
1811 if not isinstance(arg, str):
1812 raise ParseError(
1813 f"Meta-data error in kernel '{kernel_name}': 3rd "
1814 f"descriptor (stencil) of field argument with format "
1815 f"'go_stencil(...)'. Argument index {arg_idx} should "
1816 f"be a number but found '{arg}'.")
1817 if len(arg) != 3:
1818 raise ParseError(
1819 f"Meta-data error in kernel '{kernel_name}': 3rd "
1820 f"descriptor (stencil) of field argument with format "
1821 f"'go_stencil(...)'. Argument index {arg_idx} should "
1822 f"consist of 3 digits but found {len(arg)}.")
1823 # The central value is constrained to be 0 or 1
1824 if args[1][1] not in ["0", "1"]:
