44def node_is_active(node, active_variables):
45 '''Determines whether this node contains variables that are active.
46
47 :param node: the PSyIR node that is being evaluated.
48 :type node: :py:class:`psyclone.psyir.nodes.Node`
49 :param active_variables: a list of active variables either as \
50 PSyIR symbols or as strings.
51 :type active_variables: \
52 List[:py:class:`psyclone.psyir.symbols.DataSymbol`] | \
53 List[str]
54
55 :returns: True if active and False otherwise.
56 :rtype: bool
57
58 :raises TypeError: if the node argument is not a Node.
59 :raises TypeError: if the active_variables argument is not a list.
60 :raises ValueError: if the active variables list does not contain \
61 datasymbols or if the list does not contain strings.
62
63 '''
64 if not isinstance(node, Node):
65 raise TypeError(
66 f"The node argument to the node_is_active() method should be a "
67 f"PSyIR Node, but found {type(node).__name__}")
68 if not isinstance(active_variables, list):
69 raise TypeError(
70 f"The active_variables argument to the node_is_active() method "
71 f"should be a list, but found {type(active_variables).__name__}.")
72 if not (all(isinstance(item, DataSymbol) for item in active_variables) or
73 all(isinstance(item, str) for item in active_variables)):
74 item_types = [type(item).__name__ for item in active_variables]
75 raise ValueError(
76 f"Expected the active_variables argument to the node_is_active() "
77 f"method to be a list containing either solely PSyIR DataSymbols "
78 f"or solely strings, but found {item_types}.")
79
80 if active_variables and isinstance(active_variables[0], str):
81 for reference in node.walk(Reference):
82 if reference.symbol.name.lower() in active_variables:
83 return True
84 else:
85 for reference in node.walk(Reference):
86 if reference.symbol in active_variables:
87 return True
88 return False

91def node_is_passive(node, active_variables):
92 '''Determines whether this node contains only variables that are
93 passive.
94
95 :param node: the PSyIR node that is being evaluated.
96 :type node: :py:class:`psyclone.psyir.nodes.Node`
97 :param active_variables: a list of active variables either as \
98 PSyIR symbols or as strings.
99 :type active_variables: \
100 List[:py:class:`psyclone.psyir.symbols.DataSymbol`] or \
101 List[str]
102
103 :returns: True if passive and False otherwise.
104 :rtype: bool
105
106 '''
107 return not node_is_active(node, active_variables)

110def negate_expr(orig_expr):
111 '''Takes a PSyIR expression and negates it by multiplying it by minus
112 one. it is assumed that the expression returns an integer.
113
114 :param orig_expr: the PSyIR expression to negate.
115 :type orig_expr: :py:class:`psyclone.psyir.nodes.DataNode`
116
117 :returns: the expression multiplied by minus one.
118 :rytpe: :py:class:`psyclone.psyir.nodes.DataNode`
119
120 '''
121 expr = orig_expr.copy()
122 # TODO, this would be better using sympy see #1497
123 if isinstance(expr, Literal):
124 if expr.value.lstrip()[0] == "-":
125 # The literal has a negative value so make it positive.
126 # pylint: disable=protected-access
127 expr._value = expr.value.lstrip()[1:]
128 return expr
129 # Negate the literal with a unary minus.
130 return UnaryOperation.create(
131 UnaryOperation.Operator.MINUS, expr)
132 if (isinstance(expr, UnaryOperation) and
133 expr.operator == UnaryOperation.Operator.MINUS):
134 return expr.children[0].detach()
135 return BinaryOperation.create(
136 BinaryOperation.Operator.MUL, Literal("-1", INTEGER_TYPE), expr)
