57class LFRicKernMetadata(KernelType):
58 '''
59 Captures the Kernel subroutine code and metadata describing
60 the subroutine for the LFRic API.
61
62 :param ast: fparser1 AST for the kernel.
63 :type ast: :py:class:`fparser.block_statements.BeginSource`
64 :param str name: the name of this kernel.
65
66 :raises ParseError: if the metadata does not conform to the
67 rules for the LFRic API.
68 '''
69 # pylint: disable=too-many-instance-attributes
70 def __init__(self, ast, name=None):
71 # pylint: disable=too-many-branches, too-many-locals
72
73 KernelType.__init__(self, ast, name=name)
74
75 # The type of CMA operation this kernel performs (or None if
76 # no CMA operators are involved)
77 self._cma_operation = None
78
79 # Query the metadata for the evaluator shape(s) (only required if
80 # kernel uses quadrature or an evaluator). If it is not
81 # present then 'eval_shapes' will be an empty list.
82 shape = self.get_integer_variable('gh_shape')
83 if not shape:
84 # There's no scalar 'gh_shape' - is it present as an array?
85 self._eval_shapes = self.get_integer_array('gh_shape')
86 else:
87 self._eval_shapes = [shape]
88
89 # The list of function space names for which an evaluator is
90 # required. We set this up below once we have processed the
91 # metadata describing the kernel arguments.
92 self._eval_targets = []
93
94 # Whether or not this is an inter-grid kernel (i.e. has a mesh
95 # specified for each [field] argument). This property is
96 # set to True if all the checks in '_validate_inter_grid()' pass.
97 self._is_intergrid = False
98
99 # Parse the 'arg_type' metadata
100 self._arg_descriptors_arg_descriptors_arg_descriptors = []
101 for idx, arg_type in enumerate(self._inits_inits):
102 self._arg_descriptors_arg_descriptors_arg_descriptors.append(
103 LFRicArgDescriptor(arg_type, self.iterates_overiterates_over, idx))
104
105 # Get a list of the Type declarations in the metadata
106 type_declns = [cline for cline in self._ktype.content if
107 isinstance(cline, fparser.one.typedecl_statements.Type)]
108
109 # Parse the 'func_type' metadata if it exists
110 func_types = []
111 for line in type_declns:
112 for entry in line.selector:
113 if entry == "func_type":
114 func_types = getkerneldescriptors(
115 name, line, var_name="meta_funcs",
116 var_type="func_type")
117 break
118
119 self._func_descriptors = []
120 # Populate a list of function descriptor objects which we
121 # return via the 'func_descriptors' method.
122 arg_fs_names = []
123 for descriptor in self._arg_descriptors_arg_descriptors_arg_descriptors:
124 arg_fs_names.extend(descriptor.function_spaces)
125 used_fs_names = []
126 need_evaluator = False
127 for func_type in func_types:
128 descriptor = DynFuncDescriptor03(func_type)
129 fs_name = descriptor.function_space_name
130 # Check that function space names in 'meta_funcs' are specified in
131 # 'meta_args'.
132 if fs_name not in arg_fs_names:
133 raise ParseError(
134 f"In the LFRic API all function spaces specified in "
135 f"'meta_funcs' must exist in 'meta_args', but '{fs_name}' "
136 f"breaks this rule in ...\n'{self._ktype.content}'.")
137 if fs_name not in used_fs_names:
138 used_fs_names.append(fs_name)
139 else:
140 raise ParseError(
141 f"In the LFRic API function spaces specified in "
142 f"'meta_funcs' must be unique, but '{fs_name}' is "
143 f"replicated.")
144
145 const = LFRicConstants()
146 # Check that a valid shape has been specified if
147 # this function space requires a basis or differential basis
148 for op_name in descriptor.operator_names:
149 if op_name in const.VALID_EVALUATOR_NAMES:
150 need_evaluator = True
151 if not self._eval_shapes:
152 raise ParseError(
153 f"In the LFRic API any kernel requiring "
154 f"quadrature or an evaluator "
155 f"({const.VALID_EVALUATOR_NAMES}) must also "
156 f"supply the shape of that evaluator by setting "
157 f"'gh_shape' in the kernel metadata but "
158 f"this is missing for kernel '{self.name}'")
159 shape_set = set(self._eval_shapes)
160 if not shape_set.issubset(
161 set(const.VALID_EVALUATOR_SHAPES)):
162 raise ParseError(
163 f"In the LFRic API a kernel requiring either "
164 f"quadrature or an evaluator must request one or "
165 f"more valid 'gh_shapes' (one of "
166 f"{const.VALID_EVALUATOR_SHAPES}) but got "
167 f"'{self._eval_shapes}' for kernel '{self.name}'")
168
169 self._func_descriptors.append(descriptor)
170
171 # Check to see whether the optional 'gh_evaluator_targets'
172 # has been supplied. This lists the function spaces for which
173 # any evaluators (gh_shape=gh_evaluator) should be provided.
174 _targets = self.get_integer_array('gh_evaluator_targets')
175 if not _targets and \
176 self._eval_shapes and "gh_evaluator" in self._eval_shapes:
177 # Use the FS of the kernel arguments that are updated
178 write_accesses = AccessType.all_write_accesses()
179 write_args = psyGen.args_filter(self._arg_descriptors_arg_descriptors_arg_descriptors,
180 arg_accesses=write_accesses)
181 # We want the 'to' space of any operator arguments so get
182 # the first FS associated with the kernel argument.
183 _targets = [arg.function_spaces[0] for arg in write_args]
184 # Ensure that '_eval_targets' entries are not duplicated
185 for target in _targets:
186 if target not in self._eval_targets:
187 self._eval_targets.append(target)
188
189 # Does this kernel require any properties of the reference element?
190 self.reference_element = RefElementMetaData(self.name, type_declns)
191
192 # Does this kernel require any properties of the mesh?
193 self.mesh = MeshPropertiesMetaData(self.name, type_declns)
194
195 # Perform further checks that the metadata we've parsed
196 # conforms to the rules for this API
197 self._validate(need_evaluator)
198
199 def _validate(self, need_evaluator):
200 '''
201 Check that the metadata conforms to LFRic rules for a user-provided
202 kernel or a built-in.
203
204 :param bool need_evaluator: whether this kernel requires an
205 evaluator/quadrature.
206 :raises ParseError: if the kernel metadata specifies writing to the
207 read-only function space.
208 :raises ParseError: if a user-supplied LFRic kernel updates/writes
209 to a scalar argument.
210 :raises ParseError: if a kernel does not have at least one argument
211 that is updated/written to.
212 :raises ParseError: if a kernel does not require basis or
213 differential basis functions but specifies one
214 or more gh_shapes.
215 :raises ParseError: if a kernel does not require basis or
216 differential basis functions but specifies
217 gh_evaluator_targets.
218 :raises ParseError: if a kernel specifies gh_evaluator_targets
219 but does not need an evaluator.
220 :raises ParseError: if a kernel requires an evaluator on a
221 specific function space but does not have an
222 argument on that space.
223 :raises ParseError: if a kernel that has LMA operator arguments
224 also has a field argument with an invalid
225 data type (other than 'gh_real').
226
227 '''
228 # pylint: disable=too-many-branches
229 # We must have at least one argument that is written to
230 const = LFRicConstants()
231 write_count = 0
232 for arg in self._arg_descriptors_arg_descriptors_arg_descriptors:
233 if arg.access != AccessType.READ:
234 write_count += 1
235 # We must not write to a field on a read-only function space
236 if arg.argument_type in const.VALID_FIELD_NAMES \
237 and arg.function_spaces[0] in \
238 const.READ_ONLY_FUNCTION_SPACES:
239 raise ParseError(
240 f"Found kernel metadata in '{self.name}' that "
241 f"specifies writing to the read-only function space "
242 f"'{arg.function_spaces[0]}'.")
243
244 # We must not write to scalar arguments if it's not a
245 # built-in
246 if self.name not in BUILTIN_MAP and \
247 arg.argument_type in const.VALID_SCALAR_NAMES:
248 raise ParseError(
249 f"A user-supplied LFRic kernel must not write/update "
250 f"a scalar argument but kernel '{self.name}' has a "
251 f"scalar argument with "
252 f"'{arg.access.api_specific_name()}' access.")
253 if write_count == 0:
254 raise ParseError(f"An LFRic kernel must have at least one "
255 f"argument that is updated (written to) but "
256 f"found none for kernel '{self.name}'.")
257
258 # Check that no shape has been supplied if no basis or
259 # differential basis functions are required for the kernel
260 if not need_evaluator and self._eval_shapes:
261 raise ParseError(
262 f"Kernel '{self.name}' specifies one or more 'gh_shapes' "
263 f"({self._eval_shapes}) but does not need an evaluator because"
264 f" no basis or differential basis functions are required")
265 # Check that 'gh_evaluator_targets' is only present if required
266 if self._eval_targets:
267 if not need_evaluator:
268 raise ParseError(
269 f"Kernel '{self.name}' specifies 'gh_evaluator_targets' "
270 f"({self._eval_targets}) but does not need an evaluator "
271 f"because no basis or differential basis functions are "
272 f"required")
273 if "gh_evaluator" not in self._eval_shapes:
274 raise ParseError(
275 f"Kernel '{self.name}' specifies 'gh_evaluator_targets' "
276 f"({self._eval_targets}) but does not need an evaluator "
277 f"because gh_shape={self._eval_shapes}")
278 # Check that there is a kernel argument on each of the
279 # specified spaces...
280 # Create a list (set) of the function spaces associated with
281 # the kernel arguments
282 fs_list = set()
283 for arg in self._arg_descriptors_arg_descriptors_arg_descriptors:
284 fs_list.update(arg.function_spaces)
285 # Check each evaluator_target against this list
286 for eval_fs in self._eval_targets:
287 if eval_fs not in fs_list:
288 raise ParseError(
289 f"Kernel '{self.name}' specifies that an evaluator is "
290 f"required on '{eval_fs}' but does not have an "
291 f"argument on this space.")
292
293 # If we have an LMA operator as argument then only field arguments
294 # with 'gh_real' data type are permitted
295 lma_ops = psyGen.args_filter(self._arg_descriptors_arg_descriptors_arg_descriptors,
296 arg_types=["gh_operator"])
297 if lma_ops:
298 for arg in self._arg_descriptors_arg_descriptors_arg_descriptors:
299 if (arg.argument_type in const.VALID_FIELD_NAMES
300 and arg.data_type != "gh_real"):
301 raise ParseError(
302 f"In the LFRic API a kernel that has an LMA operator "
303 f"argument must only have field arguments with "
304 f"'gh_real' data type but kernel '{self.name}' has a "
305 f"field argument with '{arg.data_type}' data type.")
306
307 # If we have a columnwise operator as argument then we need to
308 # identify the operation that this kernel performs (one of
309 # assemble, apply/apply-inverse and matrix-matrix)
310 cwise_ops = psyGen.args_filter(self._arg_descriptors_arg_descriptors_arg_descriptors,
311 arg_types=["gh_columnwise_operator"])
312 if cwise_ops:
313 self._cma_operation = self._identify_cma_op(cwise_ops)
314
315 # Perform checks for inter-grid kernels
316 self._validate_inter_grid()
317
318 # Perform checks for a kernel with operates_on == domain
319 self._validate_operates_on_domain(need_evaluator)
320
321 # Perform checks for a coded kernel with operates_on == dof
322 self._validate_operates_on_dof(need_evaluator)

324 def _validate_inter_grid(self):
325 '''
326 Checks that the kernel metadata obeys the rules for LFRic inter-grid
327 kernels. If none of the kernel arguments has a mesh associated with it
328 then it is not an inter-grid kernel and this routine silently returns.
329
330 :raises: ParseError: if metadata breaks inter-grid rules.
331 '''
332 # pylint: disable=too-many-branches
333 # Dictionary of meshes associated with arguments (for inter-grid
334 # kernels). Keys are the meshes, values are lists of function spaces
335 # of the corresponding field arguments.
336 mesh_dict = OrderedDict()
337 # Whether or not any field args are missing the 'mesh_arg' specifier
338 missing_mesh = False
339 # If this is an inter-grid kernel then it must only have field
340 # arguments. Keep a record of any non-field arguments for the benefit
341 # of a verbose error message.
342 non_field_arg_types = set()
343 const = LFRicConstants()
344 for arg in self._arg_descriptors_arg_descriptors_arg_descriptors:
345 # Collect info so that we can check inter-grid kernels
346 if arg.argument_type in const.VALID_FIELD_NAMES:
347 if arg.mesh:
348 # Argument has a mesh associated with it so this must
349 # be an inter-grid kernel
350 if arg.mesh in mesh_dict:
351 mesh_dict[arg.mesh].append(arg.function_space)
352 else:
353 mesh_dict[arg.mesh] = [arg.function_space]
354 else:
355 # Record the fact that we have a field without a
356 # mesh specifier (in case this is an inter-grid kernel)
357 missing_mesh = True
358 else:
359 # Inter-grid kernels are only permitted to have field args
360 # so collect a list of other types
361 non_field_arg_types.add(arg.argument_type)
362
363 mesh_list = mesh_dict.keys()
364 if not mesh_list:
365 # There are no meshes associated with any of the arguments so
366 # this is not an inter-grid kernel
367 return
368
369 if len(const.VALID_MESH_TYPES) != 2:
370 # Sanity check that nobody has messed with the number of
371 # grid types that we recognise. This is here because the
372 # implementation assumes that there are just two grids
373 # (coarse and fine).
374 raise InternalError(
375 f"The implementation of inter-grid support in the LFRic "
376 f"API assumes there are exactly two mesh types but "
377 f"LFRicConstants.VALID_MESH_TYPES contains "
378 f"{len(const.VALID_MESH_TYPES)}: {const.VALID_MESH_TYPES}")
379 if len(mesh_list) != len(const.VALID_MESH_TYPES):
380 raise ParseError(
381 f"Inter-grid kernels in the LFRic API must have at least "
382 f"one field argument on each of the mesh types "
383 f"({const.VALID_MESH_TYPES}). However, kernel {self.name} has "
384 f"arguments only on {[str(name) for name in mesh_list]}")
385 # Inter-grid kernels must only have field arguments
386 if non_field_arg_types:
387 raise ParseError(
388 f"Inter-grid kernels in the LFRic API are only permitted "
389 f"to have field arguments but kernel {self.name} also has "
390 f"arguments of type "
391 f"{[str(name) for name in non_field_arg_types]}")
392 # Check that all arguments have a mesh specified
393 if missing_mesh:
394 raise ParseError(
395 f"Inter-grid kernels in the LFRic API must specify which mesh "
396 f"each field argument is on but kernel {self.name} has at "
397 f"least one field argument for which 'mesh_arg' is missing.")
398 # Check that arguments on different meshes are on different
399 # function spaces. We do this by checking that no function space
400 # is listed as being associated with (arguments on) both meshes.
401 fs_sets = []
402 for mesh in mesh_dict:
403 fs_sets.append(set(mesh_dict[mesh]))
404 # Check that the sets of spaces (one for each mesh type) have
405 # no intersection
406 fs_common = fs_sets[0] & fs_sets[1]
407 if fs_common:
408 raise ParseError(
409 f"In the LFRic API field arguments to inter-grid kernels "
410 f"must be on different function spaces if they are on "
411 f"different meshes. However kernel {self.name} has a field on "
412 f"function space(s) {[str(name) for name in fs_common]} on "
413 f"each of the mesh types {[str(name) for name in mesh_list]}.")
414 # Finally, record that this is a valid inter-grid kernel
415 self._is_intergrid = True

417 def _identify_cma_op(self, cwise_ops):
418 '''
419 Identify and return the type of CMA-operator-related operation
420 this kernel performs (one of "assemble", "apply" or "matrix-matrix")
421
422 :param cwise_ops: all column-wise operator arguments in a kernel.
423 :type cwise_ops: list of str
424
425 :returns: the type of CMA-operator-related operation that this
426 kernel performs.
427 :rtype: str
428
429 :raises ParseError: if the kernel metadata does not conform to the
430 LFRic rules for a kernel with a CMA operator.
431
432 '''
433 # pylint: disable=too-many-branches
434 const = LFRicConstants()
435 for arg in self._arg_descriptors_arg_descriptors_arg_descriptors:
436 # No vector arguments are permitted
437 if arg.vector_size > 1:
438 raise ParseError(
439 f"Kernel '{self.name}' takes a CMA operator but has a "
440 f"vector argument '{arg.argument_type}*{arg.vector_size}'."
441 f" This is forbidden.")
442 # No stencil accesses are permitted
443 if arg.stencil:
444 raise ParseError(
445 f"Kernel '{self.name}' takes a CMA operator but has an "
446 f"argument with a stencil access "
447 f"('{arg.stencil['type']}'). This is forbidden.")
448 # Only field arguments with 'gh_real' data type are permitted
449 if (arg.argument_type in const.VALID_FIELD_NAMES and
450 arg.data_type != "gh_real"):
451 raise ParseError(
452 f"In the LFRic API a kernel that takes a CMA operator "
453 f"argument must only have field arguments with 'gh_real' "
454 f"data type but kernel '{self.name}' has a field argument "
455 f"with '{arg.data_type}' data type.")
456
457 # Count the number of CMA operators that are written to
458 write_count = 0
459 for cop in cwise_ops:
460 if cop.access in AccessType.all_write_accesses():
461 write_count += 1
462
463 if write_count == 0:
464 # This kernel only reads from CMA operators and must
465 # therefore be an apply (or apply-inverse). It must
466 # have one CMA operator, one read-only field and one
467 # written field as arguments
468 if len(cwise_ops) != 1:
469 raise ParseError(
470 f"In the LFRic API a kernel that applies a CMA operator "
471 f"must only have one such operator in its list of "
472 f"arguments but found {len(cwise_ops)} for kernel "
473 f"'{self.name}'.")
474 cma_op = cwise_ops[0]
475 if len(self._arg_descriptors_arg_descriptors_arg_descriptors) != 3:
476 raise ParseError(
477 f"In the LFRic API a kernel that applies a CMA operator "
478 f"must have 3 arguments (the operator and two fields) but "
479 f"kernel '{self.name}' has {len(self._arg_descriptors)} "
480 f"arguments.")
481 # Check that the other two arguments are fields
482 farg_read = psyGen.args_filter(
483 self._arg_descriptors_arg_descriptors_arg_descriptors,
484 arg_types=const.VALID_FIELD_NAMES,
485 arg_accesses=[AccessType.READ])
486 write_accesses = AccessType.all_write_accesses()
487 farg_write = psyGen.args_filter(
488 self._arg_descriptors_arg_descriptors_arg_descriptors,
489 arg_types=const.VALID_FIELD_NAMES,
490 arg_accesses=write_accesses)
491 if len(farg_read) != 1:
492 raise ParseError(
493 f"Kernel '{self.name}' has a read-only CMA operator. In "
494 f"order to apply it the kernel must have one read-only "
495 f"field argument.")
496 if len(farg_write) != 1:
497 raise ParseError(
498 f"Kernel '{self.name}' has a read-only CMA operator. In "
499 f"order to apply it the kernel must write to one field "
500 f"argument.")
501 # Check that the function spaces match up
502 if farg_read[0].function_space != cma_op.function_space_from:
503 raise ParseError(
504 f"Kernel '{self.name}' applies a CMA operator but the "
505 f"function space of the field argument it reads from "
506 f"('{farg_read[0].function_space}') does not match the "
507 f"'from' space of the operator "
508 f"('{cma_op.function_space_from}').")
509 if farg_write[0].function_space != cma_op.function_space_to:
510 raise ParseError(
511 f"Kernel '{self.name}' applies a CMA operator but the "
512 f"function space of the field argument it writes to "
513 f"('{farg_write[0].function_space}') does not match the "
514 f"'to' space of the operator "
515 f"('{cma_op.function_space_to}').")
516 # This is a valid CMA-apply or CMA-apply-inverse kernel
517 return "apply"
518
519 if write_count == 1:
520 # This kernel writes to a single CMA operator and therefore
521 # must either be assembling a CMA operator
522 # or performing a matrix-matrix operation...
523 # The kernel must not write to any args other than the CMA
524 # operator
525 write_accesses = AccessType.all_write_accesses()
526 write_args = psyGen.args_filter(self._arg_descriptors_arg_descriptors_arg_descriptors,
527 arg_accesses=write_accesses)
528 if len(write_args) > 1:
529 # Remove the one CMA operator from the list of arguments
530 # that are written to so that we can produce a nice
531 # error message
532 for arg in write_args[:]:
533 if arg.argument_type == 'gh_columnwise_operator':
534 write_args.remove(arg)
535 break
536 raise ParseError(
537 f"Kernel '{self.name}' writes to a column-wise operator "
538 f"but also writes to "
539 f"{[str(arg.argument_type) for arg in write_args]} "
540 f"argument(s). This is not allowed.")
541 if len(cwise_ops) == 1:
542
543 # If this is a valid assembly kernel then we need at least one
544 # read-only LMA operator
545 lma_read_ops = psyGen.args_filter(
546 self._arg_descriptors_arg_descriptors_arg_descriptors,
547 arg_types=["gh_operator"],
548 arg_accesses=[AccessType.READ])
549 if lma_read_ops:
550 return "assembly"
551 raise ParseError(
552 f"Kernel '{self.name}' has a single column-wise operator "
553 f"argument but does not conform to the rules for an "
554 f"Assembly kernel because it does not have any read-only "
555 f"LMA operator arguments.")
556 # A valid matrix-matrix kernel must only have CMA operators
557 # and scalars as arguments.
558 scalar_args = psyGen.args_filter(
559 self._arg_descriptors_arg_descriptors_arg_descriptors,
560 arg_types=const.VALID_SCALAR_NAMES)
561 if (len(scalar_args) + len(cwise_ops)) != \
562 len(self._arg_descriptors_arg_descriptors_arg_descriptors):
563 raise ParseError(
564 f"A column-wise matrix-matrix kernel must have only "
565 f"column-wise operators and scalars as arguments but "
566 f"kernel '{self.name}' has: "
567 f"{[str(a.argument_type) for a in self._arg_descriptors]}."
568 )
569 return "matrix-matrix"
570 raise ParseError(
571 f"An LFRic kernel cannot update more than one CMA (column-wise) "
572 f"operator but kernel '{self.name}' updates {write_count}.")

574 def _validate_operates_on_domain(self, need_evaluator):
575 '''
576 Check whether a kernel that has operates_on == domain obeys
577 the rules for the LFRic API.
578
579 :param bool need_evaluator: whether a kernel requires an
580 evaluator/quadrature.
581
582 :raises ParseError: if the kernel metadata does not obey the rules
583 for an LFRic kernel with operates_on = domain.
584 '''
585 if self.iterates_overiterates_over != "domain":
586 return
587
588 self._validate_only_scalars_and_fields()

596 Check whether a kernel that has operates_on == dof obeys

598
599 :param bool need_evaluator: whether a kernel requires an
600 evaluator/quadrature.
601

604 '''
605 if self.iterates_overiterates_over != "dof":
606 return

608 # Only validating coded kerns, not builtins
609 if self.name in BUILTIN_MAP:
610 return

612 self._validate_only_scalars_and_fields()
613 self._validate_no_evaluator(need_evaluator)
614 self._validate_no_reference_element()
615 self._validate_no_mesh_properties()
616 self._validate_not_intergrid()
617
618 # Check function spaces are the same
619 # list out all function spaces
620 arg_fs_names = set()
621 for arg in self._arg_descriptors_arg_descriptors_arg_descriptors:
622 for fs_name in arg.function_spaces:
623 arg_fs_names.add(fs_name)

633 def _validate_only_scalars_and_fields(self):
634 '''
635 Checks metadata arguments are either fields or scalars by comparing
636 the kernel's argument descriptors to the valid names for these types
637 found in LFRic constants.
638

640 scalar types.
641 '''
642 const = LFRicConstants()
643 # A kernel which operates on the 'domain' is currently restricted
644 # to only accepting scalar and field arguments.
