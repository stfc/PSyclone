128 def find_container_psyir(self, local_node=None):
129 ''' Searches for the Container that this Symbol refers to. If it is
130 not available, use the interface to import the container. If
131 `local_node` is supplied then the PSyIR tree below it is searched for
132 the container first.
133
134 :param local_node: root of PSyIR sub-tree to include in search for
135 the container.
136 :type local_node: Optional[:py:class:`psyclone.psyir.nodes.Node`]
137
138 :returns: referenced container.
139 :rtype: :py:class:`psyclone.psyir.nodes.Container`
140
141 '''
142 if not self._reference_reference:
143 # First check in the current PSyIR tree (if supplied).
144 if local_node:
145 lowered_name = self.namename.lower()
146 from psyclone.psyir.nodes.container import Container
147 from psyclone.psyir.nodes.routine import Routine
148 for local in local_node.root.walk(Container,
149 stop_type=Routine):
150 if lowered_name == local.name.lower():
151 self._reference_reference = local
152 return self._reference_reference
153 # We didn't find it so now attempt to import the container.
154 self._reference_reference = self._interface.get_container(self._name)
155 return self._reference_reference

232 ''' Imports a Fortran module as a PSyIR Container (via the

234
235 :param str name: name of the module to be imported.
236
237 :returns: container associated with the given name.

250 mod_manager.add_search_path(Config.get().include_paths)
251
252 minfo = None
253 try:
254 minfo = mod_manager.get_module_info(name)
255 except FileNotFoundError:
256 pass
257 if not minfo:
258 raise SymbolError(
259 f"Module '{name}' not found in any of the include_paths "
260 f"directories {Config.get().include_paths}.")
261 return minfo.get_psyir()
