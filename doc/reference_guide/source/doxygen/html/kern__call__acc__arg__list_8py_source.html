38'''This module implements a class that manages all of the data references
39that must be copied over to a GPU before executing the kernel. Ordering
40of the parameters does not matter apart from where we have members of
41derived types. In that case, the derived type itself must be specified
42first before any members.
43'''

61 def cell_map(self, var_accesses=None):
62 '''Add cell-map to the list of required arrays.
63
64 :param var_accesses: optional VariablesAccessInfo instance to store
65 the information about variable accesses.
66 :type var_accesses: Optional[
67 :py:class:`psyclone.core.VariablesAccessInfo`]
68
69 '''
70 cargs = psyGen.args_filter(self._kern_kern.args, arg_meshes=["gh_coarse"])
71 if len(cargs) > 1:
72 raise InternalError(
73 f"An LFRic intergrid kernel should have only one coarse mesh "
74 f"but '{self._kern.name}' has {len(cargs)}")
75 carg = cargs[0]
76 # Add the cell map to our argument list
77 base_name = "cell_map_" + carg.name
78 self.append(base_name)
79 # We'll need the current cell to index into this cell map.
80 self.cell_positioncell_positioncell_position(var_accesses)

83 '''Adds a cell argument to the argument list and if supplied stores

88 :param var_accesses: optional VariablesAccessInfo instance to store

97 def field(self, arg, var_accesses=None):
98 '''Add the field array associated with the argument 'arg' to the
99 argument list. If supplied it also stores this access in var_accesses.
100
101 :param arg: the field to be added.
102 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
103 :param var_accesses: optional VariablesAccessInfo instance to store
104 the information about variable accesses.
105 :type var_accesses: :py:class:`psyclone.core.VariablesAccessInfo`
106
107 '''
108 const = LFRicConstants()
109 suffix = const.ARG_TYPE_SUFFIX_MAPPING[arg.argument_type]
110 # Look-up the name of the variable that stores the reference to
111 # the data in this field.
112 sym = self._symtab.lookup_with_tag(f"{arg.name}:{suffix}")
113
114 # Add the field data array as being read.
115 self.append(sym.name, var_accesses, var_access_name=sym.name,
116 mode=arg.access, metadata_posn=arg.metadata_index)

193 def operator(self, arg, var_accesses=None):
194 '''Add the operator arguments if they have not already been
195 added. OpenACC requires the derived type and the dereferenced
196 data to be specified. If supplied it also stores this access in
197 var_accesses.
198
199 :param arg: the meta-data description of the operator.
200 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
201 :param var_accesses: optional VariablesAccessInfo instance to store
202 the information about variable accesses.
203 :type var_accesses: Optional[
204 :py:class:`psyclone.core.VariablesAccessInfo`]
205
206 '''
207 # In case of OpenACC we do not want to transfer the same
208 # data to GPU twice.
209 if arg.proxy_name_indexed not in self.arglistarglist:
210 self.append(arg.proxy_name_indexed, var_accesses)
211 # This adds ncell_3d and local_stencil after the derived type:

214 def fs_compulsory_field(self, function_space, var_accesses=None):
215 '''Add compulsory arguments associated with this function space to
216 the list. OpenACC requires the full function-space map
217 to be specified. If supplied it also stores this access in
218 var_accesses.
219
220 :param function_space: the function space for which the compulsory
221 arguments are added.
222 :type function_space: :py:class:`psyclone.domain.lfric.FunctionSpace`
223 :param var_accesses: optional VariablesAccessInfo instance to store
224 the information about variable accesses.
225 :type var_accesses: Optional[
226 :py:class:`psyclone.core.VariablesAccessInfo`]
227
228 '''
229 if not self._kern_kern.iterates_over.endswith("cell_column"):
230 return
231 self.append(function_space.undf_name, var_accesses)
232 # The base class only adds one dimension to the list, while OpenACC
233 # needs the whole field, so we cannot call the base class.
234 self.append(function_space.map_name, var_accesses)

236 def fs_intergrid(self, function_space, var_accesses=None):
237 '''Add arrays that need to be uploaded for inter-grid kernels.
238 These arrays contain the mapping between fine and coarse meshes.
239
240 :param function_space: the function space associated with the mesh.
241 :type function_space: :py:class:`psyclone.domain.lfric.FunctionSpace`
242 :param var_accesses: optional VariablesAccessInfo instance to store
243 the information about variable accesses.
244 :type var_accesses: Optional[
245 :py:class:`psyclone.core.VariablesAccessInfo`]
246
247 '''
248 # Is this FS associated with the coarse or fine mesh? (All fields
249 # on a given mesh must be on the same FS.)
250 arg = self._kern_kern.arguments.get_arg_on_space(function_space)
251 if arg.mesh == "gh_fine":
252 # For the fine mesh, we need the *whole* dofmap
253 map_name = function_space.map_name
254 self.append(map_name, var_accesses)
255 else:
256 # For the coarse mesh we only need undf and the dofmap for
257 # the current column
258 self.fs_compulsory_fieldfs_compulsory_fieldfs_compulsory_field(function_space,
259 var_accesses=var_accesses)
