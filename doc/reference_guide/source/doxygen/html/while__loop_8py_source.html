45class WhileLoop(Statement):
46 '''
47 Node representing a while loop within the PSyIR. It has two mandatory
48 children: the first one represents the loop condition and the second one
49 the loop body.
50
51 '''
52 # The valid annotations for this node:
53 # 'was_unconditional' to tag do loops with no condition in Fortran.
54 valid_annotations = ('was_unconditional',)
55 # Textual description of the node.
56 _children_valid_format = "DataNode, Schedule"
57 _colour = "red"

59 @staticmethod
60 def _validate_child(position, child):
61 '''
62 :param int position: the position to be validated.
63 :param child: a child to be validated.
64 :type child: :py:class:`psyclone.psyir.nodes.Node`
65
66 :return: whether the given child and position are valid for this node.
67 :rtype: bool
68
69 '''
70 return (position == 0 and isinstance(child, DataNode)) or (
71 position == 1 and isinstance(child, Schedule))

107 @staticmethod
108 def create(loop_condition, loop_body):
109 '''Create a WhileLoop instance given valid instances of a
110 loop_condition and a loop_body.
111
112 :param loop_condition: the PSyIR node containing the loop \
113 condition of the while loop statement.
114 :type loop_condition: :py:class:`psyclone.psyir.nodes.Node`
115 :param loop_body: the PSyIR nodes representing the loop body of \
116 the loop statement.
117 :type loop_body: List[:py:class:`psyclone.psyir.nodes.Node`]
118
119 :returns: a WhileLoop instance.
120 :rtype: :py:class:`psyclone.psyir.nodes.WhileLoop`
121
122 :raises GenerationError: if the arguments to the create method \
123 are not of the expected type.
124
125 '''
126 if not isinstance(loop_body, list):
127 raise GenerationError(
128 f"loop_body argument in create method of WhileLoop class "
129 f"should be a list but found '{type(loop_body).__name__}'.")
130
131 loop_stmt = WhileLoop()
132 loop_schedule = Schedule(parent=loop_stmt, children=loop_body)
133 loop_stmt.children = [loop_condition, loop_schedule]
134 return loop_stmt
