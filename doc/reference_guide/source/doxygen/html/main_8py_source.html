39'''Top-level driver functions for PSyAD : the PSyclone Adjoint
40support. Transforms an LFRic tangent linear kernel to its adjoint.
41
42'''
43import argparse
44import logging
45import os
46import re
47import sys
48
49from psyclone.configuration import Config, LFRIC_API_NAMES
50from psyclone.line_length import FortLineLength
51from psyclone.psyad.tl2ad import generate_adjoint_str
52from psyclone.psyad.transformations import TangentLinearError
53
54
55def main(args):
56 '''Takes an LFRic tangent linear kernel source file as input and
57 produces its adjoint.
58
59 :param list args: the list of command-line arguments that PSyAD has \
60 been invoked with.
61
62 '''
63 # Make sure we have the supported APIs defined in the Config singleton,
64 # but postpone loading the config file till the command line was parsed
65 # in case that the user specifies a different config file.
66 Config.get(do_not_load_file=True)
67
68 # pylint: disable=too-many-statements, too-many-branches
69 # TODO #1863 - expose line-length limiting as a command-line flag.
70 line_length_limit = True
71
72 logger = logging.getLogger(__name__)
73
74 # There is a well known bug in argparse when mixing positional and
75 # variable arguments. The simplest solution is to separate them
76 # with --. The usage message is therefore manually updated to
77 # reflect this workaround.
78 def msg():
79 '''Function to overide the argpass usage message'''
80 return ("psyad [-h] [-oad OAD] [-v] [-t] [-api API] "
81 "[-coord-arg COORD_ARG] [-panel-id-arg PANEL_ID_ARG] "
82 "[-otest TEST_FILENAME] [-c CONFIG] "
83 "-a ACTIVE [ACTIVE ...] -- filename")
84
85 parser = argparse.ArgumentParser(
86 description="Run the PSyclone adjoint code generator on a "
87 "tangent-linear kernel file", usage=msg())
88 parser.add_argument(
89 '-a', '--active', nargs='+', help='names of active variables',
90 required=True)
91 parser.add_argument(
92 '-c', '--config', help='config file with PSyclone specific options')
93 parser.add_argument(
94 '-v', '--verbose', help='increase the verbosity of the output',
95 action='store_true')
96 parser.add_argument(
97 '-t', '--gen-test',
98 help='generate a standalone unit test for the adjoint code',
99 action='store_true')
100 parser.add_argument('-api', default=None,
101 help='the PSyclone API that the TL kernel conforms '
102 'to (if any)')
103 parser.add_argument('-coord-arg', default=None, type=int,
104 help='the position of the coordinate (chi) field in '
105 'the meta_args list of arguments in the kernel '
106 'metadata (LFRic only)')
107 parser.add_argument('-panel-id-arg', default=None, type=int,
108 help='the position of the panel-ID field in the '
109 'meta_args list of arguments in the kernel metadata '
110 '(LFRic only)')
111 parser.add_argument('-otest', default=None,
112 help='filename for the unit test (implies -t)',
113 dest='test_filename')
114 parser.add_argument('-oad', help='filename for the transformed code')
115 parser.add_argument('filename', help='tangent-linear kernel source')
116
117 args = parser.parse_args(args)
118
119 # If no config file name is specified, args.config is none
120 # and config will load the default config file.
121 Config.get().load(args.config)
122
123 if args.verbose:
124 logging.basicConfig(level=logging.DEBUG)
125
126 # Specifying an output file for the test harness is taken to mean that
127 # the user wants us to generate it.
128 generate_test = args.gen_test or args.test_filename
129
130 # Check the command-line arguments for consistency.
131 if args.api not in LFRIC_API_NAMES:
132 if args.coord_arg is not None:
133 logger.error(
134 "The '-coord-arg' argument is only applicable to the 'lfric' "
135 "API.")
136 sys.exit(1)
137 if args.panel_id_arg is not None:
138 logger.error("The '-panel-id-arg' argument is only applicable to "
139 "the 'lfric' API.")
140 sys.exit(1)
141
142 # Processing test filename
143 test_name = "adjoint_test"
144 if generate_test:
145 if args.api in LFRIC_API_NAMES:
146 filename_standard = "adjt_.+_alg_mod.[Xx]90|atlt_.+_alg_mod.[Xx]90"
147 regex_search = re.search(filename_standard, args.test_filename)
148 if regex_search is None:
149 logger.error("Filename '%s' with 'lfric' API "
150 "must be of the form "
151 "<path>/adjt_<name>_alg_mod.[Xx]90 or "
152 "<path>/atlt_<name>_alg_mod.[Xx]90.",
153 args.test_filename)
154 sys.exit(1)
155 # At this stage filename should be valid, so we take the base name
156 test_name = os.path.basename(args.test_filename).split("_mod.")[0]
157
158 # TL Fortran code
159 filename = args.filename
160 logger.info("Reading kernel file %s", filename)
161 try:
162 with open(filename, mode="r", encoding="utf-8") as my_file:
163 tl_fortran_str = my_file.read()
164 tl_fortran_str = str(tl_fortran_str)
165 except FileNotFoundError:
166 logger.error("psyad error: file '%s', not found.", filename)
167 sys.exit(1)
168
169 try:
170 # Create the adjoint (and associated test framework if requested)
171 ad_fortran_str, test_fortran_str = generate_adjoint_str(
172 tl_fortran_str, args.active, api=args.api,
173 coord_arg_index=args.coord_arg,
174 panel_id_arg_index=args.panel_id_arg,
175 create_test=generate_test,
176 test_name=test_name)
177 except TangentLinearError as info:
178 print(str(info.value))
179 sys.exit(1)
180 except (KeyError, NotImplementedError) as info:
181 print(str(info))
182 sys.exit(1)
183
184 fll = FortLineLength()
185
186 # Output the Fortran code for the adjoint kernel
187 if line_length_limit:
188 ad_fortran_str = fll.process(ad_fortran_str)
189
190 if args.oad:
191 logger.info("Writing adjoint of kernel to file %s", args.oad)
192 with open(args.oad, mode="w", encoding="utf-8") as adj_file:
193 adj_file.write(ad_fortran_str)
194 else:
195 print(ad_fortran_str, file=sys.stdout)
196
197 # Output test harness if requested
198 if generate_test:
199
200 if line_length_limit:
201 test_fortran_str = fll.process(test_fortran_str)
202
203 if args.test_filename:
204 logger.info("Writing test harness for adjoint kernel to file %s",
205 args.test_filename)
206 with open(args.test_filename, mode="w", encoding="utf-8") as tfile:
207 tfile.write(test_fortran_str)
208 else:
209 print(test_fortran_str, file=sys.stdout)
210
