53def find_symbol(node, name):
54 '''Utility method to find the symbol associated with the supplied
55 name. The supplied node and all of its descendants are searched
56 and the symbol and its scoping node are returned if the symbol is
57 found. If not found, None is returned for the symbol and scoping
58 node.
59
60 :param node: the starting node for the search.
61 :type: :py:class:`psyclone.psyir.nodes.node`
62 :param str name: the name of the symbol being searched.
63
64 :returns: a tuple containing the symbol with the same name as the \
65 supplied name and the scoping node for that symbol.
66 :rtype: Tuple[Optional[:py:class:`psyclone.psyir.symbols.datasymbol], \
67 Optional[:py:class:`psyclone.psyir.nodes.node`]]
68
69 '''
70 for test_node in node.walk(ScopingNode):
71 if name in test_node.symbol_table:
72 symbol = test_node.symbol_table.lookup(name)
73 scoping_node = test_node
74 return (symbol, scoping_node)
75 return (None, None)

78class RaisePSyIR2LFRicKernTrans(Transformation):
79 '''Raise a generic PSyIR representation of a kernel-layer routine and
80 metadata to an LFRic version with specialised domain-specific
81 nodes and symbols. This is currently limited to the specialisation
82 of kernel metadata.
83
84 >>> from psyclone.configuration import Config
85 >>> from psyclone.domain.lfric.transformations import \
86 RaisePSyIR2LFRicKernTrans
87 >>> from psyclone.psyir.frontend.fortran import FortranReader
88 >>> config = Config.get().api_conf("lfric")
89 >>> CODE = ("""
90 ... MODULE example
91 ... TYPE, EXTENDS(kernel_type) :: compute_cu
92 ... TYPE(arg_type), DIMENSION(4) :: meta_args = (/ &
93 ... arg_type(GH_FIELD, GH_REAL, GH_INC, W1), &
94 ... arg_type(GH_FIELD, GH_REAL, GH_READ, W3), &
95 ... arg_type(GH_FIELD, GH_REAL, GH_READ, W3), &
96 ... arg_type(GH_FIELD, GH_REAL, GH_READ, W3)/)
97 ... INTEGER :: OPERATES_ON = CELL_COLUMN
98 ... CONTAINS
99 ... PROCEDURE, NOPASS :: code => compute_cu_code
100 ... END TYPE compute_cu
101 ... contains
102 ... subroutine compute_cu_code()
103 ... end subroutine
104 ... end module""")
105 >>> fortran_reader = FortranReader()
106 >>> kernel_container = fortran_reader.psyir_from_source(CODE)
107 >>> trans = RaisePSyIR2LFRicKernTrans()
108 >>> trans.apply(kernel_container, {"metadata_name": "compute_cu"})
109
110 '''
111 def __init__(self):
112 super().__init__()

114 def validate(self, node, options=None):
115 '''Validate the supplied PSyIR tree.
116
117 :param node: a PSyIR node that is the root of a PSyIR tree.
118 :type node: :py:class:`psyclone.psyir.node.Container`
119 :param options: a dictionary with options for transformations.
120 :type options: Optional[Dict[str: str]]
121
122 :raises TransformationError: if the supplied node is not a
123 Container.
124 :raises TransformationError: if the supplied node argument has
125 a parent.
126 :raises TransformationError: if the metadata name has not been
127 provided in the options argument.
128 :raises TransformationError: if the metadata name has not been
129 set or does not exist in the code.
130 :raises TransformationError: if the metadata symbol does not
131 reside in a Container (as opposed to a FileContainer).
132
133 '''
134 super().validate(node, options=options)
135
136 if not isinstance(node, Container):
137 raise TransformationError(
138 f"Error in {self.name} transformation. The supplied node "
139 f"should be a Container but found '{type(node).__name__}'.")
140
141 if node.parent:
142 raise TransformationError(
143 f"Error in {self.name} transformation. The supplied node "
144 f"should be the root of a PSyIR tree but this node has a "
145 f"parent ({type(node.parent).__name__}).")
146
147 try:
148 metadata_name = options["metadata_name"]
149 except (TypeError, KeyError) as info:
150 names = list(options.keys()) if options else []
151 raise TransformationError(
152 f"Error in {self.name} transformation. This "
153 f"transformation requires the name of the variable "
154 f"containing the metadata to be provided in the options "
155 f"argument with lookup name 'metadata_name', but found "
156 f"'{names}'.") from info
157
158 try:
159 FortranReader.validate_name(metadata_name)
160 except (TypeError, ValueError) as err:
161 raise TransformationError(
162 f"Error in {self.name} transformation. This "
163 f"transformation requires the name of the variable "
164 f"containing the metadata to be set to a "
165 f"valid Fortran name, but found '{metadata_name}'.") from err
166
167 metadata_symbol, scoping_node = find_symbol(node, metadata_name)
168 if not metadata_symbol:
169 raise TransformationError(
170 f"Error in {self.name} transformation. The metadata name "
171 f"'{metadata_name}' provided to the transformation "
172 f"does not correspond to a symbol in the supplied PSyIR.")
173
174 # Find the nearest ancestor container including self. There
175 # will always be at least one ancestor container as otherwise
176 # an earlier test will fail.
177 # TODO issue #1886. Avoid replicating tests (see gocean
178 # raising transformation).
179 container = scoping_node.ancestor(Container, include_self=True)
180 if isinstance(container, FileContainer):
181 raise TransformationError(
182 f"Error in {self.name} transformation. The Container in "
183 f"which the metadata symbol resides is a FileContainer, "
184 f"but should be a generic Container.")
185
186 # Check that the metadata can be generated without any errors.
187 _ = LFRicKernelMetadata.create_from_psyir(metadata_symbol)

189 def apply(self, node, options=None):
190 '''Raise the supplied language-level kernel to LFRic-specific kernel
191 PSyIR. Specialises the kernel container to an LFRic-specific
192 subclass, populates this subclass with the kernel metadata
193 extracted from the metadata symbol as specified in
194 metadata_name (which is supplied via the options argument) and
195 removes the symbol from the symbol table.
196
197 :param node: a kernel represented in generic PSyIR.
198 :type node: :py:class:`psyclone.psyir.node.Container`
199 :param options: a dictionary with options for transformations. \
200 This is expected to contain the metadata_name.
201 :type options: Optional[Dict[str: str]]
202
203 '''
204 self.validatevalidate(node, options=options)
205
206 # The name of the PSyIR symbol containing the metadata.
207 metadata_name = options["metadata_name"]
208
209 # Find the metadata symbol based on the supplied name.
210 metadata_symbol, scoping_node = find_symbol(node, metadata_name)
211
212 # Find the container in which this metadata resides.
213 container = scoping_node.ancestor(Container, include_self=True)
214
215 # Create the metadata.
216 metadata = LFRicKernelMetadata.create_from_psyir(metadata_symbol)
217
218 # Remove metadata symbol.
219 # TODO issue #898: support needs to be added for removing a
220 # DataSymbol from the symbol table. At the moment we need to
221 # use internal methods.
222 symbol_table = scoping_node.symbol_table
223 # pylint: disable=protected-access
224 norm_name = symbol_table._normalize(metadata_symbol.name)
225 symbol_table._symbols.pop(norm_name)
226 # pylint: enable=protected-access
227
228 # Replace container
229 children = container.pop_all_children()
230 lfric_container = LFRicKernelContainer.create(
231 container.name, metadata, container.symbol_table.detach(),
232 children)
233 container.replace_with(lfric_container)
