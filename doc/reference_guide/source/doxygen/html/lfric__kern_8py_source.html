65class LFRicKern(CodedKern):
66 ''' Stores information about LFRic Kernels as specified by the

68 information to generate appropriate PSy layer code for the Kernel
69 instance or to generate a Kernel stub.

90 def __init__(self):
91 # The super-init is called from the _setup() method which in turn
92 # is called from load().
93 # pylint: disable=super-init-not-called
94 # Import here to avoid circular dependency
95 # pylint: disable=import-outside-toplevel
96 if False: # pylint: disable=using-constant-test
97 from psyclone.dynamo0p3 import DynKernelArguments
98 self._arguments = DynKernelArguments(None, None) # for pyreverse
99 self._parent = None
100 self._base_name = ""
101 self._func_descriptors = None
102 self._fs_descriptors = None
103 # Whether this kernel requires quadrature
104 self._qr_required = False
105 # Whether this kernel requires basis functions
106 self._basis_required = False
107 # What shapes of evaluator/quadrature this kernel requires (if any)
108 self._eval_shapes = []
109 # The function spaces on which to *evaluate* basis/diff-basis
110 # functions if an evaluator is required for this kernel. Is a dict with
111 # (mangled) FS names as keys and associated kernel argument as value.
112 self._eval_targets = OrderedDict()
113 # Will hold a dict of QRRule namedtuple objects, one for each QR
114 # rule required by a kernel, indexed by shape. Needs to be ordered
115 # because we must preserve the ordering specified in the metadata.
116 self._qr_rules = OrderedDict()
117 self._cma_operation = None
118 # Reference to the DynInterGrid object holding any inter-grid aspects
119 # of this kernel or None if it is not an intergrid kernel
120 self._intergrid_ref = None # Reference to this kernel inter-grid
121 # The reference-element properties required by this kernel
122 self._reference_element = None
123 # The mesh properties required by this kernel
124 self._mesh_properties = None
125 # The depth of halo that this kernel expects to operate on. (Only
126 # applicable to kernels with operates_on=HALO_CELL_COLUMN or
127 # OWNED_AND_HALO_CELL_COLUMN.)
128 self._halo_depth = None
129 # Initialise kinds (precisions) of all kernel arguments (start
130 # with 'real' and 'integer' kinds)
131 api_config = Config.get().api_conf("lfric")
132 self._argument_kinds = {api_config.default_kind["real"],
133 api_config.default_kind["integer"]}

155 def load(self, call, parent=None):
156 '''
157 Sets up kernel information with the call object which is
158 created by the parser. This object includes information about
159 the invoke call and the associated kernel.
160
161 :param call: The KernelCall object from which to extract information
162 about this kernel
163 :type call: :py:class:`psyclone.parse.algorithm.KernelCall`
164 :param parent: The parent node of the kernel call in the AST
165 we are constructing. This will be a loop.
166 :type parent: :py:class:`psyclone.domain.lfric.LFRicLoop`
167 '''
168 self._setup_basis(call.ktype)
169 self._setup(call.ktype, call.module_name, call.args, parent)

171 def load_meta(self, ktype):
172 '''
173 Sets up kernel information with the kernel type object
174 which is created by the parser. The object includes the
175 metadata describing the kernel code.
176
177 :param ktype: the kernel metadata object produced by the parser
178 :type ktype: :py:class:`psyclone.domain.lfric.LFRicKernMetadata`
179
180 :raises InternalError: for an invalid data type of a scalar argument.
181 :raises GenerationError: if an invalid argument type is found \
182 in the kernel.
183
184 '''
185 # pylint: disable=too-many-branches
186 # Create a name for each argument
187 args = []
188 const = LFRicConstants()
189 for idx, descriptor in enumerate(ktype.arg_descriptors):
190 pre = None
191 if descriptor.argument_type.lower() == "gh_operator":
192 pre = "op_"
193 elif descriptor.argument_type.lower() == "gh_columnwise_operator":
194 pre = "cma_op_"
195 elif descriptor.argument_type.lower() == "gh_field":
196 pre = "field_"
197 elif (descriptor.argument_type.lower() in
198 const.VALID_SCALAR_NAMES):
199 if descriptor.data_type.lower() == "gh_real":
200 pre = "rscalar_"
201 elif descriptor.data_type.lower() == "gh_integer":
202 pre = "iscalar_"
203 elif descriptor.data_type.lower() == "gh_logical":
204 pre = "lscalar_"
205 else:
206 raise InternalError(
207 f"Expected one of {const.VALID_SCALAR_DATA_TYPES} "
208 f"data types for a scalar argument but found "
209 f"'{descriptor.data_type}'.")
210 else:
211 raise GenerationError(
212 f"LFRicKern.load_meta() expected one of "
213 f"{const.VALID_ARG_TYPE_NAMES} but found "
214 f"'{descriptor.argument_type}'")
215 args.append(Arg("variable", pre+str(idx+1)))
216
217 if descriptor.stencil:
218 if not descriptor.stencil["extent"]:
219 # Stencil size (in cells) is passed in
220 args.append(Arg("variable",
221 pre+str(idx+1)+"_stencil_size"))
222 if descriptor.stencil["type"] == "xory1d":
223 # Direction is passed in
224 args.append(Arg("variable", pre+str(idx+1)+"_direction"))
225
226 # Initialise basis/diff basis so we can test whether quadrature
227 # or an evaluator is required
228 self._setup_basis(ktype)
229 if self._basis_required:
230 for shape in self._eval_shapes_eval_shapes:
231 if shape in const.VALID_QUADRATURE_SHAPES:
232 # Add a quadrature argument for each required quadrature
233 # rule.
234 args.append(Arg("variable", "qr_"+shape))
235
236 # If this kernel operates on halo cells then it takes an additional
237 # argument specifying the halo depth.
238 if "halo" in ktype.iterates_over:
239 args.append(Arg("variable", "halo_depth"))
240
241 self._setup(ktype, "dummy_name", args, None, check=False)
242
243 def _setup_basis(self, kmetadata):
244 '''
245 Initialisation of the basis/diff basis information. This may be
246 needed before general setup so is computed in a separate method.
247
248 :param kmetadata: The kernel metadata object produced by the parser.
249 :type kmetadata: :py:class:`psyclone.domain.lfric.LFRicKernMetadata`
250 '''
251 for descriptor in kmetadata.func_descriptors:
252 if len(descriptor.operator_names) > 0:
253 self._basis_required = True
254 self._eval_shapes_eval_shapes = kmetadata.eval_shapes[:]
255 break

257 def _setup(self, ktype, module_name, args, parent, check=True):
258 # pylint: disable=too-many-arguments
259 # pylint: disable=too-many-branches, too-many-locals
260 '''Internal setup of kernel information.
261
262 :param ktype: object holding information on the parsed metadata for \
263 this kernel.
264 :type ktype: :py:class:`psyclone.domain.lfric.LFRicKernMetadata`
265 :param str module_name: the name of the Fortran module that contains \
266 the source of this Kernel.
267 :param args: list of Arg objects produced by the parser for the \
268 arguments of this kernel call.
269 :type args: List[:py:class:`psyclone.parse.algorithm.Arg`]
270 :param parent: the parent of this kernel call in the generated \
271 AST (will be a loop object).
272 :type parent: :py:class:`psyclone.domain.lfric.LFRicLoop`
273 :param bool check: whether to check for consistency between the \
274 kernel metadata and the algorithm layer. Defaults to True.
275
276 '''
277 # Import here to avoid circular dependency
278 # pylint: disable=import-outside-toplevel
279 from psyclone.dynamo0p3 import DynKernelArguments, FSDescriptors
280 super().__init__(DynKernelArguments,
281 KernelCall(module_name, ktype, args),
282 parent, check)
283
284 # Remove "_code" from the name if it exists to determine the
285 # base name which (if LFRic naming conventions are
286 # followed) is used as the root for the module and subroutine
287 # names.
288 if self.namenamename.lower().endswith("_code"):
289 self._base_name = self.namenamename[:-5]
290 else:
291 # TODO: #11 add a warning here when logging is added
292 self._base_name = self.namenamename
293 self._func_descriptors = ktype.func_descriptors
294 # Keep a record of the type of CMA kernel identified when
295 # parsing the kernel metadata
296 self._cma_operation = ktype.cma_operation
297 self._fs_descriptors = FSDescriptors(ktype.func_descriptors)
298
299 # If the kernel metadata specifies that this is an inter-grid kernel
300 # create the associated DynInterGrid
301 if ktype.is_intergrid:
302 if not self.ancestor(InvokeSchedule):
303 raise NotImplementedError(
304 f"Intergrid kernels can only be setup inside an "
305 f"InvokeSchedule, but attempted '{self.name}' without it.")
306 fine_args = args_filter(self.arguments.args,
307 arg_meshes=["gh_fine"])
308 coarse_args = args_filter(self.arguments.args,
309 arg_meshes=["gh_coarse"])
310
311 from psyclone.dynamo0p3 import DynInterGrid
312 intergrid = DynInterGrid(fine_args[0], coarse_args[0])
313 self._intergrid_ref = intergrid
314
315 const = LFRicConstants()
316 # Check that all specified evaluator shapes are recognised
317 invalid_shapes = set(self._eval_shapes_eval_shapes) \
318 - set(const.VALID_EVALUATOR_SHAPES)
319 if invalid_shapes:
320 raise InternalError(
321 f"Evaluator shape(s) {list(invalid_shapes)} is/are not "
322 f"recognised. Must be one of {const.VALID_EVALUATOR_SHAPES}.")
323
324 # If this kernel operates into the halo then it must be passed a
325 # halo depth. This is currently restricted to being either a simple
326 # variable name or a literal value.
327 freader = FortranReader()
328 invoke_schedule = self.ancestor(InvokeSchedule)
329 table = invoke_schedule.symbol_table if invoke_schedule else None
330 if "halo" in ktype.iterates_over:
331 self._halo_depth_halo_depth = freader.psyir_from_expression(
332 args[-1].text.lower(), symbol_table=table)
333 if isinstance(self._halo_depth_halo_depth, Reference):
334 # If we got a Reference, check whether we need to specialise
335 # the associated Symbol.
336 sym = self._halo_depth_halo_depth.symbol
337 if not hasattr(sym, "datatype"):
338 self._halo_depth_halo_depth.symbol.specialise(
339 DataSymbol,
340 datatype=LFRicTypes("LFRicIntegerScalarDataType")())
341 # If there are any quadrature rule(s), what are the names of the
342 # corresponding algorithm arguments? Can't use set() here because
343 # we need to preserve the ordering specified in the metadata.
344 qr_shapes = [shape for shape in self._eval_shapes_eval_shapes if
345 shape in const.VALID_QUADRATURE_SHAPES]
346
347 # The quadrature-related arguments to a kernel always come last so
348 # construct an enumerator with start value -<no. of qr rules>
349 start_value = -len(qr_shapes)
350 if self._halo_depth_halo_depth:
351 start_value -= 1
352 for idx, shape in enumerate(qr_shapes, start_value):
353
354 qr_arg = args[idx]
355
356 # Use the InvokeSchedule symbol_table to create a unique symbol
357 # name for the whole Invoke.
358 if qr_arg.varname:
359 tag = "AlgArgs_" + qr_arg.text
360 qr_name = table.find_or_create_integer_symbol(qr_arg.varname,
361 tag=tag).name
362 else:
363 # If we don't have a name then we must be doing kernel-stub
364 # generation so create a suitable name.
365 # TODO #719 we don't yet have a symbol table to prevent
366 # clashes.
367 qr_name = "qr_"+shape.split("_")[-1]
368
369 # LFRic api kernels require quadrature rule arguments to be
370 # passed in if one or more basis functions are used by the kernel
371 # and gh_shape == "gh_quadrature_***".
372 # if self._eval_shape == "gh_quadrature_xyz":
373 # self._qr_args = ["np_xyz", "weights_xyz"]
374 if shape == "gh_quadrature_xyoz":
375 qr_args = ["np_xy", "np_z", "weights_xy", "weights_z"]
376 # elif self._eval_shape == "gh_quadrature_xoyoz":
377 # qr_args = ["np_x", "np_y", "np_z",
378 # "weights_x", "weights_y", "weights_z"]
379 elif shape == "gh_quadrature_face":
380 qr_args = ["nfaces", "np_xyz", "weights_xyz"]
381 elif shape == "gh_quadrature_edge":
382 qr_args = ["nedges", "np_xyz", "weights_xyz"]
383 else:
384 raise InternalError(f"Unsupported quadrature shape "
385 f"('{shape}') found in LFRicKern._setup")
386
387 # Append the name of the qr argument to the names of the qr-related
388 # variables.
389 qr_args = [arg + "_" + qr_name for arg in qr_args]
390
391 self._qr_rules[shape] = self.QRRule(qr_arg.text, qr_name, qr_args)
392
393 if "gh_evaluator" in self._eval_shapes_eval_shapes:
394 # Kernel has an evaluator. If gh_evaluator_targets is present
395 # then that specifies the function spaces for which the evaluator
396 # is required. Otherwise, the FS of the updated argument(s) tells
397 # us upon which nodal points the evaluator will be required
398 for fs_name in ktype.eval_targets:
399 arg, fspace = self.arguments.get_arg_on_space_name(fs_name)
400 # Set up our dict of evaluator targets, one entry per
401 # target FS.
402 if fspace.mangled_name not in self._eval_targets_eval_targets:
403 self._eval_targets_eval_targets[fspace.mangled_name] = (fspace, arg)
404
405 # Properties of the reference element required by this kernel
406 self._reference_element = ktype.reference_element
407
408 # Properties of the mesh required by this kernel
409 self._mesh_properties = ktype.mesh

636 @property
637 def gen_stub(self):
638 '''
639 Create the fparser1 AST for a kernel stub.
640
641 :returns: root of fparser1 AST for the stub routine.
642 :rtype: :py:class:`fparser.one.block_statements.Module`
643
644 :raises GenerationError: if the supplied kernel stub does not operate
645 on a supported subset of the domain (currently only those that
646 end with "cell_column").
647
648 '''
649 # The operates-on/iterates-over values supported by the stub generator.
650 const = LFRicConstants()
651 supported_operates_on = const.USER_KERNEL_ITERATION_SPACES[:]
652 # TODO #925 Add support for 'domain' kernels
653 # TODO #1351 Add support for 'dof' kernels
654 supported_operates_on.remove("domain")
655 supported_operates_on.remove("dof")
656
657 # Check operates-on (iteration space) before generating code

661 f"operate on one of {supported_operates_on} but found "
662 f"'{self.iterates_over}' in kernel '{self.name}'.")

665 psy_module = ModuleGen(self._base_name+"_mod")

669 implicitnone=True)

673 # pylint: disable=import-outside-toplevel
674 from psyclone.domain.lfric import (

705 def get_kernel_schedule(self):
706 '''Returns a PSyIR Schedule representing the kernel code. The base
707 class creates the PSyIR schedule on first invocation which is
708 then checked for consistency with the kernel metadata
709 here. The Schedule is just generated on first invocation, this
710 allows us to retain transformations that may subsequently be
711 applied to the Schedule.
712
713 Once issue #935 is implemented, this routine will return the
714 PSyIR Schedule using LFRic-specific PSyIR where possible.
715
716 :returns: Schedule representing the kernel code.
717 :rtype: :py:class:`psyclone.psyGen.KernelSchedule`
718
719 :raises GenerationError: if no subroutine matching this kernel can \
720 be found in the parse tree of the associated source code.
721 '''
722 if self._kern_schedule_kern_schedule:
723 return self._kern_schedule_kern_schedule
724
725 # Get the PSyIR Kernel Schedule(s)
726 routines = Fparser2Reader().get_routine_schedules(self.namenamename, self.astastast)
727 for routine in routines:
728 # If one of the symbols is not declared in a routine then
729 # this is only picked up when writing out the routine
730 # (raising a VisitorError), so we check here so that
731 # invalid code is not inlined. We use debug_string() to
732 # minimise the overhead.
733
734 # TODO #2271 could potentially avoid the need for
735 # debug_string() within. Sergi suggests that we may be
736 # missing the traversal of the declaration init
737 # expressions and that might solve the problem. I'm not so
738 # sure as we are talking about unknown symbols that will
739 # only be resolved in the back-end (or not). If I am right
740 # then one option would be to use the FortranWriter, but
741 # that would be bigger overhead, or perhaps just the
742 # declarations part of FortranWriter if that is possible.
743 # Also see TODO issue #2336 which captures the specific
744 # problem in LFRic that this fixes.
745 routine.debug_string()
746
747 if len(routines) == 1:
748 sched = routines[0]
749 # TODO #928: We don't validate the arguments yet because the
750 # validation has many false negatives.
751 # self.validate_kernel_code_args(sched.symbol_table)
752 else:
753 # The kernel name corresponds to an interface block. Find which
754 # of the routines matches the precision of the arguments.
755 for routine in routines:
756 try:
757 # The validity check for the kernel arguments will raise
758 # an exception if the precisions don't match.
759 self.validate_kernel_code_args(routine.symbol_table)
760 sched = routine
761 break
762 except GenerationError:
763 pass
764 else:
765 raise GenerationError(
766 f"Failed to find a kernel implementation with an interface"
767 f" that matches the invoke of '{self.name}'. (Tried "
768 f"routines {[item.name for item in routines]}.)")
769
770 # TODO #935 - replace the PSyIR argument data symbols with LFRic data
771 # symbols. For the moment we just return the unmodified PSyIR schedule
772 # but this should use RaisePSyIR2LFRicKernTrans once KernelInterface
773 # is fully functional (#928).
774 ksched = KernelSchedule(sched.symbol,
775 symbol_table=sched.symbol_table.detach())
776 for child in sched.pop_all_children():
777 ksched.addchild(child)
778 sched.replace_with(ksched)
779
780 self._kern_schedule_kern_schedule = ksched
781
782 return self._kern_schedule_kern_schedule

784 def validate_kernel_code_args(self, table):
785 '''Check that the arguments in the kernel code match the expected
786 arguments as defined by the kernel metadata and the LFRic
787 API.
788
789 :param table: the symbol table to validate against the metadata.
790 :type table: :py:class:`psyclone.psyir.symbols.SymbolTable`
791
792 :raises GenerationError: if the number of arguments indicated by the \
793 kernel metadata doesn't match the actual number of arguments in \
794 the symbol table.
795
796 '''
797 # Get the kernel subroutine arguments
798 kern_code_args = table.argument_list
799
800 # Get the kernel code interface according to the kernel
801 # metadata and LFRic API
802 interface_info = KernelInterface(self)
803 interface_info.generate()
804 interface_args = interface_info.arglist
805
806 # 1: Check the the number of arguments match
807 actual_n_args = len(kern_code_args)
808 expected_n_args = len(interface_args)
809 if actual_n_args != expected_n_args:
810 raise GenerationError(
811 f"In kernel '{self.name}' the number of arguments indicated by"
812 f" the kernel metadata is {expected_n_args} but the actual "
813 f"number of kernel arguments found is {actual_n_args}.")
814
815 # 2: Check that the properties of each argument match.
816 for idx, kern_code_arg in enumerate(kern_code_args):
817 interface_arg = interface_args[idx]
818 try:
819 alg_idx = interface_info.metadata_index_from_actual_index(idx)
820 alg_arg = self.arguments.args[alg_idx]
821 except KeyError:
822 # There's no algorithm argument directly associated with this
823 # kernel argument. (We only care about the data associated
824 # with scalar, field and operator arguments.)
825 alg_arg = None
826 self._validate_kernel_code_arg(kern_code_arg, interface_arg,
827 alg_arg)

829 def _validate_kernel_code_arg(self, kern_code_arg, interface_arg,
830 alg_arg=None):
831 '''Internal method to check that the supplied argument descriptions
832 match and raise appropriate exceptions if not.
833
834 :param kern_code_arg: kernel code argument.
835 :type kern_code_arg: :py:class:`psyclone.psyir.symbols.DataSymbol`
836 :param interface_arg: expected argument.
837 :type interface_arg: :py:class:`psyclone.psyir.symbols.DataSymbol`
838 :param alg_arg: the associated argument in the Algorithm layer. Note \
839 that only kernel arguments holding the data associated with \
840 scalar, field and operator arguments directly correspond to \
841 arguments that appear in the Algorithm layer.
842 :type alg_arg: \
843 Optional[:py:class`psyclone.dynamo0p3.DynKernelArgument`]
844
845 :raises GenerationError: if the contents of the arguments do \
846 not match.
847 :raises InternalError: if an unexpected datatype is found.
848
849 '''
850 # 1: intrinsic datatype
851 actual_datatype = kern_code_arg.datatype.intrinsic
852 expected_datatype = interface_arg.datatype.intrinsic
853 if actual_datatype != expected_datatype:
854 raise GenerationError(
855 f"Kernel argument '{kern_code_arg.name}' has datatype "
856 f"'{actual_datatype}' in kernel '{self.name}' but the LFRic "
857 f"API expects '{expected_datatype}'.")
858 # 2: precision. An LFRic kernel is only permitted to have a precision
859 # specified by a recognised type parameter or a no. of bytes.
860 actual_precision = kern_code_arg.datatype.precision
861 api_config = Config.get().api_conf("lfric")
862 if isinstance(actual_precision, DataSymbol):
863 # Convert precision into number of bytes to support
864 # mixed-precision kernels.
865 # TODO #1941: it would be better if the LFRic constants_mod.f90
866 # was the single source of truth for precision values.
867 actual_precision = api_config.precision_map[
868 actual_precision.name]
869 elif not isinstance(actual_precision, int):
870 raise GenerationError(
871 f"An argument to an LFRic kernel must have a precision defined"
872 f" by either a recognised LFRic type parameter (one of "
873 f"{sorted(api_config.precision_map.keys())}) or an integer"
874 f" number of bytes but argument '{kern_code_arg.name}' to "
875 f"kernel '{self.name}' has precision {actual_precision}.")
876
877 if alg_arg:
878 # We have information on the corresponding argument in the
879 # Algorithm layer so we can check that the precision matches.
880 # This is used to identify the correct kernel subroutine for a
881 # mixed-precision kernel.
882 alg_precision = api_config.precision_map[alg_arg.precision]
883 if alg_precision != actual_precision:
884 raise GenerationError(
885 f"Precision ({alg_precision} bytes) of algorithm-layer "
886 f"argument '{alg_arg.name}' does not match that "
887 f"({actual_precision} bytes) of the corresponding kernel "
888 f"subroutine argument '{kern_code_arg.name}' for kernel "
889 f"'{self.name}'.")
890
891 # 3: intent
892 actual_intent = kern_code_arg.interface.access
893 expected_intent = interface_arg.interface.access
894 if actual_intent.name != expected_intent.name:
895 raise GenerationError(
896 f"Kernel argument '{kern_code_arg.name}' has intent "
897 f"'{actual_intent.name}' in kernel '{self.name}' but the "
898 f"LFRic API expects intent '{expected_intent.name}'.")
899 # 4: scalar or array
900 if interface_arg.is_scalar:
901 if not kern_code_arg.is_scalar:
902 raise GenerationError(
903 f"Argument '{kern_code_arg.name}' to kernel '{self.name}' "
904 f"should be a scalar according to the LFRic API, but it "
905 f"is not.")
906 elif interface_arg.is_array:
907 if not kern_code_arg.is_array:
908 raise GenerationError(
909 f"Argument '{kern_code_arg.name}' to kernel '{self.name}' "
910 f"should be an array according to the LFRic API, but it "
911 f"is not.")
912 # 4.1: array dimensions
913 if len(interface_arg.shape) != len(kern_code_arg.shape):
914 raise GenerationError(
915 f"Argument '{kern_code_arg.name}' to kernel '{self.name}' "
916 f"should be an array with {len(interface_arg.shape)} "
917 f"dimension(s) according to the LFRic API, but "
918 f"found {len(kern_code_arg.shape)}.")
919 for dim_idx, kern_code_arg_dim in enumerate(kern_code_arg.shape):
920 if not isinstance(kern_code_arg_dim, ArrayType.ArrayBounds):
921 continue
922 if (not isinstance(kern_code_arg_dim.lower, Literal) or
923 kern_code_arg_dim.lower.value != "1"):
924 raise GenerationError(
925 f"All array arguments to LFRic kernels must have lower"
926 f" bounds of 1 for all dimensions. However, array "
927 f"'{kern_code_arg.name}' has a lower bound of "
928 f"'{kern_code_arg_dim.lower}' for dimension {dim_idx}")
929 kern_code_arg_upper_dim = kern_code_arg_dim.upper
930 interface_arg_upper_dim = interface_arg.shape[dim_idx].upper
931 if (isinstance(kern_code_arg_upper_dim, Reference) and
932 isinstance(interface_arg_upper_dim, Reference) and
933 isinstance(kern_code_arg_upper_dim.symbol,
934 DataSymbol) and
935 isinstance(interface_arg_upper_dim.symbol,
936 DataSymbol)):
937 # Only check when there is a symbol. Unspecified
938 # dimensions, dimensions with scalar values,
939 # offsets, or dimensions that include arithmetic
940 # are skipped.
941 try:
942 self._validate_kernel_code_arg(
943 kern_code_arg_upper_dim.symbol,
944 interface_arg_upper_dim.symbol)
945 except GenerationError as info:
946 raise GenerationError(
947 f"For dimension {dim_idx+1} in array argument "
948 f"'{kern_code_arg.name}' to kernel '{self.name}' "
949 f"the following error was found: "
950 f"{info.args[0]}") from info
951 else:
952 raise InternalError(
953 f"Unexpected argument type found for '{kern_code_arg.name}' in"
954 f" kernel '{self.name}'. Expecting a scalar or an array.")

956 def validate_global_constraints(self):
957 '''
958 Perform validation checks for any global constraints (that require the
959 tree to be complete).
960
961 :raises GenerationError: if this kernel does not have a supported
962 operates-on.
963 :raises GenerationError: if the loop goes beyond the level 1
964 halo and an operator is accessed.
965 :raises GenerationError: if a kernel in the loop has an inc access
966 and the loop is not coloured but is within an OpenMP parallel
967 region.
968 '''
969 # Check operates-on (iteration space) before generating code
970 const = LFRicConstants()
971 if self.iterates_over not in const.USER_KERNEL_ITERATION_SPACES:
972 raise GenerationError(
973 f"The LFRic API supports calls to user-supplied kernels that "
974 f"operate on one of {const.USER_KERNEL_ITERATION_SPACES}, but "
975 f"kernel '{self.name}' operates on '{self.iterates_over}'.")
976
977 # pylint: disable=import-outside-toplevel
978 from psyclone.domain.lfric import LFRicLoop
979 parent_loop = self.ancestor(LFRicLoop)
980
981 # Check whether this kernel reads from an operator
982 op_args = parent_loop.args_filter(
983 arg_types=const.VALID_OPERATOR_NAMES,
984 arg_accesses=[AccessType.READ, AccessType.READWRITE])
985 if op_args:
986 # It does. We must check that our parent loop does not
987 # go beyond the L1 halo.
988 if (parent_loop.upper_bound_name == "cell_halo" and
989 parent_loop.upper_bound_halo_depth != Literal(
990 "1", INTEGER_TYPE)):
991 raise GenerationError(
992 f"Kernel '{self._name}' reads from an operator and "
993 f"therefore cannot be used for cells beyond the level 1 "
994 f"halo. However the containing loop goes out to level "
995 f"{parent_loop.upper_bound_halo_depth.debug_string()}")
996
997 if not self.is_coloured():
998 # This kernel call has not been coloured
999 # - is it OpenMP parallel, i.e. are we a child of
1000 # an OpenMP directive?
1001 if self.is_openmp_parallel():
1002 try:
1003 # It is OpenMP parallel - does it have an argument
1004 # with INC access?
1005 _ = self.incremented_arg()
1006 raise GenerationError(f"Kernel '{self._name}' has an "
1007 f"argument with INC access and "
1008 f"therefore must be coloured in "
1009 f"order to be parallelised with "
1010 f"OpenMP.")
1011 except FieldNotFoundError:
1012 pass
1013
1014 super().validate_global_constraints()
