73def object_index(alist, item):
74 '''
75 A version of the `list.index()` method that checks object identity
76 rather that the content of the object.
77
78 TODO this is a workaround for the fact that fparser2 overrides the
79 comparison operator for all nodes in the parse tree. See fparser
80 issue 174.
81
82 :param alist: single object or list of objects to search.
83 :type alist: list or :py:class:`fparser.two.utils.Base`
84 :param obj item: object to search for in the list.
85 :returns: index of the item in the list.
86 :rtype: int
87 :raises ValueError: if object is not in the list.
88 '''
89 if item is None:
90 raise InternalError("Cannot search for None item in list.")
91 for idx, entry in enumerate(alist):
92 if entry is item:
93 return idx
94 raise ValueError(f"Item '{item}' not found in list: {alist}")

109def args_filter(arg_list, arg_types=None, arg_accesses=None, arg_meshes=None,
110 include_literals=True):
111 '''
112 Return all arguments in the supplied list that are of type
113 arg_types and with access in arg_accesses. If these are not set
114 then return all arguments.
115
116 :param arg_list: list of kernel arguments to filter.
117 :type arg_list: list of :py:class:`psyclone.parse.kernel.Descriptor`
118 :param arg_types: list of argument types (e.g. "GH_FIELD").
119 :type arg_types: list of str
120 :param arg_accesses: list of access types that arguments must have.
121 :type arg_accesses: list of \
122 :py:class:`psyclone.core.access_type.AccessType`
123 :param arg_meshes: list of meshes that arguments must be on.
124 :type arg_meshes: list of str
125 :param bool include_literals: whether or not to include literal arguments \
126 in the returned list.
127
128 :returns: list of kernel arguments matching the requirements.
129 :rtype: list of :py:class:`psyclone.parse.kernel.Descriptor`
130
131 '''
132 arguments = []
133 for argument in arg_list:
134 if arg_types:
135 if argument.argument_type.lower() not in arg_types:
136 continue
137 if arg_accesses:
138 if argument.access not in arg_accesses:
139 continue
140 if arg_meshes:
141 if argument.mesh not in arg_meshes:
142 continue
143 if not include_literals:
144 # We're not including literal arguments so skip this argument
145 # if it is literal.
146 if argument.is_literal:
147 continue
148 arguments.append(argument)
149 return arguments

312 def get(self, invoke_name):
313 '''
314 Gets the Invoke with the supplied name. If the name does not already
315 begin with ``invoke_`` then a new name with this prepended is included
316 in the search if no exact match is found initially.
317
318 :param str invoke_name: the name of the Invoke to get (not case-
319 sensitive).
320
321 :returns: the invoke with the specified name.
322 :rtype: :py:class:`psyclone.psyGen.Invoke`
323
324 :raises RuntimeError: if no Invoke with the supplied name (with or
325 without ``invoke_`` prepended) exists.
326 '''
327 search_names = [invoke_name.lower()]
328 if not search_names[0].startswith("invoke_"):
329 search_names.append("invoke_"+search_names[0])
330 for name in search_names:
331 try:
332 return self.invoke_map[name]
333 except KeyError:
334 pass
335
336 search_list = " or ".join(f"'{name}'" for name in search_names)
337 raise RuntimeError(f"Cannot find an invoke named {search_list} "
338 f"in {list(self.names)}")

359class Invoke():
360 r'''Manage an individual invoke call.
361
362 :param alg_invocation: metadata from the parsed code capturing \
363 information for this Invoke instance.
364 :type alg_invocation: :py:class:`psyclone.parse.algorithm.InvokeCall`
365 :param int idx: position/index of this invoke call in the subroutine. \
366 If not None, this number is added to the name ("invoke\_").
367 :param schedule_class: the schedule class to create for this invoke.
368 :type schedule_class: :py:class:`psyclone.psyGen.InvokeSchedule`
369 :param invokes: the Invokes instance that contains this Invoke \
370 instance.
371 :type invokes: :py:class:`psyclone.psyGen.Invokes`
372 :param reserved_names: optional list of reserved names, i.e. names that \
373 should not be used e.g. as a PSyclone-created \
374 variable name.
375 :type reserved_names: list of str
376
377 '''
378 def __init__(self, alg_invocation, idx, schedule_class, invokes,
379 reserved_names=None):
380 '''Construct an invoke object.'''
381
382 self._invokes = invokes
383 self._name_name_name = "invoke"
384 self._alg_unique_args = []
385
386 if alg_invocation is None and idx is None:
387 return
388
389 # create a name for the call if one does not already exist
390 if alg_invocation.name is not None:
391 # In Python2 unicode strings must be converted to str()
392 self._name_name_name = str(alg_invocation.name)
393 elif len(alg_invocation.kcalls) == 1 and \
394 alg_invocation.kcalls[0].type == "kernelCall":
395 # use the name of the kernel call with the position appended.
396 # Appended position is needed in case we have two separate invokes
397 # in the same algorithm code containing the same (single) kernel
398 self._name_name_name = "invoke_" + str(idx) + "_" + \
399 alg_invocation.kcalls[0].ktype.name
400 else:
401 # use the position of the invoke
402 self._name_name_name = "invoke_" + str(idx)
403
404 if not reserved_names:
405 reserved_names = []
406
407 # Get a reference to the parent container, if any
408 container = None
409 if self.invokesinvokes:
410 container = self.invokesinvokes.psy.container
411
412 # create the schedule (Routine sub-class). Routines use a
413 # symbol input argument.
414 schedule_symbol = RoutineSymbol(self._name_name_name)
415 self._schedule = schedule_class(schedule_symbol,
416 alg_invocation.kcalls,
417 reserved_names, parent=container)
418
419 # Add the new Schedule to the top-level PSy Container
420 if container:
421 container.addchild(self._schedule)
422
423 # let the schedule have access to me
424 self._schedule.invoke = self
425
426 # extract the argument list for the algorithm call and psy
427 # layer subroutine.
428 self._alg_unique_args = []
429 self._psy_unique_vars = []
430 tmp_arg_names = []
431 for call in self.scheduleschedule.kernels():
432 for arg in call.arguments.args:
433 if arg.text is not None:
434 if arg.text not in self._alg_unique_args:
435 self._alg_unique_args.append(arg.text)
436 if arg.name not in tmp_arg_names:
437 tmp_arg_names.append(arg.name)
438 self._psy_unique_vars.append(arg)
439 else:
440 # literals have no name
441 pass
442
443 # work out the unique dofs required in this subroutine
444 self._dofs_dofs = {}
445 for kern_call in self._schedule.coded_kernels():
446 dofs = kern_call.arguments.dofs
447 for dof in dofs:
448 if dof not in self._dofs_dofs:
449 # Only keep the first occurrence for the moment. We will
450 # need to change this logic at some point as we need to
451 # cope with writes determining the dofs that are used.
452 self._dofs_dofs[dof] = [kern_call, dofs[dof][0]]
453
454 def __str__(self):
455 return self._name_name_name+"("+", ".join([str(arg) for arg in
456 self._alg_unique_args])+")"

494 def unique_declarations(self, argument_types, access=None,
495 intrinsic_type=None):
496 '''
497 Returns a list of all required declarations for the specified
498 API argument types. If access is supplied (e.g. "write") then
499 only declarations with that access are returned. If an intrinsic
500 type is supplied then only declarations with that intrinsic type
501 are returned.
502
503 :param argument_types: the types of the kernel argument for the \
504 particular API.
505 :type argument_types: list of str
506 :param access: optional AccessType that the declaration should have.
507 :type access: :py:class:`psyclone.core.access_type.AccessType`
508 :param intrinsic_type: optional intrinsic type of argument data.
509 :type intrinsic_type: str
510
511 :returns: a list of all declared kernel arguments.
512 :rtype: list of :py:class:`psyclone.psyGen.KernelArgument`
513
514 :raises InternalError: if at least one kernel argument type is \
515 not valid for the particular API.
516 :raises InternalError: if an invalid access is specified.
517 :raises InternalError: if an invalid intrinsic type is specified.
518
519 '''
520 # First check for invalid argument types, access and intrinsic type
521 const = Config.get().api_conf().get_constants()
522 if any(argtype not in const.VALID_ARG_TYPE_NAMES for
523 argtype in argument_types):
524 raise InternalError(
525 f"Invoke.unique_declarations() called with at least one "
526 f"invalid argument type. Expected one of "
527 f"{const.VALID_ARG_TYPE_NAMES} but found {argument_types}.")
528
529 if access and not isinstance(access, AccessType):
530 raise InternalError(
531 f"Invoke.unique_declarations() called with an invalid "
532 f"access type. Type is '{access}' instead of AccessType.")
533
534 if (intrinsic_type and intrinsic_type not in
535 const.VALID_INTRINSIC_TYPES):
536 raise InternalError(
537 f"Invoke.unique_declarations() called with an invalid "
538 f"intrinsic argument data type. Expected one of "
539 f"{const.VALID_INTRINSIC_TYPES} but found '{intrinsic_type}'.")
540
541 # Initialise dictionary of kernel arguments to get the
542 # argument list from
543 declarations = OrderedDict()
544 # Find unique kernel arguments using their declaration names
545 for call in self.scheduleschedule.kernels():
546 for arg in call.arguments.args:
547 if not intrinsic_type or arg.intrinsic_type == intrinsic_type:
548 if not access or arg.access == access:
549 if arg.text is not None:
550 if arg.argument_type in argument_types:
551 test_name = arg.declaration_name
552 if test_name not in declarations:
553 declarations[test_name] = arg
554 return list(declarations.values())
555
556 def first_access(self, arg_name):
557 ''' Returns the first argument with the specified name passed to
558 a kernel in our schedule '''
559 for call in self.scheduleschedule.kernels():
560 for arg in call.arguments.args:
561 if arg.text is not None:
562 if arg.declaration_name == arg_name:
563 return arg
564 raise GenerationError(f"Failed to find any kernel argument with name "
565 f"'{arg_name}'")

567 def unique_declns_by_intent(self, argument_types, intrinsic_type=None):
568 '''
569 Returns a dictionary listing all required declarations for each
570 type of intent ('inout', 'out' and 'in').
571
572 :param argument_types: the types of the kernel argument for the \
573 particular API for which the intent is required.
574 :type argument_types: list of str
575 :param intrinsic_type: optional intrinsic type of argument data.
576 :type intrinsic_type: str
577
578 :returns: dictionary containing 'intent' keys holding the kernel \
579 arguments as values for each type of intent.
580 :rtype: dict of :py:class:`psyclone.psyGen.KernelArgument`
581
582 :raises InternalError: if at least one kernel argument type is \
583 not valid for the particular API.
584 :raises InternalError: if an invalid intrinsic type is specified.
585
586 '''
587 # First check for invalid argument types and intrinsic type
588 const = Config.get().api_conf().get_constants()
589 if any(argtype not in const.VALID_ARG_TYPE_NAMES for
590 argtype in argument_types):
591 raise InternalError(
592 f"Invoke.unique_declns_by_intent() called with at least one "
593 f"invalid argument type. Expected one of "
594 f"{const.VALID_ARG_TYPE_NAMES} but found {argument_types}.")
595
596 if (intrinsic_type and intrinsic_type not in
597 const.VALID_INTRINSIC_TYPES):
598 raise InternalError(
599 f"Invoke.unique_declns_by_intent() called with an invalid "
600 f"intrinsic argument data type. Expected one of "
601 f"{const.VALID_INTRINSIC_TYPES} but found '{intrinsic_type}'.")
602
603 # We will return a dictionary containing as many lists
604 # as there are types of intent
605 declns = {}
606 for intent in FORTRAN_INTENT_NAMES:
607 declns[intent] = []
608
609 for arg in self.unique_declarations(argument_types,
610 intrinsic_type=intrinsic_type):
611 first_arg = self.first_access(arg.declaration_name)
612 if first_arg.access in [AccessType.WRITE, AccessType.SUM]:
613 # If the first access is a write then the intent is
614 # out irrespective of any other accesses. Note,
615 # sum_args behave as if they are write_args from the
616 # PSy-layer's perspective.
617 declns["out"].append(arg)
618 continue
619 # if all accesses are read, then the intent is in,
620 # otherwise the intent is inout (as we have already
621 # dealt with intent out).
622 read_only = True
623 for call in self.scheduleschedule.kernels():
624 for tmp_arg in call.arguments.args:
625 if tmp_arg.text is not None and \
626 tmp_arg.declaration_name == arg.declaration_name:
627 if tmp_arg.access != AccessType.READ:
628 # readwrite_args behave in the
629 # same way as inc_args from the
630 # perspective of intents
631 read_only = False
632 break
633 if not read_only:
634 break
635 if read_only:
636 declns["in"].append(arg)
637 else:
638 declns["inout"].append(arg)
639 return declns

660class InvokeSchedule(Routine):
661 '''
662 Stores schedule information for an invocation call. Schedules can be
663 optimised using transformations.
664
665 >>> from psyclone.parse.algorithm import parse
666 >>> ast, info = parse("algorithm.f90")
667 >>> from psyclone.psyGen import PSyFactory
668 >>> api = "..."
669 >>> psy = PSyFactory(api).create(info)
670 >>> invokes = psy.invokes
671 >>> invokes.names
672 >>> invoke = invokes.get("name")
673 >>> schedule = invoke.schedule
674 >>> print(schedule.view())
675
676 :param symbol: RoutineSymbol representing the invoke.
677 :type symbol: :py:class:`psyclone.psyir.symbols.RoutineSymbol`
678 :param type KernFactory: class instance of the factory to use when \
679 creating Kernels. e.g. \
680 :py:class:`psyclone.domain.lfric.LFRicKernCallFactory`.
681 :param type BuiltInFactory: class instance of the factory to use when \
682 creating built-ins. e.g. \
683 :py:class:`psyclone.domain.lfric.lfric_builtins.LFRicBuiltInCallFactory`.
684 :param alg_calls: list of Kernel calls in the schedule.
685 :type alg_calls: list of :py:class:`psyclone.parse.algorithm.KernelCall`
686 :param kwargs: additional keyword arguments provided to the super class.
687 :type kwargs: unwrapped dict.
688
689 '''
690 # Textual description of the node.
691 _text_name = "InvokeSchedule"
692
693 def __init__(self, symbol, KernFactory, BuiltInFactory, alg_calls=None,
694 reserved_names=None, **kwargs):
695 super().__init__(symbol, **kwargs)
696
697 self._invoke = None
698
699 # Populate the Schedule Symbol Table with the reserved names.

1601 def rename_and_write(self):
1602 '''
1603 Writes the (transformed) AST of this kernel to file and resets the
1604 'modified' flag to False. By default (config.kernel_naming ==
1605 "multiple"), the kernel is re-named so as to be unique within
1606 the kernel output directory stored within the configuration
1607 object. Alternatively, if config.kernel_naming is "single"
1608 then no re-naming and output is performed if there is already
1609 a transformed copy of the kernel in the output dir. (In this
1610 case a check is performed that the transformed kernel already
1611 present is identical to the one that we would otherwise write
1612 to file. If this is not the case then we raise a GenerationError.)
1613
1614 :raises GenerationError: if config.kernel_naming == "single" and a \
1615 different, transformed version of this \
1616 kernel is already in the output directory.
1617 :raises NotImplementedError: if the kernel has been transformed but \
1618 is also flagged for module-inlining.
1619
1620 '''
1621 from psyclone.line_length import FortLineLength
1622
1623 config = Config.get()
1624
1625 # If this kernel has not been transformed we do nothing, also if the
1626 # kernel has been module-inlined, the routine already exist in the
1627 # PSyIR and we don't need to generate a new file with it.
1628 if not self.modifiedmodifiedmodified or self.module_inlinemodule_inlinemodule_inline:
1629 return
1630
1631 # Remove any "_mod" if the file follows the PSyclone naming convention
1632 orig_mod_name = self.module_name[:]
1633 if orig_mod_name.lower().endswith("_mod"):
1634 old_base_name = orig_mod_name[:-4]
1635 else:
1636 old_base_name = orig_mod_name[:]
1637
1638 # We could create a hash of a string built from the name of the
1639 # Algorithm (module), the name/position of the Invoke and the
1640 # index of this kernel within that Invoke. However, that creates
1641 # a very long name so we simply ensure that kernel names are unique
1642 # within the user-supplied kernel-output directory.
1643 name_idx = -1
1644 fdesc = None
1645 while not fdesc:
1646 name_idx += 1
1647 new_suffix = ""
1648
1649 new_suffix += f"_{name_idx}"
1650 new_name = old_base_name + new_suffix + "_mod.f90"
1651
1652 try:
1653 # Atomically attempt to open the new kernel file (in case
1654 # this is part of a parallel build)
1655 fdesc = os.open(
1656 os.path.join(config.kernel_output_dir, new_name),
1657 os.O_CREAT | os.O_WRONLY | os.O_EXCL)
1658 except (OSError, IOError):
1659 # The os.O_CREATE and os.O_EXCL flags in combination mean
1660 # that open() raises an error if the file exists
1661 if config.kernel_naming == "single":
1662 # If the kernel-renaming scheme is such that we only ever
1663 # create one copy of a transformed kernel then we're done
1664 break
1665 continue
1666
1667 # Use the suffix we have determined to rename all relevant quantities
1668 # within the AST of the kernel code.
1669 self._rename_psyir(new_suffix)
1670
1671 # Kernel is now self-consistent so unset the modified flag
1672 self.modifiedmodifiedmodified = False
1673
1674 # If we reach this point the kernel needs to be written out into a
1675 # file using a PSyIR back-end. At the moment there is no way to choose
1676 # which back-end to use, so simply use the Fortran one (and limit the
1677 # line length).
1678 fortran_writer = FortranWriter(
1679 check_global_constraints=config.backend_checks_enabled)
1680 # Start from the root of the schedule as we want to output
1681 # any module information surrounding the kernel subroutine
1682 # as well as the subroutine itself.
1683 new_kern_code = fortran_writer(self.get_kernel_schedule().root)
1684 fll = FortLineLength()
1685 new_kern_code = fll.process(new_kern_code)
1686
1687 if not fdesc:
1688 # If we've not got a file descriptor at this point then that's
1689 # because the file already exists and the kernel-naming scheme
1690 # ("single") means we're not creating a new one.
1691 # Check that what we've got is the same as what's in the file
1692 with open(os.path.join(config.kernel_output_dir,
1693 new_name), "r") as ffile:
1694 kern_code = ffile.read()
1695 if kern_code != new_kern_code:
1696 raise GenerationError(
1697 f"A transformed version of this Kernel "
1698 f"'{self._module_name + '''.f90'''}' already exists "
1699 f"in the kernel-output directory "
1700 f"({config.kernel_output_dir}) but is not the "
1701 f"same as the current, transformed kernel and the "
1702 f"kernel-renaming scheme is set to "
1703 f"'{config.kernel_naming}'. (If you wish to"
1704 f" generate a new, unique kernel for every kernel "
1705 f"that is transformed then use "
1706 f"'--kernel-renaming multiple'.)")
1707 else:
1708 # Write the modified AST out to file
1709 os.write(fdesc, new_kern_code.encode())
1710 # Close the new kernel file
1711 os.close(fdesc)

1977 def overlaps(self, arg):
1978 '''Determine whether the accesses to the provided argument overlap
1979 with the accesses of the source argument. Overlap means that
1980 the accesses share at least one memory location. For example,
1981 the arguments both access the 1st index of the same field.
1982
1983 We do not currently deal with accesses to a subset of an
1984 argument (unless it is a vector). This distinction will need
1985 to be added once loop splitting is supported.
1986
1987 :param arg: the argument to compare with our internal argument
1988 :type arg: :py:class:`psyclone.psyGen.Argument`
1989 :return bool: True if there are overlapping accesses between \
1990 arguments (i.e. accesses share at least one memory \
1991 location) and False if not.
1992
1993 '''
1994 if self._arg.name != arg.name:
1995 # the arguments are different args so do not overlap
1996 return False
1997
1998 if isinstance(self._call, HaloExchange) and \
1999 isinstance(arg.call, HaloExchange) and \
2000 (self._arg.vector_size > 1 or arg.vector_size > 1):
2001 # This is a vector field and both accesses come from halo
2002 # exchanges. As halo exchanges only access a particular
2003 # vector, the accesses do not overlap if the vector indices
2004 # being accessed differ.
2005
2006 # sanity check
2007 if self._arg.vector_size != arg.vector_size:
2008 raise InternalError(
2009 f"DataAccess.overlaps(): vector sizes differ for field "
2010 f"'{arg.name}' in two halo exchange calls. Found "
2011 f"'{self._arg.vector_size}' and '{arg.vector_size}'")
2012 if self._call.vector_index != arg.call.vector_index:
2013 # accesses are to different vector indices so do not overlap
2014 return False
2015 # accesses do overlap
2016 return True

2513 def _depends_on(self, argument):
2514 '''If there is a dependency between the argument and self then return
2515 True, otherwise return False. We consider there to be a
2516 dependency between two arguments if the names are the same and
2517 if one reads and one writes, or if both write. Dependencies
2518 are often defined as being read-after-write (RAW),
2519 write-after-read (WAR) and write after write (WAW). These
2520 dependencies can be considered to be forward dependencies, in
2521 the sense that RAW means that the read is after the write in
2522 the schedule. Similarly for WAR and WAW. We capture these
2523 dependencies in this method. However we also capture
2524 dependencies in the opposite direction (backward
2525 dependencies). These are the same dependencies as forward
2526 dependencies but are reversed. One could consider these to be
2527 read-before-write, write-before-read, and
2528 write-before-write. The terminology of forward and backward to
2529 indicate whether the argument we depend on is after or before
2530 us in the schedule is borrowed from loop dependence analysis
2531 where a forward dependence indicates a dependence in a future
2532 loop iteration and a backward dependence indicates a
2533 dependence on a previous loop iteration. Note, we currently
2534 assume that any read or write to an argument results in a
2535 dependence i.e. we do not consider the internal structure of
2536 the argument (e.g. it may be an array). However, this
2537 assumption is OK as all elements of an array are typically
2538 accessed. However, we may need to revisit this when we change
2539 the iteration spaces of loops e.g. for overlapping
2540 communication and computation.
2541
2542 :param argument: the argument we will check to see whether \
2543 there is a dependence on this argument instance (self).
2544 :type argument: :py:class:`psyclone.psyGen.Argument`
2545
2546 :returns: True if there is a dependence and False if not.
2547 :rtype: bool
2548
2549 '''
2550 if argument.name == self._name:
2551 if self.accessaccess in AccessType.all_write_accesses() and \
2552 argument.access in AccessType.all_read_accesses():
2553 return True
2554 if self.accessaccess in AccessType.all_read_accesses() and \
2555 argument.access in AccessType.all_write_accesses():
2556 return True
2557 if self.accessaccess in AccessType.all_write_accesses() and \
2558 argument.access in AccessType.all_write_accesses():
2559 return True
2560 return False
