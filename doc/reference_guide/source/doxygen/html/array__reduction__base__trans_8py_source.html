54class ArrayReductionBaseTrans(Transformation, ABC):
55 '''An abstract parent class providing common functionality to
56 array-reduction intrinsic transformations which translate the
57 intrinsics into an equivalent loop structure.
58
59 '''
60 _INTRINSIC_NAME = None
61 _INTRINSIC_TYPE = None

63 @staticmethod
64 def _get_args(node):
65 '''Utility method that returns the array-reduction intrinsic arguments
66 (array reference, dimension and mask).
67
68 :param node: an array-reduction intrinsic.
69 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
70
71 returns: a tuple containing the 3 arguments.
72 rtype: Tuple[py:class:`psyclone.psyir.nodes.reference.Reference`,
73 py:class:`psyclone.psyir.nodes.Literal` |
74 :py:class:`psyclone.psyir.nodes.Reference`,
75 Optional[:py:class:`psyclone.psyir.nodes.Node`]]
76
77 '''
78 # Determine the arguments to the intrinsic
79 args = [None, None, None]
80 arg_names_map = {"array": 0, "dim": 1, "mask": 2}
81 for idx, child in enumerate(node.arguments):
82 if not node.argument_names[idx]:
83 # positional arg
84 args[idx] = child
85 else:
86 # named arg
87 name = node.argument_names[idx].lower()
88 args[arg_names_map[name]] = child
89 return tuple(args)

91 def __str__(self):
92 return (f"Convert the PSyIR {self._INTRINSIC_NAME} intrinsic "
93 "to equivalent PSyIR code.")

95 # pylint: disable=too-many-branches
96 def validate(self, node, options=None):
97 '''Check that the input node is valid before applying the
98 transformation.
99
100 :param node: an array-reduction intrinsic.
101 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
102 :param options: options for the transformation.
103 :type options: Optional[Dict[str, Any]]
104
105 :raises TransformationError: if the supplied node is not an
106 intrinsic.
107 :raises TransformationError: if the supplied node is not an
108 array-reduction intrinsic.
109 :raises TransformationError: if there is a dimension argument.
110 :raises TransformationError: if the array argument is not an array.
111 :raises TransformationError: if the shape of the array is not
112 supported.
113 :raises TransformationError: if the array datatype is not
114 supported.
115 :raises TransformationError: if the intrinsic is not part of
116 an assignment.
117
118 '''
119 if not isinstance(node, IntrinsicCall):
120 raise TransformationError(
121 f"Error in {self.name} transformation. The supplied node "
122 f"argument is not an intrinsic, found "
123 f"'{type(node).__name__}'.")
124
125 if node.routine.name.upper() != self._INTRINSIC_NAME:
126 raise TransformationError(
127 f"Error in {self.name} transformation. The supplied node "
128 f"argument is not a {self._INTRINSIC_NAME.lower()} "
129 f"intrinsic, found '{node.routine.name}'.")
130
131 array_ref, dim_ref, _ = self._get_args(node)
132
133 # dim_ref is not yet supported by this transformation.
134 if dim_ref:
135 raise TransformationError(
136 f"The dimension argument to {self._INTRINSIC_NAME} is not "
137 f"yet supported.")
138
139 # There should be at least one arrayreference or reference to
140 # an array in the expression
141 # pylint: disable=unidiomatic-typecheck
142 for reference in array_ref.walk(Reference):
143 if (isinstance(reference, ArrayReference) or
144 type(reference) is Reference and
145 reference.symbol.is_array):
146 break
147 else:
148 raise TransformationError(
149 f"Error, no ArrayReference's found in the expression "
150 f"'{array_ref.debug_string()}'.")
151
152 if not node.ancestor(Assignment):
153 raise TransformationError(
154 f"{self.name} only works when the intrinsic is part "
155 f"of an Assignment.")
156
157 assignment = array_ref.ancestor(Assignment)
158 for this_node in assignment.lhs.walk(Node):
159 if this_node == array_ref:
160 raise TransformationError(
161 "Error, intrinsics on the lhs of an assignment are not "
162 "currently supported.")
163
164 if len(array_ref.children) == 0:
165 for shape in array_ref.symbol.shape:
166 if not (shape in [
167 ArrayType.Extent.DEFERRED, ArrayType.Extent.ATTRIBUTE]
168 or isinstance(shape, ArrayType.ArrayBounds)):
169 raise TransformationError(
170 f"Unexpected shape for array. Expecting one of "
171 f"Deferred, Attribute or Bounds but found '{shape}'.")
172
173 # If the lhs symbol is used anywhere on the assignment rhs, we need
174 # to create a temporary, and for this we need to resolve its datatype
175 for rhs_reference in assignment.rhs.walk(Reference):
176 if rhs_reference.symbol is assignment.lhs.symbol:
177 if not (isinstance(assignment.lhs.symbol, DataSymbol) and
178 isinstance(assignment.lhs.datatype, ScalarType)):
179 line = assignment.debug_string().strip('\n')
180 raise TransformationError(
181 f"To loopify '{line}'"
182 f" we need a temporary variable, but the type of "
183 f"'{assignment.lhs.debug_string()}' can not be "
184 f"resolved or is unsupported.")
185
186 # pylint: disable=too-many-locals
187 def apply(self, node, options=None):
188 '''Apply the array-reduction intrinsic conversion transformation to
189 the specified node. This node must be one of these intrinsic
190 operations which is converted to an equivalent loop structure.
191
192 :param node: an array-reduction intrinsic.
193 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
194 :param options: options for the transformation.
195 :type options: Optional[Dict[str, Any]]
196
197 '''
198 self.validatevalidate(node, options)
199
200 orig_lhs = node.ancestor(Assignment).lhs.copy()
201 orig_rhs = node.ancestor(Assignment).rhs.copy()
202
203 # Determine whether the assignment is an increment (as we have
204 # to use a temporary if so) e.g. x = x + MAXVAL(a) and store a
205 # reference to the appropriate variable in new_lhs for future
206 # use.
207 lhs_symbol = orig_lhs.symbol
208 increment = False
209 for rhs_reference in orig_rhs.walk(Reference):
210 if rhs_reference.symbol is lhs_symbol:
211 increment = True
212 if increment:
213 new_lhs_symbol = node.scope.symbol_table.new_symbol(
214 root_name="tmp_var", symbol_type=DataSymbol,
215 datatype=orig_lhs.datatype)
216 new_lhs = Reference(new_lhs_symbol)
217 else:
218 new_lhs = orig_lhs.copy()
219
220 expr, _, mask_ref = self._get_args(node)
221
222 # Step 1: replace all references to arrays within the
223 # intrinsic expressions and mask argument (if it exists) to
224 # array ranges. For example, 'maxval(a+b, mask=mod(c,2.0)==1)'
225 # becomes 'maxval(a(:,:)+b(:,:), mask=mod(c(:,:),2.0)==1)' if
226 # 'a', 'b' and 'c' are 2 dimensional arrays.
227 rhs = expr.copy()
228 _ = UnaryOperation.create(UnaryOperation.Operator.PLUS, rhs)
229 reference2arrayrange = Reference2ArrayRangeTrans()
230 # The reference to rhs becomes invalid in the following
231 # transformation so we keep a copy of the parent here and
232 # reset rhs to rhs_parent.children[0] after the
233 # transformation.
234 rhs_parent = rhs.parent
235 for reference in rhs.walk(Reference):
236 try:
237 reference2arrayrange.apply(reference)
238 except TransformationError:
239 pass
240 # Reset rhs from its parent as the previous transformation
241 # makes the value of rhs become invalid. We know there is only
242 # one child so can safely use children[0].
243 rhs = rhs_parent.children[0]
244 if mask_ref:
245 mask_ref_parent = mask_ref.parent
246 mask_ref_index = mask_ref.position
247 for reference in mask_ref.walk(Reference):
248 try:
249 reference2arrayrange.apply(reference)
250 except TransformationError:
251 pass
252 mask_ref = mask_ref_parent.children[mask_ref_index]
253
254 # Step 2: Put the intrinsic's extracted expression (stored in
255 # the 'rhs' variable) on the rhs of an argument with one of
256 # the arrays within the expression being added to the lhs of
257 # the argument. For example if:
258 # x = maxval(a(:,:)+b(:,:))
259 # then
260 # rhs = a(:,:)+b(:,:)
261 # resulting in the following code being created:
262 # a(:,:) = a(:,:)+b(:,:)
263 array_refs = rhs.walk(ArrayReference)
264 # The lhs of the created expression needs to be an array
265 # reference from the expression itself because the
266 # ArrayAssignment2Loops transformation uses it to obtain the loop
267 # bounds.
268 lhs = array_refs[0].copy()
269
270 assignment = Assignment.create(lhs, rhs.detach())
271 # Replace existing code so the new code gets access to symbol
272 # tables etc.
273 orig_assignment = node.ancestor(Assignment)
274 orig_assignment.replace_with(assignment)
275
276 # Step 3 call ArrayAssignment2Loops to create loop bounds
277 # and array indexing from the array ranges created in step 2
278 # (keeping track of where the new loop nest is created). Also
279 # extract the mask if it exists. For example:
280 # a(:,:) = a(:,:)+b(:,:)
281 # becomes
282 # do idx2 = LBOUND(a,2), UBOUND(a,2)
283 # do idx = LBOUND(a,1), UBOUND(a,1)
284 # a(idx,idx2) = a(idx,idx2) + b(idx,idx2)
285 # enddo
286 # enddo
287 if mask_ref:
288 # add mask to the rhs of the assignment
289 assignment_rhs = BinaryOperation.create(
290 BinaryOperation.Operator.AND, assignment.rhs.copy(),
291 mask_ref.copy())
292 assignment.rhs.replace_with(assignment_rhs)
293
294 assignment_parent = assignment.parent
295 assignment_position = assignment.position
296 # Must be placed here to avoid circular imports
297 # pylint: disable=import-outside-toplevel
298 from psyclone.psyir.transformations import ArrayAssignment2LoopsTrans
299 try:
300 ArrayAssignment2LoopsTrans().apply(assignment)
301 except TransformationError as err:
302 # The ArrayAssignment2LoopsTrans could fail to convert the ranges,
303 # unfortunately this can not be tested before modifications to the
304 # tree (e.g. in the validate), so the best we can do is reverting
305 # to the orginal statement (with maybe some leftover tmp variable)
306 # and produce the error here.
307 assignment.replace_with(orig_assignment)
308 # pylint: disable=raise-missing-from
309 raise TransformationError(
310 f"ArrayAssignment2LoopsTrans could not convert the "
311 f"expression:\n{assignment.debug_string()}\n into a loop "
312 f"because:\n{err.value}")
313 outer_loop = assignment_parent.children[assignment_position]
314 if mask_ref:
315 # remove mask from the rhs of the assignment
316 orig_assignment = assignment_rhs.children[0].copy()
317 indexed_mask_ref = assignment_rhs.children[1].copy()
318 assignment_rhs.replace_with(orig_assignment)
319
320 # Step 4 convert the original assignment (now within a loop
321 # and indexed) to its intrinsic form by replacing the
322 # assignment with new_lhs=INTRINSIC(orig_lhs,<expr>). Also
323 # add in the mask if one has been specified. For example:
324 # do idx2 = LBOUND(a,2), UBOUND(a,2)
325 # do idx = LBOUND(a,1), UBOUND(a,1)
326 # a(idx,idx2) = a(idx,idx2) + b(idx,idx2)
327 # enddo
328 # enddo
329 # becomes
330 # do idx2 = LBOUND(a,2), UBOUND(a,2)
331 # do idx = LBOUND(a,1), UBOUND(a,1)
332 # if (mod(c(idx,idx2),2.0)==1) then
333 # x = max(x, a(idx,idx2) + b(idx,idx2))
334 # end if
335 # enddo
336 # enddo
337 new_assignment = Assignment.create(
338 new_lhs.copy(), self._loop_body(
339 new_lhs.copy(), assignment.rhs.copy()))
340 if mask_ref:
341 # Place the indexed mask around the statement.
342 new_assignment = IfBlock.create(
343 indexed_mask_ref.copy(), [new_assignment])
344
345 assignment.replace_with(new_assignment)
346
347 # Step 5 initialise the variable and place it before the newly
348 # created outer loop (in step 2) and deal with any additional
349 # arguments on the rhs of the original expression. For
350 # example, if the original code looks like the following:
351 # x = value1 + maxval(a+b, mask=mod(c,2.0)==1) * value2
352 # and the newly created loop looks like the following:
353 # do idx2 = LBOUND(a,2), UBOUND(a,2)
354 # do idx = LBOUND(a,1), UBOUND(a,1)
355 # if (mod(c(idx,idx2),2.0)==1) then
356 # x = max(x, a(idx,idx2) + b(idx,idx2))
357 # end ifx
358 # enddo
359 # enddo
360 # then the result becomes:
361 # x = tiny(x)
362 # do idx2 = LBOUND(a,2), UBOUND(a,2)
363 # do idx = LBOUND(a,1), UBOUND(a,1)
364 # if (mod(c(idx,idx2),2.0)==1) then
365 # x = max(x, a(idx,idx2) + b(idx,idx2))
366 # end if
367 # enddo
368 # enddo
369 # x = value1 + x * value2
370 lhs = new_lhs.copy()
371 rhs = self._init_var(lhs)
372 assignment = Assignment.create(lhs, rhs)
373 outer_loop.parent.children.insert(outer_loop.position, assignment)
374 if not (isinstance(orig_rhs, IntrinsicCall) and
375 orig_rhs.intrinsic is self._INTRINSIC_TYPE):
376 # The intrinsic call is not the only thing on the rhs of
377 # the expression, so we need to deal with the additional
378 # computation.
379 rhs = orig_rhs.copy()
380 for child in rhs.walk(IntrinsicCall):
381 if child.intrinsic is self._INTRINSIC_TYPE:
382 child.replace_with(new_lhs.copy())
383 break
384 assignment = Assignment.create(orig_lhs.copy(), rhs)
385 outer_loop.parent.children.insert(
386 outer_loop.position+1, assignment)
