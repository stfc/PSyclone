51class CodeBlock(Statement, DataNode):
52 '''Node representing some generic Fortran code that PSyclone does not
53 attempt to manipulate. As such it is a leaf in the PSyIR and therefore
54 has no children.
55
56 :param fp2_nodes: the fparser2 parse-tree nodes representing the
57 Fortran code constituting the code block.
58 :type fp2_nodes: list[:py:class:`fparser.two.utils.Base`]
59 :param structure: argument indicating whether this code block is a
60 statement or an expression.
61 :type structure: :py:class:`psyclone.psyir.nodes.CodeBlock.Structure`
62 :param parent: the parent node of this code block in the PSyIR.
63 :type parent: :py:class:`psyclone.psyir.nodes.Node`
64 :param annotations: tags that provide additional information about
65 the node. The node should still be functionally correct when
66 ignoring these tags.
67 :type annotations: list[str | NoneType]
68
69 '''
70 #: Textual description of the node.
71 _children_valid_format = "<LeafNode>"
72 _text_name = "CodeBlock"
73 _colour = "red"
74 #: The annotations that are supported by this node.
75 #: psy-data-start - this node has replaced a PSyDataNode during the
76 #: lowering of the PSyIR to language level.
77 valid_annotations = ("psy-data-start")

91 def __init__(self, fp2_nodes, structure, parent=None, annotations=None):
92 super(CodeBlock, self).__init__(parent=parent, annotations=annotations)
93 # Store a list of the parser objects holding the code associated
94 # with this block. We make a copy of the contents of the list because
95 # the list itself is a temporary product of the process of converting
96 # from the fparser2 parse tree to the PSyIR.
97 self._fp2_nodes = fp2_nodes[:]
98 # Store references back into the fparser2 parse tree.
99 if fp2_nodes:
100 self.ast = self._fp2_nodes[0]
101 self.ast_end = self._fp2_nodes[-1]
102 else:
103 self.ast = None
104 self.ast_end = None
105 # Store the structure of the code block.
106 self._structure = structure

108 def __eq__(self, other):
109 '''
110 Checks whether two nodes are equal. Two CodeBlock nodes are equal
111 if they are the same type, their ast_nodes lists are equal (which
112 means the same instance) and have the same structure.
113
114 :param object other: the object to check equality to.
115
116 :returns: whether other is equal to self.
117 :rtype: bool
118 '''
119 is_eq = super().__eq__(other)
120 is_eq = is_eq and self.get_ast_nodes == other.get_ast_nodes
121 is_eq = is_eq and self.structure == other.structure
122
123 return is_eq

156 def get_symbol_names(self) -> List[str]:
157 '''
158 Analyses the fparser2 parse tree associated with this CodeBlock and
159 returns the names of all symbols accessed within it. Since, by
160 definition, we do not understand the contents of a CodeBlock, we do not
161 attempt to analyse how these symbols are accessed - they are all marked
162 as being READWRITE (this includes the names of any routines that might
163 be called).
164
165 Note that the names of any Fortran intrinsics are *not* included in the
166 result. If the original code has unwisely overridden a Fortran
167 intrinsic then fparser *may* incorrectly identify the use of such a
168 variable/routine as still being an intrinsic call and, as such, it will
169 be omitted from the names returned by this method.
170
171 TODO #2863 - these limitations (blanket use of READWRITE and the
172 ignoring of Fortran intrinsics) need to be re-visited.
173
174 :returns: the symbol names used inside the CodeBock.
175 '''
176 parse_tree = self.get_ast_nodes
177 result = []
178 for node in walk(parse_tree, Fortran2003.Name):
179 if isinstance(node.parent, Fortran2003.Else_If_Stmt):
180 # Need to make sure we include any Symbol in the conditional
181 # part but not a label (which would be the second child in the
182 # parse tree). We cannot simply do
183 # `node.parent.children.index(node)` because of fparser #174.
184 if (len(node.parent.children) == 1 or
185 node is node.parent.children[0]):
186 result.append(node.string)
187 elif not isinstance(node.parent,
188 (Fortran2003.Cycle_Stmt,
189 Fortran2003.End_Do_Stmt,
190 Fortran2003.Exit_Stmt,
191 Fortran2003.Else_Stmt,
192 Fortran2003.End_If_Stmt)):
193 # We don't want labels associated with loop or branch control.
194 result.append(node.string)
195 # Precision on literals requires special attention since they are just
196 # stored in the tree as str (fparser/#456).
197 for node in walk(parse_tree, (Fortran2003.Int_Literal_Constant,
198 Fortran2003.Real_Literal_Constant,
199 Fortran2003.Logical_Literal_Constant,
200 Fortran2003.Char_Literal_Constant)):
201 if node.items[1]:
202 result.append(node.items[1])
203 # Complex literals require even more special attention.
204 for node in walk(parse_tree, Fortran2003.Complex_Literal_Constant):
205 # A complex literal constant has a real part and an imaginary part.
206 # Each of these can have a kind.
207 for part in node.items:
208 if part.items[1]:
209 result.append(part.items[1])
210 return result

212 def reference_accesses(self, var_accesses: VariablesAccessInfo):
213 '''
214 Get all variable access information. Since this is a CodeBlock we
215 only know the names of symbols accessed within it but not how they
216 are accessed. Therefore we err on the side of caution and mark
217 them all as READWRITE, unfortunately, this will include the names of
218 any routines that are called.
219
220 TODO #2863 - it would be better to use AccessType.UNKNOWN here but
221 currently VariablesAccessInfo does not consider that type of access.
222
223 This method makes use of
224 :py:meth:`~psyclone.psyir.nodes.CodeBlock.get_symbol_names` and is
225 therefore subject to the same limitations as that method.
226
227 :param var_accesses: VariablesAccessInfo instance that stores the
228 information about variable accesses.
229
230 '''
231 for name in self.get_symbol_names():
232 var_accesses.add_access(Signature(name), AccessType.READWRITE,
233 self)
