55class HoistLocalArraysTrans(Transformation):
56 '''This transformation takes a Routine and promotes any local, 'automatic'
57 arrays to Container scope:
58
59 >>> from psyclone.psyir.backend.fortran import FortranWriter
60 >>> from psyclone.psyir.frontend.fortran import FortranReader
61 >>> from psyclone.psyir.nodes import Assignment
62 >>> from psyclone.psyir.transformations import HoistLocalArraysTrans
63 >>> code = ("module test_mod\\n"
64 ... "contains\\n"
65 ... " subroutine test_sub(n)\\n"
66 ... " integer :: i,j,n\\n"
67 ... " real :: a(n,n)\\n"
68 ... " real :: value = 1.0\\n"
69 ... " do i=1,n\\n"
70 ... " do j=1,n\\n"
71 ... " a(i,j) = value\\n"
72 ... " end do\\n"
73 ... " end do\\n"
74 ... " end subroutine test_sub\\n"
75 ... "end module test_mod\\n")
76 >>> psyir = FortranReader().psyir_from_source(code)
77 >>> hoist = HoistLocalArraysTrans()
78 >>> hoist.apply(psyir.walk(Routine)[0])
79 >>> print(FortranWriter()(psyir).lower())
80 module test_mod
81 implicit none
82 real, allocatable, dimension(:,:), private :: a
83 public
84 <BLANKLINE>
85 public :: test_sub
86 <BLANKLINE>
87 contains
88 subroutine test_sub(n)
89 integer :: n
90 integer :: i
91 integer :: j
92 real :: value = 1.0
93 <BLANKLINE>
94 if (.not.allocated(a) .or. ubound(a, 1) /= n .or. ubound(a, 2) /= n) \
95then
96 if (allocated(a)) then
97 deallocate(a)
98 end if
99 allocate(a(1 : n, 1 : n))
100 end if
101 do i = 1, n, 1
102 do j = 1, n, 1
103 a(i,j) = value
104 enddo
105 enddo
106 <BLANKLINE>
107 end subroutine test_sub
108 <BLANKLINE>
109 end module test_mod
110 <BLANKLINE>
111
112 By default, the target routine will be rejected if it is found to contain
113 an ACCRoutineDirective since this usually implies that the routine will be
114 launched in parallel on the OpenACC device. This check can be disabled
115 by setting 'allow_accroutine' to True in the `options` dictionary.
116
117 '''
118 def apply(self, node, options=None):
119 '''Applies the transformation to the supplied Routine node,
120 moving any local arrays up to Container scope and adding
121 a suitable allocation when they are first accessed. If there
122 are no local arrays or the supplied Routine is a program then
123 this method does nothing.
124
125 :param node: target PSyIR node.
126 :type node: :py:class:`psyclone.psyir.nodes.Routine`
127 :param options: a dictionary with options for transformations.
128 :param bool options["allow_accroutine"]: permit the target routine \
129 to contain an ACCRoutineDirective. These are forbidden by default \
130 because their presence usually indicates that the routine will be \
131 run in parallel on the OpenACC device.
132 :type options: Optional[Dict[str, Any]]
133
134 '''
135 self.validatevalidate(node, options)
136
137 if node.is_program:
138 # Cannot hoist arrays out of a program so do nothing.
139 return
140
141 container = node.ancestor(Container)
142
143 # Identify all arrays that are local to the target routine,
144 # do not explicitly use dynamic memory allocation and are not
145 # accessed within a CodeBlock.
146 automatic_arrays = self._get_local_arrays(node)
147
148 if not automatic_arrays:
149 # No automatic arrays found so nothing to do.
150 return
151
152 # Get the reversed tags map so that we can lookup the tag (if any)
153 # associated with the symbol being hoisted.
154 tags_dict = node.symbol_table.get_reverse_tags_dict()
155
156 for sym in automatic_arrays:
157 # Keep a copy of the original shape of the array.
158 orig_shape = sym.datatype.shape[:]
159 # Modify the *existing* symbol so that any references to it
160 # remain valid.
161 new_type = copy.copy(sym.datatype)
162 # pylint: disable=protected-access
163 new_type._shape = len(orig_shape)*[ArrayType.Extent.DEFERRED]
164 # pylint: enable=protected-access
165 sym.datatype = new_type
166 # Ensure that the promoted symbol is private to the container.
167 sym.visibility = Symbol.Visibility.PRIVATE
168 # We must allow for the situation where there's a clash with a
169 # symbol name already present at container scope. (The validate()
170 # method will already have checked for tag clashes.)
171 try:
172 container.symbol_table.add(sym, tag=tags_dict.get(sym))
173 except KeyError:
174 new_name = container.symbol_table.next_available_name(
175 sym.name, other_table=node.symbol_table)
176 node.symbol_table.rename_symbol(sym, new_name)
177 container.symbol_table.add(sym, tag=tags_dict.get(sym))
178
179 # Create the array reference that will be the argument to the
180 # new memory allocation statement.
181 dim_list = [Range.create(dim.lower.copy(), dim.upper.copy())
182 for dim in orig_shape]
183 aref = ArrayReference.create(sym, dim_list)
184
185 # Add a conditional expression to avoid repeating the allocation
186 # if its already done
187 allocated_expr = IntrinsicCall.create(
188 IntrinsicCall.Intrinsic.ALLOCATED,
189 [Reference(sym)])
190 cond_expr = UnaryOperation.create(
191 UnaryOperation.Operator.NOT, allocated_expr)
192
193 # Add runtime checks to verify that the boundaries haven't changed
194 # (we skip literals as we know they can't have changed)
195 check_added = False
196 for idx, dim in enumerate(orig_shape):
197 if not isinstance(dim.lower, Literal):
198 expr = BinaryOperation.create(
199 BinaryOperation.Operator.NE,
200 IntrinsicCall.create(
201 IntrinsicCall.Intrinsic.LBOUND,
202 [Reference(sym),
203 ("dim", Literal(str(idx+1), INTEGER_TYPE))]),
204 dim.lower.copy())
205 # We chain the new check to the already existing cond_expr
206 # which starts with the 'not allocated' condition added
207 # before this loop.
208 cond_expr = BinaryOperation.create(
209 BinaryOperation.Operator.OR,
210 cond_expr, expr)
211 check_added = True
212 if not isinstance(dim.upper, Literal):
213 expr = BinaryOperation.create(
214 BinaryOperation.Operator.NE,
215 IntrinsicCall.create(
216 IntrinsicCall.Intrinsic.UBOUND,
217 [Reference(sym),
218 ("dim", Literal(str(idx+1), INTEGER_TYPE))]),
219 dim.upper.copy())
220 # We chain the new check to the already existing cond_expr
221 # which starts with the 'not allocated' condition added
222 # before this loop.
223 cond_expr = BinaryOperation.create(
224 BinaryOperation.Operator.OR,
225 cond_expr, expr)
226 check_added = True
227
228 body = []
229 if check_added:
230 body.append(
231 IfBlock.create(
232 allocated_expr.copy(),
233 [IntrinsicCall.create(
234 IntrinsicCall.Intrinsic.DEALLOCATE,
235 [Reference(sym)])]))
236 body.append(
237 IntrinsicCall.create(IntrinsicCall.Intrinsic.ALLOCATE,
238 [aref]))
239 # Insert the conditional allocation at the start of the supplied
240 # routine.
241 node.children.insert(0, IfBlock.create(cond_expr, body))
242
243 # Finally, remove the hoisted symbols (and any associated tags)
244 # from the routine scope.
245 for sym in automatic_arrays:
246 # TODO #898: Currently the SymbolTable.remove() method does not
247 # support DataSymbols.
248 # pylint: disable=protected-access
249 del node.symbol_table._symbols[sym.name]
250 tag = tags_dict.get(sym)
251 if tag:
252 del node.symbol_table._tags[tag]

254 @staticmethod
255 def _get_local_arrays(node):
256 '''
257 Identify all arrays that are local to the target routine, all their
258 bounds/kind/type symbols are also local, do not represent a function's
259 return value, are not constant and do not explicitly use
260 dynamic memory allocation. Also excludes any such arrays that are
261 accessed within CodeBlocks or RESHAPE intrinsics.
262
263 :param node: target PSyIR node.
264 :type node: subclass of :py:class:`psyclone.psyir.nodes.Routine`
265
266 :returns: symbols representing routine-local arrays.
267 :rtype: list[:py:class:`psyclone.psyir.symbols.DataSymbol`]
268
269 '''
270 local_arrays = {}
271 for sym in node.symbol_table.automatic_datasymbols:
272 # Check that the array is not the functions return symbol, or
273 # a constant or has other references in its type declaration.
274 if (sym is node.return_symbol or not sym.is_array or
275 sym.is_constant):
276 continue
277 if isinstance(sym.datatype.intrinsic, DataTypeSymbol):
278 continue
279 if isinstance(sym.datatype.precision, DataSymbol):
280 continue
281 # Check whether all of the bounds of the array are defined - an
282 # allocatable array will have array dimensions of
283 # ArrayType.Extent.DEFERRED
284 if all(isinstance(dim, ArrayType.ArrayBounds)
285 for dim in sym.shape):
286 local_arrays[sym.name] = sym
287
288 # Exclude any arrays that are accessed within a CodeBlock (as they
289 # may get renamed as part of the transformation).
290 cblocks = node.walk(CodeBlock)
291 for cblock in cblocks:
292 cblock_names = set(cblock.get_symbol_names())
293 array_names = set(local_arrays.keys())
294 names_in_cblock = cblock_names.intersection(array_names)
295 # TODO #11 - log the fact that we can't hoist the arrays
296 # listed in 'names_in_cblock'.
297 for name in names_in_cblock:
298 del local_arrays[name]
299
300 for intrinsic in node.walk(IntrinsicCall):
301 # Exclude arrays that are used in a RESHAPE expression
302 if intrinsic.intrinsic == IntrinsicCall.Intrinsic.RESHAPE:
303 for ref in intrinsic.walk(Reference):
304 if ref.symbol.name in local_arrays:
305 del local_arrays[ref.symbol.name]
306
307 return list(local_arrays.values())

309 def validate(self, node, options=None):
310 '''Checks that the supplied node is a valid target for a hoist-
311 local-arrays transformation. It must be a Routine that is within
312 a Container (that is not a FileContainer).
313
314 :param node: target PSyIR node.
315 :type node: subclass of :py:class:`psyclone.psyir.nodes.Routine`
316 :param options: any options for the transformation.
317 :type options: Optional[Dict[str, Any]]
318
319 :raises TransformationError: if the supplied node is not a Routine.
320 :raises TransformationError: if the Routine is not within a Container \
321 (that is not a FileContainer).
322 :raises TransformationError: if the routine contains an OpenACC \
323 routine directive and options['allow_accroutine'] is not True.
324 :raises TransformationError: if any symbols corresponding to local \
325 arrays have a tag that already exists in the table of the parent \
326 Container.
327
328 '''
329 super().validate(node, options=options)
330
331 # The node should be a Routine.
332 if not isinstance(node, Routine):
333 raise TransformationError(
334 f"The target of the HoistLocalArraysTrans transformation "
335 f"should be a Routine but found '{type(node).__name__}'.")
336
337 if node.is_program:
338 # We silently ignore routines that are programs - this
339 # transformation will do nothing.
340 return
341
342 # The Routine must be within a Container (otherwise we have nowhere
343 # to hoist any array declarations to).
344 container = node.ancestor(Container)
345 if not container:
346 raise TransformationError(
347 f"The supplied routine '{node.name}' should "
348 f"be within a Container but none was found.")
349 if isinstance(container, FileContainer):
350 raise TransformationError(
351 f"The supplied routine '{node.name}' should be within a "
352 f"Container but the enclosing container is a "
353 f"FileContainer (named '{container.name}').")
354
355 if not (options and options.get("allow_accroutine")):
356 if node.walk(ACCRoutineDirective):
357 raise TransformationError(
358 f"The supplied routine '{node.name}' contains an ACC "
359 f"Routine directive which implies it will be run in "
360 f"parallel. Hoisting local arrays to global scope may "
361 f"create race conditions in this case. If this routine "
362 f"will be run in serial on the device then this check can "
363 f"be disabled by setting 'allow_accroutine' to "
364 f"True in the transformation options.")
365
366 # Check for clashing tags in the container scope.
367 auto_arrays = self._get_local_arrays(node)
368 tags_dict = node.symbol_table.get_reverse_tags_dict()
369 cont_tags_dict = container.symbol_table.tags_dict
370 for sym in auto_arrays:
371 tag = tags_dict.get(sym)
372 if tag in cont_tags_dict:
373 raise TransformationError(
374 f"The supplied routine '{node.name}' contains a local "
375 f"array '{sym.name}' with tag '{tag}' but this tag is "
376 f"also present in the symbol table of the parent "
377 f"Container (associated with variable "
378 f"'{cont_tags_dict[tag].name}').")
379
380 def __str__(self):
381 return "Hoist all local, automatic arrays to container scope."
