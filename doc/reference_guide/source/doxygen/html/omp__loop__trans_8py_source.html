48MAP_STR_TO_LOOP_DIRECTIVES = {

59class OMPLoopTrans(ParallelLoopTrans):
60 '''
61 Adds an OpenMP directive to parallelise this loop. It can insert different
62 directives such as "omp do/for", "omp parallel do/for", "omp teams
63 distribute parallel do/for" or "omp loop" depending on the provided
64 parameters.
65 The OpenMP schedule to use can also be specified, but this will be ignored
66 in case of the "omp loop" (as the 'schedule' clause is not valid for this
67 specific directive). The configuration-defined 'reprod' parameter
68 also specifies whether a manual reproducible reproduction is to be used.
69 Note, reproducible in this case means obtaining the same results with the
70 same number of OpenMP threads, not for different numbers of OpenMP threads.
71
72 :param str omp_schedule: the OpenMP schedule to use. Defaults to 'auto'.
73 :param str omp_directive: choose which OpenMP loop directive to use. \
74 Defaults to "omp do"
75
76 For example:
77
78 >>> from psyclone.psyir.frontend.fortran import FortranReader
79 >>> from psyclone.psyir.backend.fortran import FortranWriter
80 >>> from psyclone.psyir.nodes import Loop
81 >>> from psyclone.transformations import OMPLoopTrans, OMPParallelTrans
82 >>>
83 >>> psyir = FortranReader().psyir_from_source("""
84 ... subroutine my_subroutine()
85 ... integer, dimension(10, 10) :: A
86 ... integer :: i
87 ... integer :: j
88 ... do i = 1, 10
89 ... do j = 1, 10
90 ... A(i, j) = 0
91 ... end do
92 ... end do
93 ... end subroutine
94 ... """)
95 >>> loop = psyir.walk(Loop)[0]
96 >>> omplooptrans1 = OMPLoopTrans(omp_schedule="dynamic",
97 ... omp_directive="paralleldo")
98 >>> omplooptrans1.apply(loop)
99 >>> print(FortranWriter()(psyir))
100 subroutine my_subroutine()
101 integer, dimension(10,10) :: a
102 integer :: i
103 integer :: j
104 <BLANKLINE>
105 !$omp parallel do default(shared), private(i,j), schedule(dynamic)
106 do i = 1, 10, 1
107 do j = 1, 10, 1
108 a(i,j) = 0
109 enddo
110 enddo
111 !$omp end parallel do
112 <BLANKLINE>
113 end subroutine my_subroutine
114 <BLANKLINE>
115
116 '''

167 @omp_schedule.setter
168 def omp_schedule(self, value):
169 '''
170 :param str value: Sets the OpenMP schedule value that will be \
171 specified by this transformation, unless adding an OMP Loop \
172 directive (in which case it is not applicable).
173
174 :raises TypeError: if the provided value is not a string.
175 :raises ValueError: if the provided string is not a valid OpenMP \
176 schedule format.
177 '''
178
179 if not isinstance(value, str):
180 raise TypeError(
181 f"The OMPLoopTrans.omp_schedule property must be a 'str'"
182 f" but found a '{type(value).__name__}'.")
183
184 # Some schedules have an optional chunk size following a ','
185 value_parts = value.split(',')
186 if value_parts[0].lower() not in OMPScheduleClause.VALID_OMP_SCHEDULES:
187 raise ValueError(
188 f"Valid OpenMP schedules are "
189 f"{OMPScheduleClause.VALID_OMP_SCHEDULES} but got "
190 f"'{value_parts[0]}'.")
191
192 if len(value_parts) > 1:
193 if value_parts[0] == "auto":
194 raise ValueError("Cannot specify a chunk size when using an "
195 "OpenMP schedule of 'auto'.")
196 try:
197 int(value_parts[1].strip())
198 except ValueError as err:
199 raise ValueError(f"Supplied OpenMP schedule '{value}' has an "
200 f"invalid chunk-size.") from err
201
202 self._omp_schedule = value

204 def _directive(self, children, collapse=None):
205 ''' Creates the type of directive needed for this sub-class of
206 transformation.
207
208 :param children: list of Nodes that will be the children of \
209 the created directive.
210 :type children: List[:py:class:`psyclone.psyir.nodes.Node`]
211 :param int collapse: number of nested loops to collapse or None if \
212 no collapse attribute is required.
213
214 :returns: the new node representing the directive in the AST
215 :rtype: :py:class:`psyclone.psyir.nodes.OMPDoDirective` | \
216 :py:class:`psyclone.psyir.nodes.OMPParallelDoDirective` | \
217 :py:class:`psyclone.psyir.nodes. \
218 OMPTeamsDistributeParallelDoDirective` | \
219 :py:class:`psyclone.psyir.nodes.OMPLoopDirective`
220 '''
221 node = MAP_STR_TO_LOOP_DIRECTIVES[self._omp_directive](
222 children=children,
223 collapse=collapse)
224 # OMP loop does not support 'schedule' or 'reprod' attributes, so we do
225 # not attempt to set these properties for this specific directive
226 if self._omp_directive != "loop":
227 node.omp_schedule = self._omp_schedule
228 node.reprod = self._reprod
229 return node

231 def apply(self, node, options=None):
232 '''Apply the OMPLoopTrans transformation to the specified PSyIR Loop.
233
234 :param node: the supplied node to which we will apply the \
235 OMPLoopTrans transformation
236 :type node: :py:class:`psyclone.psyir.nodes.Node`
237 :param options: a dictionary with options for transformations\
238 and validation.
239 :type options: Optional[Dict[str, Any]]
240 :param bool options["reprod"]:
241 indicating whether reproducible reductions should be used. \
242 By default the value from the config file will be used.
243
244 '''
245 if not options:
246 options = {}
247 self._reprod = options.get("reprod",
248 Config.get().reproducible_reductions)
249
250 if self._reprod:
251 # When reprod is True, the variables th_idx and nthreads are
252 # expected to be declared in the scope.
253 root = node.ancestor(Routine)
254
255 symtab = root.symbol_table
256 try:
257 symtab.lookup_with_tag("omp_thread_index")
258 except KeyError:
259 symtab.new_symbol(
260 "th_idx", tag="omp_thread_index",
261 symbol_type=DataSymbol, datatype=INTEGER_TYPE)
262 try:
263 symtab.lookup_with_tag("omp_num_threads")
264 except KeyError:
265 symtab.new_symbol(
266 "nthreads", tag="omp_num_threads",
267 symbol_type=DataSymbol, datatype=INTEGER_TYPE)
268
269 super().apply(node, options)
