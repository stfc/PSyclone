53class OMPTaskwaitTrans(Transformation):
54 '''
55 Adds zero or more OpenMP Taskwait directives to an OMP parallel region.
56 This transformation will add directives to satisfy dependencies between
57 Taskloop directives without an associated taskgroup (i.e. no nogroup
58 clause). It also tries to minimise the number added to maximise available
59 parallelism.
60
61 For example:
62
63 >>> from pysclone.parse.algorithm import parse
64 >>> from psyclone.psyGen import PSyFactory
65 >>> api = "gocean"
66 >>> filename = "nemolite2d_alg.f90"
67 >>> ast, invokeInfo = parse(filename, api=api, invoke_name="invoke")
68 >>> psy = PSyFactory(api).create(invokeInfo)
69 >>>
70 >>> from psyclone.transformations import OMPParallelTrans, OMPSingleTrans
71 >>> from psyclone.transformations import OMPTaskloopTrans
72 >>> from psyclone.psyir.transformations import OMPTaskwaitTrans
73 >>> singletrans = OMPSingleTrans()
74 >>> paralleltrans = OMPParallelTrans()
75 >>> tasklooptrans = OMPTaskloopTrans()
76 >>> taskwaittrans = OMPTaskwaitTrans()
77 >>>
78 >>> schedule = psy.invokes.get('invoke_0').schedule
79 >>> print(schedule.view())
80 >>>
81 >>> # Apply the OpenMP Taskloop transformation to *every* loop
82 >>> # in the schedule.
83 >>> # This ignores loop dependencies. These are handled by the
84 >>> # taskwait transformation.
85 >>> for child in schedule.children:
86 >>> tasklooptrans.apply(child, nogroup = true)
87 >>> # Enclose all of these loops within a single OpenMP
88 >>> # SINGLE region
89 >>> singletrans.apply(schedule.children)
90 >>> # Enclose all of these loops within a single OpenMP
91 >>> # PARALLEL region
92 >>> paralleltrans.apply(schedule.children)
93 >>> taskwaittrans.apply(schedule.children)
94 >>> print(schedule.view())
95
96 '''
97 def __str__(self):
98 rval = ("Adds 'OpenMP TASKWAIT' directives to an OpenMP parallel "
99 "region to satisfy 'OpenMP TASKLOOP' dependencies")
100 return rval

102 def validate(self, node, options=None):
103 '''
104 Validity checks for input arguments.
105
106 :param node: the OMPParallelDirective node to validate.
107 :type node: :py:class:`psyclone.psyir.nodes.OMPParallelDirective`
108 :param options: a dictionary with options for transformations.
109 :type options: Optional[Dict[str, Any]]
110 :param bool options["fail_on_no_taskloop"]:
111 indicating whether this should throw an error if no \
112 OMPTaskloop nodes are found in this tree. This can be \
113 safely disabled as if there are no Taskloop nodes the \
114 result of this transformation is valid OpenMP code. Default \
115 is True.
116
117 :raises TransformationError: If the supplied node is not an \
118 OMPParallelDirective
119 :raises TransformationError: If there are no OMPTaskloopDirective \
120 nodes in this tree.
121 :raises TransformationError: If taskloop dependencies can't be \
122 satisfied due to dependencies across \
123 barrierless OpenMP Serial Regions.
124 '''
125 fail_on_no_taskloop = True
126 if options is not None:
127 fail_on_no_taskloop = options.get("fail_on_no_taskloop", True)
128 # Check the supplied node is an OMPParallelDirective
129 if not isinstance(node, nodes.OMPParallelDirective):
130 raise TransformationError(f"OMPTaskwaitTrans was supplied a "
131 f"'{node.__class__.__name__}'"
132 f" node, but expected an "
133 f"OMPParallelDirective")
134
135 # Find all the taskloops
136 taskloops = node.walk(OMPTaskloopDirective)
137
138 # Walk the tree to find any OMPTaskloopTrans
139 if fail_on_no_taskloop and taskloops == []:
140 raise TransformationError("OMPTaskwaitTrans was supplied an "
141 "OMPParallelDirective that does not "
142 "contain any OMPTaskloopDirectives")
143
144 # Check that all of the dependencies are satisfiable
145 for taskloop in taskloops:
146 # Find the first RaW or WaR dependency for this taskloop.
147 forward_dep = OMPTaskwaitTrans.get_forward_dependence(taskloop,
148 node)
149 if forward_dep is None:
150 continue
151 # Check if the taskloop and its forward dependence are in the
152 # same serial region
153 if (taskloop.ancestor(OMPSerialDirective) is not
154 forward_dep.ancestor(OMPSerialDirective)):
155 # They're not in the same serial region. Check if our
156 # taskloop is in a waiting Single Directive
157 ancestor = taskloop.ancestor(OMPSerialDirective)
158 valid = isinstance(ancestor, nodes.OMPSingleDirective)
159 if valid:
160 valid = not ancestor.nowait
161 # If not valid, then we're in a Master Directive or a
162 # Single Directive with nowait. In this case we can't
163 # safely guarantee our forward dependency so throw an error
164 if not valid:
165 # pylint: disable=cell-var-from-loop
166 # Since "Backslashes may not appear inside the expression
167 # portions of f-strings" via PEP 498, use chr(10) for '\n'
168 raise TransformationError(LazyString(
169 lambda: f"Couldn't satisfy the dependencies due to "
170 f"taskloop dependencies across barrierless "
171 f"OMP serial regions. Dependency is from\n"
172 f"{taskloop.debug_string().rstrip(chr(10))}"
173 f"\nto\n"
174 f"{forward_dep.debug_string().rstrip(chr(10))}"
175 ))

177 @staticmethod
178 def get_forward_dependence(taskloop, root):
179 '''
180 Returns the next forward dependence for a taskloop using the
181 dependence-analysis functionality provided by
182 psyclone.psyir.tools.dependency_tools.
183 Forward dependencies can be of the following types:
184 Loop
185 OMPDoDirective
186 OMPTaskloopDirective
187 OMPTaskwaitDirective (If in same OMPSingle and that single has
188 nowait=False)
189 OMPSingleDirective (If ancestor OMPSingle has nowait=False)
190
191 Loop, OMPDoDirective, OMPTaskloopDirective types are returned when
192 a following directive is found which has a RaW, WaR or WaW dependency
193 to taskloop.
194
195 An OMPTaskwaitDirective type is returned when a following directive
196 is found inside the same parent OMPSingleDirective which has no
197 nowait clause applied.
198
199 An OMPSingleDirective type is returned when the first dependency is
200 within a different OMPSerialDirective, and the ancestor of taskloop
201 is an OMPSingleDirective with no nowait clause.
202
203 The forward dependency is never a child of taskloop, and must have
204 abs_position > taskloop.abs_position
205
206 :param taskloop: the taskloop node for which to find the \
207 forward_dependence.
208 :type taskloop: :py:class:`psyclone.psyir.nodes.OMPTaskloopDirective`
209 :param root: the tree in which to search for the forward_dependence.
210 :type root: :py:class:`psyclone.psyir.nodes.OMPParallelDirective`
211
212 :returns: the forward_dependence of taskloop.
213 :rtype: :py:class:`psyclone.f2pygen.Node`
214
215 '''
216 # Check supplied the correct type for root
217 if not isinstance(root, OMPParallelDirective):
218 raise TransformationError(f"Expected the root of the tree in which"
219 f" to look for a forward dependence to "
220 f"be an instance of OMPParallelDirective"
221 f", but was supplied an instance of "
222 f"'{type(root).__name__}'")
223 # We only look for specific types
224 node_list = root.walk((Loop, OMPDoDirective, OMPTaskloopDirective,
225 OMPTaskwaitDirective))
226 # Find the taskloop's variable access info. We need to skip over the
227 # Loop variable writes from the Loop, so we skip the Loop children.
228 taskloop_vars = VariablesAccessInfo()
229 for child in taskloop.walk(nodes.Node):
230 if child is not taskloop and not isinstance(child,
231 (Schedule, Loop)):
232 taskloop_vars.merge(VariablesAccessInfo(child))
233 taskloop_signatures = taskloop_vars.all_signatures
234 # Find our parent serial region if it has a barrier
235 parent_single = taskloop.ancestor(OMPSingleDirective)
236 # Cache the parent single for use in later if statements
237 cached_parent_single = parent_single
238 # If the parent single region has a nowait clause it can never
239 # act as the dependency for this taskloop, so we set it to None.
240 # The same behaviour occurs implicitly if the parent
241 # OMPSerialDirective is an OMPMasterDirective. Without a blocking
242 # parent region we can never guarantee synchronicity if dependencies
243 # are outside of the parent region.
244 if parent_single is not None and parent_single.nowait:
245 parent_single = None
246 # Find our parent parallel region
247 parent_parallel = taskloop.ancestor(OMPParallelDirective)
248 # Raise an error if there is no parent_parallel region
249 if parent_parallel is None:
250 # Since "Backslashes may not appear inside the expression
251 # portions of f-strings" via PEP 498, use chr(10) for '\n'
252 raise InternalError(LazyString(
253 lambda: f"No parent parallel directive was found for the "
254 f"taskloop region: "
255 f"{taskloop.debug_string().rstrip(chr(10))}"))
256
257 for node in node_list:
258 if node.abs_position <= taskloop.abs_position:
259 continue
260 # Ignore any children of the taskloop directive
261 anc = node.ancestor(OMPTaskloopDirective)
262 if anc is taskloop:
263 continue
264 node_vars = None
265 if (isinstance(node, OMPTaskwaitDirective) and
266 (cached_parent_single is
267 node.ancestor(OMPSingleDirective)) and
268 cached_parent_single is not None and
269 cached_parent_single.nowait is False):
270 # If we find a taskwait barrier inside the same
271 # OMPSingleDirective and that OMPSingleDirective has a no
272 # nowait clause, then it acts as a barrier for dependencies
273 # as well, so we return it
274 return node
275 if not isinstance(node, OMPTaskwaitDirective):
276 # For all our other node types we calculate their own
277 # variable accesses
278 node_vars = VariablesAccessInfo()
279 for child in node.walk(nodes.Node):
280 if child is not node and not isinstance(child,
281 (Schedule, Loop)):
282 refs = VariablesAccessInfo(child)
283 if refs is not None:
284 node_vars.merge(refs)
285 node_signatures = node_vars.all_signatures
286 # Once we have the node's variable accesses, check for collisions
287 for sig1 in taskloop_signatures:
288 # If this signature is not in the node signatures then continue
289 if sig1 not in node_signatures:
290 continue
291 access1 = taskloop_vars[sig1]
292 access2 = node_vars[sig1]
293 # If both are only read we can ignore this signature
294 # Otherwise, one of them writes so return this node as
295 # we have a WaW, WaR or RaW dependency
296 if access1.is_written() or access2.is_written():
297 # If we have a different parent serial node, and
298 # parent_single is not None then our parent_single is our
299 # dependency, otherwise this node is the dependency
300 if (taskloop.ancestor(OMPSerialDirective) is not
301 node.ancestor(OMPSerialDirective) and
302 parent_single is not None):
303 return parent_single
304 return node
305 # If we found no dependencies, then return that!
306 return None

308 @staticmethod
309 def _eliminate_unneeded_dependencies(taskloop_positions,
310 dependence_positions,
311 dependence_nodes):
312 '''
313 Eliminates unneeded dependencies from a set of taskloop positions,
314 dependence_positions and dependence_nodes for a region of code.
315
316 :param taskloop_positions: positions of the taskloops.
317 :type taskloop_positions: list of int
318 :param dependence_positions: positions of the taskloops' dependencies.
319 :type dependence_positions: list of int
320 :param dependence_nodes: the nodes respresenting the forward \
321 dependency of each taskloop node.
322 :type dependence_nodes: list of :py:class:`psyclone.psyir.nodes.Node`
323
324 :returns: the updated dependence_positions and dependence_nodes arrays
325 :rval: 2-tuple of (list of int, list of
326 :py:class:`psyclone.psyir.nodes.Node`)
327
328 '''
329 # We loop over each dependence and perform the same operation:
330 # For each dependence we loop over all other
331 # taskloops whose position is after this taskloop, but before
332 # this taskloop's forward dependency. If that dependence is
333 # fulfilled by this dependence (i.e.
334 # dependence_position[this] < dependence_position[that]) then
335 # we remove that dependence. If this dependence is fulfilled
336 # by that dependence (i.e. dependence_position[that] <
337 # dependence_position[this]) then we remove this dependence.
338 # This assumes that taskloop_positions is in ascending order,
339 # which I believe to be true by construction.
340 for i, dep_pos in enumerate(dependence_positions):
341 # Corresponding taskloop has no dependency or was satisfied
342 # by another dependency being resolved.
343 if dep_pos is None:
344 continue
345 # Grab the position of the next dependence
346 next_dependence = dep_pos
347 # Loop over the other taskloops
348 for j in range(i+1, len(dependence_positions)):
349 # If this taskloop happens after the next_dependence
350 # then we can just stop.
351 if taskloop_positions[j] >= next_dependence:
352 break
353 # If the jth taskloop has no dependency then continue
354 if dependence_positions[j] is None:
355 continue
356 # Check if next_dependence will satisfy the jth
357 # taskloops dependency
358 if next_dependence <= dependence_positions[j]:
359 dependence_positions[j] = None
360 dependence_nodes[j] = None
361 continue
362 # Check if the jth taskloop's dependence will satisfy
363 # the next_dependence
364 if dependence_positions[j] < next_dependence:
365 dependence_positions[i] = None
366 dependence_nodes[i] = None
367 # If it does then we can move to the next taskloop
368 break
369 return dependence_positions, dependence_nodes

371 def apply(self, node, options=None):
372 '''
373 Apply an OMPTaskwait Transformation to the supplied node
374 (which must be an OMPParallelDirective). In the generated code this
375 corresponds to adding zero or more OMPTaskwaitDirectives as
376 appropriate:
377
378 .. code-block:: fortran
379
380 !$OMP PARALLEL
381 ...
382 !$OMP TASKWAIT
383 ...
384 !$OMP TASKWAIT
385 ...
386 !$OMP END PARALLEL
387
388 :param node: the node to which to apply the transformation.
389 :type node: :py:class:`psyclone.psyir.nodes.OMPParallelDirective`
390 :param options: a dictionary with options for transformations\
391 and validation.
392 :type options: Optional[Dict[str, Any]]
393 :param bool options["fail_on_no_taskloop"]:
394 indicating whether this should throw an error if no \
395 OMPTaskloop nodes are found in this tree. This can be \
396 safely disabled as if there are no Taskloop nodes the \
397 result of this transformation is valid OpenMP code. Default \
398 is True
399
400 '''
401 self.validatevalidate(node, options=options)
402
403 # Find all the OpenMP Single & Master regions
404 task_regions = node.walk(OMPSerialDirective)
405
406 # Loop over the task regions
407 for task_region in task_regions:
408 create_endtaskwait = False
409 endwaits = []
410 # Find all of the taskloops
411 taskloops = task_region.walk(OMPTaskloopDirective)
412 # Get the positions of all of the taskloops
413 taskloop_positions = [-1] * len(taskloops)
414 # Get the forward_dependence position of all of the taskloops
415 dependence_position = [None] * len(taskloops)
416 dependence_node = [None] * len(taskloops)
417 for i, taskloop in enumerate(taskloops):
418 taskloop_positions[i] = taskloop.abs_position
419 # Only set forward_dep for taskloops with nogroup set
420 forward_dep = None
421 if taskloop.nogroup:
422 forward_dep = OMPTaskwaitTrans.get_forward_dependence(
423 taskloop, node)
424 # If the forward_dependence is one of our parents then we
425 # should ignore it
426 if (forward_dep is not None and
427 forward_dep.abs_position < taskloop.abs_position):
428 # If we're in a blocking single region and the
429 # dependency for any of its tasks points to its
430 # parent single region, then its "real" next dependency
431 # is outside of the single region. To ensure we
432 # synchronize before this dependency, we must have a
433 # task synchronization construct before the spawning
434 # thread leaves the single region. It is possible that
435 # this dependency could be handled by another
436 # intermediary taskwait, however I can't work out a
437 # good way to detect that now (since there is no
438 # "end of single region" abs_position to map to).
439 # I could store a list of all the taskloops that
440 # require this final taskwait, and walk from each one
441 # to see if a taskwait has been placed to satisfy their
442 # dependency, but that does not seem elegant somehow.
443 if forward_dep is task_region:
444 endwaits.append(taskloop)
445 create_endtaskwait = True
446 continue
447 if forward_dep is None:
448 continue
449 # Check if the taskloop and its forward dependence are in the
450 # same serial region
451 if (taskloops[i].ancestor(OMPSerialDirective) is
452 forward_dep.ancestor(OMPSerialDirective)):
453 # We're in the same OMPSerialDirective so store the
454 # position of the forward dependency
455 dependence_position[i] = forward_dep.abs_position
456 dependence_node[i] = forward_dep
457 # Forward dependency positions are now computed for this region.
458 dependence_position, dependence_node = \
459 OMPTaskwaitTrans._eliminate_unneeded_dependencies(
460 taskloop_positions, dependence_position,
461 dependence_node)
462 # dependence_position now contains only the required dependencies
463 # to satisfy the full superset of dependencies. We can loop over
464 # these by index, and if dependence_position[i] is not None then
465 # go to its forward dependency, and insert a TaskwaitDirective
466 # immediately before.
467 for i, dep_pos in enumerate(dependence_position):
468 if dep_pos is not None:
469 forward_dep = dependence_node[i]
470 fdep_parent = forward_dep.parent
471 # Find the position of the forward_dep in its parent's
472 # children list
473 loc = forward_dep.position
474 # We've found the position, so we now insert an
475 # OMPTaskwaitDirective in that location instead
476 fdep_parent.addchild(OMPTaskwaitDirective(), loc)
477 if create_endtaskwait:
478 # For each taskloop that needed a taskwait at the end
479 for taskloop in endwaits:
480 taskloop_pos = taskloop.abs_position
481 # Find all the taskwaits in the task_region
482 node_list = task_region.walk(OMPTaskwaitDirective)
483 # If every taskwait appears before this taskloop then
484 # add a taskwait at the end of the serial region and
485 # stop.
486 if all(node1.abs_position < taskloop_pos
487 for node1 in node_list):
488 task_region.dir_body.addchild(OMPTaskwaitDirective())
489 break
