67def index_of_object(alist, obj):
68 '''Effectively implements list.index(obj) but returns the index of
69 the first item in the list that *is* the supplied object (rather than
70 comparing values) '''
71 for idx, body in enumerate(alist):
72 if body is obj:
73 return idx
74 raise Exception(f"Object {obj} not found in list")

81class Directive(Comment):
82 '''
83 Base class for directives so we can reason about them when walking
84 the tree. Sub-classes the fparser1 Comment class.
85
86 :param root: the parent node in the AST to which we are adding the \
87 directive
88 :type root: subclass of :py:class:`fparser.common.base_classes.Statement`
89 :param line: the fparser object which we will manipulate to create \
90 the desired directive.
91 :type line: :py:class:`fparser.common.readfortran.Comment`
92 :param str position: e.g. 'begin' or 'end' (language specific)
93 :param str dir_type: the type of directive that this is (e.g. \
94 'parallel do')
95 '''
96 def __init__(self, root, line, position, dir_type):
97 if dir_type not in self._types:
98 raise RuntimeError(f"Error, unrecognised directive type "
99 f"'{dir_type}'. Should be one of {self._types}")
100 if position not in self._positions:
101 raise RuntimeError(f"Error, unrecognised position '{position}'. "
102 f"Should be one of {self._positions}")
103 self._my_type = dir_type
104 self._position = position
105 Comment.__init__(self, root, line)
106
107 @property
108 def type(self):
109 '''
110 :returns: the type of this Directive.
111 :rtype: str
112 '''
113 return self._my_type
114
115 @property
116 def position(self):
117 '''
118 :returns: the position of this Directive ('begin' or 'end').
119 :rtype: str
120 '''
121 return self._position

175 ''' The base class for all classes that are responsible for generating
176 distinct code elements (modules, subroutines, do loops etc.) '''

197 def add(self, new_object, position=None):
198 '''Adds a new object to the tree. The actual position is determined by
199 the position argument. Note, there are two trees, the first is
200 the f2pygen object tree, the other is the f2py generated code
201 tree. These are similar but different. At the moment we
202 specify where to add things in terms of the f2pygen tree
203 (which is a higher level api) but we also insert into the f2py
204 tree at exactly the same location which needs to be sorted out
205 at some point.
206
207 '''
208
209 # By default the position is 'append'. We set it up this way for
210 # safety because in python, default arguments are instantiated
211 # as objects at the time of definition. If this object is
212 # subsequently modified then the value of the default argument
213 # is modified for subsequent calls of this routine.
214 if position is None:
215 position = ["append"]
216
217 if position[0] == "auto":
218 raise Exception("Error: BaseGen:add: auto option must be "
219 "implemented by the sub class!")
220 options = ["append", "first", "after", "before", "insert",
221 "before_index", "after_index"]
222 if position[0] not in options:
223 raise Exception(f"Error: BaseGen:add: supported positions are "
224 f"{options} but found {position[0]}")
225 if position[0] == "append":
226 self.root.content.append(new_object.root)
227 elif position[0] == "first":
228 self.root.content.insert(0, new_object.root)
229 elif position[0] == "insert":
230 index = position[1]
231 self.root.content.insert(index, new_object.root)
232 elif position[0] == "after":
233 idx = index_of_object(self.root.content, position[1])
234 self.root.content.insert(idx+1, new_object.root)
235 elif position[0] == "after_index":
236 self.root.content.insert(position[1]+1, new_object.root)
237 elif position[0] == "before_index":
238 self.root.content.insert(position[1], new_object.root)
239 elif position[0] == "before":
240 try:
241 idx = index_of_object(self.root.content, position[1])
242 except Exception as err:
243 print(str(err))
244 raise RuntimeError(
245 "Failed to find supplied object in existing content - "
246 "is it a child of the parent?")
247 self.root.content.insert(idx, new_object.root)
248 else:
249 raise Exception("Error: BaseGen:add: internal error, should "
250 "not get to here")
251 self.children.append(new_object)

253 def previous_loop(self):
254 ''' Returns the *last* occurrence of a loop in the list of
255 siblings of this node '''
256 from fparser.one.block_statements import Do
257 for sibling in reversed(self.root.content):
258 if isinstance(sibling, Do):
259 return sibling
260 raise RuntimeError("Error, no loop found - there is no previous loop")

263 '''Returns the *last* occurrence of a Declaration in the list of

266 '''

268 for sibling in reversed(self.root.content):
269 if isinstance(sibling, TypeDeclarationStatement):

272 raise RuntimeError("Error, no variable declarations found")

274 def start_parent_loop(self, debug=False):
275 ''' Searches for the outer-most loop containing this object. Returns
276 the index of that line in the content of the parent. '''
277 from fparser.one.block_statements import Do
278 if debug:
279 print("Entered before_parent_loop")
280 print(f"The type of the current node is {type(self.root)}")
281 print(("If the current node is a Do loop then move up to the "
282 "top of the do loop nest"))
283
284 # First off, check that we do actually have an enclosing Do loop
285 current = self.root
286 while not isinstance(current, Do) and getattr(current, 'parent', None):
287 current = current.parent
288 if not isinstance(current, Do):
289 raise RuntimeError("This node has no enclosing Do loop")
290
291 current = self.root
292 local_current = self
293 while isinstance(current.parent, Do):
294 if debug:
295 print("Parent is a do loop so moving to the parent")
296 current = current.parent
297 local_current = local_current.parent
298 if debug:
299 print("The type of the current node is now " + str(type(current)))
300 print("The type of parent is " + str(type(current.parent)))
301 print("Finding the loops position in its parent ...")
302 index = current.parent.content.index(current)
303 if debug:
304 print("The loop's index is ", index)
305 parent = current.parent
306 local_current = local_current.parent
307 if debug:
308 print("The type of the object at the index is " +
309 str(type(parent.content[index])))
310 print("If preceding node is a directive then move back one")
311 if index == 0:
312 if debug:
313 print("current index is 0 so finish")
314 elif isinstance(parent.content[index-1], Directive):
315 if debug:
316 print(
317 f"preceding node is a directive so find out what type ..."
318 f"\n type is {parent.content[index-1].position}"
319 f"\n diretive is {parent.content[index-1]}")
320 if parent.content[index-1].position == "begin":
321 if debug:
322 print("type of directive is begin so move back one")
323 index -= 1
324 else:
325 if debug:
326 print("directive type is not begin so finish")
327 else:
328 if debug:
329 print("preceding node is not a directive so finish")
330 if debug:
331 print("type of final location ", type(parent.content[index]))
332 print("code for final location ", str(parent.content[index]))
333 return local_current, parent.content[index]

342 def add(self, content, position=None, bubble_up=False):
343 '''
344 Specialise the add method to provide module- and subroutine-
345 -specific intelligent adding of use statements, implicit
346 none statements and declarations if the position argument
347 is set to auto (which is the default).
348
349 :param content: the Node (or sub-tree of Nodes) to add in to \
350 the AST.
351 :type content: :py:class:`psyclone.f2pygen.BaseGen`
352 :param list position: where to insert the node. One of "append", \
353 "first", "insert", "after", "after_index", \
354 "before_index", "before" or "auto". For the \
355 *_index options, the second element of the \
356 list holds the integer index.
357 :param bool bubble_up: whether or not object (content) is in the \
358 process of being bubbled-up.
359 '''
360 # By default the position is 'auto'. We set it up this way for
361 # safety because in python, default arguments are instantiated
362 # as objects at the time of definition. If this object is
363 # subsequently modified then the value of the default argument
364 # is modified for subsequent calls of this routine.
365 if position is None:
366 position = ["auto"]
367
368 # For an object to be added to another we require that they
369 # share a common ancestor. This means that the added object must
370 # have the current object or one of its ancestors as an ancestor.
371 # Loop over the ancestors of this object (starting with itself)
372 self_ancestor = self.root
373 while self_ancestor:
374 # Loop over the ancestors of the object being added
375 obj_parent = content.root.parent
376 while (obj_parent != self_ancestor and
377 getattr(obj_parent, 'parent', None)):
378 obj_parent = obj_parent.parent
379 if obj_parent == self_ancestor:
380 break
381 # Object being added is not an ancestor of the current
382 # self_ancestor so move one level back up the tree and
383 # try again
384 if getattr(self_ancestor, 'parent', None):
385 self_ancestor = self_ancestor.parent
386 else:
387 break
388
389 if obj_parent != self_ancestor:
390 raise RuntimeError(
391 f"Cannot add '{content}' to '{self}' because it is not a "
392 f"descendant of it or of any of its ancestors.")
393
394 if bubble_up:
395 # If content has been passed on (is being bubbled up) then change
396 # its parent to be this object
397 content.root.parent = self.root
398
399 if position[0] != "auto":
400 # position[0] is not 'auto' so the baseclass can deal with it
401 BaseGen.add(self, content, position)
402 else:
403 # position[0] == "auto" so insert in a context sensitive way
404 if isinstance(content, BaseDeclGen):
405
406 if isinstance(content, (DeclGen, CharDeclGen)):
407 # have I already been declared?
408 for child in self._children_children_children:
409 if isinstance(child, (DeclGen, CharDeclGen)):
410 # is this declaration the same type as me?
411 if child.root.name == content.root.name:
412 # we are modifying the list so we need
413 # to iterate over a copy
414 for var_name in content.root.entity_decls[:]:
415 for child_name in child.root.entity_decls:
416 if var_name.lower() == \
417 child_name.lower():
418 content.root.entity_decls.\
419 remove(var_name)
420 if not content.root.entity_decls:
421 # return as all variables in
422 # this declaration already
423 # exist
424 return
425 if isinstance(content, TypeDeclGen):
426 # have I already been declared?
427 for child in self._children_children_children:
428 if isinstance(child, TypeDeclGen):
429 # is this declaration the same type as me?
430 if child.root.selector[1] == \
431 content.root.selector[1]:
432 # we are modifying the list so we need
433 # to iterate over a copy
434 for var_name in content.root.entity_decls[:]:
435 for child_name in child.root.entity_decls:
436 if var_name.lower() == \
437 child_name.lower():
438 content.root.entity_decls.\
439 remove(var_name)
440 if not content.root.entity_decls:
441 # return as all variables in
442 # this declaration already
443 # exist
444 return
445
446 index = 0
447 # skip over any use statements
448 index = self._skip_use_and_comments(index)
449 # skip over implicit none if it exists
450 index = self._skip_imp_none_and_comments(index)
451 # skip over any declarations which have an intent
452 try:
453 intent = True
454 while intent:
455 intent = False
456 for attr in self.root.content[index].attrspec:
457 if attr.find("intent") == 0:
458 intent = True
459 index += 1
460 break
461 except AttributeError:
462 pass
463 elif isinstance(content.root, fparser1.statements.Use):
464 # have I already been declared?
465 for child in self._children_children_children:
466 if isinstance(child, UseGen):
467 if child.root.name == content.root.name:
468 # found an existing use with the same name
469 if not child.root.isonly and not \
470 content.root.isonly:
471 # both are generic use statements so
472 # skip this declaration
473 return
474 if child.root.isonly and not content.root.isonly:
475 # new use is generic and existing use
476 # is specific so we can safely add
477 pass
478 if not child.root.isonly and content.root.isonly:
479 # existing use is generic and new use
480 # is specific so we can skip this
481 # declaration
482 return
483 if child.root.isonly and content.root.isonly:
484 # we are modifying the list so we need
485 # to iterate over a copy
486 for new_name in content.root.items[:]:
487 for existing_name in child.root.items:
488 if existing_name.lower() == \
489 new_name.lower():
490 content.root.items.remove(new_name)
491 if not content.root.items:
492 return
493 index = 0
494 elif isinstance(content, ImplicitNoneGen):
495 # does implicit none already exist?
496 for child in self._children_children_children:
497 if isinstance(child, ImplicitNoneGen):
498 return
499 # skip over any use statements
500 index = 0
501 index = self._skip_use_and_comments(index)
502 else:
503 index = len(self.root.content) - 1
504 self.root.content.insert(index, content.root)
505 self._children_children_children.append(content)
506
507 def _skip_use_and_comments(self, index):
508 ''' skip over any use statements and comments in the ast '''
509 while isinstance(self.root.content[index],
510 fparser1.statements.Use) or\
511 isinstance(self.root.content[index],
512 fparser1.statements.Comment):
513 index += 1
514 # now roll back to previous Use
515 while isinstance(self.root.content[index-1],
516 fparser1.statements.Comment):
517 index -= 1
518 return index
519
520 def _skip_imp_none_and_comments(self, index):
521 ''' skip over an implicit none statement if it exists and any
522 comments before it '''
523 end_index = index
524 while isinstance(self.root.content[index],
525 fparser1.typedecl_statements.Implicit) or\
526 isinstance(self.root.content[index],
527 fparser1.statements.Comment):
528 if isinstance(self.root.content[index],
529 fparser1.typedecl_statements.Implicit):
530 end_index = index + 1
531 break
532 else:
533 index = index + 1
534 return end_index

537class PSyIRGen(BaseGen):
538 ''' Create a Fortran block of code that comes from a given PSyIR tree.
539
540 :param parent: node in AST to which we are adding the PSyIR block.
541 :type parent: :py:class:`psyclone.f2pygen.BaseGen`
542 :param content: the PSyIR tree we are adding.
543 :type content: :py:class:`psyclone.psyir.nodes.Node`
544
545 '''
546
547 def __init__(self, parent, content):
548 # Import FortranWriter here to avoid circular-dependency
549 # pylint: disable=import-outside-toplevel
550 from psyclone.psyir.backend.fortran import FortranWriter
551 # We need the Config object in order to see whether or not to disable
552 # the validation performed in the PSyIR backend.
553 config = Config.get()
554
555 # Use the PSyIR Fortran backend to generate Fortran code of the
556 # supplied PSyIR tree and pass the resulting code to the fparser1
557 # Fortran parser.
558 fortran_writer = FortranWriter(
559 check_global_constraints=config.backend_checks_enabled)
560 reader = FortranStringReader(fortran_writer(content),
561 ignore_comments=False)
562 # Set reader as free form, strict
563 reader.set_format(FortranFormat(True, True))
564 fparser1_parser = FortranParser(reader, ignore_comments=False)
565 fparser1_parser.parse()
566
567 # If the fparser content is larger than 1, add all the nodes but
568 # the last one as siblings of self. This is done because self
569 # can only represent one node.
570 for fparser_node in fparser1_parser.block.content[:-1]:
571 f2pygen_node = BaseGen(parent, fparser_node)
572 f2pygen_node.root.parent = parent.root
573 parent.add(f2pygen_node)
574
575 # Update this f2pygen node to be equivalent to the last of the
576 # fparser nodes that represent the provided content.
577 BaseGen.__init__(self, parent, fparser1_parser.block.content[-1])
578 self.root.parent = parent.root

638class DirectiveGen(BaseGen):
639 '''
640 Class for creating a Fortran directive, e.g. OpenMP or OpenACC.
641
642 :param parent: node in AST to which to add directive as a child.
643 :type parent: :py:class:`psyclone.f2pygen.BaseGen`
644 :param str language: the type of directive (e.g. OMP or ACC).
645 :param str position: "end" if this is the end of a directive block.
646 :param str directive_type: the directive itself (e.g. "PARALLEL DO").
647 :param str content: any additional arguments to add to the directive \
648 (e.g. "PRIVATE(ji)").
649
650 :raises RuntimeError: if an unrecognised directive language is specified.
651 '''
652 def __init__(self, parent, language, position, directive_type, content=""):
653 self._supported_languages = ["omp", "acc"]
654 self._language = language
655 self._directive_type = directive_type
656
657 reader = FortranStringReader("! content\n", ignore_comments=False)
658 reader.set_format(FortranFormat(True, True)) # free form, strict
659 subline = reader.next()
660
661 if language == "omp":
662 my_comment = OMPDirective(parent.root, subline, position,
663 directive_type)
664 my_comment.content = "$omp"
665 elif language == "acc":
666 my_comment = ACCDirective(parent.root, subline, position,
667 directive_type)
668 my_comment.content = "$acc"
669 else:
670 raise RuntimeError(
671 f"Error, unsupported directive language. Expecting one of "
672 f"{self._supported_languages} but found '{language}'")
673 if position == "end":
674 my_comment.content += " end"
675 my_comment.content += " " + directive_type
676 if content != "":
677 my_comment.content += " " + content
678
679 BaseGen.__init__(self, parent, my_comment)

686 :param parent: node in AST to which to add 'implicit none' as a child

904 ALLOCATABLE quantity.

917 :raises NotImplementedError: if initial values are supplied for array \

921 _decl = None # Will hold the declaration object created by sub-class
922
923 def __init__(self, parent, datatype="", entity_decls=None, intent="",
924 pointer=False, dimension="", allocatable=False,
925 save=False, target=False, initial_values=None, private=False):
926 if entity_decls is None:

928 "Cannot create a variable declaration without specifying the "
929 "name(s) of the variable(s)")
930

932 # are the right number of them and that they are consistent
933 # with the type of the variable(s) being declared.
934 if initial_values:
935 if len(initial_values) != len(entity_decls):
936 raise RuntimeError(

968 else:

971 # Construct the list of attributes

982 my_attrspec.append("save")

984 my_attrspec.append("private")

1101 def _check_initial_values(self, dtype, values):
1102 '''
1103 Check that the supplied values are consistent with the requested
1104 data type. Note that this checking is fairly basic and does not
1105 support a number of valid Fortran forms (e.g. arithmetic expressions
1106 involving constants or parameters).
1107
1108 :param str dtype: Fortran intrinsic type.
1109 :param list values: list of values as strings.
1110 :raises RuntimeError: if the supplied values are not consistent \
1111 with the specified data type or are not \
1112 supported.
1113 '''
1114 from fparser.two.pattern_tools import abs_name, \
1115 abs_logical_literal_constant, abs_signed_int_literal_constant, \
1116 abs_signed_real_literal_constant
1117 if dtype == "logical":
1118 # Can be .true., .false. or a valid Fortran variable name
1119 for val in values:
1120 if not abs_logical_literal_constant.match(val) and \
1121 not abs_name.match(val):
1122 raise RuntimeError(
1123 f"Initial value of '{val}' for a logical variable is "
1124 f"invalid or unsupported")
1125 elif dtype == "integer":
1126 # Can be a an integer expression or a valid Fortran variable name
1127 for val in values:
1128 if not abs_signed_int_literal_constant.match(val) and \
1129 not abs_name.match(val):
1130 raise RuntimeError(
1131 f"Initial value of '{val}' for an integer variable is "
1132 f"invalid or unsupported")
1133 elif dtype == "real":
1134 # Can be a floating-point expression or a valid Fortran name
1135 for val in values:
1136 if not abs_signed_real_literal_constant.match(val) and \
1137 not abs_name.match(val):
1138 raise RuntimeError(
1139 f"Initial value of '{val}' for a real variable is "
1140 f"invalid or unsupported")
1141 else:
1142 # We should never get to here because we check that the type
1143 # is supported before calling this routine.
1144 raise InternalError(
1145 f"unsupported type '{dtype}' - should be "
1146 f"one of {DeclGen.SUPPORTED_TYPES}")
