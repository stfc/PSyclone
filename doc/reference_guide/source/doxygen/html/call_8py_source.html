111 '''Create an instance of class cls given valid instances of a routine
112 symbol, and a list of child nodes (or name and node tuple) for
113 its arguments.
114
115 :param routine: the routine that class cls calls.
116 :type routine: py:class:`psyclone.psyir.symbols.RoutineSymbol` |
117 py:class:`psyclone.psyir.nodes.Reference`
118 :param arguments: optional list of arguments for this call, these
119 can be PSyIR nodes or tuples of string,Node for named arguments.
120 :type arguments: Optional[Iterable[\
121 Union[:py:class:`psyclone.psyir.nodes.DataNode`,\
122 Tuple[str, :py:class:`psyclone.psyir.nodes.DataNode`]]]]
123
124 :returns: an instance of cls.
125 :rtype: :py:class:`psyclone.psyir.nodes.Call` or a subclass thereof.

128 :raises GenerationError: if the arguments argument is not an Iterable.

145 else:

295 def reference_accesses(self, var_accesses):
296 '''
297 Updates the supplied var_accesses object with information on the
298 arguments passed to this call.
299
300 TODO #446 - all arguments that are passed by reference are currently
301 marked as having READWRITE access (unless we know that the routine is
302 PURE). We could do better than this if we have the PSyIR of the called
303 Routine.
304
305 :param var_accesses: VariablesAccessInfo instance that stores the
306 information about variable accesses.
307 :type var_accesses: :py:class:`psyclone.core.VariablesAccessInfo`
308
309 '''
310 if self.is_pureis_pure:
311 # If the called routine is pure then any arguments are only
312 # read.
313 default_access = AccessType.READ
314 else:
315 # We conservatively default to READWRITE otherwise (TODO #446).
316 default_access = AccessType.READWRITE
317
318 # The RoutineSymbol has a CALL access.
319 sig, indices_list = self.routineroutine.get_signature_and_indices()
320 var_accesses.add_access(sig, AccessType.CALL, self.routineroutine)
321 # Continue processing references in any index expressions.
322 for indices in indices_list:
323 for idx in indices:
324 idx.reference_accesses(var_accesses)
325
326 for arg in self.argumentsarguments:
327 if isinstance(arg, Reference):
328 # This argument is pass-by-reference.
329 sig, indices_list = arg.get_signature_and_indices()
330 var_accesses.add_access(sig, default_access, arg)
331 # Continue processing references in any index expressions.
332 for indices in indices_list:
333 for idx in indices:
334 idx.reference_accesses(var_accesses)
335 else:
336 # This argument is not a Reference so continue to walk down the
337 # tree. (e.g. it could be/contain a Call to
338 # an impure routine in which case any arguments to that Call
339 # will have READWRITE access.)
340 arg.reference_accesses(var_accesses)
341 # Make sure that the next statement will be on the next location
342 var_accesses.next_location()

440 def copy(self):
441 '''Return a copy of this node. This is a bespoke implementation for
442 a Call node that ensures that any internal id's are
443 consistent before and after copying.
444
445 :returns: a copy of this node and its children.
446 :rtype: :py:class:`psyclone.psyir.node.Node`
447
448 '''
449 # ensure _argument_names is consistent with actual arguments
450 # before copying.
451 self._reconcile()
452 # copy
453 new_copy = super().copy()
454 # Fix invalid id's in _argument_names after copying.
455 # pylint: disable=protected-access
456 new_list = []
457 for idx, child in enumerate(new_copy.arguments):
458 my_tuple = (id(child), new_copy._argument_names[idx][1])
459 new_list.append(my_tuple)
460 new_copy._argument_names = new_list
461
462 return new_copy

476 def _location_txt(node):
477 '''
478 Utility to generate meaningful location text.
479
480 :param node: a PSyIR node.
481 :type node: :py:class:`psyclone.psyir.nodes.Node`
482
483 :returns: description of location of node.
484 :rtype: str
485 '''
486 if isinstance(node, Container):
487 return f"Container '{node.name}'"
488 out_lines = node.debug_string().split("\n")
489 idx = -1
490 while not out_lines[idx]:
491 idx -= 1
492 last_line = out_lines[idx]
493 return f"code:\n'{out_lines[0]}\n...\n{last_line}'"
494
495 rsym = self.routineroutine.symbol
496 if rsym.is_unresolved:
497
498 # Check for any "raw" Routines, i.e. ones that are not
499 # in a Container. Such Routines would exist in the PSyIR
500 # as a child of a FileContainer (if the PSyIR contains a
501 # FileContainer). Note, if the PSyIR does contain a
502 # FileContainer, it will be the root node of the PSyIR.
503 for routine in self.root.children:
504 if (isinstance(routine, Routine) and
505 routine.name.lower() == rsym.name.lower()):
506 return [routine]
507
508 # Now check for any wildcard imports and see if they can
509 # be used to resolve the symbol.
510 wildcard_names = []
511 containers_not_found = []
512 current_table = self.scope.symbol_table
513 while current_table:
514 for container_symbol in current_table.containersymbols:
515 if container_symbol.wildcard_import:
516 wildcard_names.append(container_symbol.name)
517 try:
518 container = container_symbol.find_container_psyir(
519 local_node=self)
520 except SymbolError:
521 container = None
522 if not container:
523 # Failed to find/process this Container.
524 containers_not_found.append(container_symbol.name)
525 continue
526 routines = []
527 for name in container.resolve_routine(rsym.name):
528 psyir = container.find_routine_psyir(name)
529 if psyir:
530 routines.append(psyir)
531 if routines:
532 return routines
533 current_table = current_table.parent_symbol_table()
534 if not wildcard_names:
535 wc_text = "there are no wildcard imports"
536 else:
537 if containers_not_found:
538 wc_text = (
539 f"attempted to resolve the wildcard imports from"
540 f" {wildcard_names}. However, failed to find the "
541 f"source for {containers_not_found}. The module search"
542 f" path is set to {Config.get().include_paths}")
543 else:
544 wc_text = (f"wildcard imports from {wildcard_names}")
545 raise NotImplementedError(
546 f"Failed to find the source code of the unresolved routine "
547 f"'{rsym.name}' - looked at any routines in the same source "
548 f"file and {wc_text}. Searching for external routines "
549 f"that are only resolved at link time is not supported.")
550
551 root_node = self.ancestor(Container)
552 if not root_node:
553 root_node = self.root
554 container = root_node
555 can_be_private = True
556
557 if rsym.is_import:
558 cursor = rsym
559 # A Routine imported from another Container must be public in that
560 # Container.
561 can_be_private = False
562 while cursor.is_import:
563 csym = cursor.interface.container_symbol
564 try:
565 container = csym.find_container_psyir(local_node=self)
566 except SymbolError:
567 raise NotImplementedError(
568 f"RoutineSymbol '{rsym.name}' is imported from "
569 f"Container '{csym.name}' but the source defining "
570 f"that container could not be found. The module search"
571 f" path is set to {Config.get().include_paths}")
572 imported_sym = container.symbol_table.lookup(cursor.name)
573 if imported_sym.visibility != Symbol.Visibility.PUBLIC:
574 # The required Symbol must be shadowed with a PRIVATE
575 # Symbol in this Container. This means that the one we
576 # actually want is brought into scope via a wildcard
577 # import.
578 # TODO #924 - Use ModuleManager to search?
579 raise NotImplementedError(
580 f"RoutineSymbol '{rsym.name}' is imported from "
581 f"Container '{csym.name}' but that Container defines "
582 f"a private Symbol of the same name. Searching for the"
583 f" Container that defines a public Routine with that "
584 f"name is not yet supported - TODO #924")
585 if not isinstance(imported_sym, RoutineSymbol):
586 # We now know that this is a RoutineSymbol so specialise it
587 # in place.
588 imported_sym.specialise(RoutineSymbol)
589 cursor = imported_sym
590 rsym = cursor
591 root_node = container
592
593 if isinstance(rsym.datatype, UnsupportedFortranType):
594 # TODO #924 - an UnsupportedFortranType here typically indicates
595 # that the target is actually an interface.
596 raise NotImplementedError(
597 f"RoutineSymbol '{rsym.name}' exists in "
598 f"{_location_txt(root_node)} but is of "
599 f"UnsupportedFortranType:\n{rsym.datatype.declaration}\n"
600 f"Cannot get the PSyIR of such a routine.")
601
602 if isinstance(container, Container):
603 routines = []
604 for name in container.resolve_routine(rsym.name):
605 psyir = container.find_routine_psyir(
606 name, allow_private=can_be_private)
607 if psyir:
608 routines.append(psyir)
609 if routines:
610 return routines
611
612 raise SymbolError(
613 f"Failed to find a Routine named '{rsym.name}' in "
614 f"{_location_txt(root_node)}. This is normally because the routine"
615 f" is within a CodeBlock.")

634 routine_arg.datatype, UnsupportedFortranType

639 call_arg.datatype

655 def _get_argument_routine_match(self, routine: Routine):
656 '''Return a list of integers giving for each argument of the call
657 the index of the corresponding entry in the argument list of the
658 supplied routine.
659
660 :return: None if no match was found, otherwise list of integers
661 referring to matching arguments.
662 :rtype: None|List[int]
663 '''
664
665 # Create a copy of the list of actual arguments to the routine.
666 # Once an argument has been successfully matched, set it to 'None'
667 routine_argument_list: List[DataSymbol] = (
668 routine.symbol_table.argument_list[:]
669 )
670
671 if len(self.argumentsarguments) > len(routine.symbol_table.argument_list):
672 raise CallMatchingArgumentsNotFound(
673 f"More arguments in call ('{self.debug_string()}')"
674 f" than callee (routine '{routine.name}')"
675 )
676
677 # Iterate over all arguments to the call
678 ret_arg_idx_list = []
679 for call_arg_idx, call_arg in enumerate(self.argumentsarguments):
680 call_arg_idx: int
681 call_arg: DataSymbol
682
683 # If the associated name is None, it's a positional argument
684 # => Just return the index if the types match
685 if self.argument_namesargument_names[call_arg_idx] is None:
686 routine_arg = routine_argument_list[call_arg_idx]
687 routine_arg: DataSymbol
688
689 self._check_argument_type_matches(call_arg, routine_arg)
690
691 ret_arg_idx_list.append(call_arg_idx)
692 routine_argument_list[call_arg_idx] = None
693 continue
694
695 #
696 # Next, we handle all named arguments
697 #
698 arg_name = self.argument_namesargument_names[call_arg_idx]
699 routine_arg_idx = None
700
701 for routine_arg_idx, routine_arg in enumerate(
702 routine_argument_list
703 ):
704 routine_arg: DataSymbol
705
706 # Check if argument was already processed
707 if routine_arg is None:
708 continue
709
710 if arg_name == routine_arg.name:
711 self._check_argument_type_matches(call_arg, routine_arg)
712 ret_arg_idx_list.append(routine_arg_idx)
713 break
714
715 else:
716 # It doesn't match => Raise exception
717 raise CallMatchingArgumentsNotFound(
718 f"Named argument '{arg_name}' not found for routine"
719 f" '{routine.name}' in call '{self.debug_string()}'"
