39''' This module creates an LFRic-specific Invokes object which controls all
40 the required invocation calls. It also overrides the PSy gen method so
41 that LFRic-specific PSy module code is generated.
42 '''

47from psyclone.domain.lfric import (LFRicConstants, LFRicSymbolTable,
48 LFRicInvokes)
49from psyclone.f2pygen import ModuleGen, UseGen, PSyIRGen
50from psyclone.psyGen import PSy, InvokeSchedule
51from psyclone.psyir.nodes import ScopingNode

54class LFRicPSy(PSy):
55 '''
56 The LFRic-specific PSy class. This creates an LFRic-specific
57 Invokes object (which controls all the required invocation calls).
58 It also overrides the PSy gen method so that we generate
59 LFRic-specific PSy module code.
60
61 :param invoke_info: object containing the required invocation information
62 for code optimisation and generation.
63 :type invoke_info: :py:class:`psyclone.parse.algorithm.FileInfo`
64
65 '''
66 def __init__(self, invoke_info):
67 # Make sure the scoping node creates LFRicSymbolTables
68 # TODO #1954: Remove the protected access using a factory
69 ScopingNode._symbol_table_class = LFRicSymbolTable
70 Config.get().api = "lfric"
71 PSy.__init__(self, invoke_info)
72 self._invokes_invokes = LFRicInvokes(invoke_info.calls, self)
73 # Initialise the dictionary that holds the names of the required
74 # LFRic constants, data structures and data structure proxies for
75 # the "use" statements in modules that contain PSy-layer routines.
76 const = LFRicConstants()
77 const_mod = const.UTILITIES_MOD_MAP["constants"]["module"]
78 infmod_list = [const_mod]
79 # Add all field and operator modules that might be used in the
80 # algorithm layer. These do not appear in the code unless a
81 # variable is added to the "only" part of the
82 # '_infrastructure_modules' map.
83 for data_type_info in const.DATA_TYPE_MAP.values():
84 infmod_list.append(data_type_info["module"])
85
86 # This also removes any duplicates from infmod_list
87 self._infrastructure_modules = OrderedDict(
88 (k, set()) for k in infmod_list)
89
90 kind_names = set()
91
92 # The infrastructure declares integer types with default
93 # precision so always add this.
94 api_config = Config.get().api_conf("lfric")
95 kind_names.add(api_config.default_kind["integer"])
96
97 # Datatypes declare precision information themselves. However,
98 # that is not the case for literals. Therefore deal
99 # with these separately here.
100 for invoke in self.invokes.invoke_list:
101 schedule = invoke.schedule
102 for kernel in schedule.kernels():
103 for arg in kernel.args:
104 if arg.is_literal:
105 kind_names.add(arg.precision)
106 # Add precision names to the dictionary storing the required
107 # LFRic constants.
108 self._infrastructure_modules[const_mod] = kind_names
109
110 @property
111 def name(self):
112 '''
113 :returns: a name for the PSy layer. This is used as the PSy module
114 name. We override the default value as the Met Office
115 prefer "_psy" to be appended, rather than prepended.
116 :rtype: str
117
118 '''
119 return self._name + "_psy"
120
121 @property
122 def orig_name(self):
123 '''
124 :returns: the unmodified PSy-layer name.
125 :rtype: str
126
127 '''
128 return self._name
129
130 @property
131 def infrastructure_modules(self):
132 '''
133 :returns: the dictionary that holds the names of the required
134 LFRic infrastructure modules to create "use"
135 statements in the PSy-layer modules.
136 :rtype: dict of set
137
138 '''
139 return self._infrastructure_modules
140
141 @property
142 def gen(self):
143 '''
144 Generate PSy code for the LFRic API.
145
146 :returns: root node of generated Fortran AST.
147 :rtype: :py:class:`psyir.nodes.Node`
148
149 '''
150 # Create an empty PSy layer module
151 psy_module = ModuleGen(self.namenamename)
152
153 # If the container has a Routine that is not an InvokeSchedule
154 # it should also be added to the generated module.
155 for routine in self.container.children:
156 if not isinstance(routine, InvokeSchedule):
157 psy_module.add(PSyIRGen(psy_module, routine))
158
159 # Add all invoke-specific information
160 self.invokes.gen_code(psy_module)
161
162 # Include required constants and infrastructure modules. The sets of
163 # required LFRic data structures and their proxies are updated in
164 # the relevant field and operator subclasses of LFRicCollection.
165 # Here we sort the inputs in reverse order to have "_type" before
166 # "_proxy_type" and "operator_" before "columnwise_operator_".
167 # We also iterate through the dictionary in reverse order so the
168 # "use" statements for field types are before the "use" statements
169 # for operator types.
170 for infmod in reversed(self._infrastructure_modules):
171 if self._infrastructure_modules[infmod]:
172 infmod_types = sorted(
173 list(self._infrastructure_modules[infmod]), reverse=True)
174 psy_module.add(UseGen(psy_module, name=infmod,
175 only=True, funcnames=infmod_types))
176
177 # Return the root node of the generated code
178 return psy_module.root
