95 def get_user_type(self, module_name, user_type, name, tag=None):
96 # pylint: disable=too-many-arguments
97 '''Returns the symbol for a user-defined type. If required, the
98 required import statements will all be generated.
99
100 :param str module_name: the name of the module from which the \
101 user-defined type must be imported.
102 :param str user_type: the name of the user-defined type.
103 :param str name: the name of the variable to be used in the Reference.
104 :param Optional[str] tag: tag to use for the variable, defaults to \
105 the name
106
107 :return: the symbol that is used in the reference
108 :rtype: :py:class:`psyclone.psyir.symbols.Symbol`
109
110 '''
111 if not tag:
112 tag = name
113
114 try:
115 sym = self._symtab.lookup_with_tag(tag)
116 return sym
117 except KeyError:
118 pass
119
120 # The symbol does not exist already. So we potentially need to
121 # create the import statement for the type:
122 try:
123 # Check if the module is already declared:
124 module = self._symtab.lookup(module_name)
125 except KeyError:
126 module = self._symtab.new_symbol(module_name,
127 symbol_type=ContainerSymbol)
128
129 # Get the symbol table in which the module is declared:
130 mod_sym_tab = module.find_symbol_table(self._kern_kern)
131
132 # The user-defined type must be declared in the same symbol
133 # table as the container (otherwise errors will happen later):
134 user_type_symbol = mod_sym_tab.find_or_create(
135 user_type,
136 symbol_type=DataTypeSymbol,
137 datatype=UnresolvedType(),
138 interface=ImportInterface(module))
139 # Declare the actual user symbol in the local symbol table, using
140 # the datatype from the root table:
141 sym = self._symtab.new_symbol(name, tag=tag,
142 symbol_type=DataSymbol,
143 datatype=user_type_symbol)
144 return sym

146 def append_structure_reference(self, module_name, user_type, member_list,
147 name, tag=None, overwrite_datatype=None):
148 # pylint: disable=too-many-arguments
149 '''Creates a reference to a variable of a user-defined type. If
150 required, the required import statements will all be generated.
151
152 :param str module_name: the name of the module from which the
153 user-defined type must be imported.
154 :param str user_type: the name of the user-defined type.
155 :param member_list: the members used hierarchically.
156 :type member_list: List[str]
157 :param str name: the name of the variable to be used in the Reference.
158 :param Optional[str] tag: tag to use for the variable, defaults to
159 the name
160 :param overwrite_datatype: the datatype for the reference, which will
161 overwrite the value determined by analysing the corresponding
162 user defined type. This is useful when e.g. the module that
163 declares the structure cannot be accessed.
164 :type overwrite_datatype:
165 Optional[:py:class:`psyclone.psyir.symbols.DataType`]
166
167 :return: the symbol that is used in the reference
168 :rtype: :py:class:`psyclone.psyir.symbols.Symbol`
169
170 '''
171 sym = self.get_user_type(module_name, user_type, name, tag)
172 self.psyir_append(StructureReference.
173 create(sym, member_list,
174 overwrite_datatype=overwrite_datatype))
175 return sym

191 def cell_map(self, var_accesses=None):
192 '''Add cell-map and related cell counts (for inter-grid kernels)
193 to the argument list. If supplied it also stores these accesses to the
194 var_access object.
195
196 :param var_accesses: optional VariablesAccessInfo instance to store \
197 the information about variable accesses.
198 :type var_accesses: \
199 :py:class:`psyclone.core.VariablesAccessInfo`
200
201 '''
202 cargs = psyGen.args_filter(self._kern_kern.args, arg_meshes=["gh_coarse"])
203 carg = cargs[0]
204 fargs = psyGen.args_filter(self._kern_kern.args, arg_meshes=["gh_fine"])
205 farg = fargs[0]
206 base_name = "cell_map_" + carg.name
207
208 # Add the cell map to our argument list
209 cell_ref_name, cell_ref = self.cell_ref_name(var_accesses)
210 sym = self.append_array_reference(base_name, [":", ":", cell_ref],
211 ScalarType.Intrinsic.INTEGER)
212 self.append(f"{sym.name}(:,:,{cell_ref_name})",
213 var_accesses=var_accesses, var_access_name=sym.name)
214
215 # No. of fine cells per coarse cell in x
216 base_name = f"ncpc_{farg.name}_{carg.name}_x"
217 sym = self.append_integer_reference(base_name)
218 self.append(sym.name, var_accesses)
219 # No. of fine cells per coarse cell in y
220 base_name = f"ncpc_{farg.name}_{carg.name}_y"
221 sym = self.append_integer_reference(base_name)
222 self.append(sym.name, var_accesses)
223 # No. of columns in the fine mesh
224 base_name = f"ncell_{farg.name}"
225 sym = self.append_integer_reference(base_name)
226 self.append(sym.name, var_accesses)

300 def cma_operator(self, arg, var_accesses=None):
301 '''Add the CMA operator and associated scalars to the argument
302 list and optionally add them to the variable access
303 information.
304
305 :param arg: the CMA operator argument.
306 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
307 :param var_accesses: optional VariablesAccessInfo instance to store \
308 the information about variable accesses.
309 :type var_accesses: \
310 :py:class:`psyclone.core.VariablesAccessInfo`
311
312 '''
313 components = ["matrix"]
314 # Avoid circular import:
315 # pylint: disable=import-outside-toplevel
316 from psyclone.dynamo0p3 import DynCMAOperators
317 if arg.function_space_to.orig_name != (arg.function_space_from.
318 orig_name):
319 components += DynCMAOperators.cma_diff_fs_params
320 else:
321 components += DynCMAOperators.cma_same_fs_params
322
323 const = LFRicConstants()
324 suffix = const.ARG_TYPE_SUFFIX_MAPPING["gh_columnwise_operator"]
325
326 for component in components:
327 # Matrix takes the access from the declaration of the argument
328 # (i.e. read, write, ...), the rest are always read-only parameters
329 if component == "matrix":
330 # Matrix is a pointer to a 3d array
331 # REAL(KIND=r_solver), pointer:: cma_op1_matrix(:,:,:)
332 # = > null()
333 mode = arg.access
334 sym = self._symtab.lookup_with_tag(f"{arg.name}:{suffix}")
335 self.psyir_append(ArrayReference.create(sym, [":", ":", ":"]))
336 else:
337 # All other variables are scalar integers
338 name = self._symtab.lookup_with_tag(
339 f"{arg.name}:{component}:{suffix}").name
340 mode = AccessType.READ
341 sym = self.append_integer_reference(
342 name, tag=f"{arg.name}:{component}:{suffix}")
343
344 self.append(sym.name, var_accesses, mode=mode,
345 metadata_posn=arg.metadata_index)

347 def field_vector(self, argvect, var_accesses=None):
348 '''Add the field vector associated with the argument 'argvect' to the
349 argument list. If supplied it also stores these accesses to the
350 var_access object.
351
352 :param argvect: the field vector to add.
353 :type argvect: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
354 :param var_accesses: optional VariablesAccessInfo instance to store
355 the information about variable accesses.
356 :type var_accesses: :py:class:`psyclone.core.VariablesAccessInfo`
357
358 '''
359 suffix = LFRicConstants().ARG_TYPE_SUFFIX_MAPPING[
360 argvect.argument_type]
361
362 # The range function below returns values from
363 # 1 to the vector size which is what we
364 # require in our Fortran code
365 for idx in range(1, argvect.vector_size + 1):
366 # Look-up the name of the variable that stores the reference to
367 # the data in this field.
368 cmpt_sym = self._symtab.lookup_with_tag(
369 f"{argvect.name}_{idx}:{suffix}")
370 if self._kern_kern.iterates_over == "dof":
371 # If dof kernel, add access to the field by dof ref
372 dof_sym = self._symtab.find_or_create_integer_symbol(
373 "df", tag="dof_loop_idx")
374 # TODO #1010 removes the need to declare type and
375 # allows this to be fixed
376 self.append_array_reference(cmpt_sym.name,
377 [Reference(dof_sym)],
378 ScalarType.Intrinsic.INTEGER,
379 symbol=cmpt_sym)
380 # Append the dof symbol
381 text = f"{cmpt_sym.name}({dof_sym.name})"
382 else:
383 self.psyir_append(Reference(cmpt_sym))
384 text = cmpt_sym.name
385 self.append(text, metadata_posn=argvect.metadata_index)
386
387 if var_accesses is not None:
388 # We add the whole field-vector, not the individual accesses.
389 var_accesses.add_access(Signature(argvect.name), argvect.access,
390 self._kern_kern)

392 def field(self, arg, var_accesses=None):
393 '''Add the field array associated with the argument 'arg' to the
394 argument list. If supplied it also stores this access in var_accesses.
395
396 :param arg: the field to be added.
397 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
398 :param var_accesses: optional VariablesAccessInfo instance to store
399 the information about variable accesses.
400 :type var_accesses: :py:class:`psyclone.core.VariablesAccessInfo`
401
402 '''
403 const = LFRicConstants()
404 suffix = const.ARG_TYPE_SUFFIX_MAPPING[arg.argument_type]
405 # Look-up the name of the variable that stores the reference to
406 # the data in this field.
407 sym = self._symtab.lookup_with_tag(f"{arg.name}:{suffix}")
408
409 if self._kern_kern.iterates_over == "dof":
410 # If dof kernel, add access to the field by dof ref
411 dof_sym = self._symtab.find_or_create_integer_symbol(
412 "df", tag="dof_loop_idx")
413 # TODO #1010 removes the need to declare type and
414 # allows this to be fixed
415 self.append_array_reference(sym.name, [Reference(dof_sym)],
416 ScalarType.Intrinsic.INTEGER,
417 symbol=sym)
418 # Then append our symbol
419 name = f"{sym.name}({dof_sym.name})"
420 self.append(name, var_accesses, var_access_name=sym.name)
421 else:
422 # Add the field data array as being read.
423 self.append(sym.name, var_accesses, var_access_name=sym.name,
424 mode=arg.access, metadata_posn=arg.metadata_index)
425 self.psyir_append(Reference(sym))

849 def quad_rule(self, var_accesses=None):
850 '''Add quadrature-related information to the kernel argument list.
851 Adds the necessary arguments to the argument list, and optionally
852 adds variable access information to the var_accesses object.
853
854 :param var_accesses: optional VariablesAccessInfo instance to store \
855 the information about variable accesses.
856 :type var_accesses: \
857 :py:class:`psyclone.core.VariablesAccessInfo`
858
859 '''
860 # The QR shapes that this routine supports
861 supported_qr_shapes = ["gh_quadrature_xyoz", "gh_quadrature_edge",
862 "gh_quadrature_face"]
863
864 for shape, rule in self._kern_kern.qr_rules.items():
865 if shape == "gh_quadrature_xyoz":
866 # XYoZ quadrature requires the number of quadrature points in
867 # the horizontal and in the vertical.
868 self._nqp_positions.append(
869 {"horizontal": self.num_argsnum_args + 1,
870 "vertical": self.num_argsnum_args + 2})
871 self.extend(rule.kernel_args, var_accesses)
872 elif shape == "gh_quadrature_edge":
873 # TODO #705 support transformations supplying the number of
874 # quadrature points for edge quadrature.
875 self.extend(rule.kernel_args, var_accesses)
876 elif shape == "gh_quadrature_face":
877 # TODO #705 support transformations supplying the number of
878 # quadrature points for face quadrature.
879 self.extend(rule.kernel_args, var_accesses)
880 else:
881 raise NotImplementedError(
882 f"quad_rule: no support implemented for quadrature with a "
883 f"shape of '{shape}'. Supported shapes are: "
884 f"{supported_qr_shapes}.")
885 # Now define the arguments using PSyIR:
886 for arg in rule.kernel_args:
887 # Each rule has a `psy_name` (e.g. qr_xyoz), which is appended
888 # to all variable names (e.g. np_xy_qr_xyoz). Remove this
889 # suffix to get the 'generic' name, from which we derive
890 # the correct type:
891 generic_name = arg[:-len(rule.psy_name)-1]
892 if generic_name in ["np_xy", "np_z", "nfaces", "np_xyz",
893 "nedges"]:
894 # np_xy, np_z, nfaces, np_xyz, nedges are all integers:
895 self.append_integer_reference(arg)
896 elif generic_name in ["weights_xy", "weights_z"]:
897 # 1d arrays:
898 # TODO # 1910: These should be pointers
899 self.append_array_reference(arg, [":"],
900 ScalarType.Intrinsic.REAL)
901 elif generic_name in ["weights_xyz"]:
902 # 2d arrays:
903 # TODO #1910: These should be pointers
904 self.append_array_reference(arg, [":", ":"],
905 ScalarType.Intrinsic.REAL)
906 else:
907 raise InternalError(f"Found invalid kernel argument "
908 f"'{arg}'.")
