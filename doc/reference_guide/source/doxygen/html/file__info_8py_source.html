79class FileInfo:
80 """This class stores mostly cached information about source code:
81 - it stores the original filename
82 - it will read the source of the file and cache it
83 - it will parse it with fparser and cache it
84 - it will construct the PSyIR (depends on TODO #2786 "and cache it")
85
86 :param filepath: Path to the file that this
87 object holds information on. Can also be set to 'None' in case of
88 providing fparser / PSyIR node in a different way.
89 :param use_caching: Use caching of intermediate representations
90
91 """
92 def __init__(self, filepath: str, use_caching: bool = False):
93
94 # Full path to file
95 self._filename: str = filepath
96
97 # Use cache features
98 self._use_caching: bool = use_caching
99
100 # Source code:
101 self._source_code: str = None
102
103 # Source code hash sum:
104 self._source_code_hash_sum: hashlib._Hash = None
105
106 # Fparser node
107 self._fparser_tree_fparser_tree: Fortran2003.Program = None
108
109 # Flag indicating that, based on a previous attempt,
110 # the fparser tree can't be generated due to an error
111 self._fparser_tree_triggers_error_fparser_tree_triggers_error: bool = False
112
113 # Psyir node
114 self._psyir_node_psyir_node: FileContainer = None
115
116 # Single cache file
117 (path, ext) = os.path.splitext(self._filename)
118 self._filepath_cache = path + ".psycache"
119
120 # This reference to `_CacheFileInfo` is created when loading
121 # cached information from a cache file.
122 # In case the checksums mismatch, no object will be referenced.
123 # Consequently, this object will always have a checksum matching
124 # the one from the source code.
125 self._cache_data_load: _CacheFileInfo = None
126
127 # This reference is used whenever writing cache data to the
128 # persistent storage.
129 # It will also be partly updated if the `psyir` or
130 # `fparser tree` was created in the meantime and a cache update
131 # is requested.
132 self._cache_data_save_cache_data_save: _CacheFileInfo = None
133
134 @property
135 def basename(self):
136 '''
137 :returns: the base name (i.e. without path or suffix) of the filename
138 that this FileInfo object represents.
139 :rtype: str
140 '''
141
142 # Remove the path from the filename.
143 basename = os.path.basename(self._filename)
144 # splitext returns (root, ext) and it's `root` that we want.
145 return os.path.splitext(basename)[0]
146
147 # ------------------------------------------------------------------------
148 @property
149 def filename(self):
150 '''
151 :returns: the full filename that this FileInfo object represents.
152 :rtype: str
153 '''
154 return self._filename

156 def get_source_code(self, verbose: bool = False) -> str:
157 '''Returns the source code of the file. The first time, it
158 will be read from the file, but the data is then cached.
159
160 If any decoding errors are encountered then the associated character(s)
161 are simply skipped. This is because this class is intended for reading
162 Fortran source and the only way such characters can appear is if they
163 are in comments.
164
165 :param verbose: If `True`, produce some verbose output
166
167 :returns: the contents of the file (utf-8 encoding).
168
169 '''
170 if self._source_code:
171 return self._source_code
172
173 if verbose:
174 # TODO #11: Use logging for this
175 print(
176 f"- Source file '{self._filename}': "
177 f"Loading source code"
178 )
179
180 try:
181 # Specifying errors='ignore' simply skips any characters that
182 # result in decoding errors. (Comments in a code may contain all
183 # sorts of weird things.)
184 with open(
185 self._filename, "r", encoding="utf-8", errors="ignore"
186 ) as file_in:
187 self._source_code = file_in.read()
188 except FileNotFoundError as err:
189 raise FileNotFoundError(
190 f"FileInfo: No such file or directory '{self._filename}'."
191 ) from err
192
193 if self._use_caching:
194 # Only update if caching is used.
195 # Compute hash sum which will be used to
196 # check cache of fparser tree
197 self._source_code_hash_sum = hashlib.md5(
198 self._source_code.encode()
199 ).hexdigest()
200
201 return self._source_code

203 def _cache_load(
204 self,
205 verbose: bool = False,
206 ) -> _CacheFileInfo:
207 """Load fparser parse tree from the cache file if possible.
208 This also checks for matching checksums after loading the data
209 from the cache.
210 The checksum is based solely on a hashsum of the source code itself,
211 see code below.
212
213 :param verbose: Produce some verbose output
214 """
215
216 if not self._use_caching:
217 return
218
219 # Load the source code in case it's not yet loaded.
220 # This also fills in the hash sum
221 self.get_source_code()
222
223 # Check whether cache was already loaded
224 if self._cache_data_load is not None:
225 return self._cache_data_load
226
227 # Load cache file.
228 # Warning: There could be race conditions, e.g., in parallel builds.
229 # In the worst case some content is read which is incomplete or
230 # basically garbage. This will lead either to an Exception from the
231 # unpickling or a non-matching checksum which is both caught below.
232 try:
233 filehandler = open(self._filepath_cache, "rb")
234 except FileNotFoundError:
235 if verbose:
236 # TODO #11: Use logging for this
237 print(f" - No cache file '{self._filepath_cache}' found")
238 return None
239
240 # Unpack cache file
241 try:
242 cache: _CacheFileInfo = pickle.load(filehandler)
243 except Exception as ex:
244 print(f" - Error while reading cache file - ignoring: {str(ex)}")
245 return None
246
247 # Verify checksums
248 if cache._source_code_hash_sum != self._source_code_hash_sum:
249 if verbose:
250 # TODO #11: Use logging for this
251 print(
252 f" - Cache hashsum mismatch: "
253 f"source {self._source_code_hash_sum} "
254 f"vs. cache {cache._source_code_hash_sum}"
255 )
256 return None
257
258 self._cache_data_load = cache

260 def _cache_save(
261 self,
262 verbose: bool = False,
263 ) -> None:
264 """Save the following elements to a cache file:
265 - hash sum of code
266 - fparser tree
267 - in future work, potentially also psyir nodes too
268 (requires TODO #2786).
269
270 :param verbose: Produce some verbose output
271 """
272
273 if not self._use_caching:
274 return None
275
276 if self._source_code_hash_sum is None:
277 # Nothing to cache
278 return None
279
280 cache_updated = False
281 if self._cache_data_save_cache_data_save is None:
282 # Cache doesn't exist => prepare data to write to file
283 self._cache_data_save_cache_data_save = _CacheFileInfo()
284 self._cache_data_save_cache_data_save._source_code_hash_sum = (
285 self._source_code_hash_sum)
286
287 if (
288 self._cache_data_save_cache_data_save._fparser_tree is None
289 and self._fparser_tree_fparser_tree is not None
290 ):
291 # No fparser tree was loaded so far into the cache object
292 # AND
293 # an fparser tree was loaded and stored to Fileinfo.
294 #
295 # Consequently, we cache the fparser tree to the cache file.
296 # We create a deepcopy of this fparser tree to ensure that we cache
297 # the original tree and not a modified fparser tree node if the
298 # cache is updated (based on potentially future work of also
299 # caching the PSyIR)
300
301 self._cache_data_save_cache_data_save._fparser_tree = \
302 copy.deepcopy(self._fparser_tree_fparser_tree)
303 cache_updated = True
304
305 if self._cache_data_save_cache_data_save._psyir_node is None and (
306 self._psyir_node_psyir_node is not None):
307 # TODO #2786: Serialization of psyir tree not possible
308 #
309 # E.g., this call fails: copy.deepcopy(self._psyir_node)
310 #
311 # Uncomment this code if serialization of psyir tree is
312 # possible and it will work.
313 # self._cache._psyir_node = copy.deepcopy(self._psyir_node)
314 # cache_updated = True
315 pass
316
317 if not cache_updated:
318 return None
319
320 # Open cache file
321 try:
322 # Atomically attempt to open the new kernel file (in case
323 # this is part of a parallel build)
324 # We first remove the cache file and then open it.
325 # If the file exists, it throws an exception.
326 # This is not a perfect solution, but avoids parallel
327 # writing access of the same file.
328
329 # We first remove a potentially existing file
330 try:
331 os.remove(self._filepath_cache)
332 except FileNotFoundError:
333 pass
334
335 # Then we open it in exclusive mode.
336 # If it already exists, an exception would be raised.
337 fd = os.open(self._filepath_cache,
338 os.O_CREAT | os.O_WRONLY | os.O_EXCL)
339
340 filehandler = os.fdopen(fd, "wb")
341 except Exception as err:
342 if verbose:
343 # TODO #11: Use logging for this
344 print(" - Unable to write to cache file: " + str(err))
345 return None
346
347 # Dump to cache file
348 try:
349 pickle.dump(self._cache_data_save_cache_data_save, filehandler)
350 except Exception as err:
351 # Invalidate cache
352 self._cache_data_save_cache_data_save = None
353 print("Error while storing cache data - ignoring: " + str(err))
354 return None
355
356 if verbose:
357 print(
358 f" - Cache file updated with "
359 f"hashsum '{self._cache_data_save._source_code_hash_sum}"
360 )

362 def get_fparser_tree(
363 self,
364 verbose: bool = False,
365 save_to_cache_if_cache_active: bool = True
366 ) -> Fortran2003.Program:
367 """Returns the fparser Fortran2003.Program representation of the
368 source code (including Fortran2008).
369
370 :param save_to_cache_if_cache_active: Cache is updated if fparser was
371 not loaded from cache.
372 :param verbose: Produce some verbose output
373
374 :returns: fparser representation.
375
376 :raises FileInfoFParserError: if fparser had issues
377 """
378 if self._fparser_tree_fparser_tree is not None:
379 return self._fparser_tree_fparser_tree
380
381 if self._fparser_tree_triggers_error_fparser_tree_triggers_error:
382 # Raises an exception if we were not able to create the
383 # fparser tree before.
384 raise FileInfoFParserError(
385 "Failed to create fparser tree (previous attempt failed)"
386 )
387
388 if verbose:
389 # TODO #11: Use logging for this
390 print(f"- Source file '{self._filename}': " f"Running fparser")
391
392 try:
393 source_code = self.get_source_code()
394 except FileNotFoundError as err:
395 raise FileInfoFParserError(
396 f"File '{self._filename}' not found:\n{str(err)}")
397
398 # Check for cache
399 self._cache_load(verbose=verbose)
400
401 if self._cache_data_load is not None:
402 if self._cache_data_load._fparser_tree is not None:
403 if verbose:
404 # TODO #11: Use logging for this
405 print(
406 f" - Using cache of fparser tree with hashsum"
407 f" {self._cache_data_load._source_code_hash_sum}"
408 )
409
410 # Use cached version
411 self._fparser_tree_fparser_tree = self._cache_data_load._fparser_tree
412 return self._fparser_tree_fparser_tree
413
414 try:
415 config = Config.get()
416 reader = FortranStringReader(
417 source_code, include_dirs=config.include_paths
418 )
419 parser = ParserFactory().create(std=config.fortran_standard)
420 self._fparser_tree_fparser_tree = parser(reader)
421
422 except Exception as err:
423 self._fparser_tree_triggers_error_fparser_tree_triggers_error = True
424 raise FileInfoFParserError(
425 "Failed to create fparser tree: " + str(err)
426 ) from err
427
428 # We directly call the cache saving routine here in case that the
429 # fparser tree will be modified later on.
430 if save_to_cache_if_cache_active:
431 self._cache_save(verbose=verbose)
432
433 return self._fparser_tree_fparser_tree
