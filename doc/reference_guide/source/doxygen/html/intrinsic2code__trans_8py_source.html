51class Intrinsic2CodeTrans(Transformation, metaclass=abc.ABCMeta):
52 '''Provides support for transformations from PSyIR IntrinsicCall
53 nodes to equivalent PSyIR code in a PSyIR tree. Such
54 transformations can be useful when the intrinsic is not supported
55 by a particular backend or if it is more efficient to have
56 explicit code.
57
58 '''
59 def __init__(self):
60 super().__init__()
61 self._intrinsic = None
62
63 def __str__(self):
64 return (f"Convert the PSyIR '{self._intrinsic.name}' "
65 f"intrinsic to equivalent PSyIR code.")
66
67 def validate(self, node, options=None):
68 '''Perform various checks to ensure that it is valid to apply
69 an intrinsic transformation to the supplied Node.
70
71 :param node: the node that is being checked.
72 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
73 :param options: a dictionary with options for transformations.
74 :type options: Optional[Dict[str, Any]]
75
76 :raises TransformationError: if the node argument is not the \
77 expected type.
78 :raises TransformationError: if the IntrinsicCall node does \
79 not have an Assignment Node as an ancestor.
80
81 '''
82 # Check that the node is one of the expected types.
83 if not isinstance(node, IntrinsicCall):
84 raise TransformationError(
85 f"Error in {self.name} transformation. The supplied node must "
86 f"be an 'IntrinsicCall', but found '{type(node).__name__}'.")
87 if node.intrinsic != self._intrinsic:
88 raise TransformationError(
89 f"Error in {self.name} transformation. The supplied "
90 f"IntrinsicCall must be a '{self._intrinsic.name}' but found: "
91 f"'{node.intrinsic.name}'.")
92 # Check that there is an Assignment node that is an ancestor
93 # of this Operation.
94 if not node.ancestor(Assignment):
95 raise TransformationError(
96 f"Error in {self.name} transformation. This transformation "
97 f"requires the operator to be part of an assignment "
98 f"statement, but no such assignment was found.")
99
100 def _validate_scalar_arg(self, node, options=None):
101 '''
102 Check that the argument to the intrinsic is a scalar of known type.
103
104 :param node: the target intrinsic call.
105 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
106 :param options: any options for the transformation.
107 :type options: dict[str, Any]
108
109 :raises TransformationError: if the supplied SIGN call operates on
110 an argument of array type or unsupported/unresolved type.
111
112 '''
113 result_type = node.arguments[0].datatype
114 if isinstance(result_type, ArrayType):
115 raise TransformationError(
116 f"Transformation {self.name} cannot be applied to SIGN calls "
117 f"which have an array as argument but "
118 f"'{node.arguments[0].debug_string()}' is of array type. It "
119 f"may be possible to use the ArrayAssignment2LoopsTrans "
120 f"to convert this to a scalar argument.")
121 if not isinstance(result_type, ScalarType):
122 raise TransformationError(
123 f"Transformation {self.name} cannot be applied to "
124 f"'{node.debug_string()} because the type of the "
125 f"argument '{node.arguments[0].debug_string()}' is "
126 f"{result_type}")
