60def generate_lfric_adjoint(tl_psyir, active_variables):
61 '''Takes an LFRic tangent-linear kernel represented in language-level PSyIR
62 and returns its adjoint represented in language-level PSyIR.
63
64 :param tl_psyir: language-level PSyIR containing the LFRic
65 tangent-linear kernel.
66 :type tl_psyir: :py:class:`psyclone.psyir.Node`
67 :param list[str] active_variables: names of the active variables.
68
69 :returns: language-level PSyIR containing the adjoint of the
70 supplied tangent-linear kernel.
71 :rtype: :py:class:`psyclone.psyir.Node`
72
73 :raises InternalError: if the PSyIR does not contain any kernel metadata.
74 :raises InternalError: if the PSyIR does not contain any Routines.
75
76 '''
77 logger = logging.getLogger(__name__)
78
79 # Find the name of the symbol containing the metadata for the tangent-
80 # linear kernel.
81 tl_container = find_container(tl_psyir)
82 for sym in tl_container.symbol_table.datatypesymbols:
83 if (isinstance(sym.datatype, UnsupportedFortranType) and
84 "extends(kernel_type)" in sym.datatype.declaration.lower()):
85 tl_metadata_name = sym.name
86 break
87 else:
88 raise InternalError("The supplied PSyIR does not contain any kernel "
89 "metadata")
90
91 # Translate from TL to AD
92 logger.debug("Translating from TL to AD.")
93 adjoint_visitor = AdjointVisitor(active_variables)
94 ad_psyir = adjoint_visitor(tl_psyir)
95
96 # Re-name the Container for the adjoint code if there is a name
97 # clash. Use the symbol table for the existing TL code so that we
98 # don't accidentally clash with e.g. the name of the kernel
99 # routine.
100 ad_container = find_container(ad_psyir)
101 ad_container.name = ad_container.symbol_table.next_available_name(
102 create_adjoint_name(ad_container.name))
103
104 routines = ad_psyir.walk(Routine)
105 if not routines:
106 raise InternalError("The supplied PSyIR does not contain any "
107 "routines.")
108
109 # TODO issue #1800 if there are multiple
110 # modules/subroutines/program implementations in the file then
111 # raise an exception unless a particular name is specified (by
112 # e.g. command line -kernel_name=xyz. The name can be for a routine
113 # or an interface.)
114
115 # Until we can query the kernel metadata to see whether it points
116 # to a kernel or an interface (issue #1807), we simply assume that we
117 # should allow multiple routines as they imply an interface. We
118 # further assume that the implementation of the routines in the
119 # interface use the same variable names which allows us to
120 # continue to use a single command line list of active
121 # variables. This is the case for the implementations we care
122 # about but in general may not be the case. Issue #1595 should
123 # help fix this problem as it would only be arguments that would
124 # need to have the same names.
125
126 ctr_table = ad_container.symbol_table
127
128 # Re-name the routines that we've adjointed.
129 for routine in routines:
130 adj_kernel_name = create_adjoint_name(routine.name, table=ctr_table)
131 # Renaming the routines updates the symbol names
132 routine.name = adj_kernel_name
133
134 logger.debug("AD LFRic kernel will be named '%s'", routine.name)
135
136 # Now we modify the kernel metadata.
137
138 # Find the argument symbols. These will be used to determine the
139 # metadata index of a particular symbol.
140 arg_symbols = ad_container.children[0].symbol_table.argument_list
141
142 # Raise the PSyIR from generic to LFRic-specific. This gives us
143 # access to the metadata.
144 raise_trans = RaisePSyIR2LFRicKernTrans()
145 raise_trans.apply(
146 ad_psyir, options={"metadata_name": tl_metadata_name})
147
148 # Find the kernel metadata. Get ad_container again as it may have
149 # become invalid when we raised the PSyIR.
150 ad_container = find_container(ad_psyir)
151 metadata = ad_container.metadata
152
153 # Change the type and procedure names
154 metadata.name = create_adjoint_name(metadata.name)
155 if metadata.procedure_name:
156 metadata.procedure_name = create_adjoint_name(metadata.procedure_name)
157 else:
158 # Issue #2236. We are not yet able to to raise multi-precision
159 # metadata to LFRic-specific metadata, so return without
160 # making any further modifications.
161 return ad_psyir
162
163 # Change the meta_args access metadata
164 for var_name in active_variables:
165 access = _update_access_metadata(var_name, arg_symbols, metadata)
166 if access:
167 # Add in any new access symbols.
168 _check_or_add_access_symbol(ad_container, access)
169 return ad_psyir

173def _update_access_metadata(var_name, arg_symbols, metadata):
174 '''If the access metadata for the variable var_name is incorrect then
175 update it.
176
177 :param str var_name: the name of the active variable whose
178 metadata we are going to update if required.
179 :param arg_symbols: a list containing all of the argument symbols
180 that are passed into the associated kernel.
181 :type arg_symbols: List[:py:class:`psyclone.psyir.symbols.DataSymbol`]
182 :param metadata: the LFRic metadata.
183 :type metadata:
184 :py:class:`psyclone.domain.lfric.kernel.LFRicKernelMetadata`
185
186 :returns: the access metadata or None if the active variable is
187 not passed by argument.
188 :rtype: Optional[str]
189
190 :raises GenerationError: if the argument position does not match
191 the metadata.
192 :raises InternalError: if an unexpected metadata type is found.
193 :raises InternalError: if an unexpected access type is found.
194
195 '''
196 # Determine the order of the kernel arguments from the
197 # metadata. This is needed to find the metadata associated with
198 # the specified active variable.
199 meta_arg_index_from_arg_index = ArgIndexToMetadataIndex.mapping(metadata)
200
201 # Determine whether this active variable is passed by argument.
202 found_symbol = None
203 for arg_symbol in arg_symbols:
204 if arg_symbol.name.lower() == var_name.lower():
205 found_symbol = arg_symbol
206 break
207 else:
208 # No, the active variable is not passed by argument.
209 return None
210
211 arg_index = arg_symbols.index(found_symbol)
212 try:
213 meta_arg_index = meta_arg_index_from_arg_index[arg_index]
214 except KeyError as exc:
215 raise GenerationError(
216 f"The position in the kernel subroutine argument list "
217 f"'{arg_index}' of the active variable '{found_symbol.name}' "
218 f"does not match any of the positions expected by the kernel "
219 f"argument ('meta_arg') metadata descriptions. The expected "
220 f"mapping of kernel subroutine argument positions to kernel "
221 f"meta_arg positions is '{meta_arg_index_from_arg_index}'. "
222 f"The most likely reason for this is that the kernel subroutine "
223 f"argument list does not conform to the LFRic rules - perhaps "
224 f"it is a PSyKAl-lite kernel?") from exc
225 meta_arg = metadata.meta_args[meta_arg_index]
226
227 # Determine the intent of this variable from its declaration
228 # and update the metadata appropriately.
229 var_access = found_symbol.interface.access
230 const = LFRicConstants()
231 access = None
232 if var_access == ArgumentInterface.Access.READWRITE:
233 # pylint: disable=unidiomatic-typecheck
234 if type(meta_arg) is ScalarArgMetadata:
235 # TODO #2333 - in LFRic only Builtin kernels are currently allowed
236 # to write to a scalar argument (since this implies a reduction).
237 # We therefore need to flag this case.
238 access = "gh_sum"
239 elif type(meta_arg) in [
240 OperatorArgMetadata, ColumnwiseOperatorArgMetadata]:
241 access = "gh_readwrite"
242 elif type(meta_arg) in [
243 FieldArgMetadata, FieldVectorArgMetadata] and (
244 meta_arg.function_space in
245 const.VALID_DISCONTINUOUS_NAMES):
246 access = "gh_readwrite"
247 elif type(meta_arg) in [FieldArgMetadata, FieldVectorArgMetadata]:
248 access = "gh_inc"
249 else:
250 raise InternalError(
251 f"Found unexpected meta arg class "
252 f"'{type(meta_arg).__name__}'.")
253 elif var_access == ArgumentInterface.Access.WRITE:
254 access = "gh_write"
255 elif var_access == ArgumentInterface.Access.READ:
256 access = "gh_read"
257 else:
258 raise InternalError(
259 f"Found unexpected access '{var_access}' for "
260 f"'{found_symbol.name}'.")
261 meta_arg.access = access
262 return access

265def _check_or_add_access_symbol(container, access):
266 '''Check whether the LFRic access metadata name (e.g. 'gh_write')
267 provided in argument 'access' is already declared in the symbol
268 table and if not add it.
269
270 :param container: the adjoint PSyIR.
271 :type container: :py:class:`psyclone.psyir.nodes.Container`
272 :param str access: the access metadata.
273
274 '''
275 kernel = container.children[0]
276 symbol_table = kernel.symbol_table
277 try:
278 argument_mod_symbol = symbol_table.lookup(access)
279 if not isinstance(argument_mod_symbol.interface, ImportInterface):
280 raise GenerationError(
281 f"The existing symbol '{access}' is not imported from a use "
282 "statement.")
283 if not (argument_mod_symbol.interface.container_symbol.name.lower()
284 == "argument_mod"):
285 raise GenerationError(
286 f"The existing symbol '{access}' is imported from "
287 f"'{argument_mod_symbol.interface.container_symbol.name}' but "
288 f"should be imported from 'argument_mod'.")
289 except KeyError:
290 arg_mod_symbol = symbol_table.find_or_create(
291 "argument_mod", symbol_type=ContainerSymbol)
292 symbol_table = arg_mod_symbol.find_symbol_table(kernel)
293 symbol_table.new_symbol(
294 root_name=access, interface=ImportInterface(arg_mod_symbol))
