54class ArrayOfStructuresMixin(ArrayMixin, StructureAccessorMixin,
55 metaclass=abc.ABCMeta):
56 '''
57 Abstract class that combines the ArrayMixin and the StructureAccessorMixin.
58 As such, it has a member (as child 0) and indices (starting from child 1)
59
60 '''
61 @staticmethod
62 def _validate_child(position, child):
63 '''
64 :param int position: the position to be validated.
65 :param child: a child to be validated.
66 :type child: sub-class of :py:class:`psyclone.psyir.nodes.Node`
67
68 :return: whether the given child and position are valid for this node.
69 :rtype: bool
70
71 '''
72 if position == 0:
73 # The first child must be a Member
74 return isinstance(child, Member)
75 # All subsequent children must be array-index expressions
76 return isinstance(child, (DataNode, Range))
77
78 def index_of(self, node):
79 '''
80 If the given node is one of the index expressions of the array, it
81 returns the zero-indexed dimension of the array that it belongs to.
82 Note that this is different to `node.position` because
83 ArraysOfStructures have a Member child, and it is different from
84 `array.indices.index(node)` because that would use the equality
85 operator, but sibling indices may be equal and provide unexpected
86 results.
87
88 :param node: the node to get the index of.
89 :type node: :py:class:`psyclone.psyir.nodes.Node`
90
91 :returns: the index of the given node in the array.
92 :rtype: int
93
94 :raises ValueError: if node is not an index of the array.
95
96 '''
97 if node.parent is self:
98 return node.position - 1 # -1 to account for the member child
99 raise ValueError(f"'{node}' is not a child of '{self}'")
100
101 @property
102 def indices(self):
103 '''
104 Supports semantic-navigation by returning the list of nodes
105 representing the index expressions for this array reference.
106
107 :returns: the PSyIR nodes representing the array-index expressions.
108 :rtype: list of :py:class:`psyclone.psyir.nodes.Node`
109
110 :raises InternalError: if this node does not have at least two \
111 children.
112
113 '''
114 if len(self._children_children) < 2:
115 raise InternalError(
116 f"{type(self).__name__} malformed or incomplete: must "
117 f"have one or more children representing array-index "
118 f"expressions but found none.")
119 for idx, child in enumerate(self._children_children[1:], start=1):
120 if not self._validate_child_validate_child(idx, child):
121 raise InternalError(
122 f"{type(self).__name__} malformed or incomplete: child "
123 f"{idx} must represent an array-index expression but "
124 f"found '{type(child).__name__}' instead of "
125 f"psyir.nodes.DataNode or Range")
126 return self._children_children[1:]
127
128 def get_signature_and_indices(self):
129 ''':returns: the Signature of this array of structure reference, \
130 and a list of lists of the indices used for each component.
131 :rtype: tuple(:py:class:`psyclone.core.Signature`, list of \
132 lists of indices)
133
134 '''
135 sub_sig, indices = self.children[0].get_signature_and_indices()
136 sig = Signature(self.name)
137 return (Signature(sig, sub_sig), [self.indicesindicesindices]+indices)
