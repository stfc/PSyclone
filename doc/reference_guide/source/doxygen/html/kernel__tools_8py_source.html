37'''
38 This module provides the PSyclone kernel-generation 'run' routine
39 which is intended to be driven from the bin/psyclone-kern executable
40 script. It processes the various command-line options and then calls
41 the appropriate routines to do one of the following:
42
43 1. use the metadata to construct an appropriate Fortran subroutine stub
44 for completion by a kernel developer;
45 2. construct an Algorithm-layer driver program that performs the necessary
46 setup and then calls the supplied kernel with an `invoke`.
47
48'''
49
50import argparse
51import io
52import sys
53import traceback
54
55from psyclone import gen_kernel_stub
56from psyclone.configuration import Config, ConfigurationError, LFRIC_API_NAMES
57from psyclone.domain.lfric.algorithm import LFRicAlg
58from psyclone.errors import GenerationError, InternalError
59from psyclone.line_length import FortLineLength
60from psyclone.parse.utils import ParseError
61from psyclone.psyir.backend.fortran import FortranWriter
62from psyclone.version import __VERSION__
63
64# Dictionary of supported generation modes where values are a brief
65# description of what is created (used when outputting to stdout).
66GEN_MODES = {"alg": "Algorithm code",
67 "stub": "Kernel-stub code"}
68
69
70def run(args):
71 '''
72 Driver for the psyclone-kern tool.
73
74 Parses and checks the command line arguments, calls the appropriate
75 generate function(s) if all is well, catches any errors and outputs the
76 results.
77
78 :param list args: the list of command-line arguments with which \
79 psyclone-kern has been invoked.
80 '''
81 # pylint: disable=too-many-statements,too-many-branches
82
83 # Make sure we have the supported APIs defined in the Config singleton,
84 # but postpone loading the config file till the command line was parsed
85 # in case that the user specifies a different config file.
86 Config.get(do_not_load_file=True)
87
88 # We specify the name of the program so that the expected messages are
89 # produced if running within pytest.
90 parser = argparse.ArgumentParser(
91 prog="psyclone-kern",
92 description='Run the PSyclone kernel generator on a particular file.')
93
94 parser.add_argument('-gen', choices=GEN_MODES.keys(), default="stub",
95 help="what to generate for the supplied kernel "
96 "(alg=algorithm layer, stub=kernel-stub "
97 "subroutine). Defaults to stub.")
98 parser.add_argument('-o', dest='out_file', default=None,
99 help="filename for created code.")
100 parser.add_argument('-api',
101 help=f"choose a particular API from "
102 f"{Config.get().curated_api_list}.")
103 parser.add_argument('filename', help='file containing Kernel metadata.')
104
105 # Make the default an empty list so that we can check whether the
106 # user has supplied a value(s) later
107 parser.add_argument(
108 '-I', '--include', default=[], action="append",
109 help='path to Fortran INCLUDE or module files.')
110 parser.add_argument(
111 '-l', '--limit', dest='limit', default='off',
112 choices=['off', 'all', 'output'],
113 help='limit the Fortran line length to 132 characters (default '
114 '\'%(default)s\'). Use \'all\' to apply limit to both input and '
115 'output Fortran. Use \'output\' to apply line-length limit to output '
116 'Fortran only.')
117
118 parser.add_argument("--config", "-c", help="config file with "
119 "PSyclone specific options.")
120 parser.add_argument(
121 '--version', '-v', action='version',
122 version=f'psyclone-kern version: {__VERSION__}',
123 help=f'display version information ({__VERSION__})')
124
125 args = parser.parse_args(args)
126
127 # If no config file name is specified, args.config is none
128 # and config will load the default config file.
129 Config.get().load(args.config)
130
131 # Check API, if none is specified, take the setting from the config file
132 if args.api is None:
133 # No command line option, use the one specified in Config - which
134 # is either based on a parameter in the config file, or otherwise
135 # the default:
136 api = Config.get().api
137 elif args.api not in Config.get().supported_apis:
138 print(f"Unsupported API '{args.api}' specified. Supported APIs are "
139 f"{Config.get().supported_apis}.", file=sys.stderr)
140 sys.exit(1)
141 else:
142 # There is a valid API specified on the command line. Set it
143 # as API in the config object as well.
144 api = args.api
145 Config.get().api = api
146
147 # The Configuration manager checks that the supplied path(s) is/are
148 # valid so protect with a try
149 try:
150 if args.include:
151 Config.get().include_paths = args.include
152 else:
153 # Default is to instruct fparser2 to look in the directory
154 # containing the file being parsed
155 Config.get().include_paths = ["./"]
156 except ConfigurationError as err:
157 print(str(err), file=sys.stderr)
158 sys.exit(1)
159
160 try:
161 if args.gen == "alg":
162 # Generate algorithm code.
163 if api in LFRIC_API_NAMES:
164 alg_psyir = LFRicAlg().create_from_kernel("test_alg",
165 args.filename)
166 code = FortranWriter()(alg_psyir)
167 else:
168 print(f"Algorithm generation from kernel metadata is "
169 f"not yet implemented for API '{api}'", file=sys.stderr)
170 sys.exit(1)
171 elif args.gen == "stub":
172 # Generate kernel stub
173 code = gen_kernel_stub.generate(args.filename, api=api)
174 else:
175 raise InternalError(f"Expected -gen option to be one of "
176 f"{list(GEN_MODES.keys())} but got {args.gen}")
177
178 except (IOError, ParseError, GenerationError, RuntimeError) as error:
179 print("Error:", error, file=sys.stderr)
180 sys.exit(1)
181
182 except Exception: # pylint: disable=broad-except
183 print("Error, unexpected exception:\n", file=sys.stderr)
184 exc_type, exc_value, exc_traceback = sys.exc_info()
185 print(exc_type, file=sys.stderr)
186 print(exc_value, file=sys.stderr)
187 traceback.print_tb(exc_traceback)
188 sys.exit(1)
189
190 if args.limit != "off":
191 # Apply line-length limiting to the output code.
192 fll = FortLineLength()
193 code_str = fll.process(str(code))
194 else:
195 code_str = str(code)
196
197 if args.out_file:
198 with io.open(args.out_file, mode='w', encoding='utf-8') as fobj:
199 fobj.write(code_str)
200 else:
201 print(f"{GEN_MODES[args.gen]}:\n", code_str, file=sys.stdout)
