67class PSyDataNode(Statement):
68 # pylint: disable=too-many-instance-attributes, too-many-locals
69 '''
70 This class can be inserted into a schedule to instrument a set of nodes.
71 Instrument means that calls to an external library using the PSyData API
72 will be inserted before and after the child nodes, which will give that
73 library access to fields and the fact that a region is executed. This
74 can be used, for example, to add performance profiling calls, in-situ
75 visualisation of data, or for writing fields to a file (e.g. for creating
76 test cases, or using driver to run a certain kernel only). The node
77 allows specification of a class string which is used as a prefix for
78 the PSyData module name (prefix_psy_data_mod) and for the PSyDataType
79 (prefix_PSyDataType).
80
81 :param ast: reference into the fparser2 parse tree corresponding to \
82 this node.
83 :type ast: sub-class of :py:class:`fparser.two.Fortran2003.Base`
84 :param children: the PSyIR nodes that are children of this node. These \
85 will be made children of the child Schedule of this PSyDataNode.
86 :type children: list[:py:class:`psyclone.psyir.nodes.Node`]
87 :param parent: the parent of this node in the PSyIR tree.
88 :type parent: :py:class:`psyclone.psyir.nodes.Node`
89 :param options: a dictionary with options for transformations.
90 :type options: Optional[dict[str, Any]]
91 :param str options["prefix"]: a prefix to use for the PSyData module name \
92 (``prefix_psy_data_mod``) and the PSyDataType \
93 (``prefix_PSyDataType``) - a "_" will be added automatically. \
94 It defaults to "", which means the module name used will just be \
95 ``psy_data_mod``, and the data type ``PSyDataType``.
96 :param tuple[str,str] options["region_name"]: an optional name to \
97 use for this PSyDataNode, provided as a 2-tuple containing a \
98 module name followed by a local name. The pair of strings should \
99 uniquely identify a region unless aggregate information is required \
100 (and is supported by the runtime library).
101
102 :raises InternalError: if a prefix is specified that is not listed in the \
103 configuration file.
104
105 '''
106 #: Textual description of the node.
107 _children_valid_format = "Schedule"
108 _text_name = "PSyData"
109 _colour = "green"
110 #: The default prefix to add to the PSyData module name and PSyDataType
111 _default_prefix = ""
112
113 def __init__(self, ast=None, children=None, parent=None, options=None):
114
115 super().__init__(ast=ast, children=children, parent=parent)
116 if not options:
117 options = {}
118
119 # Store a copy of the options so the node can later access them
120 self._options = options.copy()
121
122 # _prefix stores a prefix to be used with all external PSyData
123 # symbols (i.e. data types and module name), used in the
124 # method 'add_psydata_class_prefix'.
125 prefix = options.get("prefix", self._default_prefix)
126 # Check that the prefix is one of those listed as being supported
127 # in the configuration file. If it *is* listed then it is assumed
128 # that a matching PSyData wrapper library is available at compile time.
129 # See the User Guide for more information:
130 # https://psyclone-dev.readthedocs.io/en/latest/psy_data.html#psy-data
131 if prefix and prefix not in Config.get().valid_psy_data_prefixes:
132 raise InternalError(
133 f"Invalid 'prefix' parameter: found '{prefix}', expected one "
134 f"of {Config.get().valid_psy_data_prefixes} as defined in "
135 f"{Config.get().filename}")
136 if not prefix:
137 self._prefix = ""
138 else:
139 self._prefix = prefix + "_"
140
141 # Create the list of symbol names that will be imported from
142 # the PSyData Fortran module. We use a namedtuple to improve
143 # readability. Currently there is only one imported symbol (the
144 # name of the PSyData derived type) but we keep a list for future
145 # extensibility.
146 _PSyDataSymbol = namedtuple("_PSyDataSymbol", "name symbol_type")
147 self.imported_symbols = [_PSyDataSymbol(self.type_name,
148 DataTypeSymbol)]
149
150 # Root of the name to use for variables associated with
151 # PSyData regions
152 self._psy_data_symbol_with_prefix = \
153 self.add_psydata_class_prefix("psy_data")
154
155 # The name of the PSyData variable that is used for this PSyDataNode
156 self._var_name = ""
157
158 # The region identifier caches the computed module- and region-name
159 # as a tuple of strings. This is required so that a derived class can
160 # query the actual name of a region (e.g. during generation of a driver
161 # for an extract node). If the user does not define a name, i.e.
162 # module_name and region_name are empty, a unique name will be
163 # computed in gen_code() or lower_to_language_level(). If this name was
164 # stored in module_name and region_name, and gen() is called again, the
165 # names would not be computed again, since the code detects already
166 # defined module and region names. This can then result in duplicated
167 # region names: The test 'test_region' in profile_test triggers this.
168 # gen()) is called first after one profile region is applied, then
169 # another profile region is added, and gen() is called again. The
170 # second profile region would compute a new name, which then happens
171 # to be the same as the name computed for the first region in the
172 # first gen_code call (which indeed implies that the name of the
173 # first profile region is different the second time it is computed).
174 # So in order to guarantee that the computed module and region names
175 # are unique when gen_code is called more than once, we
176 # cannot store a computed name in module_name and region_name.
177 self._region_identifier = ("", "")
178 # Name of the region.
179 self._module_name = None
180 self._region_name = None
181
182 # TODO: #1394 Fix code duplication between
183 # PSyDataTrans and this PSyDataNode
184 name = options.get("region_name", None)
185 if name:
186 # pylint: disable=too-many-boolean-expressions
187 if not isinstance(name, tuple) or not len(name) == 2 or \
188 not name[0] or not isinstance(name[0], str) or \
189 not name[1] or not isinstance(name[1], str):
190 raise InternalError(
191 "Error in PSyDataNode. The name must be a "
192 "tuple containing two non-empty strings.")
193 # pylint: enable=too-many-boolean-expressions
194 # Valid PSyData names have been provided by the user.
195 self._module_name = name[0]
196 self._region_name = name[1]
197 self.set_region_identifier(self._module_name,
198 self._region_name)

200 def __eq__(self, other):
201 '''
202 Checks the equality of this PSyDataNode with other. PSyDataNodes are
203 equal if they are the same type, and have the same prefix, var_name,
204 module_name and region_name.
205
206 :param object other: the object to check equality to.
207
208 :returns: whether other is equal to self.
209 :rtype: bool
210
211 '''
212 is_eq = super().__eq__(other)
213 is_eq = is_eq and self.prefix == other.prefix
214 is_eq = is_eq and self.var_name == other.var_name
215 is_eq = is_eq and self.module_name == other.module_name
216 is_eq = is_eq and self.region_name == other.region_name
217 return is_eq

267 @classmethod
268 def create(cls, children, symbol_table, ast=None, options=None):
269 '''
270 Creates a new (sub-class of a) PSyData node with the supplied
271 'children' nodes as its children. The symbols used by the PSyData API
272 are added to the supplied symbol table. This is a class method so that
273 it acts as a factory for the various sub-classes of PSyDataNode.
274
275 :param children: the PSyIR nodes that will become children of the \
276 new PSyData node.
277 :type children: list[:py:class:`psyclone.psyir.nodes.Node`]
278 :param symbol_table: the associated SymbolTable to which symbols \
279 must be added.
280 :type symbol_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
281 :parent ast: reference to fparser2 parse tree for the routine being \
282 instrumented with PSyData calls.
283 :type ast: :py:class:`fparser.two.Fortran2003.Base`
284 :param options: a dictionary with options for transformations.
285 :type options: Optional[dict[str, Any]]
286 :param str options[prefix"]: a prefix to use for the PSyData module \
287 name (``prefix_psy_data_mod``) and the PSyDataType \
288 (``prefix_PSyDataType``) - a "_" will be added automatically. \
289 It defaults to "", which means the module name used will just be \
290 ``psy_data_mod``, and the data type ``PSyDataType``.
291 :param tuple[str,str] options["region_name"]: an optional name to use \
292 for this PSyDataNode, provided as a 2-tuple containing a module \
293 name followed by a local name. The pair of strings should \
294 uniquely identify a region unless aggregate information is \
295 required (and is supported by the runtime library).
296
297 :raises TypeError: if the supplied children or symbol table are not \
298 of the correct type.
299
300 '''
301 if not isinstance(children, list):
302 raise TypeError(f"Error in PSyDataNode.create(). The 'children' "
303 f"argument must be a list (of PSyIR nodes) but "
304 f"got '{type(children).__name__}'")
305 if children and not all(isinstance(child, Node) for child in children):
306 raise TypeError(
307 f"Error in PSyDataNode.create(). The 'children' argument must "
308 f"be a list of PSyIR nodes but it contains: "
309 f"{[type(child).__name__ for child in children]}")
310 if not isinstance(symbol_table, SymbolTable):
311 raise TypeError(
312 f"Error in PSyDataNode.create(). The 'symbol_table' argument "
313 f"must be an instance of psyir.symbols.SymbolTable but got "
314 f"'{type(symbol_table).__name__}'.")
315
316 data_node = cls(ast=ast, options=options)
317 data_node.generate_symbols(symbol_table)
318
319 # A PSyData node always contains a Schedule
320 sched = Schedule(children=children, parent=data_node)
321 data_node.addchild(sched)
322
323 return data_node

325 def generate_symbols(self, symbol_table):
326 ''' Generate the necessary symbols to import and use this PSyDataNode
327 in the provided symbol_table if they don't already exist.
328
329 :param symbol_table: the associated SymbolTable to which symbols
330 must be added.
331 :type symbol_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
332
333 '''
334 # Ensure that we have a container symbol for the API access
335 try:
336 csym = symbol_table.lookup(self.fortran_module)
337 except KeyError:
338 # The symbol doesn't exist which means that we haven't already
339 # added this Container as part of a PSyData transformation.
340 csym = ContainerSymbol(self.fortran_module)
341 symbol_table.add(csym, tag=self.fortran_module)
342
343 if not isinstance(csym, ContainerSymbol):
344 raise InternalError(
345 f"Cannot add PSyData module '{self.fortran_module}' because "
346 f"another Symbol already exists with that name and is a "
347 f"{type(csym).__name__} rather than a ContainerSymbol.")
348
349 # Add the symbols that will be imported from the module. Use the
350 # PSyData names as tags to ensure we don't attempt to add them more
351 # than once if multiple transformations are applied.
352 for sym in self.imported_symbols:
353 existing_sym = symbol_table.lookup(sym.name, otherwise=None)
354 if existing_sym:
355 if not (existing_sym.is_import and
356 existing_sym.interface.container_symbol is csym):
357 raise InternalError(
358 f"Cannot add PSyData symbol '{sym.name}' because it "
359 f"already exists but is not imported from "
360 f"'{csym.name}'")
361 else:
362 symbol_table.find_or_create_tag(
363 sym.name, symbol_type=sym.symbol_type,
364 interface=ImportInterface(csym),
365 datatype=UnresolvedType())
366
367 # Store the name of the PSyData variable that is used for this
368 # PSyDataNode. This allows the variable name to be shown in str
369 # (and also, calling create_name in gen() would result in the name
370 # being changed every time gen() is called).
371 if not self._var_name:
372 self._var_name = symbol_table.next_available_name(
373 self._psy_data_symbol_with_prefix)
374 psydata_type = UnsupportedFortranType(
375 f"type({self.type_name}), save, target :: {self._var_name}")
376 symbol_table.new_symbol(self._var_name, symbol_type=DataSymbol,
377 datatype=psydata_type,
378 visibility=Symbol.Visibility.PRIVATE)

493 def _create_unique_names(self, var_list, symbol_table):
494 '''This function takes a list of (module_name, signature) tuple, and
495 for any name that is already in the symbol table (i.e. creating a
496 name clash), creates a new, unique symbol and adds it to the symbol
497 table with a tag "symbol@module". It returns a list of three-element
498 entries: module name, original_signature, unique_signature. The unique
499 signature and original signature are identical if the original name
500 was not in the symbol table.
501
502 :param var_list: the variable information list.
503 :type var_list: list[tuple[str, :py:class:`psyclone.core.Signature`]]
504 :param symbol_table: the symbol table used to create unique names in
505 :type symbol_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
506
507 :returns: a new list which has a unique signature name added.
508 :rtype: list[tuple[str, :py:class:`psyclone.core.Signature`,
509 :py:class:`psyclone.core.Signature`]]
510
511 '''
512 out_list = []
513 for (module_name, signature) in var_list:
514 if module_name:
515 var_symbol = \
516 symbol_table.find_or_create_tag(tag=f"{signature[0]}"
517 f"@{module_name}",
518 root_name=signature[0])
519 unique_sig = Signature(var_symbol.name, signature[1:])
520 else:
521 # This is a local variable anyway, no need to rename:
522 unique_sig = signature
523 out_list.append((module_name, signature, unique_sig))
524 return out_list

527 def gen_code(self, parent, options=None):
528 # pylint: disable=arguments-differ, too-many-branches
529 # pylint: disable=too-many-statements
530 '''Creates the PSyData code before and after the children
531 of this node.
532
533 TODO #1010: This method and the lower_to_language_level below contain
534 duplicated logic, the gen_code method will be deleted when all APIs can
535 use the PSyIR backends.
536
537 :param parent: the parent of this node in the f2pygen AST.
538 :type parent: :py:class:`psyclone.f2pygen.BaseGen`
539 :param options: a dictionary with options for transformations.
540 :type options: Optional[dict[str, Any]]
541 :param options["pre_var_list"]: container name and variable name to \
542 be supplied before the first child. The container name is \
543 supported to be able to handle variables that are imported from \
544 a different container (module in Fortran).
545 :type options["pre_var_list"]: list[tuple[str, str]]
546 :param options["post_var_list"]: container name and variable name to \
547 be supplied after the last child. The container name is \
548 supported to be able to handle variables that are imported from \
549 a different container (module in Fortran).
550 :type options["post_var_list"]: list[tuple[str, str]]
551 :param str options["pre_var_postfix"]: an optional postfix that will \
552 be added to each variable name in the pre_var_list.
553 :param str options["post_var_postfix"]: an optional postfix that will \
554 be added to each variable name in the post_var_list.
555
556 '''
557 # Avoid circular dependency
558 # pylint: disable=import-outside-toplevel
559 from psyclone.psyGen import Kern, InvokeSchedule
560 invoke = self.ancestor(InvokeSchedule).invoke
561 global_module_name = self._module_name
562 if global_module_name is None:
563 # The user has not supplied a module (location) name so
564 # return the psy-layer module name as this will be unique
565 # for each PSyclone algorithm file.
566 global_module_name = invoke.invokes.psy.name
567
568 region_name = self._region_name
569 if region_name is None:
570 # The user has not supplied a region name (to identify
571 # this particular invoke region). Use the invoke name as a
572 # starting point.
573 region_name = invoke.name
574 kerns = self.walk(Kern)
575 if len(kerns) == 1:
576 # This PSyData region only has one kernel within it,
577 # so append the kernel name.
578 region_name += f"-{kerns[0].name}"
579 # Add a region index to ensure uniqueness when there are
580 # multiple regions in an invoke.
581 psy_data_nodes = self.root.walk(PSyDataNode)
582 # We can't just use .index on the list because we are searching
583 # by identity, not by equality.
584 idx = None
585 for index, node in enumerate(psy_data_nodes):
586 if node is self:
587 idx = index
588 break
589 region_name += f"-r{idx}"
590
591 if not options:
592 options = {}
593
594 # Get the list of variables, and handle name clashes: a now newly
595 # imported symbol (from a module that is used directly or indirectly
596 # from a kernel) might clash with a local variable. Convert the lists
597 # of 2-tuples (module_name, signature) to a list of 3-tuples
598 # (module_name, signature, unique_signature):
599
600 symbol_table = self.scope.symbol_table
601 pre_variable_list = \
602 self._create_unique_names(options.get("pre_var_list", []),
603 symbol_table)
604 post_variable_list = \
605 self._create_unique_names(options.get("post_var_list", []),
606 symbol_table)
607
608 pre_suffix = options.get("pre_var_postfix", "")
609 post_suffix = options.get("post_var_postfix", "")
610 for module_name, signature, unique_signature in (pre_variable_list +
611 post_variable_list):
612 if module_name:
613 if unique_signature != signature:
614 rename = f"{unique_signature[0]}=>{signature[0]}"
615 use = UseGen(parent, module_name, only=True,
616 funcnames=[rename])
617 else:
618 use = UseGen(parent, module_name, only=True,
619 funcnames=[unique_signature[0]])
620 parent.add(use)
621
622 # Note that adding a use statement makes sure it is only
623 # added once, so we don't need to test this here!
624 use = UseGen(parent, self.fortran_module, only=True,
625 funcnames=[sym.name for sym in self.imported_symbols])
626 parent.add(use)
627 # We only set the visibility of this symbol if we are *not* within
628 # a Routine.
629 set_private = self.ancestor(Routine) is None
630 var_decl = TypeDeclGen(parent,
