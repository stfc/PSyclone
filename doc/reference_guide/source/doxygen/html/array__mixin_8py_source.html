88 def index_of(self, node):
89 '''
90 If the given node is one of the index expressions of the array, it
91 returns the zero-indexed dimension of the array that it belongs to.
92 Note that this is different to `node.position` because
93 ArraysOfStructures have a Member child, and it is different from
94 `array.indices.index(node)` because that would use the equality
95 operator, but sibling indices may be equal and provide unexpected
96 results.
97
98 :param node: the node to get the index of.
99 :type node: :py:class:`psyclone.psyir.nodes.Node`
100
101 :returns: the index of the given node in the array.
102 :rtype: int
103
104 :raises ValueError: if node is not an index of the array.
105
106 '''
107 if node.parent is self:
108 return node.position
109 raise ValueError(f"'{node}' is not a child of '{self}'")

127 def _validate_index(self, index):
128 '''Utility function that checks that the supplied index is an integer
129 and is less than the number of array dimensions.
130
131 :param int index: the array index to check.
132
133 :raises TypeError: if the index argument is not an integer.
134 :raises ValueError: if the index value is greater than the \
135 number of dimensions in the array (-1).
136
137 '''
138 if not isinstance(index, int):
139 raise TypeError(
140 f"The index argument should be an integer but found "
141 f"'{type(index).__name__}'.")
142 if index > len(self.indicesindices)-1:
143 raise ValueError(
144 f"In '{type(self).__name__}' '{self.name}' the specified "
145 f"index '{index}' must be less than the number of dimensions "
146 f"'{len(self.indices)}'.")

148 def _is_bound_op(self, expr, bound_operator, index):
149 '''Utility function that checks that the provided 'expr' argument is
150 in the form '[UL]BOUND(array_name, index)', where the type of
151 bound operation is determined by the 'bound_operator'
152 argument, array_name is the name of this array and the 'index'
153 argument provides the index value.
154
155 :param expr: a PSyIR expression.
156 :type expr: :py:class:`psyclone.psyir.nodes.Node`
157 :param bound_operator: the particular bound operation.
158 :type bound_operator:
159 :py:class:`psyclone.psyir.nodes.IntrinsicCall.Intrinsic.LBOUND` |
160 :py:class:`psyclone.psyir.nodes.IntrinsicCall.Intrinsic.UBOUND`
161 :param int index: the bounds index.
162
163 :returns: True if the expr is in the expected form and False otherwise.
164 :rtype: bool
165
166 '''
167 # pylint: disable=import-outside-toplevel
168 from psyclone.psyir.nodes.structure_accessor_mixin import (
169 StructureAccessorMixin)
170 if (isinstance(expr, IntrinsicCall) and
171 expr.intrinsic == bound_operator):
172 array = expr.arguments[0]
173 # If its a structure, we want to compare the whole accessor
174 while isinstance(array, StructureAccessorMixin) and array.member:
175 array = array.member
176 # This is the expected bound
177 if self.is_same_array(array):
178 # The arrays match
179 if (len(expr.arguments) > 1 and
180 isinstance(expr.arguments[1], Literal) and
181 expr.arguments[1].datatype.intrinsic ==
182 ScalarType.Intrinsic.INTEGER
183 and expr.arguments[1].value == str(index+1)):
184 # This is the correct index
185 return True
186 return False

196 the array is accessed in this array reference for the \

203 def _get_bound_expression(self, pos, bound):
204 '''
205 Lookup the upper or lower bound of this ArrayMixin.
206
207 :param int pos: the dimension of the array for which to lookup the
208 lower bound.
209 :param str bound: "upper" or "lower" - the bound which to lookup.
210
211 :returns: the declared bound for the specified dimension of this array
212 or a call to the {U/L}BOUND intrinsic if it is unknown.
213 :rtype: :py:class:`psyclone.psyir.nodes.Node`
214
215 :raises InternalError: if bound is neither upper or lower.
216 '''
217 if bound not in ("upper", "lower"):
218 raise InternalError(f"'bound' argument must be 'lower' or 'upper. "
219 f"Found '{bound}'.")
220
221 # First, walk up to the parent reference and get its type. For a simple
222 # ArrayReference this will just be self.
223 root_ref = self.ancestor(Reference, include_self=True)
224 if isinstance(root_ref.symbol, DataSymbol):
225 cursor_type = root_ref.symbol.datatype
226 else:
227 cursor_type = UnresolvedType()
228
229 # Walk back down the structure, looking up the type information as we
230 # go. We also collect the necessary information for creating a new
231 # Reference as argument to the {U/L}BOUND intrinsic in case the type
232 # information is not available.
233 cnames = []
234 cursor = root_ref
235 while cursor is not self:
236 cursor = cursor.member
237 # Collect member information.
238 if isinstance(cursor, ArrayMixin):
239 new_indices = [idx.copy() for idx in cursor.indices]
240 cnames.append((cursor.name.lower(), new_indices))
241 else:
242 cnames.append(cursor.name.lower())
243 # Continue to resolve datatype unless we hit an
244 # UnsupportedType or UnresolvedType.
245 if isinstance(cursor_type, ArrayType):
246 cursor_type = cursor_type.intrinsic
247 if isinstance(cursor_type, DataTypeSymbol):
248 cursor_type = cursor_type.datatype
249 if isinstance(cursor_type, (UnsupportedType, UnresolvedType)):
250 continue
251 cursor_type = cursor_type.components[cursor.name.lower()].datatype
252
253 if (isinstance(cursor_type, ArrayType) and
254 cursor_type.shape[pos] not in [ArrayType.Extent.DEFERRED,
255 ArrayType.Extent.ATTRIBUTE]):
256 # We have the full type information and the bound is known.
257 if bound == "lower":
258 return cursor_type.shape[pos].lower.copy()
259 return cursor_type.shape[pos].upper.copy()
260
261 # We've either failed to resolve the type or we don't know the extent
262 # of the array dimension so construct a call to the BOUND intrinsic.
263 if cnames:
264 # We have some sort of structure access - remove any indexing
265 # information from the ultimate member of the structure access.
266 if len(cnames[-1]) == 2:
267 cnames[-1] = cnames[-1][0]
268 # Have to import here to avoid circular dependencies.
269 # pylint: disable=import-outside-toplevel
270 from psyclone.psyir.nodes import (ArrayOfStructuresReference,
271 StructureReference)
272 if isinstance(root_ref, ArrayMixin):
273 new_indices = [idx.copy() for idx in root_ref.indices]
274 ref = ArrayOfStructuresReference.create(
275 root_ref.symbol, new_indices, cnames)
276 else:
277 ref = StructureReference.create(root_ref.symbol, cnames)
278 else:
279 # A simple Reference.
280 ref = Reference(root_ref.symbol)
281
282 if bound == "lower":
283 return IntrinsicCall.create(
284 IntrinsicCall.Intrinsic.LBOUND,
285 [ref, ("dim", Literal(str(pos+1), INTEGER_TYPE))])
286 return IntrinsicCall.create(
287 IntrinsicCall.Intrinsic.UBOUND,
288 [ref, ("dim", Literal(str(pos+1), INTEGER_TYPE))])

329 '''

362 def _is_bound(self, index, bound_type):
363 '''Attempts to determines whether this array access includes the lower
364 or upper bound (as specified by the bound_type argument).
365
366 Checks whether the specified array index contains a Range node
367 which has a starting/stopping value given by the
368 '{LU}BOUND(name,index)' intrinsic where 'name' is the name of
369 the current ArrayReference and 'index' matches the specified
370 array index. Also checks if the starting/stopping value of the
371 access matches the lower/upper value of the declaration.
372
373 For example, if a Fortran array A was declared as A(n) then
374 the stopping value is n and A(:UBOUND(A,1)), A(:n) or A(n)
375 would access that value. The starting value is 1 and
376 A(LBOUND(A,1):), A(1:) or A(1) would access that value.
377
378 :param int index: the array index to check.
379 :param str bound_type: the type of bound ("lower" or "upper")
380
381 :returns: True if the array index access includes the \
382 lower/upper bound of the array declaration and False if it \
383 does not or if it can't be determined.
384 :rtype: bool
385
386 '''
387 self._validate_index(index)
388
389 access_shape = self.indicesindices[index]
390
391 # Determine the appropriate (lower or upper) bound and check
392 # for a bounds intrinsic.
393 if isinstance(access_shape, Range):
394 if bound_type == "upper":
395 intrinsic = IntrinsicCall.Intrinsic.UBOUND
396 access_bound = access_shape.stop
397 else:
398 intrinsic = IntrinsicCall.Intrinsic.LBOUND
399 access_bound = access_shape.start
400
401 # Is this array access in the form of {UL}BOUND(array, index)?
402 if self._is_bound_op(access_bound, intrinsic, index):
403 return True
404 else:
405 access_bound = access_shape
406
407 # Try to compare the upper/lower bound of the array access
408 # with the upper/lower bound of the array declaration.
409
410 # Finding the array declaration is only supported for an
411 # ArrayReference at the moment.
412 # Import here to avoid circular dependence.
413 # pylint: disable=import-outside-toplevel
414 from psyclone.psyir.nodes.array_reference import ArrayReference
415 if not isinstance(self, ArrayReference):
416 return False
417 # pylint: enable=import-outside-toplevel
418
419 symbol = self.symbol
420 if not isinstance(symbol, DataSymbol):
421 # There is no type information for this symbol
422 # (probably because it originates from a wildcard import).
423 return False
424 datatype = symbol.datatype
425
426 if not isinstance(datatype, ArrayType):
427 # The declaration datatype could be of UnsupportedFortranType
428 # if the symbol is of e.g. character type.
429 return False
430
431 # The bound of the declaration is available.
432
433 if isinstance(datatype.shape[index], ArrayType.Extent):
434 # The size is unspecified at compile-time (but is
435 # available at run-time e.g. when the size is allocated by
436 # an allocate statement).
437 return False
438
439 # The size of the bound is available.
440 if bound_type == "upper":
441 declaration_bound = datatype.shape[index].upper
442 if isinstance(declaration_bound, ArrayType.Extent):
443 # But only at run-time.
444 return False
445 else:
446 declaration_bound = datatype.shape[index].lower
447
448 # Do the bounds match?
449 sym_maths = SymbolicMaths.get()
450 return sym_maths.equal(declaration_bound, access_bound)

452 def is_same_array(self, node):
453 '''
454 Checks that the provided array is the same as this node (including the
455 chain of parent accessor expressions if the array is part of a
456 Structure). If the array is part of a structure then any indices on
457 the innermost member access are ignored, e.g.
458 A(3)%B%C(1) will match with A(3)%B%C but not with A(2)%B%C(1)
459
460 :param node: the node representing the access that is to be compared \
461 with this node.
462 :type node: :py:class:`psyclone.psyir.nodes.Reference` or \
463 :py:class:`psyclone.psyir.nodes.Member`
464
465 :returns: True if the array accesses match, False otherwise.
466 :rtype: bool
467
468 '''
469 if not isinstance(node, (Member, Reference)):
470 return False
471
472 # First check that the base and depths are the same
473 if isinstance(self, Member):
474 self_base, depth = self.get_base_and_depth()
475 else:
476 self_base, depth = self, 0
477 if isinstance(node, Member):
478 node_base, node_depth = node.get_base_and_depth()
479 else:
480 node_base, node_depth = node, 0
481 if (not isinstance(self_base, Reference) or
482 not isinstance(node_base, Reference)):
483 return False
484 if self_base.symbol != node_base.symbol or depth != node_depth:
485 return False
486
487 # Then use the signatures to compare that each member until
488 # depth are also the same
489 self_sig, self_indices = self_base.get_signature_and_indices()
490 node_sig, node_indices = node_base.get_signature_and_indices()
491 if self_sig[:depth+1] != node_sig[:depth+1]:
492 return False
493
494 # Examine the indices, ignoring any on the innermost accesses (hence
495 # the slice to `depth` rather than `depth + 1` below).
496 for idx1, idx2 in zip(self_indices[:depth], node_indices[:depth]):
497 if idx1 != idx2:
498 return False
499 return True

501 def is_full_range(self, index):
502 '''Returns True if the specified array index is a Range Node that
503 specifies all elements in this index. In the PSyIR this is
504 specified by using LBOUND(name,index) for the lower bound of
505 the range, UBOUND(name,index) for the upper bound of the range
506 and "1" for the range step.
507
508 :param int index: the array index to check.
509
510 :returns: True if the access to this array index is a range \
511 that specifies all index elements. Otherwise returns \
512 False.
513 :rtype: bool
514
515 '''
516 self._validate_index(index)
517
518 array_dimension = self.indicesindices[index]
519 if isinstance(array_dimension, Range):
520 if self.is_lower_bound(index) and self.is_upper_bound(index):
521 step = array_dimension.children[2]
522 if (isinstance(step, Literal) and
523 step.datatype.intrinsic == ScalarType.Intrinsic.INTEGER
524 and str(step.value) == "1"):
525 return True
526 return False

534 :returns: the PSyIR nodes representing the array-index expressions.

555 def _extent(self, idx):
556 '''
557 Create PSyIR for the number of elements in dimension `idx` of this
558 array access. It is given by (stop - start)/step + 1 or, if it is for
559 the full range, by the SIZE intrinsic.
560
561 :param int idx: the array index for which to compute the number of
562 elements.
563
564 :returns: the PSyIR expression for the number of elements in the
565 specified array index.
566 :rtype: :py:class:`psyclone.psyir.nodes.BinaryOperation` |
567 :py:class:`psyclone.psyir.nodes.IntrinsicCall`
568 '''
569 expr = self.indicesindices[idx]
570 one = Literal("1", INTEGER_TYPE)
571
572 if isinstance(expr, Range):
573 start = expr.start
574 stop = expr.stop
575 step = expr.step
576 else:
577 # No range so just a single element is accessed.
578 return one
579
580 if (isinstance(start, IntrinsicCall) and
581 isinstance(stop, IntrinsicCall) and self.is_full_range(idx)):
582 # Access is to full range and start and stop are expressed in terms
583 # of LBOUND and UBOUND. Therefore, it's simpler to use SIZE.
584 return IntrinsicCall.create(
585 IntrinsicCall.Intrinsic.SIZE,
586 [start.arguments[0].copy(),
587 ("dim", Literal(str(idx+1), INTEGER_TYPE))])
588
589 if start == one and step == one:
590 # The range starts at 1 and the step is 1 so the extent is just
591 # the upper bound.
592 return stop.copy()
593
594 extent = BinaryOperation.create(BinaryOperation.Operator.SUB,
595 stop.copy(), start.copy())
596 if step != one:
597 # Step is not unity so have to divide range by it.
598 result = BinaryOperation.create(BinaryOperation.Operator.DIV,
599 extent, step.copy())
600 else:
601 result = extent
602 # Extent is currently 'stop-start' or '(stop-start)/step' so we have
603 # to add a '+ 1'
604 return BinaryOperation.create(BinaryOperation.Operator.ADD,
605 result, one.copy())

607 def _get_effective_shape(self):
608 '''
609 :returns: the shape of the array access represented by this node.
610 :rtype: list[:py:class:`psyclone.psyir.nodes.DataNode`]
611
612 :raises NotImplementedError: if any of the array-indices involve a
613 function call or are of unknown type.
614 '''
615 shape = []
616 for idx, idx_expr in enumerate(self.indicesindices):
617 if isinstance(idx_expr, Range):
618 shape.append(self._extent(idx))
619
620 elif isinstance(idx_expr, (Reference, Operation)):
621 dtype = idx_expr.datatype
622 if isinstance(dtype, ArrayType):
623 # An array slice can be defined by a 1D slice of another
624 # array, e.g. `a(b(1:4))` or `a(b)`.
625 indirect_array_shape = dtype.shape
626 if len(indirect_array_shape) > 1:
627 raise NotImplementedError(
628 f"An array defining a slice of a dimension of "
629 f"another array must be 1D but '{idx_expr.name}' "
630 f"used to index into '{self.name}' has "
631 f"{len(indirect_array_shape)} dimensions.")
632 # pylint: disable=protected-access
633 if isinstance(idx_expr, ArrayMixin):
634 shape.append(idx_expr._extent(idx))
635 else:
636 # We have some expression with a shape but no explicit
637 # indexing. The extent of this is then the SIZE of
638 # that array (expression).
639 sizeop = IntrinsicCall.create(
640 IntrinsicCall.Intrinsic.SIZE, [idx_expr.copy()])
641 shape.append(sizeop)
642 elif isinstance(dtype, (UnsupportedType, UnresolvedType)):
643 raise NotImplementedError(
644 f"The array index expression "
645 f"'{idx_expr.debug_string()}' in access "
646 f"'{self.debug_string()}' is of '{dtype}' type and "
647 f"therefore whether it is an array slice (i.e. an "
648 f"indirect access) cannot be determined.")
649 elif isinstance(idx_expr, (Call, CodeBlock)):
650 # We can't yet straightforwardly query the type of a function
651 # call - TODO #1799.
652 raise NotImplementedError(
653 f"The array index expressions for access "
654 f"'{self.debug_string()}' include a function call or "
655 f"unsupported feature. Querying the return type of "
656 f"such things is yet to be implemented.")
657
658 return shape

675 def same_range(self, index: int, array2, index2: int) -> bool:
676 ''' This method compares the range of this array node at a given index
677 with the range of a second array at a second index. This is useful to
678 verify if array operations are valid, e.g.: A(3,:,5) + B(:,2,2).
679
680 Note that this check supports symbolic comparisons, e.g.:
681 A(3:4) has the same range as B(2+1:5-1),
682 and will consider compile-time unknown dimensions as equal, e.g.:
683 A(:) has the same range as B(:).
684
685 TODO #2485. This method has false negatives: cases when the range
686 is the same but it can not be proved, so we return False.
687
688 TODO #2004. This method currently compares exact ranges, not just the
689 length of them, which could be done with "(upper-lower)/step" symbolic
690 comparisons. This is because arrayrange2loop does not account for
691 arrays declared with different lbounds, but this could be improved.

747 # The logic below assumes array expressions come from valid Fortran,
748 # (e.g. a(2:4) = b(2:5) is not valid Fortran)
749 # and therefore, we assume the length of the equivalent ranges in the
750 # same statement matches:
751 # e.g. a(3, :) and b(4:, 4)
752 # a dim 2 and b dim 1 must have the same length, but not necessarily
753 # the same range (lower and upper bounds)
754 n_range1 = len([x for x in self.children[:index]
755 if isinstance(x, Range)])
756 n_range2 = len([x for x in array2.children[:index2]
757 if isinstance(x, Range)])
758 assume_same_length = (
759 (self.ancestor(Assignment) is array2.ancestor(Assignment) or

761 )

764 # that these have the shape attribute)
765 array1_type = None
766 if (isinstance(self, Reference) and isinstance(self.symbol, DataSymbol)
767 and isinstance(self.symbol.datatype, ArrayType)):

770 if (isinstance(array2, Reference) and

773 array2_type = array2.symbol.datatype

776 range1_start = None
777 range2_start = None

779 # we need to prove that they have the same lower bound value on the
780 # declaration. For example

785 if self.is_same_array(array2) and array2.is_lower_bound(index2):

787 if not array1_type:

792 range1_start = Literal("1", INTEGER_TYPE)
793 else:

801 if array2_type.shape[index2] == ArrayType.Extent.ATTRIBUTE:

807 if not range1_start:
808 range1_start = range1.start
809 if not range2_start:
810 range2_start = range2.start

812 # Now we can do a symbolic comparison of the start values
813 if not sym_maths.equal(range1_start, range2_start):
814 return False

816 # If we can not guarantee the same length, we also need to check
817 # the upper bound
818 if not assume_same_length:
819 if self.is_upper_bound(index):
820 if array1_type:
821 range1_stop = array1_type.shape[index].upper
822 else:
823 return False
824 else:
825 range1_stop = range1.stop
826
827 if array2.is_upper_bound(index2):
828 if array2_type:
829 range2_stop = array2_type.shape[index2].upper
830 else:
831 return False
832 else:
833 range2_stop = range2.stop
834
835 if not sym_maths.equal(range1_stop, range2_stop):
836 return False
837
838 # Compare steps
839 if not sym_maths.equal(range1.step, range2.step):
840 return False
841
842 # Everything matches.
843 return True
