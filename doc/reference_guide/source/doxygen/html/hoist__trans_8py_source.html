37'''This module contains the HoistTrans transformation. HoistTrans
38moves an assignment out of a parent loop if it is safe to do so. Hoist
39is a name that is often used to describe this type of transformation.
40
41'''
42

51class HoistTrans(Transformation):
52 '''This transformation takes an assignment and moves it outside of
53 its parent loop if it is valid to do so. If as a result the loop body
54 becomes empty, the loop will be removed altogether. For example:
55
56 >>> from psyclone.psyir.backend.fortran import FortranWriter
57 >>> from psyclone.psyir.frontend.fortran import FortranReader
58 >>> from psyclone.psyir.nodes import Assignment
59 >>> from psyclone.psyir.transformations import HoistTrans
60 >>> code = ("program test\\n"
61 ... " integer :: i,j,n\\n"
62 ... " real :: a(n,n)\\n"
63 ... " real value\\n"
64 ... " do i=1,n\\n"
65 ... " value = 1.0\\n"
66 ... " do j=1,n\\n"
67 ... " a(i,j) = value\\n"
68 ... " end do\\n"
69 ... " end do\\n"
70 ... "end program\\n")
71 >>> psyir = FortranReader().psyir_from_source(code)
72 >>> hoist = HoistTrans()
73 >>> hoist.apply(psyir.walk(Assignment)[0])
74 >>> print(FortranWriter()(psyir))
75 program test
76 integer :: i
77 integer :: j
78 integer :: n
79 real, dimension(n,n) :: a
80 real :: value
81 <BLANKLINE>
82 value = 1.0
83 do i = 1, n, 1
84 do j = 1, n, 1
85 a(i,j) = value
86 enddo
87 enddo
88 <BLANKLINE>
89 end program test
90 <BLANKLINE>
91
92 '''
93 def apply(self, node, options=None):
94 '''Applies the hoist transformation to the supplied assignment node
95 within a loop, moving the assignment outside of the loop if it
96 is valid to do so. Issue #1445 will also look to extend this
97 transformation to other types of node.
98
99 :param node: target PSyIR node.
100 :type node: subclass of :py:class:`psyclone.psyir.nodes.Assignment`
101 :param options: a dictionary with options for transformations.
102 :type options: Optional[Dict[str, Any]]
103
104 '''
105 self.validatevalidate(node, options)
106
107 # Find the enclosing loop (the validate() method has already
108 # verified that there is one).
109 loop = node.ancestor(Loop)
110
111 # Remove the assignment node
112 node.detach()
113
114 # Place the assignment node before the loop.
115 loop.parent.children.insert(loop.position, node)
116
117 # Remove loop if no more children are left.
118 if not loop.loop_body.children:
119 loop.detach()

121 def validate(self, node, options=None):
122 '''Checks that the supplied node is a valid target for a hoist
123 transformation. At this stage only an assignment statement is
124 allowed to be hoisted, see #1445. It should also be tested if
125 there is a directive outside of the loop, see #1446
126
127 :param node: target PSyIR node.
128 :type node: subclass of :py:class:`psyclone.psyir.nodes.Assignment`
129 :param options: a dictionary with options for transformations.
130 :type options: Optional[Dict[str, Any]]
131
132 :raises TransformationError: if the supplied node is not an \
133 assignment.
134 :raises TransformationError: if the assignment is not within a \
135 loop.
136 :raises TransformationError: if the assignment is not a direct \
137 child of the the loop.
138
139 '''
140 # The node should be an assignment
141 if not isinstance(node, Assignment):
142 raise TransformationError(
143 f"The target of the HoistTrans transformation should be an "
144 f"assignment, but found '{type(node).__name__}'.")
145
146 # The assignment should be within a loop.
147 parent_loop = node.ancestor(Loop)
148 if not parent_loop:
149 raise TransformationError(
150 f"The supplied assignment node '{node.debug_string()}' should "
151 f"be within a loop, but no loop was found.")
152
153 # The assignment should be directly within a loop i.e. no other
154 # control logic in between.
155 current = node.parent
156 while current is not parent_loop:
157 if not isinstance(current, Schedule):
158 raise TransformationError(
159 f"The supplied assignment node '{node.debug_string()}' "
160 f"should be directly within a loop but found "
161 f"'{current.debug_string()}'.")
162 current = current.parent
163
164 # Check that the assignment has no side effects
165 for to_check in node.walk((Call, CodeBlock)):
166 if isinstance(to_check, Call) and to_check.is_pure:
167 continue # Pure calls are fine
168 raise TransformationError(
169 f"The supplied assignment should not have side effects, but "
170 f"we can't prove this for '{node.debug_string()}' since it "
171 f"contains Calls or CodeBlocks")
172
173 # Check dependency issues that might prevent hoisting:
174 self._validate_dependencies(node, parent_loop)
175
176 def _validate_dependencies(self, statement, parent_loop):
177 '''Checks if the variable usage allows the specified statement to be
178 hoisted out of the loop, i.e. no dependency on loop variable (directly
179 or indirectly). This validation does not assume that the statement
180 is an assignment, it works with other types of nodes, too. See
181 #1445 for fully supporting other and multiple statements.
182
183 :param statement: the statement that is to be hoisted out of \
184 the loop.
185 :type statement: \
186 subclass of :py:class:`psyclone.psyir.nodes.Assignment`
187 :param parent_loop: the loop out of which the statement is to \
188 be hoisted.
189 :type parent_loop: subclass of :py:class:`psyclone.psyir.nodes.Loop`
190
191 :raises TransformationError: if a variable in the statement is read \
192 and written in the statement (e.g. reduction: a=a+1).
193 :raises TransformationError: if any variable that is written in the \
194 statement is accessed previously.
195 :raises TransformationError: if any variable that is written in the \
196 statement is written more than once in the loop.
197 :raises TransformationError: if the left- or right-hand-side of \
198 the statement depends directly or indirectly on the loop \
199 variable or loop iteration.
200
201 '''
202 # pylint: disable=too-many-locals
203 # Collect all variable usages in the loop
204 all_loop_vars = VariablesAccessInfo(parent_loop)
205
206 # Collect all variables used in the statement that will be hoisted.
207 all_statement_vars = VariablesAccessInfo(statement)
208
209 # Determine the variables which are written (and potentially read)
210 # and which are read-only:
211 read_only_sigs = []
212 write_sigs = []
213 for sig in all_statement_vars.all_data_accesses:
214 if all_statement_vars[sig].is_written():
215 write_sigs.append(sig)
216 else:
217 read_only_sigs.append(sig)
218
219 for written_sig in write_sigs:
220 accesses_in_statement = all_statement_vars[written_sig]
221 # If this written variable is also read in the statement to be
222 # hoisted, we can't hoist (likely a # reduction statement: a=a+1)
223 if accesses_in_statement.is_read():
224 raise TransformationError(f"The statement can't be hoisted as "
225 f"it contains a variable "
226 f"('{written_sig}') that is both "
227 f"read and written.")
228
229 # Check if the variable is written or read before the first
230 # access in the statement to be hoisted:
231 written_node = accesses_in_statement[0].node
232 # Get all access to that variable in the whole loop before the
233 # first write access that is to be hoisted:
234 accesses_in_loop = all_loop_vars[written_sig]
235 if accesses_in_loop.is_accessed_before(written_node):
236 code = statement.debug_string().strip()
237 raise TransformationError(f"The statement '{code}' can't be "
238 f"hoisted as variable "
239 f"'{written_sig}' is accessed "
240 f"earlier within the loop.")
241
242 # Make sure that there is no additional write statement to a
243 # written variable in the loop outside of the statement to be
244 # hoisted. E.g.:
245 # a = 3; b(i) = a; a = 2; c(i) = a
246 # Hoisting any of the assignments to 'a' out is invalid.
247 # This is done by counting the write accesses to the variable
248 # in the loop and in the statement.
249 writes_in_loop = sum(access.access_type == AccessType.WRITE
250 for access in accesses_in_loop)
251 writes_in_statement = sum(access.access_type == AccessType.WRITE
252 for access in accesses_in_statement)
253 if writes_in_loop > writes_in_statement:
254 raise TransformationError(f"There is at least one additional "
255 f"write to the variable "
256 f"'{written_sig}' in the loop, "
257 f"outside the supplied statement.")
258
259 # Now check if any variable read in the statement to be hoisted is
260 # being written to somewhere in the loop. This especially includes
261 # the loop variable (which is marked as write-read in the loop
262 # statement). This will create a direct (loop variable) or indirect
263 # dependency (to a value in the loop), which prohibits hoisting.
264
265 for read_sig in read_only_sigs:
266 # Get all access to this variable in the whole loop
267 accesses_in_loop = all_loop_vars[read_sig]
268 if accesses_in_loop.is_written():
269 code = statement.debug_string().strip()
270 raise TransformationError(f"The statement '{code}' can't be "
271 f"hoisted as it reads variable "
272 f"'{read_sig}' which is written "
273 f"somewhere else in the loop.")
274
275 def __str__(self):
276 return "Hoist an assignment outside of its parent loop"
