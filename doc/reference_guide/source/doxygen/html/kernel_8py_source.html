65def get_kernel_filepath(module_name, kernel_paths, alg_filename):
66 '''Search for a kernel module file containing a module with
67 'module_name'. The assumed convention is that the name of the
68 kernel file is the name of the module with .f90 or .F90 appended.
69
70 Look in the directories and all subdirectories associated with the
71 supplied kernel paths or in the same directory as the algorithm
72 file if not found within the kernel paths.
73
74 Return the filepath if the file is found.
75
76 :param str module_name: the name of the module to search for. The \
77 assumption is that the file containing the module will have the \
78 same name as the module name with .f90 or .F90 appended.
79 :param kernel_paths: directories in which to search for the module \
80 file. If nothing is supplied then look in the same directory \
81 as the algorithm file. Directories below the specified \
82 directories are recursively searched.
83 :type list_kernels: list of str
84 :param str alg_filename: the name of the algorithm file. This is \
85 used to determine its directory location if required.
86
87 :returns: a filepath to the file containing the specified module \
88 name.
89 :rtype: str
90
91 :raises ParseError: if the supplied kernel directory does not \
92 exist.
93 :raises ParseError: if the file can not be found.
94 :raises ParseError: if more than one file with the specified name \
95 is found.
96
97 '''
98 # Only consider files with the suffixes .f90 and .F90 when
99 # searching for the kernel source (we perform a case insensitive
100 # match).
101 search_string = f"{module_name}.F90"
102 matches = []
103
104 # If a search path has been specified then look there. Otherwise
105 # look in the directory containing the algorithm definition file.
106 for kernel_path in kernel_paths:
107 # Look for the file in the supplied directory and recursively
108 # in any subdirectories.
109 cdir = os.path.abspath(kernel_path)
110
111 if not os.access(cdir, os.R_OK):
112 raise ParseError(
113 f"kernel.py:get_kernel_filepath: Supplied kernel search path "
114 f"does not exist or cannot be read: {cdir}")
115
116 # Recursively search down through the directory tree starting
117 # at the specified path.
118 for root, _, filenames in os.walk(cdir):
119 for filename in filenames:
120 # perform a case insensitive match
121 if filename.lower() == search_string.lower():
122 matches.append(os.path.join(root, filename))
123 if not kernel_paths:
124 # Look *only* in the directory that contained the algorithm
125 # file.
126 cdir = os.path.abspath(os.path.dirname(alg_filename))
127 filenames = os.listdir(cdir)
128 for filename in filenames:
129 # perform a case insensitive match
130 if filename.lower() == search_string.lower():
131 matches.append(os.path.join(cdir, filename))
132
133 if not matches:
134 # There were no matches.
135 raise ParseError(
136 f"Kernel file '{module_name}.[fF]90' not found in {cdir}")
137 if len(matches) > 1:
138 # There was more than one match
139 raise ParseError(
140 f"kernel.py:get_kernel_filepath: More than one match for kernel "
141 f"file '{module_name}.[fF]90' found!")
142 # There is a single match
143 return matches[0]

146def get_kernel_parse_tree(filepath):
147 '''Parse the file in filepath with fparser1 and return a parse tree.
148
149 :param str filepath: path to a file (hopefully) containing \
150 PSyclone kernel code.
151
152 :returns: Parse tree of the kernel code contained in the specified \
153 file.
154 :rtype: :py:class:`fparser.one.block_statements.BeginSource`
155
156 :raises ParseError: if fparser fails to parse the file
157
158 '''
159 parsefortran.FortranParser.cache.clear()
160
161 # If logging is disable during a sphinx doctest run, doctest will just
162 # stop working. So only disable logging if we are not running doctest.
163 if 'sphinx.ext.doctest' not in sys.modules:
164 fparser.logging.disable(fparser.logging.CRITICAL)
165
166 try:
167 parse_tree = fpapi.parse(filepath)
168 # parse_tree includes an extra comment line which contains
169 # file details. This line can be long which can cause line
170 # length issues. Therefore set the information (name) to be
171 # empty.
172 parse_tree.name = ""
173 except Exception as err:
174 raise ParseError(
175 f"Failed to parse kernel code '{filepath}'. Is the Fortran "
176 f"correct?") from err
177 return parse_tree

180def get_kernel_ast(module_name, alg_filename, kernel_paths, line_length):
181 '''Search for the kernel source code containing a module with the name
182 'module_name' looking in the directory and subdirectories
183 associated with the supplied 'kernel_paths' or in the same
184 directory as the 'algorithm_filename' if the kernel path is
185 empty. If the file is found then check it conforms to the
186 'line_length' restriction if this is set and then parse this file
187 and return the parsed file.
188
189 :param str module_name: the name of the module to search for.
190 :param str alg_filename: the name of the algorithm file.
191 :param kernel_paths: directory in which to search for the module \
192 file.
193 :type kernel_paths: list of str
194 :param bool line_length: whether to check that the kernel code \
195 conforms to the 132 character line length limit (True) or not \
196 (False).
197
198 :returns: Parse tree of the kernel module with the name 'module_name'.
199 :rtype: :py:class:`fparser.one.block_statements.BeginSource`
200
201 '''
202 filepath = get_kernel_filepath(module_name, kernel_paths, alg_filename)
203 if line_length:
204 check_line_length(filepath)
205 parse_tree = get_kernel_parse_tree(filepath)
206 return parse_tree

210class KernelTypeFactory():
211 '''Factory to create the required API-specific information about
212 coded-kernel metadata and a reference to its code.
213
214 :param str api: The API for which this factory is to create Kernel \
215 information. If it is not supplied then the default API, as \
216 specified in the PSyclone config file, is used.
217
218 '''
219 def __init__(self, api=''):
220 check_api(api)
221 self._type = api
222
223 def create(self, parse_tree, name=None):
224 '''Create API-specific information about the kernel metadata and a
225 reference to its code. The API is set when the factory is
226 created.
227
228 :param parse_tree: The fparser1 parse tree for the Kernel code.
229 :type parse_tree: :py:class:`fparser.one.block_statements.BeginSource`
230
231 :param name: the name of the Kernel. Defaults to None if \
232 one is not provided.
233 :type name: str or NoneType
234
235 :raises ParseError: if the supplied API is not supported.
236
237 '''
238 # Avoid circular import
239 # pylint: disable=import-outside-toplevel
240 if self._type in LFRIC_API_NAMES:
241 from psyclone.domain.lfric import LFRicKernMetadata
242 return LFRicKernMetadata(parse_tree, name=name)
243 if self._type in GOCEAN_API_NAMES:
244 from psyclone.gocean1p0 import GOKernelType1p0
245 return GOKernelType1p0(parse_tree, name=name)
246 raise ParseError(
247 f"KernelTypeFactory:create: Unsupported kernel type '{self._type}'"
248 f" found.")

251class BuiltInKernelTypeFactory(KernelTypeFactory):
252 '''Create API-specific information about the builtin metadata. The API
253 is set when the factory is created. Subclasses KernelTypeFactory
254 and makes use of its init method.
255
256 '''
257 # pylint: disable=arguments-differ
258 def create(self, builtin_names, builtin_defs_file, name=None):
259 '''Create API-specific information about the builtin metadata. This
260 method finds and parses the metadata then makes use of the
261 KernelTypeFactory parent class to return the api-specific
262 information about the builtin.
263
264 :param builtin_names: a list of valid builtin names
265 :type builtin_names: list of str
266 :param str builtin_defs_file: the file containing builtin \
267 metadata
268 :param name: the name of the builtin. Defaults to None if \
269 one is not provided.
270 :type name: str or NoneType
271
272 :raises ParseError: if the supplied name is not one of the \
273 builtin names
274 :raises ParseError: if the supplied name is recognised as a \
275 builtin but the associated file containing the required \
276 metadata can not be found.
277 :raises ParseError: if the metadata for the supplied builtin \
278 can not be parsed.
279
280 '''
281 if name not in builtin_names:
282 raise ParseError(
283 f"BuiltInKernelTypeFactory:create unrecognised built-in name. "
284 f"Got '{name}' but expected one of {builtin_names}")
285 # The meta-data for these lives in a Fortran module file
286 # passed in to this method.
287 fname = os.path.join(
288 os.path.dirname(os.path.abspath(__file__)),
289 builtin_defs_file)
290 if not os.path.isfile(fname):
291 raise ParseError(
292 f"BuiltInKernelTypeFactory:create Kernel '{name}' is a "
293 f"recognised Built-in but cannot find file '{fname}' "
294 f"containing the meta-data describing the Built-in operations "
295 f"for API '{self._type}'")
296 # Attempt to parse the meta-data
297 try:
298 parsefortran.FortranParser.cache.clear()
299 fparser.logging.disable(fparser.logging.CRITICAL)
300 parse_tree = fpapi.parse(fname)
301 except Exception as err:
302 raise ParseError(
303 f"BuiltInKernelTypeFactory:create: Failed to parse the meta-"
304 f"data for PSyclone built-ins in file '{fname}'.") from err
305
306 # Now we have the parse tree, call our parent class to create \
307 # the object
308 return KernelTypeFactory.create(self, parse_tree, name)

312def get_mesh(metadata, valid_mesh_types):
313 '''
314 Returns the mesh-type described by the supplied meta-data
315
316 :param metadata: node in parser ast
317 :type metadata: py:class:`psyclone.expression.NamedArg`
318 :param valid_mesh_types: List of valid mesh types
319 :type valid_mesh_types: list of strings
320
321 :return: the name of the mesh
322 :rtype: string
323
324 :raises ParseError: if the supplied meta-data is not a recognised \
325 mesh identifier.
326 :raises ParseError: if the mesh type is unsupported.
327
328 '''
329 if not isinstance(metadata, expr.NamedArg) or \
330 metadata.name.lower() != "mesh_arg":
331 raise ParseError(
332 f"{metadata} is not a valid mesh identifier (expected "
333 f"mesh_arg=MESH_TYPE where MESH_TYPE is one of "
334 f"{valid_mesh_types}))")
335 mesh = metadata.value.lower()
336 if mesh not in valid_mesh_types:
337 raise ParseError(f"mesh_arg must be one of {valid_mesh_types} but got "
338 f"{mesh}")
339 return mesh

342def get_stencil(metadata, valid_types):
343 '''Returns stencil_type and stencil_extent as a dictionary
344 object from stencil metadata if the metadata conforms to the
345 stencil(type[,extent]) format
346
347 :param metadata: Component of kernel meta-data stored as a node in \
348 the fparser1 AST
349 :type metadata: :py:class:`psyclone.expression.FunctionVar`
350 :param list valid_types: List of valid stencil types (strings)
351
352 :return: The stencil type and extent described in the meta-data
353 :rtype: dict with keys 'type' (str) and 'extent' (int)
354
355 :raises ParseError: if the supplied meta-data is not a recognised \
356 stencil specification
357
358 '''
359
360 if not isinstance(metadata, expr.FunctionVar):
361 raise ParseError(
362 f"Expecting format stencil(<type>[,<extent>]) but found the "
363 f"literal {metadata}")
364 if metadata.name.lower() != "stencil" or not metadata.args:
365 raise ParseError(
366 f"Expecting format stencil(<type>[,<extent>]) but found {metadata}"
367 )
368 if len(metadata.args) > 2:
369 raise ParseError(
370 f"Expecting format stencil(<type>[,<extent>]) so there must "
371 f"be at most two arguments inside the brackets {metadata}")
372 if not isinstance(metadata.args[0], expr.FunctionVar):
373 if isinstance(metadata.args[0], str):
374 raise ParseError(
375 f"Expecting format stencil(<type>[,<extent>]). However, "
376 f"the specified <type> '{metadata.args[0]}' is a literal and "
377 f"therefore is not one of the valid types '{valid_types}'")
378 raise ParseError(
379 f"Internal error, expecting either FunctionVar or str from the "
380 f"expression analyser but found {type(metadata.args[0])}")
381 if metadata.args[0].args:
382 raise ParseError(
383 "Expected format stencil(<type>[,<extent>]). However, the "
384 "specified <type> '{0}' includes brackets")
385 stencil_type = metadata.args[0].name
386 if stencil_type not in valid_types:
387 raise ParseError(
388 f"Expected format stencil(<type>[,<extent>]). However, the "
389 f"specified <type> '{stencil_type}' is not one of the valid types "
390 f"'{valid_types}'")
391
392 stencil_extent = None
393 if len(metadata.args) == 2:
394 if not isinstance(metadata.args[1], str):
395 raise ParseError(
396 f"Expected format stencil(<type>[,<extent>]). However, the "
397 f"specified <extent> '{metadata.args[1]}' is not an integer")
398 stencil_extent = int(metadata.args[1])
399 if stencil_extent < 1:
400 raise ParseError(
401 f"Expected format stencil(<type>[,<extent>]). However, the "
402 f"specified <extent> '{stencil_extent}' is less than 1")
403 raise NotImplementedError(
404 "Kernels with fixed stencil extents are not currently "
405 "supported")
406 return {"type": stencil_type, "extent": stencil_extent}

409class Descriptor():
410 '''
411 A description of how a kernel argument is accessed, constructed from
412 the kernel metadata.
413
414 :param str access: whether argument is read/write etc.
415 :param str space: which function space/grid-point type argument is on.
416 :param int metadata_index: position of this argument in the list of \
417 arguments specified in the metadata.
418 :param dict stencil: type of stencil access for this argument. \
419 Defaults to None if the argument is not supplied.
420 :param str mesh: which mesh this argument is on. Defaults to None \
421 if the argument is not supplied.
422 :param str argument_type: the type of this argument. Defaults to \
423 None if the argument is not supplied.
424
425 :raises InternalError: if the metadata_index argument is not an int or is \
426 less than zero.
427
428 '''
429 # pylint: disable=too-many-arguments
430 def __init__(self, access, space, metadata_index, stencil=None, mesh=None,
431 argument_type=None):
432 self._access = access
433 self._space = space
434 if not isinstance(metadata_index, int) or metadata_index < 0:
435 raise InternalError(
436 f"The metadata index must be an integer and greater than or "
437 f"equal to zero but got: {metadata_index}")
438 self._metadata_index = metadata_index
439 self._stencil = stencil
440 self._mesh = mesh
441 self._argument_type = argument_type
442
443 @property
444 def access(self):
445 '''
446 :returns: whether argument is read/write etc.
447 :rtype: str
448
449 '''
450 return self._access
451
452 @property
453 def function_space(self):
454 '''
455 :returns: which function space/grid-point type argument is on.
456 :rtype: str
457
458 '''
459 return self._space
460
461 @property
462 def metadata_index(self):
463 '''
464 :returns: the position of the corresponding argument descriptor in \
465 the kernel metadata.
466 :rtype: int
467 '''
468 return self._metadata_index
469
470 @property
471 def stencil(self):
472 '''
473 :returns: type of stencil access for this argument.
474 :rtype: dict or NoneType
475
476 '''
477 return self._stencil
478
479 @property
480 def mesh(self):
481 '''
482 :returns: the mesh the argument is on.
483 :rtype: str or NoneType
484
485 '''
486 return self._mesh
487
488 @property
489 def argument_type(self):
490 '''
491 :returns: the type of the argument depending on the specific \
492 API (e.g. scalar, field, grid property, operator).
493 :rtype: str or NoneType
494
495 '''
496 return self._argument_type
497
498 def __repr__(self):
499 return (f"Descriptor({self.access}, {self.function_space}, "
500 f"{self.metadata_index})")

503class KernelProcedure():
504 '''
505 Captures the parse tree and name of a kernel subroutine.
506
507 :param ktype_ast: the fparser1 parse tree for the Kernel meta-data.
508 :type ktype_ast: :py:class:`fparser.one.block_statements.Type`
509 :param str ktype_name: name of the Fortran type holding the Kernel \
510 meta-data.
511 :param modast: the fparser1 parse tree for the module containing the \
512 Kernel routine.
513 :type modast: :py:class:`fparser.one.block_statements.BeginSource`
514
515 '''
516 def __init__(self, ktype_ast, ktype_name, modast):
517 self._ast, self._name = KernelProcedure.get_procedure(
518 ktype_ast, ktype_name, modast)
519
520 # pylint: disable=too-many-branches
521 @staticmethod
522 def get_procedure(ast, name, modast):
523 '''
524 Get the name of the subroutine associated with the Kernel. This is
525 a type-bound procedure in the meta-data which may take one of three
526 forms:
527 PROCEDURE, nopass :: code => <proc_name>
528 or
529 PROCEDURE, nopass :: <proc_name>
530 or if there is no type-bound procedure, an interface may be used:
531 INTERFACE <proc_name>
532
533 :param ast: the fparser1 parse tree for the Kernel meta-data.
534 :type ast: :py:class:`fparser.one.block_statements.Type`
535 :param str name: the name of the Fortran type holding the Kernel \
536 meta-data.
537 :param modast: the fparser1 parse tree for the module containing the \
538 Kernel routine.
539 :type modast: :py:class:`fparser.one.block_statements.BeginSource`
540
541 :returns: 2-tuple of the fparser1 parse tree of the Subroutine \
542 statement and the name of that Subroutine.
543 :rtype: (:py:class:`fparser1.block_statements.Subroutine`, str)
544
545 :raises ParseError: if the supplied Kernel meta-data does not \
546 have a type-bound procedure or interface.
547 :raises ParseError: if no implementation is found for the \
548 type-bound procedure or interface module \
549 procedures.
550 :raises ParseError: if the type-bound procedure specifies a binding \
551 name but the generic name is not "code".
552 :raises InternalError: if we get an empty string for the name of the \
553 type-bound procedure.
554 '''
555 bname = None
556 # Search the the meta-data for a SpecificBinding
557 for statement in ast.content:
558 if isinstance(statement, fparser1.statements.SpecificBinding):
559 # We support:
560 # PROCEDURE, nopass :: code => <proc_name> or
561 # PROCEDURE, nopass :: <proc_name>
562 if statement.bname:
563 if statement.name.lower() != "code":
564 raise ParseError(
565 f"Kernel type {name} binds to a specific procedure"
566 f" but does not use 'code' as the generic name.")
567 bname = statement.bname.lower()
568 else:
569 bname = statement.name.lower()
570 break
571 if bname is None:
572 # If no type-bound procedure found, search for an explicit
573 # interface that has module procedures.
574 bname, subnames = get_kernel_interface(name, modast)
575 if bname is None:
576 # no interface found either
577 raise ParseError(
578 f"Kernel type {name} does not bind a specific procedure "
579 f"or provide an explicit interface")
580 elif bname == '':
581 raise InternalError(
582 f"Empty Kernel name returned for Kernel type {name}.")
583 else:
584 # add the name of the tbp to the list of strings to search for.
585 subnames = [bname]
586 # walk the AST to check the subroutine names exist.
587 procedure_count = 0
588 for subname in subnames:
589 for statement, _ in fpapi.walk(modast):
590 if isinstance(statement,
591 fparser1.block_statements.Subroutine) \
592 and statement.name.lower() \
593 == subname:
594 procedure_count = procedure_count + 1
595 if procedure_count == 1:
596 # set code to statement if there is one procedure.
597 code = statement
598 else:
599 code = None # set to None if there is more than one.
600 break
601 else:
602 raise ParseError(
603 f"kernel.py:KernelProcedure:get_procedure: Kernel "
604 f"subroutine '{subname}' not found.")
605 return code, bname
606
607 @property
608 def name(self):
609 '''
610 :returns: the name of the kernel subroutine
611 :rtype: str
612
613 '''
614 return self._name
615
616 @property
617 def ast(self):
618 '''
619 :returns: the parse tree of the kernel subroutine
620 :rtype: :py:class:`fparser.one.block_statements.Subroutine`
621
622 '''
623 return self._ast
624
625 def __repr__(self):
626 return f"KernelProcedure({self.name})"
627
628 def __str__(self):
629 return str(self._ast)

632def get_kernel_metadata(name, ast):
633 '''Takes the kernel module parse tree and returns the metadata part
634 of the parse tree (a Fortran type) with the name 'name'.
635
636 :param str name: the metadata name (of a Fortran type). Also \
637 the name referencing the kernel in the algorithm layer. The name \
638 provided and the name of the kernel in the parse tree are case \
639 insensitive in this function.
640 :param ast: parse tree of the kernel module code
641 :type ast: :py:class:`fparser.one.block_statements.BeginSource`
642
643 :returns: Parse tree of the metadata (a Fortran type with name \
644 'name')
645 :rtype: :py:class:`fparser.one.block_statements.Type`
646
647 :raises ParseError: if the metadata type name is not found in \
648 the kernel code parse tree
649
650 '''
651 ktype = None
652 for statement, _ in fpapi.walk(ast):
653 if isinstance(statement, fparser1.block_statements.Type) \
654 and statement.name.lower() == name.lower():
655 ktype = statement
656 break
657 if ktype is None:
658 raise ParseError(f"Kernel type {name} does not exist")
659 return ktype

662def get_kernel_interface(name, ast):
663 '''Takes the kernel module parse tree and returns the interface part
664 of the parse tree.
665
666 :param str name: The kernel name
667 :param ast: parse tree of the kernel module code
668 :type ast: :py:class:`fparser.one.block_statements.BeginSource`
669
670 :returns: Name of the interface block and the names of the module \
671 procedures (lower case). Or None, None if there is no \
672 interface or the interface has no nodule procedures.
673 :rtype: : `str`, list of str`.
674
675 :raises ParseError: if more than one interface is found.
676 '''
677
678 iname = None
679 sname = None
680 count = 0
681 for statement, _ in fpapi.walk(ast):
682 if isinstance(statement, fparser1.block_statements.Interface):
683 # count the interfaces, then can be only one!
684 count = count + 1
685 if count >= 2:
686 raise ParseError(f"Module containing kernel {name} has more "
687 f"than one interface, this is forbidden in "
688 f"the LFRic API.")
689 # Check the interfaces assigns one or more module procedures.
690 if statement.a.module_procedures:
691 iname = statement.name.lower()
692 # If implicit interface (no name) set to none as there is no
693 # procedure name for PSyclone to use.
694 if iname == '':
695 iname = None
696 else:
697 sname = [str(sname).lower() for sname
698 in statement.a.module_procedures]
699 return iname, sname

702def getkerneldescriptors(name, ast, var_name='meta_args', var_type=None):
703 '''Get the required argument metadata information for a kernel.
704
705 :param str name: the name of the kernel (for error messages).
706 :param ast: metadata describing kernel arguments.
707 :type ast: :py:class:`fparser.one.block_statements.Type`
708 :param str var_name: the name of the variable storing the \
709 argument metadata. this argument is optional and defaults to \
710 'meta_args'.
711 :param str var_type: the type of the structure constructor used to \
712 define the meta-data or None.
713
714 :returns: argument metadata parsed using the expression parser \
715 (as fparser1 will not parse expressions and arguments).
716 :rtype: :py:class:`psyclone.expression.LiteralArray`
717
718 :raises ParseError: if 'var_name' is not found in the metadata.
719 :raises ParseError: if 'var_name' is not an array.
720 :raises ParseError: if 'var_name' is not a 1D array.
721 :raises ParseError: if the structure constructor uses '[...]' \
722 as only '(/.../)' is supported.
723 :raises ParseError: if the argument metadata is invalid and cannot \
724 be parsed.
725 :raises ParseError: if the dimensions specified do not tally with \
726 the number of metadata arguments.
727 :raises ParseError: if var_type is specified and a structure \
728 constructor for a different type is found.
729 '''
730 descs = ast.get_variable(var_name)
731 if "INTEGER" in str(descs):
732 # INTEGER in above 'if' test is an fparser1 hack as get_variable()
733 # returns an integer if the variable is not found.
734 raise ParseError(
735 f"No variable named '{var_name}' found in the metadata for kernel "
736 f"'{name}': '{str(ast).strip()}'.")
737 try:
738 nargs = int(descs.shape[0])
739 except AttributeError as err:
740 raise ParseError(
741 f"In kernel metadata '{name}': '{var_name}' variable must be an "
742 f"array.") from err
743 if len(descs.shape) != 1:
744 raise ParseError(
745 f"In kernel metadata '{name}': '{var_name}' variable must be a 1 "
746 f"dimensional array.")
747 if descs.init.find("[") != -1 and descs.init.find("]") != -1:
748 # there is a bug in fparser1
749 raise ParseError(
750 f"Parser does not currently support '[...]' initialisation for "
751 f"'{var_name}', please use '(/.../)' instead.")
752 try:
753 inits = expr.FORT_EXPRESSION.parseString(descs.init)[0]
754 except ParseException as err:
755 raise ParseError(f"Kernel metadata has an invalid format "
756 f"{descs.init}.") from err
757 nargs = int(descs.shape[0])
758 if len(inits) != nargs:
759 raise ParseError(
760 f"In the '{var_name}' metadata, the number of items in the array "
761 f"constructor ({len(inits)}) does not match the extent of the "
762 f"array ({nargs}).")
763 if var_type:
764 # Check that each element in the list is of the correct type
765 if not all(init.name == var_type for init in inits):
766 raise ParseError(
767 f"The '{var_name}' metadata must consist of an array of "
768 f"structure constructors, all of type '{var_type}' but found: "
769 f"{[str(init.name) for init in inits]}.")
770 return inits

773class KernelType():
774 '''Base class for describing Kernel Metadata.
775
776 This contains the name of the elemental procedure and metadata associated
777 with how that procedure is mapped over mesh entities.
778
779 :param ast: fparser1 AST for the parsed kernel meta-data.
780 :type ast: :py:class:`fparser.one.block_statements.BeginSource`
781 :param str name: name of the Fortran derived type describing the kernel.
782
783 :raises ParseError: if the supplied AST does not contain a Fortran \
784 module.
785 :raises ParseError: if the module name is too short to contain \
786 '_mod' at the end.
787 :raises ParseError: if the module name does not end in '_mod'.
788
789 '''
790 def __init__(self, ast, name=None):
791
792 if name is None:
793 # if no name is supplied then use the module name to
794 # determine the type name. The assumed convention is that
795 # the module is called <name/>_mod and the type is called
796 # <name/>_type
797 found = False
798 for statement, _ in fpapi.walk(ast):
799 if isinstance(statement, fparser1.block_statements.Module):
800 module_name = statement.name
801 found = True
802 break
803 if not found:
804 raise ParseError(
805 "Error KernelType, the file does not contain a module. "
806 "Is it a Kernel file?")
807
808 mn_len = len(module_name)
809 if mn_len < 5:
810 raise ParseError(
811 f"Error, module name '{module_name}' is too short to have "
812 f"'_mod' as an extension. This convention is assumed.")
813 base_name = module_name.lower()[:mn_len-4]
814 extension_name = module_name.lower()[mn_len-4:mn_len]
815 if extension_name != "_mod":
816 raise ParseError(
817 f"Error, module name '{module_name}' does not have '_mod' "
818 f"as an extension. This convention is assumed.")
819 name = base_name + "_type"
820
821 self._name = name
822 self._ast = ast
823 self._ktype = get_kernel_metadata(name, ast)
824 # TODO #1204 since the valid form of the metadata beyond this point
825 # depends on the API, the code beyond this point should be refactored
826 # into LFRicKernMetadata and GOKernelType1p0.
827 operates_on = self.get_integer_variable("operates_on")
828 # The GOcean API still uses the 'iterates_over' metadata entry
829 # although this is deprecated in the LFRic API.
830 # Validation is left to the API-specific code in either dynamo0p3.py
831 # or gocean1p0.py.
832 iterates_over = self.get_integer_variable("iterates_over")
833 if operates_on:
834 self._iterates_over = operates_on
835 elif iterates_over:
836 self._iterates_over = iterates_over
837 else:
838 # We don't raise an error here - we leave it to the API-specific
839 # validation code.
840 self._iterates_over = None
841 # Although validation of the value given to operates_on or
842 # iterates_over is API-specific, we can check that the metadata doesn't
843 # specify both of them because that doesn't make sense.
844 if operates_on and iterates_over:
845 raise ParseError(f"The metadata for kernel '{name}' contains both "
846 f"'operates_on' and 'iterates_over'. Only one of "
847 f"these is permitted.")
848 self._procedure = KernelProcedure(self._ktype, name, ast)
849 self._inits = getkerneldescriptors(name, self._ktype)
850 self._arg_descriptors = [] # this is set up by the subclasses

902 def get_integer_variable(self, name):
903 ''' Parse the kernel meta-data and find the value of the
904 integer variable with the supplied name. Return None if no
905 matching variable is found. The search is not case sensitive.
906
907 :param str name: the name of the integer variable to find.
908
909 :return: value of the specified integer variable (lower case) or None.
910 :rtype: str
911
912 :raises ParseError: if the RHS of the assignment is not a Name.
913
914 '''
915 # Ensure the Fortran parser is initialised
916 _ = ParserFactory().create(std=Config.get().fortran_standard)
917 # Fortran is not case sensitive so nor is our matching
918 lower_name = name.lower()
919
920 for statement, _ in fpapi.walk(self._ktype):
921 if isinstance(statement, fparser1.typedecl_statements.Integer):
922 # fparser only goes down to the statement level. We use
923 # fparser2 to parse the statement itself (eventually we'll
924 # use fparser2 to parse the whole thing).
925 assign = Fortran2003.Assignment_Stmt(
926 statement.entity_decls[0])
927 if str(assign.items[0]).lower() == lower_name:
928 if not isinstance(assign.items[2], Fortran2003.Name):
929 raise ParseError(
930 f"get_integer_variable: RHS of assignment is not "
931 f"a variable name: '{assign}'")
932 return str(assign.items[2]).lower()
933 return None

935 def get_integer_array(self, name):
936 ''' Parse the kernel meta-data and find the values of the
937 integer array variable with the supplied name. Returns an empty list
938 if no matching variable is found. The search is not case sensitive.
939
940 :param str name: the name of the integer array to find.
941
942 :return: list of values (lower-case).
943 :rtype: list of str.
944
945 :raises InternalError: if we fail to parse the LHS of the array \
946 declaration or the array constructor.
947 :raises ParseError: if the array is not of rank 1.
948 :raises ParseError: if the array extent is not specified using an \
949 integer literal.
950 :raises ParseError: if the RHS of the declaration is not an array \
951 constructor.
952 :raises InternalError: if the parse tree for the array constructor \
953 does not have the expected structure.
954 :raises ParseError: if the number of items in the array constructor \
955 does not match the extent of the array.
956
957 '''
958 # Ensure the classes are setup for the Fortran parser
959 _ = ParserFactory().create(std=Config.get().fortran_standard)
960 # Fortran is not case sensitive so nor is our matching
961 lower_name = name.lower()
962
963 for statement, _ in fpapi.walk(self._ktype):
964 if not isinstance(statement, fparser1.typedecl_statements.Integer):
965 # This isn't an integer declaration so skip it
966 continue
967 # fparser only goes down to the statement level. We use fparser2 to
968 # parse the statement itself.
969 assign = Fortran2003.Assignment_Stmt(statement.entity_decls[0])
970 names = walk(assign.children, Fortran2003.Name)
971 if not names:
972 raise InternalError(f"Unsupported assignment statement: "
973 f"'{assign}'")
974
975 if str(names[0]).lower() != lower_name:
976 # This is not the variable declaration we're looking for
977 continue
978
979 if not isinstance(assign.children[0], Fortran2003.Part_Ref):
980 # Not an array declaration
981 return []
982
983 if not isinstance(assign.children[0].children[1],
984 Fortran2003.Section_Subscript_List):
985 raise InternalError(
986 f"get_integer_array: expected array declaration to have a "
987 f"Section_Subscript_List but found "
988 f"'{type(assign.children[0].children[1]).__name__}' "
989 f"for: '{assign}'")
990
991 dim_stmt = assign.children[0].children[1]
992 if len(dim_stmt.children) != 1:
993 raise ParseError(
994 f"get_integer_array: array must be 1D but found an array "
995 f"with {len(dim_stmt.children)} dimensions for name '"
996 f"{name}'")
997 if not isinstance(dim_stmt.children[0],
998 Fortran2003.Int_Literal_Constant):
999 raise ParseError(
1000 f"get_integer_array: array extent must be specified using "
1001 f"an integer literal but found '{dim_stmt.children[0]}' "
1002 f"for array '{name}'")
1003 # Get the declared size of the array
1004 array_extent = int(str(dim_stmt.children[0]))
1005
1006 if not isinstance(assign.children[2],
1007 Fortran2003.Array_Constructor):
1008 raise ParseError(
1009 f"get_integer_array: RHS of assignment is not "
1010 f"an array constructor: '{assign}'")
1011 # fparser2 AST for Array_Constructor is:
1012 # Array_Constructor('[', Ac_Value_List(',', (Name('w0'),
1013 # Name('w1'))), ']')
1014 # Construct a list of the names in the array constructor
1015 names = walk(assign.children[2].children, Fortran2003.Name)
1016 if not names:
1017 raise InternalError(f"Failed to parse array constructor: "
1018 f"'{assign.items[2]}'")
1019 if len(names) != array_extent:
1020 # Ideally fparser would catch this but it isn't yet mature
1021 # enough.
1022 raise ParseError(
1023 f"get_integer_array: declared length of array '{name}' is "
1024 f"{array_extent} but constructor only contains "
1025 f"{len(names)} names: '{assign}'")
1026 return [str(name).lower() for name in names]
1027 # No matching declaration for the provided name was found
1028 return []
