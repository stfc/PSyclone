56def _get_array_bound(vector1, vector2):
57 '''A utility function that returns the appropriate loop bounds (lower,
58 upper and step) for a vector.
59 If either of the vectors are declared with known bounds (an integer or a
60 symbol) then these bound values are used. If the size is unknown
61 (a deferred or attribute type) then the LBOUND and UBOUND PSyIR
62 nodes are used.
63
64 The validate() method in DotProduct2CodeTrans will ensure that the
65 arguments to _get_array_bound() are valid (as it is always called
66 beforehand). The validate() method also ensures that array slices
67 are for the first dimension of the array and that they are for the
68 full size of that dimension (they are limited to ":"). This
69 function makes use of these constraint, e.g. it always returns 1
70 for the stride.
71
72 :param array: the reference that we are interested in.
73 :type array: :py:class:`psyir.nodes.Reference`
74 :param int index: the (array) reference index that we are \
75 interested in.
76
77 :returns: the loop bounds for this array index.
78 :rtype: (Literal, Literal, Literal) or \
79 (BinaryOperation, BinaryOperation, Literal)
80
81 '''
82 # Look for explicit bounds in one of the array declarations
83 for vector in [vector1, vector2]:
84 symbol = vector.symbol
85 my_dim = symbol.shape[0]
86 if isinstance(my_dim, ArrayType.ArrayBounds):
87 lower_bound = my_dim.lower
88 upper_bound = my_dim.upper
89 step = Literal("1", INTEGER_TYPE)
90 return (lower_bound, upper_bound, step)
91
92 # No explicit array bound information could be found for either
93 # array so use the LBOUND and UBOUND intrinsics.
94 symbol = vector1.symbol
95 my_dim = symbol.shape[0]
96 lower_bound = IntrinsicCall.create(
97 IntrinsicCall.Intrinsic.LBOUND,
98 [Reference(symbol), ("dim", Literal("1", INTEGER_TYPE))])
99 upper_bound = IntrinsicCall.create(
100 IntrinsicCall.Intrinsic.UBOUND,
101 [Reference(symbol), ("dim", Literal("1", INTEGER_TYPE))])
102 step = Literal("1", INTEGER_TYPE)
103 return (lower_bound, upper_bound, step)

106class DotProduct2CodeTrans(Intrinsic2CodeTrans):
107 '''Provides a transformation from a PSyIR DOT_PRODUCT Operator node to
108 equivalent code in a PSyIR tree. Validity checks are also
109 performed.
110
111 If ``R`` is a scalar and ``A``, and ``B`` have dimension ``N``,
112 The transformation replaces:
113
114 .. code-block:: fortran
115
116 R = ... DOT_PRODUCT(A,B) ...
117
118 with the following code:
119
120 .. code-block:: fortran
121
122 TMP = 0.0
123 do I=1,N
124 TMP = TMP + A(i)*B(i)
125 R = ... TMP ...
126
127 For example:
128
129 >>> from psyclone.psyir.backend.fortran import FortranWriter
130 >>> from psyclone.psyir.frontend.fortran import FortranReader
131 >>> from psyclone.psyir.nodes import IntrinsicCall
132 >>> from psyclone.psyir.transformations import DotProduct2CodeTrans
133 >>> code = ("subroutine dot_product_test(v1,v2)\\n"
134 ... "real,intent(in) :: v1(10), v2(10)\\n"
135 ... "real :: result\\n"
136 ... "result = dot_product(v1,v2)\\n"
137 ... "end subroutine\\n")
138 >>> psyir = FortranReader().psyir_from_source(code)
139 >>> trans = DotProduct2CodeTrans()
140 >>> trans.apply(psyir.walk(IntrinsicCall)[0])
141 >>> print(FortranWriter()(psyir))
142 subroutine dot_product_test(v1, v2)
143 real, dimension(10), intent(in) :: v1
144 real, dimension(10), intent(in) :: v2
145 real :: result
146 integer :: i
147 real :: res_dot_product
148 <BLANKLINE>
149 res_dot_product = 0.0
150 do i = 1, 10, 1
151 res_dot_product = res_dot_product + v1(i) * v2(i)
152 enddo
153 result = res_dot_product
154 <BLANKLINE>
155 end subroutine dot_product_test
156 <BLANKLINE>
157
158 '''
159 def __init__(self):
160 super().__init__()
161 self._intrinsic_intrinsic = IntrinsicCall.Intrinsic.DOT_PRODUCT

163 def validate(self, node, options=None):
164 '''Perform checks to ensure that it is valid to apply the
165 DotProduct2CodeTran transformation to the supplied node.
166
167 Note, this validation does not check for invalid argument
168 combinations to dot_product (e.g. different precision or
169 different datatypes) as that should have already been picked
170 up when creating the PSyIR.
171
172 :param node: the node that is being checked.
173 :type node: :py:class:`psyclone.psyir.nodes.BinaryOperation`
174 :param options: a dictionary with options for transformations.
175 :type options: dict of str:str or None
176
177 :raises TransformationError: if one of the arguments is not a \
178 Reference node.
179 :raises TransformationError: if an argument does not use array \
180 slice notation and is not a 1d array.
181 :raises TransformationError: if an argument uses array slice \
182 notation but the array slice is not for the first dimension of \
183 the array.
184 :raises TransformationError: if an argument uses array slice \
185 notation but it is not for the full range of the dimension.
186
187 '''
188 super().validate(node, options)
189
190 # Both arguments should be references (or array references)
191 for arg in node.children:
192 if arg.__class__ not in [Reference, ArrayReference]:
193 raise TransformationError(
194 f"The DotProduct2CodeTrans transformation only supports "
195 f"the transformation of a dotproduct intrinsic if its "
196 f"arguments are plain arrays, but found "
197 f"{arg.debug_string()} in {node.debug_string()}.")
198
199 for arg in node.arguments:
200 # The argument should be a 1D array if the argument does
201 # not provide any array slice information (i.e. it is a
202 # Reference)
203 if arg.__class__ is Reference:
204 symbol = arg.symbol
205 # This symbol should be a 1D array
206 if not (isinstance(symbol, DataSymbol) and symbol.is_array and
207 len(symbol.shape) == 1):
208 raise TransformationError(
209 f"The DotProduct2CodeTrans transformation only "
210 f"supports the transformation of a dotproduct "
211 f"intrinsic with an argument not containing an array "
212 f"slice if the argument is a 1D array, but found "
213 f"{arg.debug_string()} with {len(symbol.shape)} "
214 f"dimensions in {node.debug_string()}.")
215
216 for arg in node.children:
217 # If the argument does provide array slice information
218 # then check the array slice is in the first dimension of
219 # the array and that the slice is for the full range of that
220 # dimension i.e. uses a ':'.
221 if isinstance(arg, ArrayReference):
222 if not isinstance(arg.indices[0], Range):
223 raise TransformationError(
224 f"The DotProduct2CodeTrans transformation only "
225 f"supports the transformation of a dotproduct "
226 f"intrinsic with an argument containing an array "
227 f"slice if the array slice is for the 1st dimension "
228 f"of the array, but found {arg.debug_string()} in "
229 f"{node.debug_string()}.")
230
231 if not arg.is_full_range(0):
232 raise TransformationError(
233 f"The DotProduct2CodeTrans transformation only "
234 f"supports the transformation of a dotproduct "
235 f"intrinsic with an argument containing an array "
236 f"slice if the argument is for the 1st dimension "
237 f"of the array and is for the full range of that "
238 f"dimension, but found {arg.debug_string()} in "
239 f"{node.debug_string()}.")
240
241 # Both arguments should be real (as other intrinsic datatypes
242 # are not supported).
243 for arg in node.arguments:
244 if arg.symbol.datatype.intrinsic != ScalarType.Intrinsic.REAL:
245 raise TransformationError(
246 f"The DotProduct2CodeTrans transformation only supports "
247 f"arrays of real data, but found {arg.debug_string()} of "
248 f"type {arg.symbol.datatype.intrinsic.name} in "
249 f"{node.debug_string()}.")

251 def apply(self, node, options=None):
252 '''Apply the DOT_PRODUCT intrinsic conversion transformation to the
253 specified node. This node must be a DOT_PRODUCT
254 BinaryOperation. If the transformation is successful then an
255 assignment which includes a DOT_PRODUCT BinaryOperation node
256 is converted to equivalent inline code.
257
258 :param node: a DOT_PRODUCT Binary-Operation node.
259 :type node: :py:class:`psyclone.psyir.nodes.BinaryOperation`
260 :param options: a dictionary with options for transformations.
261 :type options: dict of str:str or None
262
263 '''
264 self.validatevalidatevalidate(node, options)
265
266 assignment = node.ancestor(Assignment)
267 vector1 = node.arguments[0]
268 vector2 = node.arguments[1]
269 symbol_table = node.ancestor(Routine).symbol_table
270
271 # Create new i loop iterator.
272 i_loop_symbol = symbol_table.new_symbol(
273 "i", symbol_type=DataSymbol, datatype=INTEGER_TYPE)
274
275 # Create temporary result variable. Use the datatype of one of
276 # the arguments. We can do this as the validate method only
277 # allows plain real arrays.
278 vector1_datatype = vector1.symbol.datatype
279 datatype = ScalarType(
280 vector1_datatype.intrinsic, vector1_datatype.precision)
281 symbol_res_var = symbol_table.new_symbol(
282 "res_dot_product", symbol_type=DataSymbol,
283 datatype=datatype)
284
285 # Replace operation with the temporary result variable.
286 result_ref = Reference(symbol_res_var)
287 node.replace_with(result_ref)
288
289 # Create "vector1(i)"
290 vector1_dims = [Reference(i_loop_symbol)]
291 if len(vector1.children) > 1:
292 # Add any additional dimensions (in case of an array slice)
293 for child in vector1.children[1:]:
294 vector1_dims.append(child.copy())
295 vector1_array_reference = ArrayReference.create(
296 vector1.symbol, vector1_dims)
297 # Create "vector2(i)"
298 vector2_dims = [Reference(i_loop_symbol)]
299 if len(vector2.children) > 1:
300 # Add any additional dimensions (in case of an array slice)
301 for child in vector2.children[1:]:
302 vector2_dims.append(child.copy())
303 vector2_array_reference = ArrayReference.create(
304 vector2.symbol, vector2_dims)
305 # Create "vector1(i) * vector2(i)"
306 multiply = BinaryOperation.create(
307 BinaryOperation.Operator.MUL, vector1_array_reference,
308 vector2_array_reference)
309 # Create "result + vector1(i) * vector2(i)"
310 rhs = BinaryOperation.create(
311 BinaryOperation.Operator.ADD, result_ref.copy(), multiply)
312 # Create "result = result + vector1(i) * vector2(i)"
313 assign = Assignment.create(result_ref.copy(), rhs)
314 # Work out the loop bounds
315 lower_bound, upper_bound, step = _get_array_bound(vector1, vector2)
316 # Create i loop and add the above code as a child
317 iloop = Loop.create(i_loop_symbol, lower_bound.copy(),
318 upper_bound.copy(), step.copy(), [assign])
319 # Create "result = 0.0"
320 assign = Assignment.create(result_ref.copy(),
321 Literal("0.0", REAL_TYPE))
322 # Add the initialisation and loop nodes into the PSyIR tree
323 assignment.parent.children.insert(assignment.position, assign)
324 assignment.parent.children.insert(assignment.position, iloop)
