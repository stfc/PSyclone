39'''
40This module provides support for extraction of code within a specified
41invoke. The extracted code may be a single kernel, multiple occurrences of a
42kernel in an invoke, nodes in an invoke or the entire invoke (extraction
43applied to all Nodes).
44
45There is currently only one class in this module: ExtractNode (see below).
46
47Another class which contains helper functions for code extraction, such as
48wrapping up settings for generating driver for the extracted code, will
49be added in Issue #298.
50'''
51

56class ExtractNode(PSyDataNode):
57 '''
58 This class can be inserted into a Schedule to mark Nodes for
59 code extraction using the ExtractRegionTrans transformation. By
60 applying the transformation the Nodes marked for extraction become
61 children of (the Schedule of) an ExtractNode.
62
63 :param ast: reference into the fparser2 parse tree corresponding to
64 this node.
65 :type ast: sub-class of :py:class:`fparser.two.Fortran2003.Base`
66 :param children: the PSyIR nodes that are children of this node.
67 :type children: list of :py:class:`psyclone.psyir.nodes.Node`
68 :param parent: the parent of this node in the PSyIR tree.
69 :type parent: :py:class:`psyclone.psyir.nodes.Node`
70 :param options: a dictionary with options provided via transformations.
71 :type options: Optional[Dict[str, Any]]
72 :param str options["prefix"]: a prefix to use for the PSyData module name
73 (``prefix_psy_data_mod``) and the PSyDataType
74 (``prefix_PSyDataType``) - a "_" will be added automatically.
75 It defaults to "extract", which means the module name used will be
76 ``extract_psy_data_mode``, and the data type ``extract_PSyDataType``.
77 :param str options["post_var_postfix"]: a postfix to be used when
78 creating names to store values of output variable. A variable 'a'
79 would store its value as 'a', and its output values as 'a_post' with
80 the default post_var_postfix of '_post'.
81 :param options["read_write_info"]: information about variables that are
82 read and/or written in the instrumented code.
83 :type options["read_write_info"]:
84 py:class:`psyclone.psyir.tools.ReadWriteInfo`
85
86 '''
87 # Textual description of the node.
88 _text_name = "Extract"
89 _colour = "green"
90 # The default prefix to add to the PSyData module name and PSyDataType
91 _default_prefix = "extract"
92
93 def __init__(self, ast=None, children=None, parent=None, options=None):
94 super().__init__(ast=ast, children=children,
95 parent=parent, options=options)
96
97 # Define a postfix that will be added to variable that are
98 # modified to make sure the names can be distinguished between pre-
99 # and post-variables (i.e. here input and output). A variable
100 # "myvar" will be stored as "myvar" with its input value, and
101 # "myvar_post" with its output value. It is the responsibility
102 # of the transformation that inserts this node to make sure this
103 # name is consistent with the name used when creating the driver
104 # (otherwise the driver will not be able to read in the dumped
105 # valued), and also to handle any potential name clashes (e.g. a
106 # variable 'a' exists, which creates 'a_out' for the output variable,
107 # which would clash with a variable 'a_out' used in the program unit).
108
109 if options is None:
110 options = {}
111
112 self._post_name = options.get("post_var_postfix", "_post")
113
114 # Keep a copy of the argument list:
115 self._read_write_info_read_write_info = options.get("read_write_info")

148 def gen_code(self, parent):
149 # pylint: disable=arguments-differ
150 '''
151 Generates the code required for extraction of one or more Nodes.
152 It uses the PSyData API (via the base class PSyDataNode) to create
153 the required callbacks that will allow a library to write the
154 kernel data to a file.
155
156 :param parent: the parent of this Node in the PSyIR.
157 :type parent: :py:class:`psyclone.psyir.nodes.Node`.
158
159 '''
160 if self._read_write_info_read_write_info is None:
161 # Typically, _read_write_info should be set at the constructor,
162 # but some tests do not provide the required information. To
163 # support these tests, allow creation of the read_write info here.
164 # We cannot do this in the constructor, since at construction
165 # time of this node it is not yet part of the PSyIR tree, so it
166 # does not have children from which we can collect the input/output
167 # parameters.
168
169 # Avoid circular dependency
170 # pylint: disable=import-outside-toplevel
171 from psyclone.psyir.tools.call_tree_utils import CallTreeUtils
172 # Determine the variables to write:
173 ctu = CallTreeUtils()
174 self._read_write_info_read_write_info = ctu.get_in_out_parameters(self)
175
176 options = {'pre_var_list': self._read_write_info_read_write_info.read_list,
177 'post_var_list': self._read_write_info_read_write_info.write_list,
178 'post_var_postfix': self._post_name}
179
180 parent.add(CommentGen(parent, ""))
181 parent.add(CommentGen(parent, " ExtractStart"))
182 parent.add(CommentGen(parent, ""))
183 super().gen_code(parent, options)
184 parent.add(CommentGen(parent, ""))
185 parent.add(CommentGen(parent, " ExtractEnd"))
186 parent.add(CommentGen(parent, ""))

188 def lower_to_language_level(self):
189 # pylint: disable=arguments-differ
190 '''
191 Lowers this node (and all children) to language-level PSyIR. The
192 PSyIR tree is modified in-place.
193
194 :returns: the lowered version of this node.
195 :rtype: :py:class:`psyclone.psyir.node.Node`
196
197 '''
198 if self._read_write_info_read_write_info is None:
199 # Typically, _read_write_info should be set at the constructor,
200 # but some tests do not provide the required information. To
201 # support these tests, allow creation of the read_write info
202 # here (it can't be done in the constructor, since this node

206 # Avoid circular dependency

209 # Determine the variables to write:
210 ctu = CallTreeUtils()
211 self._read_write_info_read_write_info = ctu.get_in_out_parameters(
212 self, include_non_data_accesses=True)
213
214 options = {'pre_var_list': self._read_write_info_read_write_info.read_list,
215 'post_var_list': self._read_write_info_read_write_info.write_list,
216 'post_var_postfix': self._post_name}
217
218 return super().lower_to_language_level(options)
