53class ArrayReference(ArrayMixin, Reference):
54 '''
55 Node representing a reference to an element or elements of an Array.
56 The array-index expressions are stored as the children of this node.
57
58 '''
59 # Textual description of the node.
60 _children_valid_format = "[DataNode | Range]+"
61 _text_name = "ArrayReference"
62
63 @staticmethod
64 def create(symbol, indices):
65 '''Create an ArrayReference instance given a symbol and a list of Node
66 array indices. The special value ":" can be used as an index to
67 create the corresponding PSyIR Range that represents ":".
68
69 :param symbol: the symbol that this array is associated with.
70 :type symbol: :py:class:`psyclone.psyir.symbols.DataSymbol`
71 :param indices: a list of Nodes or ":" describing the array indices.
72 :type indices: List[Union[:py:class:`psyclone.psyir.nodes.Node`,":"]]
73
74 :returns: an ArrayReference instance.
75 :rtype: :py:class:`psyclone.psyir.nodes.ArrayReference`
76
77 :raises GenerationError: if the arguments to the create method \
78 are not of the expected type.
79
80 '''
81 if not isinstance(symbol, DataSymbol):
82 raise GenerationError(
83 f"symbol argument in create method of ArrayReference class "
84 f"should be a DataSymbol but found '{type(symbol).__name__}'.")
85 if not isinstance(indices, list):
86 raise GenerationError(
87 f"indices argument in create method of ArrayReference class "
88 f"should be a list but found '{type(indices).__name__}'.")
89 if not symbol.is_array:
90 # Unresolved and Unsupported types may still be arrays
91 if not isinstance(symbol.datatype, (UnresolvedType,
92 UnsupportedType)):
93 raise GenerationError(
94 f"expecting the symbol '{symbol.name}' to be an array, but"
95 f" found '{symbol.datatype}'.")
96 elif len(symbol.shape) != len(indices):
97 raise GenerationError(
98 f"the symbol '{symbol.name}' should have the same number of "
99 f"dimensions as indices (provided in the 'indices' argument). "
100 f"Expecting '{len(indices)}' but found '{len(symbol.shape)}'.")
101
102 array = ArrayReference(symbol)
103 for ind, child in enumerate(indices):
104 if child == ":":
105 lbound = IntrinsicCall.create(
106 IntrinsicCall.Intrinsic.LBOUND,
107 [Reference(symbol),
108 ("dim", Literal(f"{ind+1}", INTEGER_TYPE))])
109 ubound = IntrinsicCall.create(
110 IntrinsicCall.Intrinsic.UBOUND,
111 [Reference(symbol),
112 ("dim", Literal(f"{ind+1}", INTEGER_TYPE))])
113 my_range = Range.create(lbound, ubound)
114 array.addchild(my_range)
115 else:
116 array.addchild(child)
117 return array

125 @property
126 def datatype(self):
127 '''
128 :returns: the datatype of the accessed array element(s).
129 :rtype: :py:class:`psyclone.psyir.symbols.DataType`
130 '''
131 try:
132 shape = self._get_effective_shape()
133 except NotImplementedError:
134 return UnresolvedType()
135 if shape:
136 if type(self.symbolsymbolsymbolsymbolsymbol) is Symbol:
137 # We don't have any information on the shape of the original
138 # declaration.
139 orig_shape = None
140 elif isinstance(self.symbolsymbolsymbolsymbolsymbol.datatype, ArrayType):
141 # We have full type information so we know the shape of the
142 # original declaration.
143 orig_shape = self.symbolsymbolsymbolsymbolsymbol.datatype.shape
144 elif (isinstance(self.symbolsymbolsymbolsymbolsymbol.datatype, UnsupportedFortranType) and
145 self.symbolsymbolsymbolsymbolsymbol.datatype.partial_datatype):
146 # We have partial type information so we also know the shape
147 # of the original declaration.
148 orig_shape = self.symbolsymbolsymbolsymbolsymbol.datatype.partial_datatype.shape
149 else:
150 # We don't have any information on the shape of the original
151 # declaration.
152 orig_shape = None
153 if (orig_shape is not None and len(shape) == len(orig_shape) and
154 all(self.is_full_range(idx) for idx in range(len(shape)))):
155 # Although this access has a shape, it is in fact for the
156 # whole array and therefore the type of the result is just
157 # that of the base symbol. (This wouldn't be true for a
158 # StructureReference but they have their own implementation
159 # of this method.)
160 return self.symbolsymbolsymbolsymbolsymbol.datatype
161 if type(self.symbolsymbolsymbolsymbolsymbol) is Symbol or isinstance(self.symbolsymbolsymbolsymbolsymbol.datatype,
162 UnsupportedType):
163 # Even if an Unsupported(Fortran)Type has partial type
164 # information, we can't easily use it here because we'd need
165 # to re-write the original Fortran declaration stored in the
166 # type. We could manipulate the shape in the fparser2 parse
167 # tree if need be but, at this point, we wouldn't know what
168 # the variable name should be (TODO #2137).
169 base_type = UnresolvedType()
170 else:
171 base_type = self.symbolsymbolsymbolsymbolsymbol.datatype
172 # TODO #1857 - passing base_type as an instance of ArrayType
173 # only works because the ArrayType constructor just pulls out
174 # the intrinsic and precision properties of the type.
175 return ArrayType(base_type, shape)
176
177 # Otherwise, we're accessing a single element of the array.
178 if type(self.symbolsymbolsymbolsymbolsymbol) is Symbol:
179 return UnresolvedType()
180 if isinstance(self.symbolsymbolsymbolsymbolsymbol.datatype, UnsupportedType):
181 if (isinstance(self.symbolsymbolsymbolsymbolsymbol.datatype, UnsupportedFortranType) and
182 self.symbolsymbolsymbolsymbolsymbol.datatype.partial_datatype):
183 precision = self.symbolsymbolsymbolsymbolsymbol.datatype.partial_datatype.precision
184 intrinsic = self.symbolsymbolsymbolsymbolsymbol.datatype.partial_datatype.intrinsic
185 return ScalarType(intrinsic, precision)
186 # Since we're accessing a single element of an array of
187 # UnsupportedType we have to create a new UnsupportedFortranType.
188 # Ideally we would re-write the original Fortran
189 # declaration stored in the type. We could remove the
190 # shape in the fparser2 parse tree but, at this point, we
191 # wouldn't know what the variable name should be (TODO #2137).
192 return UnresolvedType()
193 if not isinstance(self.symbolsymbolsymbolsymbolsymbol.datatype, ArrayType):
194 # If we have an array reference to a symbol that is not considered
195 # an array by PSyIR, we don't know how to retrieve its element type
196 # (TODO #2448)
197 return UnresolvedType()
198
199 if isinstance(self.symbolsymbolsymbolsymbolsymbol.datatype.intrinsic, DataTypeSymbol):
200 return self.symbolsymbolsymbolsymbolsymbol.datatype.intrinsic
201 # TODO #1857: Really we should just be able to return
202 # self.symbol.datatype here but currently arrays of scalars are
203 # handled in a different way to all other types of array.
204 return ScalarType(self.symbolsymbolsymbolsymbolsymbol.datatype.intrinsic,
205 self.symbolsymbolsymbolsymbolsymbol.datatype.precision)
