48class LanguageWriter(PSyIRVisitor):
49 '''A convenience PSyIR visitor intermediate class. It provides
50 configuration options and functions that can be shared between different
51 language-specific visitors.
52
53 :param array_parenthesis: a tuple of two strings that contain the \
54 opening and closing parenthesis used for array accesses - e.g.:
55 ("(", ")").
56 :type array_parenthesis: 2-tuple of str
57 :param str structure_symbol: the symbol to be used to address a
58 member of a structure, e.g. "%".
59 :param kwargs: additional keyword arguments provided to the super class.
60 :type kwargs: unwrapped dict.
61
62 :raises TypeError: if any of the supplied parameters are of the wrong type.
63
64 '''
65 # pylint: disable=too-many-arguments
66 def __init__(self, array_parenthesis, structure_character, **kwargs):
67
68 super().__init__(**kwargs)
69 if not isinstance(array_parenthesis, tuple) or \
70 len(array_parenthesis) != 2:
71 raise TypeError(f"Invalid array-parenthesis parameter, must be "
72 f"a tuple of two strings, got "
73 f"'{array_parenthesis}'.")
74 if not isinstance(structure_character, str):
75 raise TypeError(f"Invalid structure_character parameter, must be "
76 f"a string, got '{array_parenthesis}'.")
77
78 self._array_parenthesis = array_parenthesis
79 self._structure_character = structure_character

85 :rtype: two element list of str

91 @property
92 def structure_character(self):
93 ''':returns: the character use to access a member of a structure in \
94 this language.
95 :rtype: str

116 def arrayreference_node(self, node):
117 '''This method is called when an ArrayReference instance is found
118 in the PSyIR tree.
119
120 :param node: an ArrayNode PSyIR node.
121 :type node: :py:class:`psyclone.psyir.nodes.ArrayNode`
122
123 :returns: the code as a string.
124 :rtype: str
125
126 :raises VisitorError: if the node does not have any children.
127
128 '''
129 if not node.children:
130 raise VisitorError(
131 f"Incomplete ArrayReference node (for symbol '{node.name}') "
132 f"found: must have one or more children.")
133 args = self.gen_indices(node.children, node.name)
134 result = f"{node.name}{self._array_parenthesis[0]}"\
135 f"{','.join(args)}{self._array_parenthesis[1]}"
136 return result

138 # ------------------------------------------------------------------------
139 def structurereference_node(self, node):
140 '''
141 Creates the code for an access to a member of a structure type.
142
143 :param node: a StructureReference PSyIR node.
144 :type node: :py:class:`psyclone.psyir.nodes.StructureReference`
145
146 :returns: the code as string.
147 :rtype: str
148
149 :raises VisitorError: if this node does not have an instance of Member\
150 as its only child.
151
152 '''
153 if len(node.children) != 1:
154 raise VisitorError(
155 f"A StructureReference must have a single child but the "
156 f"reference to symbol '{node.name}' has {len(node.children)}.")
157 if not isinstance(node.children[0], Member):
158 raise VisitorError(
159 f"A StructureReference must have a single child which is a "
160 f"sub-class of Member but the reference to symbol "
161 f"'{node.name}' has a child of type "
162 f"'{type(node.children[0]).__name__}'")
163 result = node.symbol.name + self._structure_character + \
164 self._visit(node.children[0])
165 return result

167 # ------------------------------------------------------------------------
168 def member_node(self, node):
169 '''
170 Creates the code for an access to a member of a derived type.
171
172 :param node: a Member PSyIR node.
173 :type node: :py:class:`psyclone.psyir.nodes.Member`
174
175 :returns: the code as string
176 :rtype: str
177
178 '''
179 result = node.name
180 if not node.children:
181 # A simple member access that does not access any further
182 # structures, so just return the name itself
183 return result
184
185 if isinstance(node.children[0], Member):
186 # If the first child is a member, we are accessing a structure:
187 if len(node.children) > 1:
188 # If the node has more children, any additional children are
189 # array indices. Add the indices to the output string:
190 indices = self.gen_indices(node.children[1:], node.name)
191 result += f"{self._array_parenthesis[0]}{','.join(indices)}"\
192 f"{self._array_parenthesis[1]}"
193 # Now add the first child, which is the member that is being
194 # accessed, to the output string
195 result += self._structure_character + self._visit(node.children[0])
196 else:
197 # There is no access of a structure element, add the children
198 # (which exist since this was tested above) as indices to the
199 # output string.
200 args = self.gen_indices(node.children, node.name)
201 result += f"{self._array_parenthesis[0]}{','.join(args)}"\
202 f"{self._array_parenthesis[1]}"
203 return result

207 def arrayofstructuresreference_node(self, node):
208 '''
209 Creates the code for a reference to one or more elements of an
210 array of derived types.
211
212 :param node: an ArrayOfStructuresReference PSyIR node.
213 :type node: :py:class:`psyclone.psyir.nodes.ArrayOfStructuresReference`
214
215 :returns: the code as string.
216 :rtype: str
217
218 :raises VisitorError: if the supplied node does not have the correct \
219 number and type of children.
220 '''
221 if len(node.children) < 2:
222 raise VisitorError(
223 f"An ArrayOfStructuresReference must have at least two "
224 f"children but found {len(node.children)}")
225
226 if not isinstance(node.children[0], Member):
227 raise VisitorError(
228 f"An ArrayOfStructuresReference must have a Member as its "
229 f"first child but found '{type(node.children[0]).__name__}'")
230
231 # Generate the array reference. We need to skip over the first child
232 # (as that refers to the member of the derived type being accessed).
233 args = self.gen_indices(node.children[1:])
234
235 result = (node.symbol.name + self._array_parenthesis[0] +
236 ",".join(args) + self._array_parenthesis[1] +
237 self._structure_character + self._visit(node.children[0]))
238 return result

241 def clause_node(self, node):
242 '''This method is called when a Clause instance is found in the
243 PSyIR tree. It returns the clause and its children as a string.
244
245 :param node: a Clause PSyIR node.
246 :type node: :py:class:`psyclone.psyir.nodes.Clause`
247
248 :returns: the code for this node.
249 :rtype: str
250
251 '''
252 result = node.clause_string
253
254 if len(node.children) > 0:
255 result = result + "("
256 child_list = []
257 for child in node.children:
258 child_list.append(self._visit(child))
259 result = result + ",".join(child_list)
260 result = result + ")"
261
262 return result
