58class LFRicScalarArgs(LFRicCollection):
59 '''
60 Handles the declarations of scalar kernel arguments appearing in either
61 an Invoke or a Kernel stub.
62
63 :param node: the Invoke or Kernel stub for which to manage the scalar \
64 arguments.
65 :type node: :py:class:`psyclone.domain.lfric.LFRicKern` or \
66 :py:class:`psyclone.domain.lfric.LFRicInvoke`
67
68 '''
69 def __init__(self, node):
70 super().__init__(node)
71
72 # Initialise dictionaries of 'real', 'integer' and 'logical'
73 # scalar arguments by data type and intent
74 self._scalar_args = {}
75 self._real_scalars = {}
76 self._integer_scalars = {}
77 self._logical_scalars_logical_scalars = {}
78 for intent in FORTRAN_INTENT_NAMES:
79 self._scalar_args[intent] = []
80 self._real_scalars[intent] = []
81 self._integer_scalars[intent] = []
82 self._logical_scalars_logical_scalars[intent] = []
83
84 def _invoke_declarations(self, parent):
85 '''
86 Create argument lists and declarations for all scalar arguments
87 in an Invoke.
88
89 :param parent: the f2pygen node representing the PSy-layer routine \
90 to which to add declarations.
91 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
92
93 :raises InternalError: for unsupported argument intrinsic types.
94 :raises GenerationError: if the same scalar argument has different \
95 data types in different Kernel calls \
96 within the same Invoke.
97
98 '''
99 # Create dictionary of all scalar arguments for checks
100 const = LFRicConstants()
101 self._scalar_args = self._invoke_invoke.unique_declns_by_intent(
102 const.VALID_SCALAR_NAMES)
103 # Filter scalar arguments by intent and intrinsic type
104 self._real_scalars = self._invoke_invoke.unique_declns_by_intent(
105 const.VALID_SCALAR_NAMES,
106 intrinsic_type=const.MAPPING_DATA_TYPES["gh_real"])
107 self._integer_scalars = self._invoke_invoke.unique_declns_by_intent(
108 const.VALID_SCALAR_NAMES,
109 intrinsic_type=const.MAPPING_DATA_TYPES["gh_integer"])
110 self._logical_scalars_logical_scalars = self._invoke_invoke.unique_declns_by_intent(
111 const.VALID_SCALAR_NAMES,
112 intrinsic_type=const.MAPPING_DATA_TYPES["gh_logical"])
113
114 for intent in FORTRAN_INTENT_NAMES:
115 scal = [arg.declaration_name for arg in self._scalar_args[intent]]
116 rscal = [arg.declaration_name for
117 arg in self._real_scalars[intent]]
118 iscal = [arg.declaration_name for
119 arg in self._integer_scalars[intent]]
120 lscal = [arg.declaration_name for
121 arg in self._logical_scalars_logical_scalars[intent]]
122 # Add "real", "integer" and "logical" scalar lists for checks
123 decl_scal = rscal + iscal + lscal
124 # Check for unsupported intrinsic types
125 scal_inv = sorted(set(scal) - set(decl_scal))
126 if scal_inv:
127 raise InternalError(
128 f"Found unsupported intrinsic types for the scalar "
129 f"arguments {scal_inv} to Invoke '{self._invoke.name}'. "
130 f"Supported types are {const.VALID_INTRINSIC_TYPES}.")
131 # Check that the same scalar name is not found in either of
132 # 'real', 'integer' or 'logical' scalar lists (for instance if
133 # passed to one kernel as a 'real' and to another kernel as an
134 # 'integer' scalar)
135 scal_multi_type = [item for item, count in
136 Counter(decl_scal).items() if count > 1]
137 if scal_multi_type:
138 raise GenerationError(
139 f"Scalar argument(s) {scal_multi_type} in Invoke "
140 f"'{self._invoke.name}' have different metadata for data "
141 f"type ({list(const.MAPPING_DATA_TYPES.keys())}) in "
142 f"different kernels. This is invalid.")
143
144 # Create declarations
145 self._create_declarations(parent)
146
147 def _stub_declarations(self, parent):
148 '''
149 Create and add declarations for all scalar arguments in
150 a Kernel stub.
151
152 :param parent: node in the f2pygen AST representing the Kernel stub \
153 to which to add declarations.
154 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
155
156 :raises InternalError: for an unsupported argument data type.
157
158 '''
159 # Extract all scalar arguments
160 for arg in self._calls[0].arguments.args:
161 if arg.is_scalar:
162 self._scalar_args[arg.intent].append(arg)
163
164 const = LFRicConstants()
165 # Filter scalar arguments by intent and data type
166 for intent in FORTRAN_INTENT_NAMES:
167 for arg in self._scalar_args[intent]:
168 if arg.descriptor.data_type == "gh_real":
169 self._real_scalars[intent].append(arg)
170 elif arg.descriptor.data_type == "gh_integer":
171 self._integer_scalars[intent].append(arg)
172 elif arg.descriptor.data_type == "gh_logical":
173 self._logical_scalars_logical_scalars[intent].append(arg)
174 else:
175 raise InternalError(
176 f"Found an unsupported data type "
177 f"'{arg.descriptor.data_type}' for the scalar "
178 f"argument '{arg.declaration_name}'. Supported types "
179 f"are {const.VALID_SCALAR_DATA_TYPES}.")
180
181 # Create declarations
182 self._create_declarations(parent)
183
184 def _create_declarations(self, parent):
185 '''Add declarations for the scalar arguments.
186
187 :param parent: the f2pygen node in which to insert declarations \
188 (Invoke or Kernel).
189 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
190
191 :raises InternalError: if neither self._invoke nor \
192 self._kernel are set.
193
194 '''
195 const = LFRicConstants()
196 const_mod = const.UTILITIES_MOD_MAP["constants"]["module"]
197 const_mod_uses = None
198 if self._invoke_invoke:
199 const_mod_uses = self._invoke_invoke.invokes.psy.infrastructure_modules[
200 const_mod]
201 # Real scalar arguments
202 for intent in FORTRAN_INTENT_NAMES:
203 if self._real_scalars[intent]:
204 # Filter scalars based on precision
205 real_scalars_precision_map = OrderedDict()
206 for real_scalar in self._real_scalars[intent]:
207 try:
208 real_scalars_precision_map[
209 real_scalar.precision].append(real_scalar)
210 except KeyError:
211 # This precision has not been seen before so
212 # create a new entry
213 real_scalars_precision_map[
214 real_scalar.precision] = [real_scalar]
215 # Declare scalars
216 for real_scalar_kind, real_scalars_list in \
217 real_scalars_precision_map.items():
218 real_scalar_type = real_scalars_list[0].intrinsic_type
219 real_scalar_names = [arg.declaration_name for arg
220 in real_scalars_list]
221 parent.add(
222 DeclGen(parent, datatype=real_scalar_type,
223 kind=real_scalar_kind,
224 entity_decls=real_scalar_names,
225 intent=intent))
226 if self._invoke_invoke:
227 const_mod_uses.add(real_scalar_kind)
228 elif self._kernel_kernel_kernel:
229 self._kernel_kernel_kernel.argument_kinds.add(real_scalar_kind)
230 else:
231 raise InternalError(
232 "Expected the declaration of real scalar kernel "
233 "arguments to be for either an invoke or a "
234 "kernel stub, but it is neither.")
235
236 # Integer scalar arguments
237 for intent in FORTRAN_INTENT_NAMES:
238 if self._integer_scalars[intent]:
239 dtype = self._integer_scalars[intent][0].intrinsic_type
240 dkind = self._integer_scalars[intent][0].precision
241 integer_scalar_names = [arg.declaration_name for arg
242 in self._integer_scalars[intent]]
243 parent.add(
244 DeclGen(parent, datatype=dtype, kind=dkind,
245 entity_decls=integer_scalar_names,
246 intent=intent))
247 if self._invoke_invoke:
248 const_mod_uses.add(dkind)
249 elif self._kernel_kernel_kernel:
250 self._kernel_kernel_kernel.argument_kinds.add(dkind)
251 else:
252 raise InternalError(
253 "Expected the declaration of integer scalar kernel "
254 "arguments to be for either an invoke or a "
255 "kernel stub, but it is neither.")
256
257 # Logical scalar arguments
258 for intent in FORTRAN_INTENT_NAMES:
259 if self._logical_scalars_logical_scalars[intent]:
260 dtype = self._logical_scalars_logical_scalars[intent][0].intrinsic_type
261 dkind = self._logical_scalars_logical_scalars[intent][0].precision
262 logical_scalar_names = [arg.declaration_name for arg
263 in self._logical_scalars_logical_scalars[intent]]
264 parent.add(
265 DeclGen(parent, datatype=dtype, kind=dkind,
266 entity_decls=logical_scalar_names,
267 intent=intent))
268 if self._invoke_invoke:
269 const_mod_uses.add(dkind)
270 elif self._kernel_kernel_kernel:
271 self._kernel_kernel_kernel.argument_kinds.add(dkind)
272 else:
273 raise InternalError(
274 "Expected the declaration of logical scalar kernel "
275 "arguments to be for either an invoke or a "
276 "kernel stub, but it is neither.")
