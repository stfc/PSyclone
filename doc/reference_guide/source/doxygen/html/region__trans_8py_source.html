38'''This module contains the base class RegionTrans. All transformations which
39 act on a region of code (list of PSyIR nodes) sub-class this one.
40'''

42import abc

44from psyclone.errors import LazyString
45from psyclone.psyGen import Transformation
46from psyclone.psyir.transformations.transformation_error \

51class RegionTrans(Transformation, metaclass=abc.ABCMeta):
52 # Avoid pylint warning about abstract functions (apply, name) not
53 # overwritten:
54 # pylint: disable=abstract-method,arguments-differ
55 '''
56 This abstract class is a base class for all transformations that act
57 on a list of nodes. It gives access to a validate function that
58 makes sure that the nodes in the list are in the same order as in
59 the original AST, no node is duplicated, and that all nodes have
60 the same parent. We also check that all nodes to be enclosed are
61 valid for this transformation - this requires that the sub-class
62 populate the `excluded_node_types` tuple.
63
64 '''
65 # The types of Node that are excluded from within this region. Must be
66 # populated by sub-class.
67 excluded_node_types = ()
68
69 def get_node_list(self, nodes):
70 '''This is a helper function for region based transformations.
71 The parameter for any of those transformations is either a single
72 node, a schedule, or a list of nodes. This function converts this
73 into a list of nodes according to the parameter type. This function
74 will always return a copy, to avoid issues e.g. if a child list
75 of a node should be provided, and a transformation changes the order
76 in this list (which would then also change the order of the
77 nodes in the tree).
78
79 :param nodes: can be a single node, a schedule or a list of nodes.
80 :type nodes: Union[:py:obj:`psyclone.psyir.nodes.Node`,
81 :py:obj:`psyclone.psyir.nodes.Schedule`,
82 List[:py:obj:`psyclone.psyir.nodes.Node`]
83 :param options: a dictionary with options for transformations.
84 :type options: Optional[Dict[str,Any]]
85
86 :returns: a list of nodes.
87 :rtype: List[:py:class:`psyclone.psyir.nodes.Node`]
88
89 :raises TransformationError: if the supplied parameter is neither a \
90 single Node, nor a Schedule, nor a list of Nodes.
91
92 '''
93 if isinstance(nodes, list) and \
94 all(isinstance(node, Node) for node in nodes):
95 # We still need to return a copy, since the user might have
96 # provided Node.children as parameter.
97 return nodes[:]
98 if isinstance(nodes, Schedule):
99 # We've been passed a Schedule so default to enclosing its
100 # children.
101 return nodes.children[:]
102 if isinstance(nodes, Node):
103 # Single node that's not a Schedule
104 return [nodes]
105
106 arg_type = str(type(nodes))
107 raise TransformationError(f"Error in {self.name}: "
108 f"Argument must be a single Node in a "
109 f"Schedule, a Schedule or a list of Nodes "
110 f"in a Schedule but have been passed an "
111 f"object of type: {arg_type}")
112
113 def validate(self, nodes, options=None):
114 '''Checks that the nodes in node_list are valid for a region
115 transformation.
116
117 :param nodes: can be a single node, a schedule or a list of nodes.
118 :type nodes: Union[:py:obj:`psyclone.psyir.nodes.Node`,
119 :py:obj:`psyclone.psyir.nodes.Schedule`,
120 List[:py:obj:`psyclone.psyir.nodes.Node`]
121 :param options: a dictionary with options for transformations.
122 :type options: Optional[Dict[str,Any]]
123 :param bool options["node-type-check"]: this flag controls if the \
124 type of the nodes enclosed in the region should be tested \
125 to avoid using unsupported nodes inside a region.
126
127 :raises TransformationError: if the nodes in the list are not \
128 in the original order in which they are in the AST, \
129 a node is duplicated or the nodes have different parents.
130 :raises TransformationError: if any of the nodes to be enclosed in \
131 the region are of an unsupported type.
132 :raises TransformationError: if the parent of the supplied Nodes is \
133 not a Schedule or a Directive.
134 :raises TransformationError: if the nodes are in a NEMO \
135 Schedule and the transformation acts on the child of a \
136 single-line If or Where statment.
137 :raises TransformationError: if the supplied options are not a \
138 dictionary.
139
140 '''
141 # pylint: disable=too-many-branches
142
143 # Ensure we are always validating a list of nodes, even if only
144 # one was supplied via the `node_list` argument.
145 node_list = self.get_node_list(nodes)
146
147 if not options:
148 options = {}
149 if not isinstance(options, dict):
150 raise TransformationError(
151 f"Transformation apply method options argument must be a "
152 f"dictionary but found '{type(options).__name__}'.")
153 node_parent = node_list[0].parent
154 prev_position = -1
155 for child in node_list:
156 if child.parent is not node_parent:
157 raise TransformationError(
158 f"Error in {self.name} transformation: supplied nodes "
159 f"are not children of the same parent.")
160 if prev_position >= 0 and prev_position+1 != child.position:
161 raise TransformationError(
162 f"Children are not consecutive children of one parent: "
163 f"child '{child}' has position {child.position}, but "
164 f"previous child had position {prev_position}.")
165 prev_position = child.position
166
167 # Check that the proposed region contains only supported node types
168 if options.get("node-type-check", True):
169 for child in node_list:
170 for bad in child.walk(self.excluded_node_types):
171 # Ideally we'd just use `bad.debug_string()` always
172 # but this sometimes causes errors if nodes do their
173 # own validation (rather than relying on the backend),
174 # e.g. ACCEnterDataDirective.begin_string().
175 # pylint: disable-next=import-outside-toplevel
176 from psyclone.psyir.nodes import CodeBlock
177 if isinstance(bad, CodeBlock):
178 msg = LazyString(
179 lambda: f"Nodes of type '{type(bad).__name__}' "
180 f"cannot be enclosed by a {self.name} "
181 f"transformation but found:\n{bad.debug_string()}")
182 else:
183 msg = (f"Nodes of type '{type(bad).__name__}' cannot "
184 f"be enclosed by a {self.name} transformation.")
185 raise TransformationError(msg)
186
187 # If we've been passed a list that contains one or more Schedules
188 # then something is wrong. e.g. two Schedules that are both children
189 # of an IfBlock would imply that the transformation is being applied
190 # around both the if-body and the else-body and that doesn't make
191 # sense.
192 if (len(node_list) > 1 and
193 any(isinstance(node, Schedule) for node in node_list)):
194 raise TransformationError(
195 "Cannot apply a transformation to multiple nodes when one "
196 "or more is a Schedule. Either target a single Schedule "
197 "or the children of a Schedule.")
198
199 # Sanity check that we've not been passed the condition part of
200 # an If statement or the bounds of a Loop. If the parent node is
201 # a Loop or IfBlock then we can only accept a single Schedule.
202 if not isinstance(node_parent, Schedule) and \
203 not isinstance(node_list[0], Schedule):
204 # We've already checked for lists with len > 1 that contain a
205 # Schedule above so if the first item is a Schedule then that's
206 # all the list contains.
207 raise TransformationError(
208 "Cannot apply transformation to the immediate children of a "
209 "Loop/IfBlock unless it is to a single Schedule representing"
210 " the Loop/If/Else body.")
