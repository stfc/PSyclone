Detailed Description

Examines the supplied Schedule and adds OpenACC "update" directives
for any data accessed outside of a kernels or parallel region.
For example:

>>> from psyclone.psyir.backend.fortran import FortranWriter
>>> from psyclone.psyir.frontend.fortran import FortranReader
>>> from psyclone.psyir.nodes import Routine
>>> from psyclone.psyir.transformations import ACCUpdateTrans
>>>
>>> code = """
... subroutine run_it()
...   real :: a(10)
...   a(:) = 7.0
... end subroutine run_it"""
>>>
>>> psyir = FortranReader().psyir_from_source(code)
>>> routine = psyir.walk(Routine)[0]
>>>
>>> # Add update directives
>>> uptrans = ACCUpdateTrans()
>>> uptrans.apply(routine)
>>>
>>> # Uncomment to see a text view of the new routine schedule
>>> # print(routine.view())
>>> print(FortranWriter()(routine))
subroutine run_it()
  real, dimension(10) :: a
<BLANKLINE>
  !$acc update if_present host(a)
  a(:) = 7.0
  !$acc update if_present device(a)
<BLANKLINE>
end subroutine run_it
<BLANKLINE>

170 Recursively identify those statements that are not being executed
171 on the device and add any required OpenACC update directives.

177 # We must walk through the Schedule and find those nodes
178 # representing contiguous regions of code that are not
179 # executed on the device. Any Call nodes are taken as
180 # boundaries of such regions because it may be that (part of)
181 # their bodies are executed on the device. Allocate and
182 # deallocate intrinsics are also taken as boundaries of such
183 # regions.
184 node_list = []
185 for child in sched[:]:
186 if isinstance(child, self._ACC_IGNORE):
187 continue
188 found = False
189 for node in child.walk(Node):
190 if isinstance(node, self._ACC_COMPUTE) or \
191 self._may_compute(node):
192 found = True
193 break
194 if not found:
195 node_list.append(child)
196 else:
197 self._add_update_directives(node_list)
198 node_list.clear()
199 if self._may_compute(child):
200 # Conservatively add an update host statement just
201 # before the node.
202 self._add_update_directives([child])
203 elif isinstance(child, IfBlock):
204 # Add any update host statements that are required due to
205 # (read) accesses within the condition of the If.
206 self._add_update_directives([child.condition])
207 # Recurse down into the if body
208 self._add_updates_to_schedule(child.if_body)
209 if child.else_body:
210 self._add_updates_to_schedule(child.else_body)
211 elif isinstance(child, Loop):
212 # A loop on the host with code that may be on the device.
213 # The loop start, stop and step values could all
214 # potentially have been written to on the device.
215 self._add_update_directives([child.start_expr,
216 child.stop_expr,
217 child.step_expr])

219 self._add_updates_to_schedule(child.loop_body)

222 # last nodes that represent computation on the host?

375 '''
376 Retrieves, amongst the host region signatures in host_sig, those that
377 require synchronisation because of accesses in the compute regions
378 within the statements in dep_stmts.

380 :param dep_stmts: list of statements which may include compute regions.

389 # If there is a statement (e.g. a call) among the dependent
390 # statements that may launch device kernels, we conservatively
391 # assume a dependency for all variables in the host region
392 # regardless of access mode. For the performance of the Python
393 # code there is an assumption here that _may_compute only
394 # returns True for Call nodes.
