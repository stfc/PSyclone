115 '''Lower this GOcean-specific container to language level psyir.

200 @staticmethod
201 def create_from_psyir(symbol):
202 '''Create a new instance of GOceanKernelMetadata populated with
203 metadata from a kernel in language-level PSyIR.
204
205 :param symbol: the symbol in which the metadata is stored \
206 in language-level PSyIR.
207 :type symbol: :py:class:`psyclone.psyir.symbols.DataTypeSymbol`
208
209 :returns: an instance of GOceanKernelMetadata.
210 :rtype: :py:class:`psyclone.domain.gocean.kernel.psyir.\
211 GOceanKernelMetadata`
212
213 :raises TypeError: if the symbol argument is not the expected \
214 type.
215 :raises InternalError: if the datatype of the provided symbol \
216 is not the expected type.
217
218 '''
219 if not isinstance(symbol, DataTypeSymbol):
220 raise TypeError(
221 f"Expected a DataTypeSymbol but found a "
222 f"{type(symbol).__name__}.")
223
224 datatype = symbol.datatype
225
226 if not isinstance(datatype, UnsupportedFortranType):
227 raise InternalError(
228 f"Expected kernel metadata to be stored in the PSyIR as "
229 f"an UnsupportedFortranType, but found "
230 f"{type(datatype).__name__}.")
231
232 # In an UnsupportedFortranType, the declaration is stored as a
233 # string, so use create_from_fortran_string()
234 return GOceanKernelMetadata.create_from_fortran_string(
235 datatype.declaration)

237 @staticmethod
238 def create_from_fortran_string(fortran_string):
239 '''Create a new instance of GOceanKernelMetadata populated with
240 metadata stored in a fortran string.
241
242 :param str fortran_string: the metadata stored as Fortran.
243
244 :returns: an instance of GOceanKernelMetadata.
245 :rtype: :py:class:`psyclone.domain.gocean.kernel.psyir.\
246 GOceanKernelMetadata`
247
248 :raises ValueError: if the string does not contain a fortran \
249 derived type.
250 :raises ParseError: if the metadata has an unexpected format.
251
252 '''
253 kernel_metadata = GOceanKernelMetadata()
254
255 # Ensure the Fortran parser is initialised.
256 std = Config.get().fortran_standard
257 _ = ParserFactory().create(std=std)
258 reader = FortranStringReader(fortran_string)
259 try:
260 spec_part = Fortran2003.Derived_Type_Def(reader)
261 except Fortran2003.NoMatchError:
262 # pylint: disable=raise-missing-from
263 raise ValueError(
264 f"Expected kernel metadata to be a Fortran derived type, but "
265 f"found '{fortran_string}'.")
266
267 kernel_metadata.name = spec_part.children[0].children[1].tostr()
268
269 const = GOceanConstants()
270 # Extract and store the required 'iterates_over',
271 # 'index_offset' and 'code' properties from the parse tree
272
273 # the value of iterates over (go_all_pts, ...)
274 value = GOceanKernelMetadata._get_property(
275 spec_part, "iterates_over").string
276 kernel_metadata.iterates_over = value
277
278 # the value of index offset (NE, ...)
279 value = GOceanKernelMetadata._get_property(
280 spec_part, "index_offset").string
281 kernel_metadata.index_offset = value
282
283 # the name of the procedure that this metadata refers to.
284 kernel_metadata.procedure_name = GOceanKernelMetadata._get_property(
285 spec_part, "code").string
286
287 # meta_args contains arguments which have
288 # properties. Therefore create appropriate (GridArg, ScalarArg
289 # or FieldArg) instances to capture this information.
290 meta_args = GOceanKernelMetadata._get_property(
291 spec_part, "meta_args")
292 args = walk(meta_args, Fortran2003.Ac_Value_List)
293 if not args:
294 raise ParseError(
295 f"meta_args should be a list, but found "
296 f"'{str(meta_args)}' in '{spec_part}'.")
297
298 for meta_arg in args[0].children:
299 if len(meta_arg.children[1].children) == 2:
300 # Grid args have 2 arguments
301 kernel_metadata.meta_args.append(
302 GOceanKernelMetadata.GridArg(meta_arg, kernel_metadata))
303 elif len(meta_arg.children[1].children) == 3:
304 # scalar and field args have 3 arguments
305 arg2 = meta_arg.children[1].children[1].string.lower()
306 if arg2 in const.VALID_FIELD_GRID_TYPES:
307 kernel_metadata.meta_args.append(
308 GOceanKernelMetadata.FieldArg(
309 meta_arg, kernel_metadata))
310 elif arg2 in const.VALID_SCALAR_TYPES:
311 kernel_metadata.meta_args.append(
312 GOceanKernelMetadata.ScalarArg(
313 meta_arg, kernel_metadata))
314 else:
315 raise ParseError(
316 f"Expected a 'meta_arg' entry with 3 arguments to "
317 f"either be a field or a scalar, but found '{arg2}' "
318 f"as the second argument instead of "
319 f"'{const.VALID_FIELD_GRID_TYPES}' (fields) or "
320 f"'{const.VALID_SCALAR_TYPES}' (scalars).")
321 else:
322 raise ParseError(
323 f"'meta_args' should have either 2 or 3 arguments, but "
324 f"found {len(meta_arg.children[1].children)} in "
325 f"{str(meta_arg)}.")
326
327 return kernel_metadata

329 @staticmethod
330 def _get_property(spec_part, property_name):
331 '''Internal utility that gets the property 'property_name' from an
332 fparser2 tree capturing gocean metadata. It is assumed that
333 the code property is part of a type bound procedure and that
334 the other properties are part of the data declarations.
335
336 :param spec_part: the fparser2 parse tree containing the metadata.
337 :type spec_part: :py:class:`fparser.two.Fortran2003.Derived_Type_Def`
338 :param str property_name: the name of the property whose value \
339 is being extracted from the metadata.
340
341 :returns: the value of the property.
342 :rtype: :py:class:`fparser.two.Fortran2003.Name | \
343 :py:class:`fparser.two.Fortran2003.Array_Constructor`
344
345 :raises ParseError: if the property name is not found in the \
346 metadata.
347
348 '''
349 if property_name.lower() == "code":
350 # The value of 'code' should be found in a type bound
351 # procedure (after the contains keyword)
352 type_bound_procedure = get_child(
353 spec_part, Fortran2003.Type_Bound_Procedure_Part)
354 if not type_bound_procedure:
355 raise ParseError(
356 f"No type-bound procedure found within a 'contains' "
357 f"section in '{spec_part}'.")
358 if len(type_bound_procedure.children) != 2:
359 raise ParseError(
360 f"Expecting a type-bound procedure, but found "
361 f"'{spec_part}'.")
362 specific_binding = type_bound_procedure.children[1]
363 if not isinstance(specific_binding, Fortran2003.Specific_Binding):
364 raise ParseError(
365 f"Expecting a specific binding for the type-bound "
366 f"procedure, but found '{specific_binding}' in "
367 f"'{spec_part}'.")
368 binding_name = specific_binding.children[3]
369 procedure_name = specific_binding.children[4]
370 if binding_name.string.lower() != "code" and procedure_name:
371 raise ParseError(
372 f"Expecting the type-bound procedure binding-name to be "
373 f"'code' if there is a procedure name, but found "
374 f"'{str(binding_name)}' in '{spec_part}'.")
375 if not procedure_name:
376 # Support the alternative metadata format that does
377 # not include 'code =>'
378 procedure_name = binding_name
379 return procedure_name
380
381 # The 'property_name' will be declared within Component_Part.
382 component_part = get_child(spec_part, Fortran2003.Component_Part)
383 if not component_part:
384 raise ParseError(
385 f"No declarations were found in the kernel metadata: "
386 f"'{spec_part}'.")
387 # Each name/value pair will be contained within a Component_Decl
388 for component_decl in walk(component_part, Fortran2003.Component_Decl):
389 # Component_Decl(Name('name') ...)
390 name = component_decl.children[0].string
391 if name.lower() == property_name.lower():
392 # The value will be contained in a Component_Initialization
393 comp_init = get_child(
394 component_decl, Fortran2003.Component_Initialization)
395 if not comp_init:
396 raise ParseError(
397 f"No value for property {property_name} was found "
398 f"in '{spec_part}'.")
399 # Component_Initialization('=', Name('name'))
400 return comp_init.children[1]
401 raise ParseError(
402 f"'{property_name}' was not found in {str(spec_part)}.")

404 def fortran_string(self):
405 '''
406 :returns: the metadata represented by this instance as Fortran.
407 :rtype: str
408 '''
409 go_args = []
410 for go_arg in self.meta_argsmeta_args:
411 go_args.append(go_arg.fortran_string())
412 go_args_str = ", &\n".join(go_args)
413 result = (
414 f"TYPE, EXTENDS(kernel_type) :: {self.name}\n"
415 f" TYPE(go_arg), DIMENSION({len(self.meta_args)}) :: "
416 f"meta_args = (/ &\n{go_args_str}/)\n"
417 f" INTEGER :: ITERATES_OVER = {self.iterates_over}\n"
418 f" INTEGER :: INDEX_OFFSET = {self.index_offset}\n"
419 f" CONTAINS\n"
420 f" PROCEDURE, NOPASS :: code => {self.procedure_name}\n"
421 f"END TYPE {self.name}\n")
422 return result

661 class FieldArg():
662 '''Internal class to capture Kernel metadata information for
663 a field argument.
664
665 :param meta_arg: an fparser2 tree representation of the metadata.
666 :type meta_arg: :py:class:`fparser.two.Fortran2003.Part_Ref`
667
668 :param parent: a KernelMetadataSymbol instance that captures \
669 other parts of the metadata and references this instance.
670 :type parent: :py:class`psyclone.psyir.common.kernel. \
671 KernelMetadataSymbol`
672
673 :raises ParseError: if the metadata does not contain three \
674 arguments.
675
676 '''
677 def __init__(self, meta_arg, parent):
678 self._parent = parent
679
680 arg_list = meta_arg.children[1]
681 if len(arg_list.children) != 3:
682 raise ParseError(
683 f"There should be 3 kernel metadata entries for a field "
684 f"argument, but found {len(arg_list.children)} in "
685 f"{str(meta_arg)}.")
686
687 # access descriptor (go_read, go_write, ...)
688 access = arg_list.children[0].string
689 self.accessaccessaccess = access
690
691 # grid point type (go_ct, ...)
692 grid_point_type = arg_list.children[1].string
693 self.grid_point_typegrid_point_typegrid_point_type = grid_point_type
694
695 if isinstance(arg_list.children[2], Fortran2003.Name):
696 # form of access (go_pointwise, ...)
697 form = arg_list.children[2].string
698 self.formformformform = form
699 self._stencil_stencil = None
700 else: # Stencil form (go_stencil) and stencil value
701 # (e.g. [000, 111, 000])
702 name = arg_list.children[2].children[0].string
703 self._validate_stencil_name(name)
704 self.formformformform = name
705 stencil = []
706 for stencil_dim in arg_list.children[2].children[1].children:
707 stencil.append(stencil_dim.children[0])
708 self._stencil_stencil = stencil

842 '''Check that 'value_list' is a valid list of 'stencil' elements

871 f"'{type(value).__name__}'.")
