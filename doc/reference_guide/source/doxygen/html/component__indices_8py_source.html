37'''This module provides a class to manage indices in variable accesses.'''
38
39from __future__ import print_function, absolute_import
40
41
42from psyclone.errors import InternalError
43
44
45class ComponentIndices():
46 '''This class stores index information for variable accesses. It stores
47 one index list for each component of a variable, e.g. for `a(i)%b(j)`
48 it would store `[ [i], [j] ]`. Even for scalar accesses an empty list
49 is stored, so `a` would have the component indices `[ [] ]`, and `a%b`
50 would have `[ [], [] ]`. Each member of this list of lists is the PSyIR
51 node describing the array expression used.
52
53 As a shortcut, the `indices` parameter can be None or an empty list
54 (which then creates the component indices as `[[]]`, i.e. indicating
55 a scalar access), a list `l` (which will then create the component
56 indices as `[l]`, i.e. a single component variable, which uses
57 all the indices in the list `l` as array indices).
58
59 TODO #845 - the constructor should check that the things it is passed
60 are PSyIR nodes. Currently it is sometimes given strings.
61
62 :param indices: the indices from which to create this object.
63 :type indices: None, [], a list or a list of lists of \
64 :py:class:`psyclone.psyir.nodes.Node`
65
66 :raises InternalError: if the indices parameter is not None, a list \
67 or a list of lists.
68 :raises InternalError: if the indices parameter is a list, and some \
69 but not all members are a list.
70
71 '''
72 def __init__(self, indices=None):
73 if indices is None or indices == []:
74 self._component_indices = [[]]
75 elif isinstance(indices, list):
76 if all(isinstance(indx, list) for indx in indices):
77 # All elements of the indices are a list - so we
78 # got a list of lists:
79 self._component_indices = indices
80 elif all(not isinstance(indx, list) for indx in indices):
81 self._component_indices = [indices]
82 else:
83 raise InternalError(f"ComponentIndices: Invalid "
84 f"list parameter '{indices}' - some "
85 f"elements but not all are lists")
86 else:
87 raise InternalError(f"Index object in ComponentIndices "
88 f"constructor must be None, a list or "
89 f"list of lists, got '{indices}'")
90
91 # ------------------------------------------------------------------------
92 def __str__(self):
93 '''Returns a string representing the indices.'''
94 return str(self._component_indices)
95
96 # ------------------------------------------------------------------------
97 def iterate(self):
98 '''Allows iterating over all component indices. It returns a tuple
99 with two elements, the first one indicating the component, the second
100 the dimension for which the index is. The return tuple can be used
101 in a dictionary access (see `__getitem__`) of this object.
102
103 :returns: a tuple of the component index and index.
104 :rtype: tuple(int, int)
105
106 '''
107 for comp_ind, component in enumerate(self._component_indices):
108 for indx in range(len(component)):
109 yield (comp_ind, indx)
110
111 # ------------------------------------------------------------------------
112 def __getitem__(self, indx):
113 '''Allows to use this class as a dictionary. If `indx` is an integer,
114 the list of indices for the specified component is returned. If `indx`
115 is a tuple (as returned from `iterate`), it will return the PSyIR of
116 the index for the specified component at the specified dimension.
117
118 :returns: either the list of indices for a component, or the index \
119 PSyIR node for the specified tuple.
120 :rtype: list of :py:class:`psyclone.psyir.nodes.Node`, or \
121 :py:class:`psyclone.psyir.nodes.Node`
122
123 :raises IndexError: if a tuple is given and one of the indices is \
124 outside of the valid range.
125
126 '''
127 if isinstance(indx, tuple):
128 if indx[0] < 0 or indx[0] >= len(self._component_indices):
129 raise IndexError(f"First index ({indx[0]}) of {indx} is out "
130 f"of range.")
131 if indx[1] < 0 or indx[1] >= len(self._component_indices[indx[0]]):
132 raise IndexError(f"Second index ({indx[1]}) of {indx} is out "
133 f"of range.")
134 return self._component_indices[indx[0]][indx[1]]
135 return self._component_indices[indx]
136
137 # ------------------------------------------------------------------------
138 def __len__(self):
139 ''':returns: the number of components in this class.
140 :rtype: int '''
141 return len(self._component_indices)
142
143 # ------------------------------------------------------------------------
144 @property
145 def indices_lists(self):
146 ''':returns: the component indices list of lists.
147 :rtype: list of list of :py:class:`psyclone.psyir.nodes.Node`
148 '''
149 return self._component_indices
150
151 # ------------------------------------------------------------------------
152 def is_array(self):
153 '''Test whether there is an index used in any component. E.g. an access
154 like `a(i)%b` with indices `[ [i], [] ]` would still be considered an
155 array.
156
157 :returns: whether any of the variable components uses an index, i.e.\
158 the variable is an array.
159 :rtype: bool
160 '''
161 return any(grp for grp in self._component_indices)
162
163 # ------------------------------------------------------------------------
164 def get_subscripts_of(self, set_of_vars):
165 '''This function returns a flat list of which variable from the
166 given set of variables is used in each subscript. For example, the
167 access `a(i+i2)%b(j*j+k,k)%c(l,5)` would have the component_indices
168 `[[i+i2], [j*j+k,k], [l,5]]`. If the set of variables is
169 `(i,j,k)`, then `get_subscripts_of` would return
170 `[{i},{j,k},{k},{l},{}]`.
171
172 :param set_of_vars: set with name of all variables.
173 :type set_of_vars: Set[str]
174
175 :return: a list of sets with all variables used in the corresponding \
176 array subscripts as strings.
177 :rtype: List[Set[str]]
178
179 '''
180 # Circular import
181 # pylint: disable=import-outside-toplevel
182 from psyclone.core import VariablesAccessInfo
183
184 indices = []
185 for i in self.iterate():
186 indx = self[i]
187 index_vars = VariablesAccessInfo(indx)
188 unique_vars = set(str(sig) for sig in index_vars.keys())
189 unique_vars = unique_vars.intersection(set_of_vars)
190 indices.append(unique_vars)
191 return indices
