51class ExtractTrans(PSyDataTrans):
52 '''This transformation inserts an ExtractNode or a node derived
53 from ExtractNode into the PSyIR of a schedule. At code creation
54 time this node will use the PSyData API to create code that can
55 write the input and output parameters to a file. The node might
56 also create a stand-alone driver program that can read the created
57 file and then execute the instrumented region.
58 Examples are given in the derived classes DynamoExtractTrans and
59 GOceanExtractTrans.
60
61 After applying the transformation the Nodes marked for extraction are
62 children of the ExtractNode.
63 Nodes to extract can be individual constructs within an Invoke (e.g.
64 Loops containing a Kernel or BuiltIn call) or entire Invokes. This
65 functionality does not support distributed memory.
66
67 :param node_class: The Node class of which an instance will be inserted \
68 into the tree (defaults to ExtractNode), but can be any derived class.
69 :type node_class: :py:class:`psyclone.psyir.nodes.ExtractNode` or \
70 derived class
71
72 '''
73 # The types of node that this transformation cannot enclose
74 excluded_node_types = (CodeBlock, ExtractNode,
75 HaloExchange, GlobalSum)

83 @staticmethod
84 def determine_postfix(read_write_info, postfix="_post"):
85 '''
86 This function prevents any name clashes that can occur when adding
87 the postfix to output variable names. For example, if there is an
88 output variable 'a', the driver (and the output file) will contain
89 two variables: 'a' and 'a_post'. But if there is also another variable
90 called 'a_post', a name clash would occur (two identical keys in the
91 output file, and two identical local variables in the driver). In
92 order to avoid this, the suffix 'post' is changed (to 'post0',
93 'post1', ...) until any name clashes are avoided. This works for
94 structured and non-structured types.
95
96 :param read_write_info: information about all input and output \
97 parameters.
98 :type read_write_info: :py:class:`psyclone.psyir.tools.ReadWriteInfo`
99 :param str postfix: the postfix to append to each output variable.
100
101 :returns: a postfix that can be added to each output variable without
102 generating a name clash.
103 :rtype: str
104
105 '''
106 suffix = ""
107 # Create the a set of all input and output variables (to avoid
108 # checking input+output variables more than once)
109 all_vars = read_write_info.set_of_all_used_vars
110 # The signatures in the input/output list need to be converted
111 # back to strings to easily append the suffix.
112 all_vars_string = [str(input_var) for _, input_var in all_vars]
113 while any(str(out_sig)+postfix+str(suffix) in all_vars_string
114 for out_sig in read_write_info.signatures_written):
115 if suffix == "":
116 suffix = 0
117 else:
118 suffix += 1
119 return postfix+str(suffix)

122 def validate(self, node_list, options=None):
123 # pylint: disable=arguments-renamed
124 '''Performs validation checks specific to extract-based
125 transformations.
126
127 :param node_list: the list of Node(s) we are checking.
128 :type node_list: list of :py:class:`psyclone.psyir.nodes.Node`
129 :param options: a dictionary with options for transformations.
130 :type options: Optional[Dict[str, Any]]
131
132 :raises TransformationError: if transformation is applied to a \
133 Kernel or a BuiltIn call without its \
134 parent Loop.
135 :raises TransformationError: if transformation is applied to a Loop \
136 without its parent Directive when \
137 optimisations are applied.
138 :raises TransformationError: if transformation is applied to an \
139 orphaned Directive without its parent \
140 Directive.
141 '''
142
143 # Check ExtractTrans specific constraints.
144
145 # Check constraints not covered by excluded_node_types for
146 # individual Nodes in node_list.
147 for node in node_list:
148
149 # Check that ExtractNode is not inserted between a Kernel or
150 # a BuiltIn call and its parent Loop.
151 if isinstance(node, (Kern, BuiltIn)) and \
152 isinstance(node.parent.parent, Loop):
153 raise TransformationError(
154 f"Error in {self.name}: Application to a Kernel or a "
155 f"Built-in call without its parent Loop is not allowed.")
156
157 # Check that ExtractNode is not inserted between a Loop and its
158 # parent Directive when optimisations are applied, as this may
159 # result in including the end Directive for extraction but
160 # not the beginning.
161 if isinstance(node, Loop) and isinstance(node.parent, Schedule) \
162 and isinstance(node.parent.parent, Directive):
163 raise TransformationError(
164 f"Error in {self.name}: Application to a Loop without its "
165 f"parent Directive is not allowed.")
166
167 # Check that ExtractNode is not inserted within a thread
168 # parallel region when optimisations are applied. For instance,
169 # this may be between an orphaned Directive (e.g. OMPDoDirective,
170 # ACCLoopDirective) and its ancestor Directive (e.g. ACC or OMP
171 # Parallel Directive) or within an OMPParallelDoDirective.
172 if node.ancestor((OMPParallelDirective, ACCParallelDirective)):
173 raise TransformationError(
174 f"Error in {self.name}: Application to Nodes enclosed "
175 f"within a thread-parallel region is not allowed.")
176
177 # Performs validation checks specific to PSyData-based
178 # transformations.
179 super().validate(node_list, options)
