125class UnsupportedType(DataType, metaclass=abc.ABCMeta):
126 '''
127 Indicates that a variable declaration is not supported by the PSyIR.
128 This class is abstract and must be subclassed for each language
129 supported by the PSyIR frontends.
130
131 :param str declaration_txt: the original textual declaration of
132 the symbol.
133
134 :raises TypeError: if the supplied declaration_txt is not a str.
135
136 '''

138 if not isinstance(declaration_txt, str):
139 raise TypeError(
140 f"UnsupportedType constructor expects the original variable "
141 f"declaration as a string but got an argument of type "

148

162class UnsupportedFortranType(UnsupportedType):
163 '''Indicates that a Fortran declaration is not supported by the PSyIR.
164
165 :param str declaration_txt: string containing the original variable
166 declaration.
167 :param partial_datatype: a PSyIR datatype representing the subset
168 of type attributes that are supported.
169 :type partial_datatype: Optional[
170 :py:class:`psyclone.psyir.symbols.DataType` or
171 :py:class:`psyclone.psyir.symbols.DataTypeSymbol`]
172
173 '''
174 def __init__(self, declaration_txt, partial_datatype=None):
175 super().__init__(declaration_txt)
176 # This will hold the Fortran type specification (as opposed to
177 # the whole declaration).
178 self._type_text = ""
179 if not isinstance(
180 partial_datatype, (type(None), DataType, DataTypeSymbol)):
181 raise TypeError(
182 f"partial_datatype argument in UnsupportedFortranType "
183 f"initialisation should be a DataType, DataTypeSymbol, or "
184 f"NoneType, but found '{type(partial_datatype).__name__}'.")
185 # This holds a subset of the type in a datatype if it is
186 # possible to determine enough information to create one.
187 self._partial_datatype = partial_datatype
188
189 def __str__(self):
190 return f"UnsupportedFortranType('{self._declaration}')"
191
192 @property
193 def partial_datatype(self):
194 '''
195 :returns: partial datatype information if it can be determined,
196 else None.
197 :rtype: Optional[:py:class:`psyclone.psyir.symbols.DataType` |
198 :py:class:`psyclone.symbols.symbols.DataTypeSymbol`]
199 '''
200 return self._partial_datatype
201
202 @property
203 def type_text(self):
204 '''
205 Parses the original Fortran declaration and uses the resulting
206 parse tree to extract the type information. This is returned in
207 text form and also cached.
208
209 TODO #2137 - alter Unsupported(Fortran)Type so that it is only the
210 type information that is stored as a string. i.e. remove the name
211 of the variable being declared. Once that is done this method
212 won't be required.
213
214 Note that UnsupportedFortranType is also used to hold things like
215 'SAVE :: /my_common/' and thus it is not always possible/appropriate
216 to extract a type expression.
217
218 :returns: the Fortran code specifying the type.
219 :rtype: str
220
221 :raises NotImplementedError: if declaration text cannot be
222 extracted from the original Fortran declaration.
223
224 '''
225 if self._type_text:
226 return self._type_text
227
228 # Encapsulate fparser2 functionality here.
229 # pylint:disable=import-outside-toplevel
230 from fparser.common.readfortran import FortranStringReader
231 from fparser.common.sourceinfo import FortranFormat
232 from fparser.two import Fortran2003
233 from fparser.two.parser import ParserFactory
234 string_reader = FortranStringReader(self._declaration)
235 # Set reader to free format.
236 string_reader.set_format(FortranFormat(True, False))
237 ParserFactory().create(std=Config.get().fortran_standard)
238 try:
239 ptree = Fortran2003.Specification_Part(
240 string_reader)
241 except Exception as err:
242 raise NotImplementedError(
243 f"Cannot extract the declaration part from UnsupportedFortran"
244 f"Type '{self._declaration}' because parsing (attempting to "
245 f"match a Fortran2003.Specification_Part) failed.") from err
246 node = ptree.children[0]
247 if isinstance(node, (Fortran2003.Declaration_Construct,
248 Fortran2003.Type_Declaration_Stmt)):
249 self._type_text = str(node.children[0])
250 elif isinstance(node, Fortran2003.Save_Stmt):
251 self._type_text = "SAVE"
252 elif isinstance(node, Fortran2003.Common_Stmt):
253 self._type_text = "COMMON"
254 else:
255 raise NotImplementedError(
256 f"Cannot extract the declaration part from UnsupportedFortran"
257 f"Type '{self._declaration}'. Only Declaration_Construct, "
258 f"Type_Declaration_Stmt, Save_Stmt and Common_Stmt are "
259 f"supported but got '{type(node).__name__}' from the parser.")
260 return self._type_text
261
262 def __eq__(self, other):
263 '''
264 :param Any other: the object to check equality to.
265
266 :returns: whether this type is equal to the 'other' type.
267 :rtype: bool

269 if not super().__eq__(other):
270 return False

272 return other.type_text == self.type_text

471class ArrayType(DataType):
472 '''Describes an array datatype. Can be an array of intrinsic types (e.g.
473 integer) or of structure types. For the latter, the type must currently be
474 specified as a DataTypeSymbol (see #1031).
475
476 :param datatype: the datatype of the array elements.
477 :type datatype: :py:class:`psyclone.psyir.datatypes.DataType` |
478 :py:class:`psyclone.psyir.symbols.DataTypeSymbol`
479 :param list shape: shape of the symbol in column-major order (leftmost
480 index is contiguous in memory). Each entry represents an array
481 dimension. If it is ArrayType.Extent.ATTRIBUTE the extent of that
482 dimension is unknown but the lower bound is 1 and the extent can be
483 obtained by querying the appropriate intrinsic (e.g. using the SIZE
484 intrinsic in Fortran). If it is ArrayType.Extent.DEFERRED then the
485 extent and bounds are all unknown and may or may not be possible to
486 query them using intrinsics (e.g. the array is ALLOCATABLE in Fortran).
487 Otherwise it can be an int or a DataNode (that returns an
488 int) or a 2-tuple of such quantities. If only a single value is
489 provided then that is taken to be the upper bound and the lower bound
490 defaults to 1. If a 2-tuple is provided then the two members specify
491 the lower and upper bounds, respectively, of the current dimension.
492 Note that providing an int is supported as a convenience, the provided
493 value will be stored internally as a Literal node.
494
495 :raises TypeError: if the arguments are of the wrong type.
496 :raises NotImplementedError: if a structure type does not have a
497 DataTypeSymbol as its type.
498 '''
499 class Extent(Enum):
500 '''
501 Enumeration of array shape extents that are unspecified at compile
502 time. An 'ATTRIBUTE' extent means that the lower bound is 1 with an
503 unknown extent (which can be retrieved with appropriate run-time
504 intrinsics). A 'DEFERRED' extent means that we don't know anything
505 about the bounds, and run-time intrinsics may or may not be able
506 to retrieve them (e.g. the array may need to be allocated/malloc'd).
507
508 '''
509 DEFERRED = 1
510 ATTRIBUTE = 2
511
512 def copy(self):
513 '''
514 :returns: a copy of self.
515 :rtype: :py:class:`psyclone.psyir.symbols.ArrayType.Extent`
516 '''
517 return copy.copy(self)

539 def _node_from_int(var):
540 ''' Helper routine that simply creates a Literal out of an int.
541 If the supplied arg is not an int then it is returned unchanged.
542
543 :param var: variable for which to create a Literal if necessary.
544 :type var: int | :py:class:`psyclone.psyir.nodes.DataNode` | Extent
545
546 :returns: the variable with ints converted to DataNodes.
547 :rtype: :py:class:`psyclone.psyir.nodes.DataNode` | Extent

554 def _dangling_parent(var):
555 ''' Helper routine that copies and adds a dangling parent
556 Assignment to a given node, this implicitly guarantees that the
557 node is not attached anywhere else (and is unexpectedly modified)
558 and also makes it behave like other nodes (e.g. calls inside an
559 expression do not have the "call" keyword in Fortran)
560
561 :param var: variable with a dangling parent if necessary.
562 :type var: int | :py:class:`psyclone.psyir.nodes.DataNode` | Extent
563
564 :returns: the variable with dangling parent when necessary.
565 :rtype: :py:class:`psyclone.psyir.nodes.DataNode` | Extent
566 '''
567 if isinstance(var, DataNode):
568 parent = Assignment()
569 parent.addchild(var.copy())
570 return parent.children[0]
571 return var

573 if isinstance(datatype, DataType):
574 if isinstance(datatype, StructureType):
575 # TODO #1031 remove this restriction.
576 raise NotImplementedError(
577 "When creating an array of structures, the type of "
578 "those structures must be supplied as a DataTypeSymbol "
579 "but got a StructureType instead.")
580 if not isinstance(datatype, (UnsupportedType, UnresolvedType)):
581 self._intrinsic = datatype.intrinsic
582 self._precision = datatype.precision
583 else:
584 self._intrinsic = datatype
585 self._precision = None
586 elif isinstance(datatype, DataTypeSymbol):
587 self._intrinsic = datatype
588 self._precision = None
589 else:
590 raise TypeError(
591 f"ArrayType expected 'datatype' argument to be of type "
592 f"DataType or DataTypeSymbol but found "
593 f"'{type(datatype).__name__}'.")
594 # We do not have a setter for shape as it is an immutable property,
595 # therefore we have a separate validation routine.
596 self._validate_shape(shape)
597 # Replace any ints in shape with a Literal. int's are only supported
598 # as they allow a more concise dimension declaration.
599 self._shape = []
600 one = Literal("1", INTEGER_TYPE)
601 for dim in shape:
602 if isinstance(dim, (DataNode, int)):
603 # The lower bound is 1 by default.
604 self._shape.append(
605 ArrayType.ArrayBounds(
606 _dangling_parent(one.copy()),
607 _dangling_parent(_node_from_int(dim))))
608 elif isinstance(dim, tuple):
609 self._shape.append(
610 ArrayType.ArrayBounds(
611 _dangling_parent(_node_from_int(dim[0])),
612 _dangling_parent(_node_from_int(dim[1]))))
613 else:
614 self._shape.append(dim)
615
616 self._datatype = datatype

790 # dimension must have an 'ATTRIBUTE' extent

803 def __str__(self):
804 '''
805 :returns: a description of this array datatype. If the lower bound \
806 of any dimension has the default value of 1 then it is omitted \
807 for the sake of brevity.
808 :rtype: str

810 :raises InternalError: if the shape of this datatype contains \
811 any elements that aren't ArrayBounds or ArrayType.Extent objects.

818 # Have to import locally to avoid circular dependence
819 # pylint: disable=import-outside-toplevel
820 from psyclone.psyir.nodes import Literal

822 if isinstance(dimension.lower, Literal):

840 f"'{type(dimension).__name__}'")
