84class OMPDirective(metaclass=abc.ABCMeta):
85 '''
86 Base mixin class for all OpenMP-related directives.
87
88 This class is useful to provide a unique common ancestor to all the
89 OpenMP directives, for instance when traversing the tree with
90 `node.walk(OMPDirective)`
91
92 Note that classes inheriting from it must place the OMPDirective in
93 front of the other Directive node sub-class, so that the Python
94 MRO gives preference to this class's attributes.
95 '''
96 _PREFIX = "OMP"

251 def _valid_dependence_literals(self, lit1, lit2):
252 '''
253 Compares two Nodes to check whether they are a valid dependency
254 pair. For two Nodes where at least one is a Literal, a valid
255 dependency is any pair of Literals.
256
257 :param lit1: the first node to compare.
258 :type lit1: :py:class:`psyclone.psyir.nodes.Node`
259 :param lit2: the second node to compare.
260 :type lit2: :py:class:`psyclone.psyir.nodes.Node`
261
262 :returns: whether or not these two nodes can be used as a valid
263 dependency pair in OpenMP.
264 :rtype: bool
265
266 '''
267 # Check both are Literals
268
269 # If a Literal index into an array that is a dependency
270 # has calculated dependency to a
271 # non-Literal array index, this will return False, as this is not
272 # currently supported in PSyclone.
273
274 # If literals are not the same its fine, since a(1) is not a
275 # dependency to a(2), so as long as both are Literals this is ok.
276 return isinstance(lit1, Literal) and isinstance(lit2, Literal)

278 def _valid_dependence_ranges(self, arraymixin1, arraymixin2, index):
279 '''
280 Compares two ArrayMixin Nodes to check whether they are a valid
281 dependency pair on the provided index. For two Nodes where at least
282 one has a Range at this index, they must both have Ranges, and both be
283 full ranges, i.e. ":".
284
285 :param arraymixin1: the first node to validate.
286 :type arraymixin1: :py:class:`psyclone.psyir.nodes.ArrayMixin`
287 :param arraymixin2: the second node to validate.
288 :type arraymixin2: :py:class:`psyclone.psyir.nodes.ArrayMixin`
289
290 :returns: whether or not these two nodes can be used as a valid
291 dependency pair in OpenMP, based upon the provided index.
292 :rtype: bool
293 '''
294 # We know both inputs are always ArrayMixin as this is a private
295 # function.
296 # Check both are Ranges
297 if (not isinstance(arraymixin1.indices[index], Range) or not
298 isinstance(arraymixin2.indices[index], Range)):
299 # Range index to a dependency has calculated dependency to a
300 # non-Range index, which is not currently supported in PSyclone
301 return False
302
303 # To be valid, both ranges need to be full ranges.
304
305 # If we have a range index between dependencies which does not cover
306 # the full array range, it is not currently supported in
307 # PSyclone (due to OpenMP limitations), so False will be returned.
308 return (arraymixin1.is_full_range(index) and
309 arraymixin2.is_full_range(index))

311 def _compute_accesses_get_start_stop_step(self, preceding_nodes, task,
312 symbol):
313 '''
314 Computes the start, stop and step values used in the _compute_accesses
315 function by searching through the preceding nodes for the last access
316 to the symbol.
317
318 :param preceding_nodes: a list of nodes that precede the task in the
319 tree.
320 :type preceding_nodes: List[:py:class:`psyclone.psyir.nodes.Node`]
321 :param task: the OMPTaskDirective node being used in
322 _compute_accesses.
323 :type task: :py:class:`psyclone.psyir.nodes.OMPTaskDirective`
324 :param symbol: the symbol used by the ref in _compute_accesses.
325 :type symbol: :py:class:`psyclone.psyir.symbols.Symbol`
326
327 :returns: a tuple containing the start, stop and step nodes (or None
328 if there is no value).
329 :rtype: Tuple[:py:class`psyclone.psyir.nodes.Node, None]
330 '''
331 start = None
332 stop = None
333 step = None
334 for node in preceding_nodes:
335 # Only Assignment, Loop or Call nodes can modify the symbol in our
336 # Reference
337 if not isinstance(node, (Assignment, Loop, Call)):
338 continue
339 # At the moment we allow all IntrinsicCall nodes through, and
340 # assume that all IntrinsicCall nodes we find don't modify
341 # symbols, but only read from them.
342 if isinstance(node, Call) and not isinstance(node, IntrinsicCall):
343 # Currently opting to fail on any non-intrinsic Call.
344 # Potentially it might be possible to check if the Symbol is
345 # written to and only if so then raise an error
346 raise UnresolvedDependencyError(
347 "Found a Call in preceding_nodes, which "
348 "is not yet supported.")
349 if isinstance(node, Assignment) and node.lhs.symbol == symbol:
350 start = node.rhs.copy()
351 break
352 if isinstance(node, Loop) and node.variable == symbol:
353 # If the loop is not an ancestor of the task then
354 # we don't currently support it.
355 ancestor_loop = task.ancestor(Loop, limit=self)
356 is_ancestor = False
357 while ancestor_loop is not None:
358 if ancestor_loop == node:
359 is_ancestor = True
360 break
361 ancestor_loop = ancestor_loop.ancestor(Loop, limit=self)
362 if not is_ancestor:
363 raise UnresolvedDependencyError(
364 f"Found a dependency index that "
365 f"was updated as a Loop variable "
366 f"that is not an ancestor Loop of "
367 f"the task. The variable is "
368 f"'{node.variable.name}'.")
369 # It has to be an ancestor loop, so we want to find the start,
370 # stop and step Nodes
371 start, stop, step = node.start_expr, node.stop_expr, \
372 node.step_expr
373 break
374 return (start, stop, step)

376 def _compute_accesses(self, ref, preceding_nodes, task):
377 '''
378 Computes the set of accesses for a Reference or BinaryOperation
379 Node, based upon the preceding_nodes and containing task.
380
381 The returned result is either a set of Literals, or Literals
382 and BinaryOperations.
383
384 If ref is a BinaryOperation, it needs to be of the following formats:
385 1. Reference ADD/SUB Literal
386 2. Literal ADD Reference
387 3. Binop(Literal MUL Literal) ADD Reference
388 4. Reference ADD/SUB Binop(Literal MUL Literal)
389
390
391 :param ref: the Reference or BinaryOperation node to compute
392 accesses for.
393 :type ref: Union[:py:class:`psyclone.psyir.nodes.Reference,
394 :py:class:`psyclone.psyir.nodes.BinaryOperation]
395 :param preceding_nodes: a list of nodes that precede the task in the
396 tree.
397 :type preceding_nodes: List[:py:class:`psyclone.psyir.nodes.Node`]
398 :param task: the OMPTaskDirective node containing ref as a child.
399 :type task: :py:class:`psyclone.psyir.nodes.OMPTaskDirective`
400
401 :raises UnresolvedDependencyError: If the ref contains an unsupported
402 BinaryOperation structure, such as
403 a non-ADD/SUB/MUL operator. Check
404 error message for more details.
405 :raises UnresolvedDependencyError: If preceding_nodes contains a Call
406 node.
407 :raises UnresolvedDependencyError: If ref is a BinaryOperation and
408 neither child of ref is a Literal
409 or BinaryOperation.
410 :raises UnresolvedDependencyError: If there is a dependency between
411 ref (a BinaryOperation) and a
412 previously set constant.
413 :raises UnresolvedDependencyError: If there is a dependency between
414 ref and a Loop variable that is
415 not an ancestor of task.
416 :raises UnresolvedDependencyError: If preceding_nodes contains a
417 dependent loop with a non-Literal
418 step.
419
420 :returns: a list of the dependency values for the input ref, or a
421 dict of the start, stop and step values.
422 :rtype: List[Union[:py:class:`psyclone.psyir.nodes.Literal`,
423 :py:class:`psyclone.psyir.nodes.BinaryOperation`]] or
424 Dict[str: py:class:`psyclone.psyir.nodes.Node`]
425 '''
426 if isinstance(ref, Reference):
427 symbol = ref.symbol
428 else:
429 # Get the symbol out of the Binop, and store some other
430 # important information. We store the step value of the
431 # ancestor loop (which will be the value of the Literal, or
432 # one of the Literals if an operand is a BinaryOperation).
433 # In the case that one of the operands is a BinaryOperation,
434 # we also store a "num_entries" value, which is based upon the
435 # multiplier of the step value. This is how we can handle
436 # cases such as array(i+57) if the step value is 32, the
437 # dependencies stored would be array(i+32) and array(i+64).
438 if isinstance(ref.children[0], Literal):
439 if ref.operator == BinaryOperation.Operator.ADD:
440 # Have Literal + Reference. Store the symbol of the
441 # Reference and the integer value of the Literal.
442 symbol = ref.children[1].symbol
443 binop_val = int(ref.children[0].value)
444 num_entries = 2
445 else:
446 raise UnresolvedDependencyError(
447 f"Found a dependency index that is "
448 f"a BinaryOperation where the "
449 f"format is Literal OP Reference "
450 f"with a non-ADD operand "
451 f"which is not supported. "
452 f"The operation found was "
453 f"'{ref.debug_string()}'.")
454 elif isinstance(ref.children[1], Literal):
455 # Have Reference OP Literal. Store the symbol of the
456 # Reference, and the integer value of the Literal. If the
457 # operator is negative, then we store the value negated.
458 if ref.operator in [BinaryOperation.Operator.ADD,
459 BinaryOperation.Operator.SUB]:
460 symbol = ref.children[0].symbol
461 binop_val = int(ref.children[1].value)
462 num_entries = 2
463 if ref.operator == BinaryOperation.Operator.SUB:
464 binop_val = -binop_val
465 else:
466 raise UnresolvedDependencyError(
467 f"Found a dependency index that is "
468 f"a BinaryOperation where the "
469 f"Operator is neither ADD not SUB "
470 f"which is not supported. "
471 f"The operation found was "
472 f"'{ref.debug_string()}'.")
473
474 elif isinstance(ref.children[0], BinaryOperation):
475 if ref.operator == BinaryOperation.Operator.ADD:
476 # Have Binop ADD Reference. Store the symbol of the
477 # Reference, and store the binop. The binop is of
478 # structure Literal MUL Literal, where the second
479 # Literal is to the step of a parent loop.
480 symbol = ref.children[1].symbol
481 binop = ref.children[0]
482 if binop.operator != BinaryOperation.Operator.MUL:
483 raise UnresolvedDependencyError(
484 f"Found a dependency index that is a "
485 f"BinaryOperation with a child "
486 f"BinaryOperation with a non-MUL operator "
487 f"which is not supported. "
488 f"The operation found was "
489 f"'{ref.debug_string()}'.")
490 # These binary operations are format of Literal MUL Literal
491 # where step_val is the 2nd literal and the multiplier
492 # is the first literal
493 if (not (isinstance(binop.children[0], Literal) and
494 isinstance(binop.children[1], Literal))):
495 raise UnresolvedDependencyError(
496 f"Found a dependency index that is a "
497 f"BinaryOperation with a child "
498 f"BinaryOperation with a non-Literal child "
499 f"which is not supported. "
500 f"The operation found was "
501 f"'{ref.debug_string()}'.")
502 # We store the step of the parent loop in binop_val, and
503 # use the other operand to compute how many entries we
504 # need to compute to validate this dependency list.
505 binop_val = int(binop.children[1].value)
506 num_entries = int(binop.children[0].value)+1
507 else:
508 raise UnresolvedDependencyError(
509 f"Found a dependency index that is "
510 f"a BinaryOperation where the "
511 f"format is BinaryOperator OP "
512 f"Reference with a non-ADD operand "
513 f"which is not supported. "
514 f"The operation found was "
515 f"'{ref.debug_string()}'.")
516 elif isinstance(ref.children[1], BinaryOperation):
517 # Have Reference ADD/SUB Binop. Store the symbol of the
518 # Reference, and store the binop. The binop is of
519 # structure Literal MUL Literal, where the second
520 # Literal is to the step of a parent loop.
521 if ref.operator in [BinaryOperation.Operator.ADD,
522 BinaryOperation.Operator.SUB]:
523 symbol = ref.children[0].symbol
524 binop = ref.children[1]
525 if binop.operator != BinaryOperation.Operator.MUL:
526 raise UnresolvedDependencyError(
527 f"Found a dependency index that is a "
528 f"BinaryOperation with a child "
529 f"BinaryOperation with a non-MUL operator "
530 f"which is not supported. "
531 f"The operation found was "
532 f"'{ref.debug_string()}'.")
533 # These binary operations are format of Literal MUL Literal
534 # where step_val is the 2nd literal.
535 if (not (isinstance(binop.children[0], Literal) and
536 isinstance(binop.children[1], Literal))):
537 raise UnresolvedDependencyError(
538 f"Found a dependency index that is a "
539 f"BinaryOperation with an operand "
540 f"BinaryOperation with a non-Literal operand "
541 f"which is not supported. "
542 f"The operation found was "
543 f"'{ref.debug_string()}'.")
544 # We store the step of the parent loop in binop_val, and
545 # use the other operand to compute how many entries we
546 # need to compute to validate this dependency list.
547 binop_val = int(binop.children[1].value)
548 num_entries = int(binop.children[0].value)+1
549 if ref.operator == BinaryOperation.Operator.SUB:
550 # If the operator is SUB then we use 1 less
551 # entry in the list, as Fortran arrays start
552 # from 1.
553 binop_val = -binop_val
554 num_entries = num_entries-1
555 else:
556 raise UnresolvedDependencyError(
557 f"Found a dependency index that is "
558 f"a BinaryOperation where the "
559 f"format is Reference OP "
560 f"BinaryOperation with a non-ADD, "
561 f"non-SUB operand "
562 f"which is not supported. "
563 f"The operation found was "
564 f"'{ref.debug_string()}'.")
565 else:
566 raise UnresolvedDependencyError(
567 f"Found a dependency index that is a "
568 f"BinaryOperation where neither child "
569 f"is a Literal or BinaryOperation. "
570 f"PSyclone can't validate "
571 f"this dependency. "
572 f"The operation found was "
573 f"'{ref.debug_string()}'.")
574 start, stop, step = self._compute_accesses_get_start_stop_step(
575 preceding_nodes, task, symbol)
576
577 if isinstance(ref, BinaryOperation):
578 output_list = []
579 if step is None:
580 # Found no ancestor loop, PSyclone cannot handle
581 # this case, as BinaryOperations created by OMPTaskDirective
582 # in dependencies will always be based on ancestor loops.
583 raise UnresolvedDependencyError(
584 f"Found a dependency between a "
585 f"BinaryOperation and a previously "
586 f"set constant value. "
587 f"PSyclone cannot yet handle this "
588 f"interaction. The error occurs from "
589 f"'{ref.debug_string()}'.")
590 # If the step isn't a Literal value, then we can't compute what
591 # the address accesses at compile time, so we can't validate the
592 # dependency.
593 if not isinstance(step, Literal):
594 raise UnresolvedDependencyError(
595 f"Found a dependency index that is a "
596 f"Loop variable with a non-Literal step "
597 f"which we can't resolve in PSyclone. "
598 f"Containing node is '{ref.debug_string()}'.")
599 # If the start and stop are both Literals, we can compute a set
600 # of accesses this BinaryOperation is related to precisely.
601 if (isinstance(start, Literal) and isinstance(stop, Literal)):
602 # Fill the output list with all values from start to stop
603 # incremented by step
604 startval = int(start.value)
605 stopval = int(stop.value)
606 stepval = int(step.value)
607 # We loop from startval to stopval + 1 as PSyIR loops will
608 # include stopval, wheras Python loops do not.
609 for i in range(startval, stopval + 1, stepval):
610 new_x = i + binop_val
611 output_list.append(Literal(f"{new_x}", INTEGER_TYPE))
612 return output_list
613
614 # If they are not all literals, we have a special case. In this
615 # case we return a dict containing start, stop and step and this
616 # is compared directly to the start, stop and step of a
617 # corresponding access.
618 output_list = {}
619 output_list["start"] = BinaryOperation.create(
620 BinaryOperation.Operator.ADD,
621 start.copy(),
622 Literal(f"{binop_val}", INTEGER_TYPE)
623 )
624 output_list["stop"] = stop.copy()
625 output_list["step"] = step.copy()
626 return output_list
627 if step is None:
628 # Result for an assignment.
629 output_list = [start]
630 return output_list
631 output_list = []
632 # If step is not a Literal then we probably can't resolve this
633 if not isinstance(step, Literal):
634 raise UnresolvedDependencyError(
635 "Found a dependency index that is a "
636 "Loop variable with a non-Literal step "
637 "which we can't resolve in PSyclone.")
638 # Special case when all are Literals
639 if (isinstance(start, Literal) and isinstance(stop, Literal)):
640 # Fill the output list with all values from start to stop
641 # incremented by step
642 startval = int(start.value)
643 stopval = int(stop.value)
644 stepval = int(step.value)
645 # We loop from startval to stopval + 1 as PSyIR loops will include
646 # stopval, wheras Python loops do not.
647 for i in range(startval, stopval + 1, stepval):
648 output_list.append(Literal(f"{i}", INTEGER_TYPE))
649 return output_list

651 # the sequence only. In this case, we have a non-parent loop reference
652 # which is also firstprivate (as shared indices are forbidden in
653 # OMPTaskDirective already), so is essentially a constant. In this
654 # case therefore we will have an unknown start and stop value, so we
655 # verify this dependency differently. To ensure this special case is
656 # understood as a special case, we return a dict with the 3 members.
657 output_list = {}
658 output_list["start"] = start.copy()
659 output_list["stop"] = stop.copy()
660 output_list["step"] = step.copy()
661 return output_list

663 def _check_valid_overlap(self, sympy_ref1s, sympy_ref2s):
664 '''
665 Takes two lists of SymPy expressions, and checks that any overlaps
666 between the expressions is valid for OpenMP depend clauses.
667
668 :param sympy_ref1s: the list of SymPy expressions corresponding to
669 the first dependency clause.
670 :type sympy_ref1s: List[:py:class:`sympy.core.basic.Basic`]
671 :param sympy_ref2s: the list of SymPy expressions corresponding to
672 the second dependency clause.
673 :type sympy_ref2s: List[:py:class:`sympy.core.basic.Basic`]
674
675 :returns: whether this is a valid overlap according to the OpenMP
676 standard.
677 :rtype: bool
678 '''
679 # r1_min will contain the minimum computed value for (ref + value)
680 # from the list. r1_max will contain the maximum computed value.
681 # Loop through the values in sympy_ref1s, and compute the maximum
682 # and minumum values in that list. These correspond to the maximum and
683 # minimum values used for accessing the array relative to the
684 # symbol used as a base access.
685 values = [int(member) for member in sympy_ref1s]
686 r1_min = min(values)
687 r1_max = max(values)
688 # Loop over the elements in sympy_ref2s and check that the dependency
689 # is valid in OpenMP.
690 for member in sympy_ref2s:
691 # If the value is between min and max of r1 then we check that
692 # the value is in the values list
693 val = int(member)
694 if r1_min <= val <= r1_max:
695 if val not in values:
696 # Found incompatible dependency between two
697 # array accesses, ref1 is in range r1_min
698 # to r1_max, but doesn't contain val.
699 # This can happen if we have two loops with
700 # different start values or steps.
701 return False
702 return True

704 def _valid_dependence_ref_binop(self, ref1, ref2, task1, task2):
705 '''
706 Compares two Reference/BinaryOperation Nodes to check they are a set
707 of dependencies that are valid according to OpenMP. Both these nodes
708 are array indices on the same array symbol, so for OpenMP to correctly
709 compute this dependency, we must guarantee at compile time that we
710 know the addresses/array sections covered by this index are identical.
711
712 :param ref1: the first Node to compare.
713 :type ref1: Union[:py:class:`psyclone.psyir.nodes.Reference`,
714 :py:class:`psyclone.psyir.nodes.BinaryOperation`]
715 :param ref2: the second Node to compare.
716 :type ref2: Union[:py:class:`psyclone.psyir.nodes.Reference`,
717 :py:class:`psyclone.psyir.nodes.BinaryOperation`]
718 :param task1: the task containing ref1 as a child.
719 :type task1: :py:class:`psyclone.psyir.nodes.OMPTaskDirective`
720 :param task2: the task containing ref2 as a child.
721 :type task2: :py:class:`psyclone.psyir.nodes.OMPTaskDirective`
722
723 :raises GenerationError: If ref1 and ref2 are dependencies on the
724 same array, and one does not contain a
725 Reference but the other does.
726 :raises GenerationError: If ref1 and ref2 are dependencies on the
727 same array but are References to different
728 variables.
729 :raises GenerationError: If ref1 and ref2 are dependencies on the
730 same array, but the computed index values
731 are not dependent according to OpenMP.
732
733 :returns: whether or not these two nodes can be used as a valid
734 dependency on the same array in OpenMP.
735 :rtype: bool
736
737 '''
738 # pylint: disable=import-outside-toplevel
739 from psyclone.psyir.backend.sympy_writer import SymPyWriter
740 # In this case we have two Reference/BinaryOperation as indices.
741 # We need to attempt to find their value set and check the value
742 # set matches.
743 # Find all the nodes before these tasks
744 preceding_t1 = task1.preceding(reverse=True)
745 preceding_t2 = task2.preceding(reverse=True)
746 # Get access list for each ref
747 try:
748 ref1_accesses = self._compute_accesses(ref1, preceding_t1, task1)
749 ref2_accesses = self._compute_accesses(ref2, preceding_t2, task2)
750 except UnresolvedDependencyError:
751 # If we get a UnresolvedDependencyError from compute_accesses, then
752 # we found an access that isn't able to be handled by PSyclone, so
753 # dependencies based on it need to be handled by a taskwait
754 return False
755
756 # Create our sympy_writer
757 sympy_writer = SymPyWriter()
758
759 # If either of the returned accesses are a dict, this is a special
760 # case where both must be a dict and have the same start, stop and
761 # step.
762 if isinstance(ref1_accesses, dict) or isinstance(ref2_accesses, dict):
763 # If they aren't both dicts then we need to return False as
764 # the special case isn't handled correctly.
765 if type(ref1_accesses) is not type(ref2_accesses):
766 return False
767 # If they're both dicts then we need the step to be equal for
768 # this dependency to be satisfiable.
769 if ref1_accesses["step"] != ref2_accesses["step"]:
770 return False
771 # Now we know the step is equal, we need the start values to be
772 # start1 = start2 + x * step, where x is an integer value.
773 # We use SymPy to solve this equation and perform this check.
774 sympy_start1 = sympy_writer(ref1_accesses["start"])
775 sympy_start2 = sympy_writer(ref2_accesses["start"])
776 sympy_step = sympy_writer(ref2_accesses["step"])
777 b_sym = sympy.Symbol('b')
778 result = sympy.solvers.solve(sympy_start1 - sympy_start2 +
779 b_sym * sympy_step, b_sym)
780 if not isinstance(result[0], sympy.core.numbers.Integer):
781 return False
782
783 # If we know the start and step are aligned, all possible
784 # dependencies are aligned so we don't need to check the stop
785 # value.
786 return True
787
788 # If we have a list, then we have a set of Literal values.
789 # We use the SymPyWriter to convert these objects to expressions
790 # we can use to obtain integer values for these Literals
791 sympy_ref1s = sympy_writer(ref1_accesses)
792 sympy_ref2s = sympy_writer(ref2_accesses)
793 return self._check_valid_overlap(sympy_ref1s, sympy_ref2s)

795 def _check_dependency_pairing_valid(self, node1, node2, task1, task2):
796 '''
797 Given a pair of nodes which are children of a OMPDependClause, this
798 function checks whether the described dependence is correctly
799 described by the OpenMP standard.
800 If the dependence is not going to be handled safely, this function
801 returns False, else it returns true.
802
803 :param node1: the first input node to check.
804 :type node1: :py:class:`psyclone.psyir.nodes.Reference`
805 :param node2: the second input node to check.
806 :type node2: :py:class:`psyclone.psyir.nodes.Reference`
807 :param task1: the OMPTaskDirective node containing node1 as a
808 dependency
809 :type task1: :py:class:`psyclone.psyir.nodes.OMPTaskDirective`
810 :param task2: the OMPTaskDirective node containing node2 as a
811 dependency
812 :type task2: :py:class:`psyclone.psyir.nodes.OMPTaskDirective`
813
814 :returns: whether the dependence is going to be handled safely
815 according to the OpenMP standard.
816 :rtype: bool
817 '''
818 # Checking the symbol is the same works. If the symbol is not the same
819 # then there's no dependence, so its valid.
820 if node1.symbol != node2.symbol:
821 return True
822 # The typing check handles any edge case where we have node1 and node2
823 # pointing to the same symbol, but one is a specialised reference type
824 # and the other is a base Reference type - this is unlikely to happen
825 # but we check just in case. In this case we have to assume there is
826 # an unhandled dependency
827 if type(node1) is not type(node2):
828 return False
829 # For structure reference we need to check they access
830 # the same member. If they don't, no dependence so valid.
831 if isinstance(node1, StructureReference):
832 # If either is a StructureReference here they must both be,
833 # as they access the same symbol.
834
835 # We can't just do == on the Member child, as that
836 # will recurse and check the array indices for any
837 # ArrayMixin children
838
839 # Check the signature of both StructureReference
840 # to see if they are accessing the same data
841 ref0_sig = node1.get_signature_and_indices()[0]
842 ref1_sig = node2.get_signature_and_indices()[0]
843 if ref0_sig != ref1_sig:
844 return True
845
846 # If we have (exactly) Reference objects we filter out
847 # non-matching ones with the symbol check, and matching ones
848 # are always valid since they are simple accesses.
849 # pylint: disable=unidiomatic-typecheck
850 if type(node1) is Reference:
851 return True
852
853 # All remaining objects are some sort of Array access
854 array1 = None
855 array2 = None
856
857 # PSyclone will not handle dependencies on multiple array indexes
858 # at the moment, so we return False.
859 if len(node1.walk(ArrayMixin)) > 1 or len(node2.walk(ArrayMixin)) > 1:
860 return False
861 if isinstance(node1, ArrayReference):
862 array1 = node1
863 array2 = node2
864 else:
865 array1 = node1.walk(ArrayMixin)[0]
866 array2 = node2.walk(ArrayMixin)[0]
867 for i, index in enumerate(array1.indices):
868 if (isinstance(index, Literal) or
869 isinstance(array2.indices[i], Literal)):
870 valid = self._valid_dependence_literals(
871 index, array2.indices[i])
872 elif (isinstance(index, Range) or
873 isinstance(array2.indices[i], Range)):
874 valid = self._valid_dependence_ranges(
875 array1, array2, i)
876 else:
877 # The only remaining option is that the indices are
878 # References or BinaryOperations
879 valid = self._valid_dependence_ref_binop(
880 index, array2.indices[i], task1, task2)
881 # If this was not valid then return False, else keep checking
882 # other indices
883 if not valid:
884 return False
885
886 return valid

888 def _validate_task_dependencies(self):
889 '''
890 Validates all task dependencies in this OMPSerialDirective region are
891 valid within the restraints of OpenMP & PSyclone. This is done through
892 a variety of helper functions, and checks each pair of tasks' inout,
893 outin and outout combinations.
894
895 Any task dependencies that are detected and will not be handled by
896 OpenMP's depend clause will be handled through the addition of
897 OMPTaskwaitDirective nodes.
898
899 :raises NotImplementedError: If this region contains both an
900 OMPTaskDirective and an
901 OMPTaskloopDirective.
902 '''
903 # pylint: disable=import-outside-toplevel
904 from psyclone.psyir.nodes.omp_task_directive import OMPTaskDirective
905 tasks = self.walk(OMPTaskDirective)
906 # For now we disallow Tasks and Taskloop directives in the same Serial
907 # Region
908 if len(tasks) > 0 and any(self.walk(OMPTaskloopDirective)):
909 raise NotImplementedError("OMPTaskDirectives and "
910 "OMPTaskloopDirectives are not "
911 "currently supported inside the same "
912 "parent serial region.")
913
914 pairs = itertools.combinations(tasks, 2)
915
916 # List of tuples of dependent nodes that aren't handled by OpenMP
917 unhandled_dependent_nodes = []
918 # Lowest and highest position nodes contain the abs_position of each
919 # tuple inside unhandled_dependent_nodes, used for sorting the arrays
920 # and checking if the unhandled dependency has a taskwait inbetween.
921 lowest_position_nodes = []
922 highest_position_nodes = []
923
924 for pair in pairs:
925 task1 = pair[0]
926 task2 = pair[1]
927
928 # Find all References in each tasks' depend clauses
929 # Should we cache these instead?
930 task1_in = [x for x in task1.input_depend_clause.children
931 if isinstance(x, Reference)]
932 task1_out = [x for x in task1.output_depend_clause.children
933 if isinstance(x, Reference)]
934 task2_in = [x for x in task2.input_depend_clause.children
935 if isinstance(x, Reference)]
936 task2_out = [x for x in task2.output_depend_clause.children
937 if isinstance(x, Reference)]
938
939 inout = list(itertools.product(task1_in, task2_out))
940 outin = list(itertools.product(task1_out, task2_in))
941 outout = list(itertools.product(task1_out, task2_out))
942 # Loop through each potential dependency pair and check they
943 # will be handled correctly.
944
945 # Need to predefine satisfiable in case all lists are empty.
946 satisfiable = True
947 for mem in inout + outin + outout:
948 satisfiable = \
949 self._check_dependency_pairing_valid(mem[0], mem[1],
950 task1, task2)
951 # As soon as any is not satisfiable, then we don't need to
952 # continue checking.
953 if not satisfiable:
954 break
955
956 # If we have an unsatisfiable dependency between two tasks, then we
957 # need to have a taskwait between them always. We need to loop up
958 # to find these tasks' parents which are closest to the Schedule
959 # which contains both tasks, and use them as the nodes which are
960 # dependent.
961 if not satisfiable:
962 # Find the lowest schedule containing both nodes.
963 schedule1 = task1.ancestor(Schedule, shared_with=task2)
964 # Find the closest ancestor to the common schedule.
965 task1_proxy = task1
966 while task1_proxy.parent is not schedule1:
967 task1_proxy = task1_proxy.parent
968 task2_proxy = task2
969 while task2_proxy.parent is not schedule1:
970 task2_proxy = task2_proxy.parent
971
972 # Now we have the closest nodes to the closest common ancestor
973 # schedule, so add them to the unhandled_dependent_nodes list.
974 if task1_proxy is not task2_proxy:
975 # If they end up with the same proxy, they have the same
976 # ancestor tree but are in different schedules. This means
977 # that they are in something like an if/else block with
978 # one node in an if block and the other in the else block.
979 # These dependencies we can ignore as they are not ever
980 # both executed
981 unhandled_dependent_nodes.append(
982 (task1_proxy, task2_proxy))
983 lowest_position_nodes.append(min(task1_proxy.abs_position,
984 task2_proxy.abs_position))
985 highest_position_nodes.append(
986 max(task1_proxy.abs_position,
987 task2_proxy.abs_position))
988
989 # If we have no invalid dependencies we can return early
990 if len(unhandled_dependent_nodes) == 0:
991 return
992
993 # Need to sort lists by highest_position_nodes value, and then
994 # by lowest value if tied.
995 # Based upon
996 # https://stackoverflow.com/questions/9764298/how-to-sort-two-
997 # lists-which-reference-each-other-in-the-exact-same-way
998
999 # sorted_highest_positions and sorted_lowest_positions contain
1000 # the abs_positions for the corresponding Nodes in the tuple at
1001 # the same index in sorted_dependency_pairs. The
1002 # sorted_dependency_pairs list contains each pair of unhandled
1003 # dependency nodes that were previously computed, but sorted
1004 # according to abs_position in the tree.
1005 sorted_highest_positions, sorted_lowest_positions, \
1006 sorted_dependency_pairs = (list(t) for t in
1007 zip(*sorted(zip(
1008 highest_position_nodes,
1009 lowest_position_nodes,
1010 unhandled_dependent_nodes)
1011 )))
1012 # The location of any node where need to place an OMPTaskwaitDirective
1013 # to ensure code correctness. The size of this list should be
1014 # minimised during construction as we will not add another
1015 # OMPTaskwaitDirective when a dependency will be handled already by
1016 # an existing OMPTaskwaitDirective or one that will be created during
1017 # this process.
1018 taskwait_location_nodes = []
1019 # Stores the abs_position for each of the OMPTaskwaitDirective nodes
1020 # that does or will exist.
1021 taskwait_location_abs_pos = []
1022 for taskwait in self.walk(OMPTaskwaitDirective):
1023 taskwait_location_nodes.append(taskwait)
1024 taskwait_location_abs_pos.append(taskwait.abs_position)
1025 # Add the first node to have a taskwait placed in front of it into the
1026 # list, unless one of the existing OMPTaskwaitDirective nodes already
1027 # satisfies the dependency.
1028 lo_abs_pos = sorted_lowest_positions[0]
1029 hi_abs_pos = sorted_highest_positions[0]
1030 for ind, taskwait_loc in enumerate(taskwait_location_nodes):
1031 if (taskwait_location_abs_pos[ind] <= hi_abs_pos and
1032 taskwait_location_abs_pos[ind] >= lo_abs_pos):
1033 # We potentially already satisfy this initial dependency
1034 if (sorted_dependency_pairs[0][1].ancestor(Schedule) is
1035 taskwait_loc.ancestor(Schedule)):
1036 break
1037 else:
1038 taskwait_location_nodes.append(sorted_dependency_pairs[0][1])
1039 taskwait_location_abs_pos.append(sorted_highest_positions[0])
1040
1041 for index, pairs in enumerate(sorted_dependency_pairs[1:]):
1042 # Add 1 to index here because we're looking from [1:]
1043 lo_abs_pos = sorted_lowest_positions[index+1]
1044 hi_abs_pos = sorted_highest_positions[index+1]
1045 for ind, taskwait_loc in enumerate(taskwait_location_nodes):
1046 if (taskwait_location_abs_pos[ind] <= hi_abs_pos and
1047 taskwait_location_abs_pos[ind] >= lo_abs_pos):
1048 # We have a taskwait meant to be placed here that is
1049 # potentially already satisfied. To check we need to
1050 # ensure that the ancestor schedules of the nodes
1051 # are identical
1052 if (pairs[0].ancestor(Schedule) is
1053 taskwait_loc.ancestor(Schedule)):
1054 break
1055 else:
1056 # If we didn't find a taskwait we plan to add that satisfies
1057 # this dependency, add it to the list
1058 taskwait_location_nodes.append(pairs[1])
1059 taskwait_location_abs_pos.append(hi_abs_pos)
1060 # Now loop through the list in reverse and add taskwaits unless the
1061 # node is already a taskwait
1062 taskwait_location_nodes.reverse()
1063 for taskwait_loc in taskwait_location_nodes:
1064 if isinstance(taskwait_loc, OMPTaskwaitDirective):
1065 continue
1066 node_parent = taskwait_loc.parent
1067 loc = taskwait_loc.position
1068 node_parent.addchild(OMPTaskwaitDirective(), loc)

1533 def infer_sharing_attributes(self):
1534 '''
1535 The PSyIR does not specify if each symbol inside an OpenMP region is
1536 private, firstprivate, shared or shared but needs synchronisation,
1537 the attributes are inferred looking at the usage of each symbol inside
1538 the parallel region.
1539
1540 This method analyses the directive body and automatically classifies
1541 each symbol using the following rules:
1542 - All arrays are shared unless listed in the explicitly private list.
1543 - Scalars that are accessed only once are shared.
1544 - Scalars that are read-only or written outside a loop are shared.
1545 - Scalars written in multiple iterations of a loop are private, unless:
1546
1547 * there is a write-after-read dependency in a loop iteration,
1548 in this case they are shared but need synchronisation;
1549 * they are read before in the same parallel region (but not inside
1550 the same loop iteration), in this case they are firstprivate.
1551 * they are only conditionally written in some iterations;
1552 in this case they are firstprivate.
1553
1554 This method returns the sets of private, firstprivate, and shared but
1555 needing synchronisation symbols, all symbols not in these sets are
1556 assumed shared. How to synchronise the symbols in the third set is
1557 up to the caller of this method.
1558
1559 :returns: three set of symbols that classify each of the symbols in
1560 the directive body as PRIVATE, FIRSTPRIVATE or SHARED NEEDING
1561 SYNCHRONISATION.
1562 :rtype: Tuple[Set(:py:class:`psyclone.psyir.symbols.Symbol`),
1563 Set(:py:class:`psyclone.psyir.symbols.Symbol`),
1564 Set(:py:class:`psyclone.psyir.symbols.Symbol`)]
1565
1566 :raises GenerationError: if the DefaultClauseType associated with
1567 this OMPParallelDirective is not shared.
1568
1569 '''
1570 if (self.default_clause.clause_type !=
1571 OMPDefaultClause.DefaultClauseTypes.SHARED):
1572 raise GenerationError("OMPParallelClause cannot correctly generate"
1573 " the private clause when its default "
1574 "data sharing attribute in its default "
1575 "clause is not 'shared'.")
1576
1577 # TODO #598: Improve the handling of scalar variables, there are
1578 # remaining issues when we have accesses after the parallel region
1579 # of variables that we currently declare as private. We could use
1580 # the DefinitionUseChain to prove that there are no more uses after
1581 # the loop.
1582 # e.g:
1583 # !$omp parallel do <- will set private(ji, my_index)
1584 # do ji = 1, jpk
1585 # my_index = ji+1
1586 # array(my_index) = 2
1587 # enddo
1588 # #end do
1589 # call func(my_index) <- my_index has not been updated
1590
1591 private = set()
1592 fprivate = set()
1593 need_sync = set()
1594
1595 # Determine variables that must be private, firstprivate or need_sync
1596 var_accesses = VariablesAccessInfo()
1597 self.reference_accesses(var_accesses)
1598 for signature in var_accesses.all_signatures:
1599 if not var_accesses[signature].has_data_access():
1600 continue
1601 accesses = var_accesses[signature].all_accesses
1602 # TODO #2094: var_name only captures the top-level
1603 # component in the derived type accessor. If the attributes
1604 # only apply to a sub-component, this won't be captured
1605 # appropriately.
1606 name = signature.var_name
1607 symbol = accesses[0].node.scope.symbol_table.lookup(
1608 name, otherwise=None)
1609
1610 # If it is manually marked as a local symbol, add it to private or
1611 # firstprivate set
1612 if (isinstance(symbol, DataSymbol) and
1613 isinstance(self.dir_body[0], Loop) and
1614 symbol in self.dir_body[0].explicitly_private_symbols):
1615 if any(ref.symbol is symbol for ref in self.preceding()
1616 if isinstance(ref, Reference)):
1617 # If it's used before the loop, make it firstprivate
1618 fprivate.add(symbol)
1619 else:
1620 private.add(symbol)
1621 continue
1622
1623 # All arrays not explicitly marked as threadprivate are shared
1624 if any(accs.is_array() for accs in accesses):
1625 continue
1626
1627 # If a variable is only accessed once, it is either an error
1628 # or a shared variable - anyway it is not private
1629 if len(accesses) == 1:
1630 continue
1631
1632 # TODO #598: If we only have writes, it must be need_sync:
1633 # do ji = 1, jpk
1634 # if ji=3:
1635 # found = .true.
1636 # Or lastprivate in order to maintain the serial semantics
1637 # do ji = 1, jpk
1638 # found = ji
1639
1640 # We consider private variables as being the ones that are written
1641 # in every iteration of a loop.
1642 # If one such scalar is potentially read before it is written, it
1643 # will be considered firstprivate.
1644 has_been_read = False
1645 last_read_position = 0
1646 for access in accesses:
1647 if access.access_type == AccessType.READ:
1648 has_been_read = True
1649 last_read_position = access.node.abs_position
1650
1651 if access.access_type == AccessType.WRITE:
1652
1653 # Check if the write access is outside a loop. In this case
1654 # it will be marked as shared. This is done because it is
1655 # likely to be re-used later. e.g:
1656 # !$omp parallel
1657 # jpk = 100
1658 # !omp do
1659 # do ji = 1, jpk
1660 loop_ancestor = access.node.ancestor(
1661 (Loop, WhileLoop),
1662 limit=self,
1663 include_self=True)
1664 if not loop_ancestor:
1665 # If we find it at least once outside a loop we keep it
1666 # as shared
1667 break
1668
1669 # Otherwise, the assignment to this variable is inside a
1670 # loop (and it will be repeated for each iteration), so
1671 # we declare it as private or need_synch
1672 name = signature.var_name
1673 # TODO #2094: var_name only captures the top-level
1674 # component in the derived type accessor. If the attributes
1675 # only apply to a sub-component, this won't be captured
1676 # appropriately.
1677 symbol = access.node.scope.symbol_table.lookup(name)
1678
1679 # If it has been read before we have to check if ...
1680 if has_been_read:
1681 loop_pos = loop_ancestor.loop_body.abs_position
1682 if last_read_position < loop_pos:
1683 # .. it was before the loop, so it is fprivate
1684 fprivate.add(symbol)
1685 else:
1686 # or inside the loop, in which case it needs sync
1687 need_sync.add(symbol)
1688 break
1689
1690 # If the write is not guaranteed, we make it firstprivate
1691 # so that in the case that the write doesn't happen we keep
1692 # the original value
1693 conditional_write = access.node.ancestor(
1694 IfBlock,
1695 limit=loop_ancestor,
1696 include_self=True)
1697 if conditional_write:
1698 fprivate.add(symbol)
1699 break
1700
1701 # Already found the first write and decided if it is
1702 # shared, private or firstprivate. We can stop looking.
1703 private.add(symbol)
1704 break
1705
1706 return private, fprivate, need_sync
