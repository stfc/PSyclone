47#: The prefix we will prepend to routine, container and metadata
48#: names when generating the adjoint. If the original name contains
49#: the tl prefix, then this is removed.
50ADJOINT_NAME_PREFIX = "adj_"
51TL_NAME_PREFIX = "tl_"
52#: The tolerance applied to the comparison of the inner product values in
53#: the generated test-harness code.
54#: TODO #1346 this tolerance should be user configurable.
55INNER_PRODUCT_TOLERANCE = 1500.0

58def create_adjoint_name(tl_name, table=None):
59 '''Create an adjoint name from the supplied tangent linear name. This
60 is done by stripping the TL_NAME_PREFIX from the name if it exists
61 and then adding the ADJOINT_NAME_PREFIX. The adjoint name is also
62 lower-cased. If a symbol table is supplied, then this routine also
63 ensures that the new name is not present in that table.
64
65 :param str: the tangent-linear name.
66 :param table: an optional symbol table in which the new adjoint name must
67 be unique.
68 :type table: Optional[:py:class:`psyclone.psyir.symbols.SymbolTable`]
69
70 :returns: the adjoint name.
71 :rtype: str
72
73 '''
74 adj_name = tl_name.lower()
75 if adj_name.startswith(TL_NAME_PREFIX):
76 adj_name = adj_name[len(TL_NAME_PREFIX):]
77 base_name = ADJOINT_NAME_PREFIX + adj_name
78 if table:
79 return table.next_available_name(base_name)
80 return base_name

83def create_real_comparison(sym_table, kernel, var1, var2):
84 '''Creates PSyIR that checks the values held by Symbols var1 and var2
85 for equality, allowing for machine precision and writes the
86 success or failure of the checks to stdout.
87
88 :param sym_table: the SymbolTable in which to put new Symbols.
89 :type sym_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
90 :param kernel: the routine for which this adjoint test is being performed.
91 :type kernel: :py:class:`psyclone.psyir.nodes.Routine`
92 :param var1: the symbol holding the first value for the comparison.
93 :type var1: :py:class:`psyclone.psyir.symbols.DataSymbol`
94 :param var2: the symbol holding the second value for the comparison.
95 :type var2: :py:class:`psyclone.psyir.symbols.DataSymbol`
96
97 :returns: the PSyIR nodes that perform the check and write its
98 success or failure to stdout.
99 :rtype: List[:py:class:`psyclone.psyir.nodes.Node`]
100
101 '''
102 statements = []
103 statements.extend(common_real_comparison(sym_table, var1, var2))
104 statements.extend(_common_write(sym_table, kernel, var1, var2))
105 return statements

108def common_real_comparison(sym_table, var1, var2):
109 '''
110 Creates PSyIR that checks the values held by Symbols var1 and var2 for
111 equality, allowing for machine precision.
112
113 :param sym_table: the SymbolTable in which to put new Symbols.
114 :type sym_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
115 :param var1: the symbol holding the first value for the comparison.
116 :type var1: :py:class:`psyclone.psyir.symbols.DataSymbol`
117 :param var2: the symbol holding the second value for the comparison.
118 :type var2: :py:class:`psyclone.psyir.symbols.DataSymbol`
119
120 :returns: the PSyIR nodes that perform the check.
121 :rtype: List[:py:class:`psyclone.psyir.nodes.Node`]
122
123 '''
124 freader = FortranReader()
125 statements = []
126 mtol = sym_table.new_symbol("MachineTol", symbol_type=DataSymbol,
127 datatype=var1.datatype)
128 rel_diff = sym_table.new_symbol(
129 "relative_diff", symbol_type=DataSymbol, datatype=var1.datatype,
130 tag="relative_diff")
131 # The "overall_tolerance" tag is used to get the symbol from the
132 # symbol table in subsequent routines even though the symbol
133 # itself is not used in this routine.
134 _ = sym_table.new_symbol("overall_tolerance",
135 tag="overall_tolerance",
136 symbol_type=DataSymbol,
137 datatype=var1.datatype,
138 is_constant=True,
139 initial_value=INNER_PRODUCT_TOLERANCE)
140 assign = freader.psyir_from_statement(
141 f"MachineTol = SPACING ( MAX( ABS({var1.name}), ABS({var2.name}) ) )",
142 sym_table)
143 statements.append(assign)
144 statements[-1].preceding_comment = (
145 "Test the inner-product values for equality, allowing for the "
146 "precision of the active variables")
147 sub_op = BinaryOperation.create(BinaryOperation.Operator.SUB,
148 Reference(var1), Reference(var2))
149 abs_op = IntrinsicCall.create(IntrinsicCall.Intrinsic.ABS, [sub_op])
150 div_op = BinaryOperation.create(BinaryOperation.Operator.DIV,
151 abs_op, Reference(mtol))
152 statements.append(Assignment.create(Reference(rel_diff), div_op))
153
154 return statements

157def _common_write(sym_table, kernel, var1, var2):
158 '''Creates PSyIR that writes whether the precision test passed or
159 failed to stdout.
160
161 :param sym_table: the SymbolTable in which to read existing Symbols.
162 :type sym_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
163 :param kernel: the routine for which this adjoint test is being performed.
164 :type kernel: :py:class:`psyclone.psyir.nodes.Routine`
165 :param var1: the symbol holding the first value for the comparison.
166 :type var1: :py:class:`psyclone.psyir.symbols.DataSymbol`
167 :param var2: the symbol holding the second value for the comparison.
168 :type var2: :py:class:`psyclone.psyir.symbols.DataSymbol`
169
170 :returns: PSyIR nodes that write out test success or failure to stdout.
171 :rtype: List[:py:class:`psyclone.psyir.nodes.Node`]
172
173 '''
174 # TODO #1345 make this code language agnostic.
175 freader = FortranReader()
176 statements = []
177 rel_diff = sym_table.lookup_with_tag("relative_diff")
178 overall_tol = sym_table.lookup_with_tag("overall_tolerance")
179 write1 = freader.psyir_from_statement(
180 f"write(*,*) 'Test of adjoint of ''{kernel.name}'' PASSED: ', "
181 f"{var1.name}, {var2.name}, {rel_diff.name}", sym_table)
182 write2 = freader.psyir_from_statement(
183 f"write(*,*) 'Test of adjoint of ''{kernel.name}'' FAILED: ', "
184 f"{var1.name}, {var2.name}, {rel_diff.name}", sym_table)
185
186 statements.append(
187 IfBlock.create(BinaryOperation.create(BinaryOperation.Operator.LT,
188 Reference(rel_diff),
189 Reference(overall_tol)),
190 [write1], [write2]))
191
192 return statements

195def find_container(psyir):
196 ''' Finds the first Container in the supplied PSyIR that is not a
197 FileContainer. Also validates that the PSyIR contains at most one
198 FileContainer which, if present, contains a Container.
199
200 :param psyir: the PSyIR to search for a Container.
201 :type psyir: :py:class:`psyclone.psyir.nodes.Node`
202
203 :returns: the first Container that is not a FileContainer or None if \
204 there is none.
205 :rtype: :py:class:`psyclone.psyir.nodes.Container` or NoneType
206
207 :raises InternalError: if there are two Containers and the second is a \
208 FileContainer.
209 :raises NotImplementedError: if there are two Containers and the first is \
210 not a FileContainer.
211 :raises NotImplementedError: if there are more than two Containers.
212
213 '''
214 if not isinstance(psyir, Node):
215 raise TypeError(
216 f"Expected a PSyIR Node but got '{type(psyir).__name__}'")
217
218 containers = psyir.walk(Container)
219 if not containers:
220 return None
221
222 if len(containers) == 1:
223 if isinstance(containers[0], FileContainer):
224 return None
225 return containers[0]
226
227 if len(containers) == 2:
228 if isinstance(containers[1], FileContainer):
229 raise InternalError(
230 "The supplied PSyIR contains two Containers but the innermost "
231 "is a FileContainer. This should not be possible.")
232 if not isinstance(containers[0], FileContainer):
233 raise NotImplementedError(
234 "The supplied PSyIR contains two Containers and the outermost "
235 "one is not a FileContainer. This is not supported.")
236 return containers[1]
237
238 raise NotImplementedError("The supplied PSyIR contains more than two "
239 "Containers. This is not supported.")
