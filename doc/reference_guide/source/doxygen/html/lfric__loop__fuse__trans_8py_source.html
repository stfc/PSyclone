48class LFRicLoopFuseTrans(LoopFuseTrans):
49 ''' LFRic API specialisation of the
50 :py:class:`base class <LoopFuseTrans>` in order to fuse two Dynamo
51 loops after performing validity checks. For example:
52
53 >>> from psyclone.parse.algorithm import parse
54 >>> from psyclone.psyGen import PSyFactory
55 >>>
56 >>> API = "lfric"
57 >>> FILENAME = "alg.x90"
58 >>> ast, invokeInfo = parse(FILENAME, api=API)
59 >>> psy = PSyFactory(API, distributed_memory=False).create(invoke_info)
60 >>> schedule = psy.invokes.get('invoke_0').schedule
61 >>>
62 >>> from psyclone.domain.lfric.transformations import LFRicLoopFuseTrans
63 >>> ftrans = LFRicLoopFuseTrans()
64 >>>
65 >>> ftrans.apply(schedule[0], schedule[1])
66 >>> print(schedule.view())
67
68 The optional argument `same_space` can be set as
69
70 >>> ftrans.apply(schedule[0], schedule[1], {"same_space": True})
71
72 when applying the transformation.
73
74 '''
75
76 def __str__(self):
77 return ("Fuse two adjacent loops together with Dynamo-specific "
78 "validity checks")
79
80 def validate(self, node1, node2, options=None):
81 ''' Performs various checks to ensure that it is valid to apply
82 the LFRicLoopFuseTrans transformation to the supplied loops.
83
84 :param node1: the first Loop to fuse.
85 :type node1: :py:class:`psyclone.domain.lfric.LFRicLoop`
86 :param node2: the second Loop to fuse.
87 :type node2: :py:class:`psyclone.domain.lfric.LFRicLoop`
88 :param options: a dictionary with options for transformations.
89 :type options: Optional[Dict[str, Any]]
90 :param bool options["same_space"]: this optional flag, set to `True`, \
91 asserts that an unknown iteration space (i.e. `ANY_SPACE`) \
92 matches the other iteration space. This is set at the user's own \
93 risk. If both iteration spaces are discontinuous the loops can be \
94 fused without having to use the `same_space` flag.
95
96 :raises TransformationError: if either of the supplied loops contains \
97 an inter-grid kernel.
98 :raises TransformationError: if one or both function spaces have \
99 invalid names.
100 :raises TransformationError: if the `same_space` flag was set, but \
101 does not apply because neither field \
102 is on `ANY_SPACE` or the spaces are not \
103 the same.
104 :raises TransformationError: if one or more of the iteration spaces \
105 is unknown (`ANY_SPACE`) and the \
106 `same_space` flag is not set to `True`.
107 :raises TransformationError: if the loops are over different spaces \
108 that are not both discontinuous and \
109 the loops both iterate over cells.
110 :raises TransformationError: if the loops' upper bound names are \
111 not the same.
112 :raises TransformationError: if the halo-depth indices of two loops \
113 are not the same.
114 :raises TransformationError: if each loop already contains a reduction.
115 :raises TransformationError: if the first loop has a reduction and \
116 the second loop reads the result of \
117 the reduction.
118 '''
119 # pylint: disable=too-many-locals,too-many-branches
120 # Call the parent class validation first
121
122 if not options:
123 my_options = {}
124 else:
125 my_options = options.copy()
126
127 # TODO #2498: access information for LFRic kernels do not have any
128 # index information for field accesses, and the loop fusion dependency
129 # tests will therefore fail. To avoid this, disable the dependency test
130 # in the generic loop fusion class for LFRic.
131 # TODO 257: if the loop-fusion transformation is implemented to just
132 # check that a variable with a stencil read-access is written, then
133 # the test could be enabled for LFRic as well, so the force option
134 # can be removed.
135 if "force" not in my_options:
136 my_options["force"] = True
137
138 same_space = my_options.get("same_space", False)
139 if same_space and not isinstance(same_space, bool):
140 raise TransformationError(
141 f"Error in {self.name} transformation: The value of the "
142 f"'same_space' flag must be either bool or None type, but the "
143 f"type of flag provided was '{type(same_space).__name__}'.")
144 super().validate(node1, node2, options=my_options)
145 # Now test for Dynamo-specific constraints
146
147 # 1) Check that we don't have an inter-grid kernel
148 check_intergrid(node1)
149 check_intergrid(node2)
150
151 # 2) Check function space names
152 node1_fs_name = node1.field_space.orig_name
153 node2_fs_name = node2.field_space.orig_name
154 # 2.1) Check that both function spaces are valid
155 const = LFRicConstants()
156 if not (node1_fs_name in const.VALID_FUNCTION_SPACE_NAMES and
157 node2_fs_name in const.VALID_FUNCTION_SPACE_NAMES):
158 raise TransformationError(
159 f"Error in {self.name} transformation: One or both function "
160 f"spaces '{node1_fs_name}' and '{node2_fs_name}' have invalid "
161 f"names.")
162 # Check whether any of the spaces is ANY_SPACE. Loop fusion over
163 # ANY_SPACE is allowed only when the 'same_space' flag is set
164 node_on_any_space = node1_fs_name in \
165 const.VALID_ANY_SPACE_NAMES or \
166 node2_fs_name in const.VALID_ANY_SPACE_NAMES
167 # 2.2) If 'same_space' is true check that both function spaces are
168 # the same or that at least one of the nodes is on ANY_SPACE. The
169 # former case is convenient when loop fusion is applied generically.
170
171 if same_space:
172 if node1_fs_name == node2_fs_name:
173 pass
174 elif not node_on_any_space:
175 raise TransformationError(
176 f"Error in {self.name} transformation: The 'same_space' "
177 f"flag was set, but does not apply because "
178 f"neither field is on 'ANY_SPACE'.")
179 # 2.3) If 'same_space' is not True then make further checks
180 else:
181 # 2.3.1) Check whether one or more of the function spaces
182 # is ANY_SPACE without the 'same_space' flag
183 if node_on_any_space:
184 raise TransformationError(
185 f"Error in {self.name} transformation: One or more of the "
186 f"iteration spaces is unknown ('ANY_SPACE') so loop fusion"
187 f" might be invalid. If you know the spaces are the same "
188 f"then please set the 'same_space' optional argument to "
189 f"'True'.")
190 # 2.3.2) Check whether specific function spaces are the
191 # same. If they are not, the loop fusion is still possible
192 # but only when both function spaces are discontinuous
193 # (w3, w2v, wtheta or any_discontinuous_space) and the upper
194 # loop bounds are the same (checked further below).
195 if node1_fs_name != node2_fs_name:
196 if not (node1_fs_name in
197 const.VALID_DISCONTINUOUS_NAMES and
198 node2_fs_name in
199 const.VALID_DISCONTINUOUS_NAMES):
200 raise TransformationError(
201 f"Error in {self.name} transformation: Cannot fuse "
202 f"loops that are over different spaces "
203 f"'{node1_fs_name}' and '{node2_fs_name}' unless they "
204 f"are both discontinuous.")
205
206 # 3) Check upper loop bounds
207 if node1.upper_bound_name != node2.upper_bound_name:
208 raise TransformationError(
209 f"Error in {self.name} transformation: The upper bound names "
210 f"are not the same. Found '{node1.upper_bound_name}' and "
211 f"'{node2.upper_bound_name}'.")
212
213 # 4) Check halo depths
214 if node1.upper_bound_halo_depth != node2.upper_bound_halo_depth:
215 node1_depth = (node1.upper_bound_halo_depth.debug_string() if
216 node1.upper_bound_halo_depth else "None")
217 node2_depth = (node2.upper_bound_halo_depth.debug_string() if
218 node2.upper_bound_halo_depth else "None")
219 raise TransformationError(
220 f"Error in {self.name} transformation: The halo-depth indices "
221 f"are not the same. Found "
222 f"'{node1_depth}' and '{node2_depth}'.")
223
224 # 5) Check for reductions
225 arg_types = const.VALID_SCALAR_NAMES
226 all_reductions = AccessType.get_valid_reduction_modes()
227 node1_red_args = node1.args_filter(arg_types=arg_types,
228 arg_accesses=all_reductions)
229 node2_red_args = node2.args_filter(arg_types=arg_types,
230 arg_accesses=all_reductions)
231
232 if node1_red_args and node2_red_args:
233 raise TransformationError(
234 f"Error in {self.name} transformation: Cannot fuse loops "
235 f"when each loop already contains a reduction.")
236 if node1_red_args:
237 for reduction_arg in node1_red_args:
238 other_args = node2.args_filter()
239 for arg in other_args:
240 if reduction_arg.name == arg.name:
241 raise TransformationError(
242 f"Error in {self.name} transformation: Cannot fuse"
243 f" loops as the first loop has a reduction and "
244 f"the second loop reads the result of the "
245 f"reduction.")
