Detailed Description

Handles the declarations of scalar kernel arguments appearing in either
an Invoke or a Kernel stub.

:param node: the Invoke or Kernel stub for which to manage the scalar \
             arguments.
:type node: :py:class:`psyclone.domain.lfric.LFRicKern` or \
            :py:class:`psyclone.domain.lfric.LFRicInvoke`

85 '''
86 Create argument lists and declarations for all scalar arguments

88
89 :param parent: the f2pygen node representing the PSy-layer routine \
90 to which to add declarations.
91 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`

94 :raises GenerationError: if the same scalar argument has different \
95 data types in different Kernel calls \
96 within the same Invoke.
97
98 '''
99 # Create dictionary of all scalar arguments for checks
100 const = LFRicConstants()
101 self._scalar_args = self._invoke.unique_declns_by_intent(
102 const.VALID_SCALAR_NAMES)
103 # Filter scalar arguments by intent and intrinsic type
104 self._real_scalars = self._invoke.unique_declns_by_intent(
105 const.VALID_SCALAR_NAMES,
106 intrinsic_type=const.MAPPING_DATA_TYPES["gh_real"])
107 self._integer_scalars = self._invoke.unique_declns_by_intent(
108 const.VALID_SCALAR_NAMES,
109 intrinsic_type=const.MAPPING_DATA_TYPES["gh_integer"])
110 self._logical_scalars = self._invoke.unique_declns_by_intent(
111 const.VALID_SCALAR_NAMES,
112 intrinsic_type=const.MAPPING_DATA_TYPES["gh_logical"])
113
114 for intent in FORTRAN_INTENT_NAMES:
115 scal = [arg.declaration_name for arg in self._scalar_args[intent]]
116 rscal = [arg.declaration_name for
117 arg in self._real_scalars[intent]]
118 iscal = [arg.declaration_name for
119 arg in self._integer_scalars[intent]]
120 lscal = [arg.declaration_name for
121 arg in self._logical_scalars[intent]]
122 # Add "real", "integer" and "logical" scalar lists for checks
123 decl_scal = rscal + iscal + lscal
124 # Check for unsupported intrinsic types
125 scal_inv = sorted(set(scal) - set(decl_scal))
126 if scal_inv:
127 raise InternalError(
128 f"Found unsupported intrinsic types for the scalar "
129 f"arguments {scal_inv} to Invoke '{self._invoke.name}'. "
130 f"Supported types are {const.VALID_INTRINSIC_TYPES}.")
131 # Check that the same scalar name is not found in either of
132 # 'real', 'integer' or 'logical' scalar lists (for instance if
133 # passed to one kernel as a 'real' and to another kernel as an
134 # 'integer' scalar)
135 scal_multi_type = [item for item, count in
136 Counter(decl_scal).items() if count > 1]
137 if scal_multi_type:
138 raise GenerationError(
139 f"Scalar argument(s) {scal_multi_type} in Invoke "
140 f"'{self._invoke.name}' have different metadata for data "
