45class MetaFuncsArgMetadata(CommonArgMetadata):
46 '''Class to capture the LFRic kernel metadata information for a
47 meta_funcs argument.
48
49 :param str function_space: the name of a function space.
50 :param bool basis_function: whether a basis_function is \
51 required. Defaults to False.
52 :param bool diff_basis_function: whether a differential basis \
53 function is required. Defaults to False.
54
55 '''
56 def __init__(self, function_space, basis_function=False,
57 diff_basis_function=False):
58 super().__init__()
59 self.function_spacefunction_spacefunction_space = function_space
60 # The setters for basis_function and diff_basis_function check
61 # for consistent values between the two. The
62 # diff_basis_function has not been set yet so the
63 # basis_function setter should not be used here.
64 self.check_boolean(basis_function, "basis_function argument")
65 self._basis_function = basis_function
66 self.diff_basis_functiondiff_basis_functiondiff_basis_function = diff_basis_function
67
68 @staticmethod
69 def create_from_fparser2(fparser2_tree):
70 '''Create an instance of this class from an fparser2 tree.
71
72 :param fparser2_tree: fparser2 tree containing the metadata \
73 for a meta_funcs argument.
74 :type fparser2_tree: :py:class:`fparser.two.Fortran2003.Part_Ref`
75
76 :returns: an instance of this class.
77 :rtype: :py:class:`psyclone.domain.lfric.kernel.MetaFuncsArgMetadata`
78
79 '''
80 MetaFuncsArgMetadata.check_fparser2_arg(
81 fparser2_tree, type_name="func_type")
82 nargs = MetaFuncsArgMetadata.get_nargs(fparser2_tree)
83 # There must be at least 2 and at most 3 arguments
84 if nargs < 2:
85 raise ParseError(
86 f"There must be at least 2 arguments: a function_space and "
87 f"one of basis_function or diff_basis_function, but found "
88 f"'{nargs}' arguments.")
89 if nargs > 3:
90 raise ParseError(
91 f"There must be at most 3 arguments: function_space, "
92 f"basis_function and diff_basis_function, but found "
93 f"'{nargs}'.")
94 function_space = MetaFuncsArgMetadata.get_arg(fparser2_tree, 0)
95
96 basis_function = False
97 diff_basis_function = False
98 arg1 = MetaFuncsArgMetadata.get_arg(fparser2_tree, 1)
99 MetaFuncsArgMetadata.validate_scalar_value(
100 arg1, ["gh_basis", "gh_diff_basis"],
101 "basis or differential basis")
102 if arg1.lower() == "gh_basis":
103 basis_function = True
104 else:
105 diff_basis_function = True
106
107 arg2 = None
108 if nargs == 3:
109 arg2 = MetaFuncsArgMetadata.get_arg(fparser2_tree, 2)
110 MetaFuncsArgMetadata.validate_scalar_value(
111 arg2, ["gh_basis", "gh_diff_basis"],
112 "basis or differential basis")
113 if arg1.lower() == arg2.lower():
114 raise ParseError(
115 f"The same basis or differential basis function value "
116 f"should not be repeated, but found '{arg1}' twice.")
117 basis_function = True
118 diff_basis_function = True
119 return MetaFuncsArgMetadata(
120 function_space, basis_function=basis_function,
121 diff_basis_function=diff_basis_function)

123 def fortran_string(self):
124 '''
125 :returns: the metadata represented by this class as Fortran.
126 :rtype: str
127
128 '''
129 args_str_list = [self._function_space]
130 if self._basis_function:
131 args_str_list.append("gh_basis")
132 if self._diff_basis_function:
133 args_str_list.append("gh_diff_basis")
134 args_str = ", ".join(args_str_list)
135 return f"func_type({args_str})"

195 @staticmethod
196 def _check_constraints(basis_function, diff_basis_function):
197 '''Check that at least one of basis_function or diff_basis_function
198 have been set to True.
199
200 :param bool basis_function: whether a basis_function is \
201 required.
202 :param bool diff_basis_function: whether a differential basis \
203 function is required.
204
205 :param ValueError: if neither of basis_function or \
206 diff_basis_function are set to True.
207
208 '''
209 if not basis_function and not diff_basis_function:
210 raise ValueError(
211 "At least one of basis_function or diff_basis_function must "
212 "be set to True.")
