62class ExtractDriverCreator(BaseDriverCreator):
63 '''This class provides the functionality to create a driver that
64 reads in extracted data produced by using the PSyData kernel-extraction
65 functionality.
66
67 :param integer_type: default scalar integer type to be used for integer
68 variables. Defaults to INTEGER_TYPE.
69 :type integer_type: :py:class:`psyclone.psyir.symbols.ScalarType`
70 :param real_type: default scalar real type to be used for real
71 variables. Defaults to REAL8_TYPE.
72 :type real_type: :py:class:`psyclone.psyir.symbols.ScalarType`
73
74 '''
75 def __init__(self, integer_type=INTEGER_TYPE, real_type=REAL8_TYPE):
76 super().__init__()
77 # Set the integer and real types to use.
78 # For convenience, also add the names used in the gocean config file:
79 self._default_types = {ScalarType.Intrinsic.INTEGER: integer_type,
80 "integer": integer_type,
81 ScalarType.Intrinsic.REAL: real_type,
82 "real": real_type}

85 def create_flattened_symbol(self, flattened_name, reference, symbol_table,
86 writer=FortranWriter()):
87 '''Takes a reference to a structure and creates a new Symbol of the
88 type that the reference resolves to, e.g. fld%data... will be mapped
89 to `real, dimension(:,:)`, and `fld%data$whole%xstart` to `integer`.
90
91 :param str flattened_name: the new 'flattened' name to be used for
92 the newly created symbol.
93 :param reference: the reference to a structure.
94 :type reference: :py:class:`psyclone.psyir.nodes.StructureReference`
95 :param symbol_table: the symbol table to use to make sure a unique
96 name is created (based on `flattened_name`).
97 :type symbol_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
98 :param writer: a backend visitor to convert the reference into a
99 string, which is then used to find the corresponding
100 gocean-grid-properties.
101 :type writer:
102 :py:class:`psyclone.psyir.backend.language_writer.LanguageWriter`
103
104 :raises InternalError: if the structure access is not to a
105 GOCean grid property.
106 :raises InternalError: if there is no default type defined for
107 the type of the GOCean grid property (defaults are defined in
108 the constructor of this class).
109 :raises InternalError: if the gocean grid property type is
110 neither 'array' nor 'scalar'.
111
112 :returns: the new symbol created.
113 :rtype: :py:class:`psyclone.psyir.symbol.DataSymbol`
114
115 '''
116 fortran_expression = writer(reference)
117 api_config = Config.get().api_conf("gocean")
118 grid_properties = api_config.grid_properties
119
120 for prop_name in grid_properties:
121 gocean_property = grid_properties[prop_name]
122 deref_name = gocean_property.fortran.format(reference.name)
123 if fortran_expression == deref_name:
124 break
125 else:
126 raise InternalError(
127 f"Could not find type for reference '{fortran_expression}' "
128 f"in the config file '{Config.get().filename}'.")
129 try:
130 base_type = self._default_types[gocean_property.intrinsic_type]
131 except KeyError as err:
132 raise InternalError(
133 f"Type '{gocean_property.intrinsic_type}' of "
134 f"the property reference '{fortran_expression}' "
135 f"as defined in the config file "
136 f"'{Config.get().filename}' is not supported "
137 f"in the GOcean API.") from err
138 # Handle name clashes (e.g. if the user used a variable that is
139 # the same as a flattened grid property)
140 flattened_name = symbol_table.next_available_name(flattened_name)
141 if gocean_property.type == "scalar":
142 new_symbol = DataSymbol(flattened_name, base_type)
143
144 elif gocean_property.type == "array":
145 # At this stage all gocean arrays are 2d (even integer ones)
146 # so no need to add any further tests here.
147 array = ArrayType(base_type, [ArrayType.Extent.DEFERRED,
148 ArrayType.Extent.DEFERRED])
149 new_symbol = DataSymbol(flattened_name, array)
150 else:
151 raise InternalError(f"The expression '{fortran_expression}' maps "
152 f"to an unknown GOcean property type "
153 f"'{gocean_property.type}' in the config "
154 f"file '{Config.get().filename}'.")
155
156 return new_symbol

159 @staticmethod
160 def flatten_string(fortran_string):
161 '''Replaces all `%` with `_` in the string, creating a 'flattened'
162 name.
163
164 :param str fortran_string: the Fortran string containing '%'.
165
166 :returns: a flattened string (all '%' replaced with '_'.)
167 :rtype: str
168
169 '''
170 return fortran_string.replace("%", "_")

172 # -------------------------------------------------------------------------
173 def flatten_reference(self, old_reference, symbol_table,
174 writer=FortranWriter()):
175 '''Replaces `old_reference` which is a structure type with a new
176 simple Reference and a flattened name (replacing all % with _).
177
178 :param old_reference: a reference to a structure member.
179 :type old_reference:
180 :py:class:`psyclone.psyir.nodes.StructureReference`
181 :param symbol_table: the symbol table to which to add the newly
182 defined flattened symbol.
183 :type symbol_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
184 :param writer: a Fortran writer used when flattening a
185 `StructureReference`.
186 :type writer: :py:class:`psyclone.psyir.backend.fortran.FortranWriter`
187
188 '''
189 # A field access (`fld%data`) will get the `%data` removed, since then
190 # this avoids a potential name clash (`fld` is guaranteed to
191 # be unique, since it's a variable already, but `fld_data` could clash
192 # with a user variable if the user uses `fld` and `fld_data`).
193 # Furthermore, the netcdf file declares the variable without `%data`,
194 # so removing `%data` here also simplifies code creation later on.
195 signature, _ = old_reference.get_signature_and_indices()
196 fortran_string = writer(old_reference)
197 if signature[-1] == "data":
198 # Remove %data
199 fortran_string = fortran_string[:-5]
200 try:
201 symbol = symbol_table.lookup_with_tag(fortran_string)
202 except KeyError:
203 flattened_name = self.flatten_string(fortran_string)
204 # Symbol not in table, create a new symbol
205 symbol = self.create_flattened_symbol(flattened_name,
206 old_reference, symbol_table,
207 writer)
208 symbol_table.add(symbol, tag=fortran_string)
209 # We need to create a new, flattened Reference and replace the
210 # StructureReference with it:
211 old_reference.replace_with(Reference(symbol))
212
213 # -------------------------------------------------------------------------
214 def add_all_kernel_symbols(self, sched, symbol_table,
215 writer=FortranWriter()):
216 '''This function adds all symbols used in `sched` to the symbol table.
217 It uses GOcean-specific knowledge to declare fields and flatten their
218 name.
219
220 :param sched: the schedule that will be called by this driver program.
221 :type sched: :py:class:`psyclone.psyir.nodes.Schedule`
222 :param symbol_table: the symbol table to which to add all found
223 symbols.
224 :type symbol_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
225 :param writer: a Fortran writer used when flattening a
226 `StructureReference`.
227 :type writer: :py:class:`psyclone.psyir.backend.fortan.FortranWriter`
228
229 :raises InternalError: if a non-derived type has an unknown
230 intrinsic type.
231 :raises InternalError: if an unknown derived type is
232 encountered. At this stage only the dl_esm_inf `field` type
233 is supported.
234
235 '''
236 all_references = sched.walk(Reference)
237 # First we add all non-structure names to the symbol table. This way
238 # the flattened name can be ensured not to clash with a variable name
239 # used in the program.
240 for reference in all_references:
241 if isinstance(reference.symbol, (RoutineSymbol, IntrinsicSymbol)):
242 continue
243 # For now ignore structure names, which require flattening
244 if isinstance(reference, StructureReference):
245 continue
246 old_symbol = reference.symbol
247 if old_symbol.name in symbol_table:
248 # The symbol has already been declared. We then still
249 # replace the old symbol with the new symbol to have all
250 # symbols consistent:
251 reference.symbol = symbol_table.lookup(old_symbol.name)
252 continue
253
254 # We found a new symbol, so we create a new symbol in the new
255 # symbol table here. GOcean does not support any arbitrary array
256 # as kernel argument (only fields and grid properties), so we
257 # only need to declare scalars here.
258 try:
259 new_type = self._default_types[old_symbol.datatype.intrinsic]
260 except KeyError as err:
261 fortran_string = writer(reference)
262 valid = list(self._default_types.keys())
263 # Sort to make sure we get a reproducible order for testing
264 valid.sort()
265 raise InternalError(
266 f"Error when constructing driver for '{sched.name}': "
267 f"Unknown intrinsic data type "
268 f"'{old_symbol.datatype.intrinsic}' in reference "
269 f"'{fortran_string}'. Valid types are '{valid}'.") from err
270 new_symbol = symbol_table.new_symbol(root_name=reference.name,
271 tag=reference.name,
272 symbol_type=DataSymbol,
273 datatype=new_type)
274 reference.symbol = new_symbol
275
276 # Now handle all derived type. In GOcean the only supported derived
277 # type is "r2d_field". This type might be used to access the field
278 # data, loop boundaries or other properties. We use the
279 # grid_properties information from the config file to identify which
280 # property is used. The name of a derived type is 'flattened', i.e.
281 # all '%' are replaced with '_', and this is then declared as a
282 # non-structured type. We also need to make sure that a flattened
283 # name does not clash with a variable declared by the user. We use
284 # the structured name (with '%') as tag to handle this.
285 for reference in all_references:
286 if isinstance(reference.symbol, (RoutineSymbol, IntrinsicSymbol)):
287 continue
288 if not isinstance(reference, StructureReference):
289 continue
290 old_symbol = reference.symbol
291 if old_symbol.datatype.name != "r2d_field":
292 fortran_string = writer(reference)
293 raise InternalError(
294 f"Error when constructing driver for '{sched.name}': "
295 f"Unknown derived type '{old_symbol.datatype.name}' "
296 f"in reference '{fortran_string}'.")
297 # We have a structure reference to a field, flatten it, and
298 # replace the StructureReference with a new Reference to this
299 # flattened name (e.g. `fld%data` becomes `fld_data`)
300 self.flatten_reference(reference, symbol_table, writer=writer)

303 @staticmethod
304 def create_read_in_code(program, psy_data, read_write_info, postfix):
305 '''This function creates the code that reads in the NetCDF file
306 produced during extraction. For each:
307
308 - variable that is read-only, it will declare the symbol and add code
309 that reads in the variable using the PSyData library.
310 - variable that is read and written, it will create code to read in the
311 variable that is read, and create a new variable with the same name
312 and "_post" added which is read in to store the values from the
313 NetCDF file after the instrumented region was executed. In the end,
314 the variable that was read and written should have the same value
315 as the corresponding "_post" variable.
316 - variable that is written only, it will create a variable with "_post"
317 as postfix that reads in the output data from the NetCDF file. It
318 then also declares a variable without postfix (which will be the
319 parameter to the function), allocates it based on the shape of
320 the corresponding "_post" variable, and initialises it with 0.
321
322 :param program: the PSyIR Routine to which any code must
323 be added. It also contains the symbol table to be used.
324 :type program: :py:class:`psyclone.psyir.nodes.Routine`
325 :param psy_data: the PSyData symbol to be used.
326 :type psy_data: :py:class:`psyclone.psyir.symbols.DataSymbol`
327 :param read_write_info: information about all input and output
328 parameters.
329 :type read_write_info: :py:class:`psyclone.psyir.tools.ReadWriteInfo`
330 :param str postfix: a postfix that is added to a variable to
331 create the corresponding variable that stores the output
332 value from the kernel data file.
333
334 :returns: a list with all output parameters, i.e. variables that
335 need to be verified after executing the kernel. Each entry is
336 a 2-tuple containing the symbol of the computed variable, and
337 the symbol of the variable that contains the value read from
338 the file.
339 :rtype: list[tuple[:py:class:`psyclone.psyir.symbols.Symbol`,
340 :py:class:`psyclone.psyir.symbols.Symbol`]]
341
342 '''
343 symbol_table = program.scope.symbol_table
344 read_var = f"{psy_data.name}%ReadVariable"
345
346 # Collect all output symbols to later create the tests for
347 # correctness. This list stores 2-tuples: first one the
348 # variable that stores the output from the kernel, the second
349 # one the variable that stores the output values read from the
350 # file. The content of these two variables should be identical
351 # at the end.
352 output_symbols = []
353
354 # First handle variables that are read:
355 # -------------------------------------
356 for signature in read_write_info.signatures_read:
357 # Find the right symbol for the variable. Note that all variables
358 # in the input and output list have been detected as being used
359 # when the variable accesses were analysed. Therefore, these
360 # variables have References, and will already have been declared
361 # in the symbol table (in add_all_kernel_symbols).
362 sig_str = str(signature)
363 sym = symbol_table.lookup_with_tag(sig_str)
364 name_lit = Literal(sig_str, CHARACTER_TYPE)
365 BaseDriverCreator.add_call(program, read_var,
366 [name_lit, Reference(sym)])
367
368 # Then handle all variables that are written (note that some
369 # variables might be read and written)
370 for signature in read_write_info.signatures_written:
371 # Find the right symbol for the variable. Note that all variables
372 # in the input and output list have been detected as being used
373 # when the variable accesses were analysed. Therefore, these
374 # variables have References, and will already have been declared
375 # in the symbol table (in add_all_kernel_symbols).
376 sig_str = str(signature)
377 sym = symbol_table.lookup_with_tag(sig_str)
378
379 # The variable is written (and maybe read as well)
380 # ------------------------------------------------
381 # Declare a 'post' variable of the same type and
382 # read in its value.
383 post_name = sig_str+postfix
384 post_sym = symbol_table.new_symbol(post_name,
385 symbol_type=DataSymbol,
386 datatype=sym.datatype)
387 BaseDriverCreator.add_call(program, read_var,
388 [Literal(post_name, CHARACTER_TYPE),
389 Reference(post_sym)])
390
391 # Now if a variable is written to, but not read, the variable
392 # is not allocated. So we need to allocate it and set it to 0.
393 if not read_write_info.is_read(signature):
394 if isinstance(post_sym.datatype, ArrayType):
395 alloc = IntrinsicCall.create(
396 IntrinsicCall.Intrinsic.ALLOCATE,
397 [Reference(sym), ("mold", Reference(post_sym))])
398 program.addchild(alloc)
399 set_zero = Assignment.create(Reference(sym),
400 Literal("0", INTEGER_TYPE))
401 program.addchild(set_zero)
402 output_symbols.append((sym, post_sym))
403 return output_symbols

406 @staticmethod
407 def import_modules(program, sched):
408 '''This function adds all the import statements required for the
409 actual kernel calls. It finds all calls in the PSyIR tree and
410 checks for calls with a ImportInterface. Any such call will
411 get a ContainerSymbol added for the module, and a RoutineSymbol
412 with an import interface pointing to this module.
413
414 :param program: the PSyIR Routine to which any code must
415 be added. It also contains the symbol table to be used.
416 :type program: :py:class:`psyclone.psyir.nodes.Routine`
417 :param sched: the schedule that will be called by the driver
418 program created.
419 :type sched: :py:class:`psyclone.psyir.nodes.Schedule`
420
421 '''
422 symbol_table = program.scope.symbol_table
423 for call in sched.walk(Call):
424 routine = call.routine.symbol
425 if not isinstance(routine.interface, ImportInterface):
426 continue
427 if routine.name in symbol_table:
428 # Symbol has already been added - ignore
429 continue
430 # We need to create a new symbol for the module and the routine
431 # called (the PSyIR backend will then create a suitable import
432 # statement).
433 module = ContainerSymbol(routine.interface.container_symbol.name)
434 symbol_table.add(module)
435 new_routine_sym = RoutineSymbol(routine.name, UnresolvedType(),
436 interface=ImportInterface(module))
437 symbol_table.add(new_routine_sym)

440 def create(self, nodes, read_write_info, prefix, postfix, region_name):
441 # pylint: disable=too-many-arguments
442 '''This function uses the PSyIR to create a stand-alone driver
443 that reads in a previously created file with kernel input and
444 output information, and calls the kernels specified in the 'nodes'
445 PSyIR tree with the parameters from the file. It returns the
446 file container which contains the driver.
447
448 :param nodes: a list of nodes.
449 :type nodes: list[:py:class:`psyclone.psyir.nodes.Node`]
450 :param read_write_info: information about all input and output
451 parameters.
452 :type read_write_info: :py:class:`psyclone.psyir.tools.ReadWriteInfo`
453 :param str prefix: the prefix to use for each PSyData symbol,
454 e.g. 'extract' as prefix will create symbols `extract_psydata`.
455 :param str postfix: a postfix that is appended to an output variable
456 to create the corresponding variable that stores the output
457 value from the kernel data file. The caller must guarantee that
458 no name clashes are created when adding the postfix to a variable
459 and that the postfix is consistent between extract code and
460 driver code (see 'ExtractTrans.determine_postfix()').
461 :param (str,str) region_name: an optional name to
462 use for this PSyData area, provided as a 2-tuple containing a
463 location name followed by a local name. The pair of strings
464 should uniquely identify a region.
465
466 :returns: the program PSyIR for a stand-alone driver.
467 :rtype: :py:class:`psyclone.psyir.psyir.nodes.FileContainer`
468
469 '''
470 # pylint: disable=too-many-locals
471
472 # Since this is a 'public' method of an entirely separate class,
473 # we check that the list of nodes is what it expects. This is done
474 # by invoking the validate function of the basic extract function.
475 extract_trans = ExtractTrans()
476 # We need to provide the prefix to the validation function:
477 extract_trans.validate(nodes, options={"prefix": prefix})
478
479 module_name, local_name = region_name
480 unit_name = f"{module_name}_{local_name}"
481
482 # First create the file container, which will only store the program:
483 file_container = FileContainer(unit_name)
484
485 # Create the program and add it to the file container:
486 program = Routine.create(unit_name, is_program=True)
487 program_symbol_table = program.symbol_table
488 file_container.addchild(program)
489
490 if prefix:
491 prefix = prefix + "_"
492
493 psy_data_mod = ContainerSymbol("read_kernel_data_mod")
494 program_symbol_table.add(psy_data_mod)
495 psy_data_type = DataTypeSymbol("ReadKernelDataType", UnresolvedType(),
496 interface=ImportInterface(psy_data_mod))
497 program_symbol_table.add(psy_data_type)
498
499 writer = FortranWriter()
500 # The validation of the extract transform guarantees that all nodes
501 # in the node list have the same parent.
502 schedule_copy = nodes[0].parent.copy()
503 schedule_copy.lower_to_language_level()
504 self.import_modules(program, schedule_copy)
505 self.add_all_kernel_symbols(schedule_copy, program_symbol_table,
506 writer)
507
508 root_name = prefix + "psy_data"
509 psy_data = program_symbol_table.new_symbol(root_name=root_name,
510 symbol_type=DataSymbol,
511 datatype=psy_data_type)
512
513 module_str = Literal(module_name, CHARACTER_TYPE)
514 region_str = Literal(local_name, CHARACTER_TYPE)
515 self.add_call(program, f"{psy_data.name}%OpenReadModuleRegion",
516 [module_str, region_str])
517
518 output_symbols = self.create_read_in_code(program, psy_data,
519 read_write_info, postfix)
520 # Copy over all of the executable part of the extracted region
521 all_children = schedule_copy.pop_all_children()
522 for child in all_children:
523 program.addchild(child)
524
525 self.add_result_tests(program, output_symbols)
526
527 return file_container

530 def get_driver_as_string(self, nodes, read_write_info,
531 prefix, postfix, region_name,
532 writer=FortranWriter()):
533 # pylint: disable=too-many-arguments
534 '''This function uses `create()` function to get the PSyIR of a
535 stand-alone driver, and then uses the provided language writer
536 to create a string representation in the selected language
537 (defaults to Fortran).
538
539 :param nodes: a list of nodes.
540 :type nodes: list[:py:class:`psyclone.psyir.nodes.Node`]
541 :param read_write_info: information about all input and output
542 parameters.
543 :type read_write_info: :py:class:`psyclone.psyir.tools.ReadWriteInfo`
544 :param str prefix: the prefix to use for each PSyData symbol,
545 e.g. 'extract' as prefix will create symbols `extract_psydata`.
546 :param str postfix: a postfix that is appended to an output variable
547 to create the corresponding variable that stores the output
548 value from the kernel data file. The caller must guarantee that
549 no name clashes are created when adding the postfix to a variable
550 and that the postfix is consistent between extract code and
551 driver code (see 'ExtractTrans.determine_postfix()').
552 :param (str,str) region_name: an optional name to
553 use for this PSyData area, provided as a 2-tuple containing a
554 location name followed by a local name. The pair of strings
555 should uniquely identify a region.
556 :param language_writer: a backend visitor to convert PSyIR
557 representation to the selected language. It defaults to
558 the FortranWriter.
559 :type language_writer:
560 :py:class:`psyclone.psyir.backend.language_writer.LanguageWriter`
561
562 :returns: the driver in the selected language.
563 :rtype: str
564
565 '''
566 file_container = self.create(nodes, read_write_info,
567 prefix, postfix, region_name)
568 return writer(file_container)

571 def write_driver(self, nodes, read_write_info, prefix, postfix,
572 region_name, writer=FortranWriter()):
573 # pylint: disable=too-many-arguments
574 '''This function uses the `get_driver_as_string()` function to get a
575 a stand-alone driver, and then writes this source code to a file. The
576 file name is derived from the region name:
577 "driver-"+module_name+"_"+region_name+".f90"
578
579 :param nodes: a list of nodes.
580 :type nodes: list[:py:class:`psyclone.psyir.nodes.Node`]
581 :param read_write_info: information about all input and output
582 parameters.
583 :type read_write_info: :py:class:`psyclone.psyir.tools.ReadWriteInfo`
584 :param str prefix: the prefix to use for each PSyData symbol,
585 e.g. 'extract' as prefix will create symbols `extract_psydata`.
586 :param str postfix: a postfix that is appended to an output variable
587 to create the corresponding variable that stores the output
588 value from the kernel data file. The caller must guarantee that
589 no name clashes are created when adding the postfix to a variable
590 and that the postfix is consistent between extract code and
591 driver code (see 'ExtractTrans.determine_postfix()').
592 :param (str,str) region_name: an optional name to
593 use for this PSyData area, provided as a 2-tuple containing a
594 location name followed by a local name. The pair of strings
595 should uniquely identify a region.
596 :param language_writer: a backend visitor to convert PSyIR
597 representation to the selected language. It defaults to
598 the FortranWriter.
599 :type language_writer:
600 :py:class:`psyclone.psyir.backend.language_writer.LanguageWriter`
601
602 '''
