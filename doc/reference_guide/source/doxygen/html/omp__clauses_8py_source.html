155class OMPPrivateClause(Clause):
156 '''
157 OpenMP private clause. This is used to declare variables as private
158 to an OpenMP region.
159 '''
160 _children_valid_format = "Reference*"
161
162 @staticmethod
163 def create(symbols):
164 ''' Create a OMPPrivateClause containing a Reference to each of the
165 provided symbols as children.
166
167 :param symbols: List of symbols to reference in the private clause.
168 :type symbols: List[:py:class:`psyclone.psyir.symbols.Symbol`]
169
170 :returns: A OMPPrivateClause referencing the provided symbols.
171 :rtype: py:class:`psyclone.psyir.nodes.OMPPrivateClause`
172
173 :raises TypeError: If the symbols argument is not a List that \
174 contains only PSyIR Symbols.
175
176 '''
177 if not isinstance(symbols, list):
178 raise TypeError(
179 f"OMPPrivateClause expected the 'symbols' argument to be a "
180 f"list, but found '{type(symbols).__name__}' instead.")
181 for symbol in symbols:
182 if not isinstance(symbol, Symbol):
183 raise TypeError(
184 f"OMPPrivateClause expected all the items in the 'symbols'"
185 f" list to be PSyIR Symbols, but found a "
186 f"'{type(symbol).__name__}'.")
187
188 references = [Reference(symbol) for symbol in symbols]
189 return OMPPrivateClause(children=references)
190

192 def _clause_string(self):
193 '''
194 :returns: the string that represents this clause in OpenMP (i.e.\
195 "private"). Returns an empty string to avoid generation of\
196 code if this clause has no children.
197 :rtype: str
198 '''
199 if len(self.childrenchildrenchildrenchildren) > 0:
200 return "private"
201 return ""

206 Decides whether a given child and position are valid for this node.

354class OMPScheduleClause(Clause):
355 '''
356 OpenMP Schedule clause used for OMP Do Directives.
357
358 :param str schedule: The OpenMP schedule to use with this directive. \
359 The default value is "none" which means that no explicit schedule \
360 is specified.
361 :param kwargs: additional keyword arguments provided to the PSyIR node.
362 :type kwargs: unwrapped dict.
363
364 '''
365 _children_valid_format = "None"
366
367 VALID_OMP_SCHEDULES = ["runtime", "static", "dynamic", "guided", "auto",
368 "none"]
369
370 def __init__(self, schedule="none", **kwargs):
371 self.schedulescheduleschedule = schedule
372 super().__init__(**kwargs)
373
374 @property
375 def _clause_string(self):
376 '''
377 :returns: the string that represents this clause in OpenMP (e.g. \
378 "schedule(static)"). The value inside parentheses is \
379 set to the value of the schedule of this clause unless \
380 that value is 'none' in which case an empty string is returned.
381 :rtype: str
382 '''
383 if self._schedule != "none":
384 return f"schedule({self._schedule})"
385 return ""
386
387 @property
388 def schedule(self):
389 '''
390 :returns: the schedule for this OMPScheduleClause.
391 :rtype: str
392 '''
393 return self._schedule
394
395 @schedule.setter
396 def schedule(self, schedule):
397 '''
398 :param str schedule: the schedule to use for this clause.
399
400 :raises ValueError: if the supplied value is not a recognised \
401 OpenMP schedule.
402 '''
403 if schedule not in self.VALID_OMP_SCHEDULES:
404 raise ValueError(
405 f"Schedule must be one of {self.VALID_OMP_SCHEDULES}. "
406 f"Found '{schedule}'.")
407 self._schedule = schedule
408
409 def __eq__(self, other):
410 '''
411 Two OMPScheduleClause are equal if they have the same schedule.
412
413 :param object other: the object to check equality to.
414
415 :returns: whether other is equal to self.
416 :rtype: bool
417 '''
418 is_eq = super().__eq__(other)
419 is_eq = is_eq and (self.schedulescheduleschedule == other.schedule)
420 return is_eq
