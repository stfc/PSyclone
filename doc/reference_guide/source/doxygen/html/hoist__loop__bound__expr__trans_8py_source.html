36'''This module contains the HoistLoopBoundExprTrans transformation. This
37transformation moves complex bounds expressions out of the loop construct and
38places them in integer scalar assignments before the loop.
39
40'''
41
42from psyclone.psyir.nodes import Routine, Literal, Reference, \
43 StructureReference, Assignment, Directive
44from psyclone.psyir.symbols import DataSymbol, INTEGER_TYPE
45from psyclone.psyir.transformations.loop_trans import LoopTrans, \
46 TransformationError
47
48
49class HoistLoopBoundExprTrans(LoopTrans):
50 '''This transformation moves complex bounds expressions out of the loop
51 construct and places them in integer scalar assignments before the loop.
52
53 >>> from psyclone.psyir.backend.fortran import FortranWriter
54 >>> from psyclone.psyir.frontend.fortran import FortranReader
55 >>> from psyclone.psyir.nodes import Loop
56 >>> from psyclone.psyir.transformations import HoistTrans
57 >>> code = ("program test\\n"
58 ... " use mymod, only: mytype\\n"
59 ... " integer :: i,j,n\\n"
60 ... " real :: a(n)\\n"
61 ... " do i=mytype%start, UBOUND(a,1)\\n"
62 ... " a(i) = 1.0\\n"
63 ... " end do\\n"
64 ... "end program\\n")
65 >>> psyir = FortranReader().psyir_from_source(code)
66 >>> hoist = HoistLoopBoundExprTrans()
67 >>> hoist.apply(psyir.walk(Loop)[0])
68 >>> print(FortranWriter()(psyir))
69 program test
70 use mymod, only : mytype
71 integer :: i
72 integer :: j
73 integer :: n
74 real, dimension(n) :: a
75 integer :: loop_bound
76 integer :: loop_bound_1
77 <BLANKLINE>
78 loop_bound_1 = UBOUND(a, 1)
79 loop_bound = mytype%start
80 do i = loop_bound, loop_bound_1, 1
81 a(i) = 1.0
82 enddo
83 <BLANKLINE>
84 end program test
85 <BLANKLINE>
86
87 '''
88 def apply(self, node, options=None):
89 '''Move complex bounds expressions out of the given loop construct and
90 place them in integer scalar assignments before the loop.
91
92 :param node: target PSyIR loop.
93 :type node: :py:class:`psyclone.psyir.nodes.Loop`
94 :param options: a dictionary with options for transformations.
95 :type options: Dict[str, Any]
96
97 '''
98 self.validatevalidatevalidate(node, options)
99
100 parent = node.parent
101 position = node.position
102
103 for name, bound in [("start", node.start_expr),
104 ("stop", node.stop_expr),
105 ("step", node.step_expr)]:
106
107 # Skip Literals and non-structure references. Maybe it could be
108 # more selective by allowing arithmetic expressions
109 if isinstance(bound, Literal):
110 continue
111 if isinstance(bound, Reference) and not \
112 isinstance(bound, StructureReference):
113 continue
114
115 # Create new symbol
116 symbol = node.ancestor(Routine).symbol_table.new_symbol(
117 f"loop_{name}", symbol_type=DataSymbol, datatype=INTEGER_TYPE
118 )
119 # Move bound expression to an assignment preceding the loop
120 bound.replace_with(Reference(symbol))
121 parent.addchild(Assignment.create(Reference(symbol), bound),
122 position)
123
124 def validate(self, node, options=None):
125 '''Checks that the supplied node is a valid target for the
126 transformation.
127
128 :param node: target PSyIR loop.
129 :type node: :py:class:`psyclone.psyir.nodes.Loop`
130 :param options: a dictionary with options for transformations.
131 :type options: Dict[str, Any]
132
133 :raises TransformationError: if the supplied node does not have an \
134 ancestor Routine.
135 :raises TransformationError: if the supplied node is directly inside \
136 a Directive Schedule.
137
138 '''
139 super().validate(node)
140
141 # The node should be an assignment
142 if not node.ancestor(Routine):
143 raise TransformationError(
144 "The loop provided to HoistLoopBoundExprTrans must belong to"
145 " a Routine into which the hoisted expressions can be placed.")
146
147 # TODO #1817: We can remove this restriction if we get rid of the
148 # special Schedule that some directives have.
149 # The node should not be directly inside a directive
150 if node.parent.parent and isinstance(node.parent.parent, Directive):
151 raise TransformationError(
152 f"The loop provided to HoistLoopBoundExprTrans must not be "
153 f"directly inside a Directive as its Schedule does not support"
154 f" multiple statements, but found "
155 f"'{node.parent.parent.coloured_name(False)}'.")
156
157 def __str__(self):
158 return ("Hoist complex loop bound expressions outside the loop "
159 "construct")
