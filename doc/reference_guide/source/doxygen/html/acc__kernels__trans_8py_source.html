56class ACCKernelsTrans(RegionTrans):
57 '''
58 Enclose a sub-set of nodes from a Schedule within an OpenACC kernels
59 region (i.e. within "!$acc kernels" ... "!$acc end kernels" directives).
60
61 For example:
62
63 >>> from psyclone.psyir.frontend import FortranReader
64 >>> psyir = FortranReader().psyir_from_source(NEMO_SOURCE_FILE)
65 >>>
66 >>> from psyclone.psyir.transformations import ACCKernelsTrans
67 >>> ktrans = ACCKernelsTrans()
68 >>>
69 >>> schedule = psyir.children[0]
70 >>> # Uncomment the following line to see a text view of the schedule
71 >>> # print(schedule.view())
72 >>> kernels = schedule.children[9]
73 >>> # Transform the kernel
74 >>> ktrans.apply(kernels)
75
76 '''
77 excluded_node_types = (CodeBlock, Return, PSyDataNode,
78 psyGen.HaloExchange, WhileLoop)
79
80 def apply(self, node: Union[Node, List[Node]], options: dict = None):
81 '''
82 Enclose the supplied list of PSyIR nodes within an OpenACC
83 Kernels region.
84
85 :param node: the node(s) in the PSyIR to enclose.
86 :type node: :py:class:`psyclone.psyir.nodes.Node` |
87 list[:py:class:`psyclone.psyir.nodes.Node`]
88 :param options: a dictionary with options for transformations.
89 :type options: Optional[Dict[str, Any]]
90 :param bool options["default_present"]: whether or not the kernels
91 region should have the 'default present' attribute (indicating
92 that data is already on the accelerator). When using managed
93 memory this option should be False.
94 :param bool options["allow_string"]: whether to allow the
95 transformation on assignments involving character types. Defaults
96 to False.
97 :param bool options["verbose"]: log the reason the validation failed,
98 at the moment with a comment in the provided PSyIR node.
99
100 '''
101 # Ensure we are always working with a list of nodes, even if only
102 # one was supplied via the `node` argument.
103 node_list = self.get_node_list(node)
104
105 self.validatevalidatevalidate(node_list, options)
106
107 parent = node_list[0].parent
108 start_index = node_list[0].position
109
110 if not options:
111 options = {}
112 default_present = options.get("default_present", False)
113
114 # Create a directive containing the nodes in node_list and insert it.
115 directive = ACCKernelsDirective(
116 parent=parent, children=[node.detach() for node in node_list],
117 default_present=default_present)
118
119 parent.children.insert(start_index, directive)

121 def validate(self, nodes: Union[Node, List[Node]],
122 options: dict = None) -> None:
123 # pylint: disable=signature-differs
124 '''
125 Check that we can safely enclose the supplied node or list of nodes
126 within OpenACC kernels ... end kernels directives.
127
128 :param nodes: the proposed PSyIR node or nodes to enclose in the
129 kernels region.
130 :type nodes: (list of) :py:class:`psyclone.psyir.nodes.Node`
131 :param options: a dictionary with options for transformations.
132 :type options: Optional[Dict[str, Any]]
133 :param bool options["disable_loop_check"]: whether to disable the
134 check that the supplied region contains 1 or more loops. Default
135 is False (i.e. the check is enabled).
136 :param bool options["allow_string"]: whether to allow the
137 transformation on assignments involving character types. Defaults
138 to False.
139 :param bool options["verbose"]: log the reason the validation failed,
140 at the moment with a comment in the provided PSyIR node.
141
142 :raises NotImplementedError: if the supplied Nodes belong to
143 a GOInvokeSchedule.
144 :raises TransformationError: if there is an access to an assumed-size
145 character variable within the region.
146 :raises TransformationError: if the proposed region contains a call to
147 a routine that is not available on the accelerator.
148 :raises TransformationError: if there are no Loops within the
149 proposed region and options["disable_loop_check"] is not True.
150 :raises TransformationError: if any assignments in the region contain a
151 character type child and options["allow_string"] is not True.
152
153 '''
154 if not options:
155 options = {}
156
157 # Ensure we are always working with a list of nodes, even if only
158 # one was supplied via the `nodes` argument.
159 node_list = self.get_node_list(nodes)
160
161 # Check that the front-end is valid
162 # pylint: disable-next=import-outside-toplevel
163 from psyclone.gocean1p0 import GOInvokeSchedule
164 if node_list[0].ancestor(GOInvokeSchedule):
165 raise NotImplementedError(
166 "OpenACC kernels regions are not currently supported for "
167 "GOcean InvokeSchedules")
168 super().validate(node_list, options)
169
170 # The regex we use to determine whether a character declaration is
171 # of assumed size ('LEN=*' or '*(*)').
172 # TODO #2612 - improve the fparser2 frontend support for character
173 # declarations.
174 assumed_size = re.compile(r"\‍(\s*len\s*=\s*\*\s*\‍)|\*\s*\‍(\s*\*\s*\‍)")
175
176 # Construct a list of any symbols that correspond to assumed-size
177 # character strings. These can only be routine arguments.
178 char_syms = []
179 parent_routine = node_list[0].ancestor(Routine)
180 if parent_routine:
181 arg_syms = parent_routine.symbol_table.argument_datasymbols
182 for sym in arg_syms:
183 # Currently the fparser2 frontend does not support any type
184 # of LEN= specification on a character variable so we resort
185 # to a regex to check whether it is assumed-size.
186 if isinstance(sym.datatype, UnsupportedFortranType):
187 type_txt = sym.datatype.type_text.lower()
188 if (type_txt.startswith("character") and
189 assumed_size.search(type_txt)):
190 char_syms.append(sym)
191
192 for node in node_list:
193 # Check that there are no assumed-size character variables as these
194 # cause an Internal Compiler Error with (at least) NVHPC <= 24.5.
195 for ref in node.walk(Reference):
196 if ref.symbol in char_syms:
197 stmt = ref.ancestor(Statement)
198 raise TransformationError(
199 f"Assumed-size character variables cannot be enclosed "
200 f"in an OpenACC region but found "
201 f"'{stmt.debug_string()}'")
202 # Check there are no character assignments in the region as these
203 # cause various problems with (at least) NVHPC <= 24.5
204 if not options.get("allow_string", False):
205 message = (
206 f"{self.name} does not permit assignments involving "
207 f"character variables by default (use the 'allow_string' "
208 f"option to include them)")
209 for assign in node.walk(Assignment):
210 ArrayAssignment2LoopsTrans.validate_no_char(
211 assign, message, options)
212
213 # Check that any called routines are supported on the device.
214 for icall in node.walk(Call):
215 if not icall.is_available_on_device():
216 raise TransformationError(
217 f"Cannot include '{icall.debug_string()}' in an "
218 f"OpenACC region because it is not available on GPU.")
219
220 # Check that we have at least one loop or array range within
221 # the proposed region unless this has been disabled.
222 if options and options.get("disable_loop_check", False):
223 return
224
225 for node in node_list:
226 if (any(assign for assign in node.walk(Assignment)
227 if assign.is_array_assignment) or node.walk(Loop)):
228 break
229 else:
230 # Branch executed if loop does not exit with a break
231 raise TransformationError(
232 "A kernels transformation must enclose at least one loop or "
233 "array range but none were found.")
