51class ChunkLoopTrans(LoopTrans):
52 '''
53 Apply a chunking transformation to a loop (in order to permit a
54 chunked parallelisation). For example:
55
56 >>> from psyclone.psyir.frontend.fortran import FortranReader
57 >>> from psyclone.psyir.nodes import Loop
58 >>> from psyclone.psyir.transformations import ChunkLoopTrans
59 >>> psyir = FortranReader().psyir_from_source("""
60 ... subroutine sub()
61 ... integer :: ji, tmp(100)
62 ... do ji=1, 100
63 ... tmp(ji) = 2 * ji
64 ... enddo
65 ... end subroutine sub""")
66 >>> loop = psyir.walk(Loop)[0]
67 >>> ChunkLoopTrans().apply(loop)
68
69 will generate:
70
71 .. code-block:: fortran
72
73 subroutine sub()
74 integer :: ji
75 integer, dimension(100) :: tmp
76 integer :: ji_el_inner
77 integer :: ji_out_var
78 do ji_out_var = 1, 100, 32
79 ji_el_inner = MIN(ji_out_var + (32 - 1), 100)
80 do ji = ji_out_var, ji_el_inner, 1
81 tmp(ji) = 2 * ji
82 enddo
83 enddo
84 end subroutine sub
85
86 '''
87 def __str__(self):
88 return "Split a loop into a chunked loop pair"

90 def validate(self, node, options=None):
91 '''
92 Validates that the given Loop node can have a ChunkLoopTrans applied.
93
94 :param node: the loop to validate.
95 :type node: :py:class:`psyclone.psyir.nodes.Loop`
96 :param options: a dict with options for transformation.
97 :type options: Optional[Dict[str, Any]]
98 :param int options["chunksize"]: The size to chunk over for this \
99 transformation. If not specified, the value 32 is used.
100
101 :raises TransformationError: if the supplied Loop has a step size \
102 which is not a constant value.
103 :raises TransformationError: if the supplied Loop has a non-integer \
104 step size.
105 :raises TransformationError: if the supplied Loop has a step size \
106 larger than the chosen chunk size.
107 :raises TransformationError: if the supplied Loop is a chunked loop.
108 :raises TransformationError: if the supplied Loop has a step size \
109 of 0.
110 :raises TransformationError: if the supplied Loop writes to Loop \
111 variables inside the Loop body.
112 :raises TransformationError: if the supplied Loop contains a \
113 CodeBlock node.
114 :raises TransformationError: if an unsupported option has been \
115 provided.
116 :raises TransformationError: if the provided tilesize is not a \
117 positive integer.
118 '''
119 if options is None:
120 options = {}
121 super().validate(node, options=options)
122
123 # Validate options map
124 # TODO #613: Hardcoding the valid_options does not allow for
125 # subclassing this transformation and adding new options, this
126 # should be fixed.
127 valid_options = ['chunksize']
128 for key, value in options.items():
129 if key in valid_options:
130 if key == "chunksize" and not isinstance(value, int):
131 raise TransformationError(
132 f"The ChunkLoopTrans chunksize option must be a "
133 f"positive integer but found a "
134 f"'{type(value).__name__}'.")
135 if key == "chunksize" and value <= 0:
136 raise TransformationError(
137 f"The ChunkLoopTrans chunksize option must be a "
138 f"positive integer but found '{value}'.")
139 else:
140 raise TransformationError(
141 f"The ChunkLoopTrans does not support the "
142 f"transformation option '{key}', the supported options "
143 f"are: {valid_options}.")
144
145 if not isinstance(node.step_expr, nodes.Literal):
146 # If step is a variable we don't support it.
147 raise TransformationError(
148 f"Cannot apply a ChunkLoopTrans to a loop with a non-literal "
149 f"step size, but a step expression node of type "
150 f"'{type(node).__name__}' was found.")
151 if node.step_expr.datatype.intrinsic is not \
152 ScalarType.Intrinsic.INTEGER:
153 raise TransformationError(
154 f"Cannot apply a ChunkLoopTrans to a loop with a non-integer "
155 f"step size, but a step expression of type "
156 f"'{node.step_expr.datatype.intrinsic.name}' was found.")
157 chunk_size = options.get("chunksize", 32)
158 if abs(int(node.step_expr.value)) > abs(chunk_size):
159 raise TransformationError(
160 f"Cannot apply a ChunkLoopTrans to a loop with larger step "
161 f"size ({node.step_expr.value}) than the chosen chunk size "
162 f"({chunk_size}).")
163 if 'chunked' in node.annotations:
164 raise TransformationError("Cannot apply a ChunkLoopTrans to "
165 "an already chunked loop.")
166
167 if int(node.step_expr.value) == 0:
168 raise TransformationError("Cannot apply a ChunkLoopTrans to "
169 "a loop with a step size of 0.")
170
171 if len(node.loop_body.walk(CodeBlock)) != 0:
172 raise TransformationError("Cannot apply a ChunkLoopTrans to "
173 "a loop which contains a CodeBlock "
174 "node.")
175 # Other checks needed for validation
176 # Dependency analysis, following rules:
177 # No child has a write dependency to the loop variable.
178 # Find variable access info for the loop variable and step
179 refs = VariablesAccessInfo(node.start_expr)
180 bounds_ref = VariablesAccessInfo()
181 if refs is not None:
182 bounds_ref.merge(refs)
183 refs = VariablesAccessInfo(node.stop_expr)
184 if refs is not None:
185 bounds_ref.merge(refs)
186 # The current implementation of ChunkLoopTrans does not allow
187 # the step size to be non-constant, so it is ignored.
188
189 # Add the access pattern to the node variable name
190 bounds_ref.add_access(Signature(node.variable.name),
191 AccessType.READWRITE, self)
192
193 bounds_sigs = bounds_ref.all_signatures
194
195 # Find the Loop code's signatures
196 body_refs = VariablesAccessInfo(node.loop_body)
197 body_sigs = body_refs.all_signatures
198
199 for ref1 in bounds_sigs:
200 if ref1 not in body_sigs:
201 continue
202 access2 = body_refs[ref1]
203
204 # If access2 is a write then we write to a loop variable
205 if access2.is_written():
206 raise TransformationError(
207 f"Cannot apply a ChunkLoopTrans to this loop because "
208 f"the boundary variable '{access2.signature.var_name}' "
209 f"is written to inside the loop body.")

211 def apply(self, node, options=None):
212 '''
213 Converts the given Loop node into a nested loop where the outer
214 loop is over chunks and the inner loop is over each individual element
215 of the chunk.
216
217 :param node: the loop to transform.
218 :type node: :py:class:`psyclone.psyir.nodes.Loop`
219 :param options: a dict with options for transformations.
220 :type options: Optional[Dict[str, Any]]
221 :param int options["chunksize"]: The size to chunk over for this \
222 transformation. If not specified, the value 32 is used.
223
224 '''
225
226 self.validatevalidatevalidate(node, options)
227 if options is None:
228 options = {}
229 chunk_size = options.get("chunksize", 32)
230 # Create (or find) the symbols we need for the chunking transformation
231 routine = node.ancestor(nodes.Routine)
232 end_inner_loop = routine.symbol_table.find_or_create_tag(
233 f"{node.variable.name}_el_inner",
234 symbol_type=DataSymbol,
235 datatype=node.variable.datatype)
236 outer_loop_variable = routine.symbol_table.find_or_create_tag(
237 f"{node.variable.name}_out_var",
238 symbol_type=DataSymbol,
239 datatype=node.variable.datatype)
240 # We currently don't allow ChunkLoops to be ancestors of ChunkLoop
241 # so our ancestors cannot use these variables.
242
243 # Store the node's parent for replacing later and the start and end
244 # indices
245 start = node.start_expr
246 stop = node.stop_expr
247
248 # For positive steps we do:
249 # el_inner = min(out_var+chunk_size-1, el_outer)
250 if int(node.step_expr.value) > 0:
251 add = BinaryOperation.create(
252 BinaryOperation.Operator.ADD,
253 Reference(outer_loop_variable),
254 BinaryOperation.create(
255 BinaryOperation.Operator.SUB,
256 Literal(f"{chunk_size}", node.variable.datatype),
257 Literal("1", node.variable.datatype)))
258 minop = IntrinsicCall.create(IntrinsicCall.Intrinsic.MIN,
259 [add, stop.copy()])
260 inner_loop_end = Assignment.create(Reference(end_inner_loop),
261 minop)
262 # For negative steps we do:
263 # el_inner = max(out_var-chunk_size+1, el_outer)
264 elif int(node.step_expr.value) < 0:
265 sub = BinaryOperation.create(
266 BinaryOperation.Operator.SUB,
267 Reference(outer_loop_variable),
268 BinaryOperation.create(
269 BinaryOperation.Operator.ADD,
270 Literal(f"{chunk_size}", node.variable.datatype),
271 Literal("1", node.variable.datatype)))
272 maxop = IntrinsicCall.create(IntrinsicCall.Intrinsic.MAX,
273 [sub, stop.copy()])
274 inner_loop_end = Assignment.create(Reference(end_inner_loop),
275 maxop)
276 # chunk_size needs to be negative if we're reducing
277 chunk_size = -chunk_size
278 # step size of 0 is caught by the validate call
279
280 # Replace the inner loop start and end with the chunking ones
281 start.replace_with(Reference(outer_loop_variable))
282 stop.replace_with(Reference(end_inner_loop))
283
284 # Create the outerloop as a bare Loop construct
285 outerloop = Loop(variable=outer_loop_variable)
286 outerloop.children = [start, stop,
287 Literal(f"{chunk_size}",
288 outer_loop_variable.datatype),
289 Schedule(parent=outerloop,
290 children=[inner_loop_end])]
291 # Add the chunked annotation
292 outerloop.annotations.append('chunked')
293 node.annotations.append('chunked')
294 # Replace this loop with the outerloop
295 node.replace_with(outerloop)
296 # Add the loop to the innerloop's schedule
297 outerloop.loop_body.addchild(node)
