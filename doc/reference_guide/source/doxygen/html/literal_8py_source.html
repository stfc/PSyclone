49class Literal(DataNode):
50 '''
51 Node representing a Literal. The value and datatype properties of
52 this node are immutable.
53
54 If the node represents "real" data and the value is expressed with
55 an exponent (e.g. 3.2e4 or 0.1E-3) then the stored value always uses
56 a lower case "e".
57
58 :param str value: the value of the literal.
59 :param datatype: the datatype of this literal.
60 :type datatype: :py:class:`psyclone.psyir.symbols.DataType`
61 :param parent: the parent node of this Literal in the PSyIR.
62 :type parent: :py:class:`psyclone.psyir.nodes.Node`
63
64 :raises TypeError: if the datatype is not an instance of \
65 :py:class:`psyclone.psyir.symbols.DataType`.
66 :raises ValueError: if the datatype is not one of self.VALID_DATA_TYPES.
67 :raises TypeError: if the supplied value is not a string.
68 :raises ValueError: if the supplied value is an empty string and the \
69 Literal is not a CHARACTER.
70 :raises ValueError: if the Literal is a BOOLEAN and the value is not \
71 'true' or 'false'.
72 :raises ValueError: if the Literal is a REAL but does not conform to \
73 the supported format defined by the `_real_value` property.
74 :raises ValueError: if the Literal is an INTEGER but does not conform \
75 to the supported format defined by the `_int_value` property.
76
77 '''
78 # Textual description of the node.
79 _children_valid_format = "<LeafNode>"
80 _text_name = "Literal"
81 _colour = "yellow"
82 _real_value = r'^[+-]?[0-9]+(\.[0-9]*)?([eE][+-]?[0-9]+)?$'
83 _int_value = r'(([+-]?[0-9]+)|(NOT_INITIALISED))'

85 def __init__(self, value, datatype, parent=None):
86 super().__init__(parent=parent)
87
88 # Checks for the datatype
89 if not isinstance(datatype, (ScalarType, ArrayType)):
90 raise TypeError(
91 f"The datatype of a Literal must be an instance of "
92 f"psyir.symbols.ScalarType or psyir.symbols.ArrayType "
93 f"but found '{type(datatype).__name__}'")
94
95 if not isinstance(value, str):
96 raise TypeError(
97 f"Literals must be supplied with a value encoded as a string "
98 f"but found '{type(value).__name__}'")
99
100 if not value and datatype.intrinsic != ScalarType.Intrinsic.CHARACTER:
101 raise ValueError("A non-character literal value cannot be empty.")
102
103 if (isinstance(datatype, ScalarType) and
104 datatype.intrinsic == ScalarType.Intrinsic.BOOLEAN and
105 value not in ("true", "false")):
106 raise ValueError(
107 f"A scalar boolean literal can only be: 'true' or "
108 f"'false' but found '{value}'.")
109
110 if datatype.intrinsic == ScalarType.Intrinsic.REAL:
111 if not re.match(Literal._real_value, value):
112 raise ValueError(
113 f"A scalar real literal value must conform to the "
114 f"supported format ('{Literal._real_value}') but found "
115 f"'{value}'.")
116 # Ensure we always store any exponent with a lowercase 'e'
117 self._value = value.replace("E", "e", 1)
118 elif isinstance(datatype, ScalarType) and datatype.intrinsic \
119 == ScalarType.Intrinsic.INTEGER:
120 if not re.fullmatch(Literal._int_value, value):
121 raise ValueError(
122 f"A scalar integer literal value must conform to the "
123 f"supported format ('{Literal._int_value}') but found "
124 f"'{value}'.")
125 self._value = value
126 else:
127 self._value = value
128 self._datatype = datatype

131 '''Checks the equality of this Literal with other. Literals are
132 equal if they are the same type, and have the same datatype and
133 value string (for now only compared with ==).
134
135 :param object other: the object to check equality to.
136
137 :returns: whether other is equal to self.
138 :rtype: bool
139 '''
140 is_eq = super().__eq__(other)
141 is_eq = is_eq and self.datatypedatatypedatatype == other.datatype
142 is_eq = is_eq and self.value == other.value
143 return is_eq
