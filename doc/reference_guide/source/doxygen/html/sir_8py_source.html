48# Mapping from PSyIR data types to SIR types.
49# SIR does not seem to support the Character datatype. Boolean does
50# seem to be supported but there are no examples with the data value
51# so we don't include it here.
52# We do not yet deal with precision e.g. the SIR supports a DOUBLE
53# type which would probably be equivalent to PSyIR's
54# Precision.DOUBLE. This is the subject of issue #741.
55
56TYPE_MAP_TO_SIR = {ScalarType.Intrinsic.REAL: "BuiltinType.Float",
57 ScalarType.Intrinsic.INTEGER: "BuiltinType.Integer"}

60def gen_stencil(node):
61 '''Given an array access as input, determine the form of stencil
62 access and return it in the form expected by the SIR as a
63 string. Raise an exception if the array access is not a recognised
64 stencil access.
65
66 :param node: an array access.
67 :type node: :py:class:`psyclone.psyir.nodes.ArrayReference`
68
69 :returns: the SIR stencil access format for the array access.
70 :rtype: str
71
72 :raises VisitorError: if the node is not the expected type or the \
73 array access is not in a recognised stencil form.
74
75 '''
76 if not isinstance(node, ArrayReference):
77 raise VisitorError(
78 f"gen_stencil expected an ArrayReference as input but found "
79 f"'{type(node)}'.")
80 dims = []
81 for child in node.children:
82 if isinstance(child, Reference):
83 dims.append("0")
84 elif isinstance(child, BinaryOperation):
85 if isinstance(child.children[0], Reference) and \
86 isinstance(child.children[1], Literal):
87 if child.operator == BinaryOperation.Operator.SUB:
88 dims.append("-"+child.children[1].value)
89 elif child.operator == BinaryOperation.Operator.ADD:
90 dims.append(child.children[1].value)
91 else:
92 raise VisitorError(
93 f"gen_stencil unsupported stencil operator found "
94 f"'{child.operator.name}'. Expecting '+' or '-'.")
95 else:
96 raise VisitorError(
97 f"gen_stencil unsupported stencil index found '{child}'.")
98 else:
99 raise VisitorError(
100 f"gen_stencil unsupported (non-stencil) index found '{child}'.")
101 return f"[{', '.join(dims)}]"

104class SIRWriter(PSyIRVisitor):
105 '''Implements a PSyIR-to-SIR back end for PSyIR kernel code (not
106 currently PSyIR PSy-layer code which has its own gen method for
107 generating Fortran).
108
109 :param bool skip_nodes: if skip_nodes is False then an exception \
110 is raised if a visitor method for a PSyIR node has not been \
111 implemented, otherwise the visitor continues, printing out a \
112 representation of the unsupported node. This is an optional \
113 argument which defaults to False.
114 :param str indent_string: specifies what to use for indentation. This \
115 is an optional argument that defaults to two spaces.
116 :param int initial_indent_depth: Specifies how much indentation to \
117 start with. This is an optional argument that defaults to 0.
118
119 '''
120 def __init__(self, skip_nodes=False, indent_string=" ",
121 initial_indent_depth=0):
122 super().__init__(skip_nodes, indent_string, initial_indent_depth)
123 # The _field_names variable stores the unique field names
124 # found in the PSyIR. This is required as the SIR declares
125 # field names after the computation.
126 self._field_names = set()
127 # The _scalar_names variable stores the unique scalar names
128 # found in the PSyIR. The current assumption is that scalars
129 # are temporaries. This is not necessarily correct and this
130 # problem is captured in issue #521. Scalar temporaries can be
131 # declared as field temporaries as the Dawn backend works out
132 # what is required.
133 self._scalar_names_scalar_names = set()
134
135 def node_node(self, node):
136 '''Catch any unsupported nodes, output their class names and continue
137 down the node hierarchy (if skip_node is set to True). This is
138 useful for debugging and differs from the base class
139 implementation of skip_nodes which silently continues. If
140 skip_nodes is set to False then raise an exception if an
141 unsupported node is found.
142
143 :param node: an unsupported PSyIR node.
144 :type node: subclass of :py:class:`psyclone.psyir.nodes.Node`
145
146 :returns: the SIR Python code.
147 :rtype: str
148
149 :raises VisitorError: if skip_nodes is set to False.
150
151 '''
152 if not self._skip_nodes_skip_nodes_skip_nodes:
153 raise VisitorError(
154 f"Class SIRWriter method node_node(), unsupported node "
155 f"found '{type(node)}'")
156 result = f"{self._nindent}[ {type(node).__name__} start ]\n"
157 self._depth += 1
158 for child in node.children:
159 result += self._visit(child)
160 self._depth -= 1
161 result += f"{self._nindent}[ {type(node).__name__} end ]\n"
162 return result

164 def loop_node(self, loop_node):
165 '''Supported loops are triply nested with particular indices (not
166 yet checked) and should contain only computation. If this is not the
167 case then it is not possible to translate so an exception is
168 raised.
169
170 :param loop_node: a Loop PSyIR node.
171 :type loop_node: subclass of :py:class:`psyclone.psyir.nodes.Loop`
172
173 :returns: the SIR Python code.
174 :rtype: str
175
176 :raises VisitorError: if the loop is not triply nested with computation
177 within the triply nested loop.
178
179 '''
180 loops = loop_node.walk(Loop)
181 if len(loops) != 3:
182 raise VisitorError("Only triply-nested loops are supported.")
183
184 # Check first loop has a single loop as a child.
185 loop_content = loops[0].loop_body.children
186 if not (len(loop_content) == 1 and
187 isinstance(loop_content[0], Loop)):
188 raise VisitorError("Child of loop should be a single loop.")
189
190 # Check second loop has a single loop as a child.
191 loop2_content = loops[1].loop_body.children
192 if not (len(loop2_content) == 1 and
193 isinstance(loop2_content[0], Loop)):
194 raise VisitorError(
195 "Child of child of loop should be a single loop.")
196
197 # Check that the innermost loop does not contain any CodeBlocks.
198 loop3 = loops[2]
199 if loop3.walk(CodeBlock):
200 raise VisitorError(
201 f"A loop nest containing a CodeBlock cannot be translated to "
202 f"SIR:\n"
203 f"{loop3.debug_string()}")
204
205 # The interval values are hardcoded for the moment (see #470).
206 result = f"{self._nindent}interval = "\
207 f"make_interval(Interval.Start, Interval.End, 0, 0)\n"
208 result += f"{self._nindent}body_ast = make_ast([\n"
209 self._depth += 1
210 for child in loop3.loop_body:
211 result += self._visit(child)
212 self._depth -= 1
213 # Remove the trailing comma if there is one as this is the
214 # last entry in make_ast.
215 result = result.rstrip(",\n") + "\n"
216 result += f"{self._nindent}])\n"
217 # For the moment there is a hard coded assumption that the
218 # vertical looping is in the forward (1..n) direction (see
219 # #470).
220 result += f"{self._nindent}vertical_region_fns.append("\
221 f"make_vertical_region_decl_stmt(body_ast, interval, "\
222 f"VerticalRegion.Forward))\n"
223 return result

225 def routine_node(self, node):
226 '''This method is called when a Routine instance is found
227 in the PSyIR tree.
228
229 :param node: a KernelSchedule PSyIR node.
230 :type node: :py:class:`psyclone.psyir.nodes.KernelSchedule`
231
232 :returns: the SIR Python code.
233 :rtype: str
234
235 '''
236 result = "# PSyclone autogenerated SIR Python\n"
237 result += "vertical_region_fns = []\n"
238 # The stencil name is currently hardcoded.
239 result += "stencil_name = \"psyclone\"\n"
240
241 exec_statements = ""
242 for child in node.children:
243 exec_statements += self._visit(child)
244 result += f"{exec_statements}\n"
245 # The file name is hard coded at the moment.
246 result += (self._nindent + "hir = make_sir(stencil_name+\".cpp\", "
247 "AST.GridType.Value(\"Cartesian\"), [\n"
248 + self._nindent + self._indent + "make_stencil(\n"
249 + self._nindent + self._indent * 2 + "stencil_name,\n"
250 + self._nindent + self._indent * 2
251 + "make_ast(vertical_region_fns),\n"
252 + self._nindent + self._indent * 2 + "[")
253 functions = []
254 for name in self._field_names:
255 functions.append(
256 f"make_field(\"{name}\", make_field_dimensions_cartesian())")
257 # The current assumption is that scalars are temporaries. This
258 # is not necessarily correct and this problem is captured in
259 # issue #521. Scalar temporaries can be declared as field
260 # temporaries as the Dawn backend works out what is required.
261 for name in self._scalar_names_scalar_names:
262 functions.append(
263 f"make_field(\"{name}\", make_field_dimensions_cartesian(), "
264 f"is_temporary=True)")
265 result += ", ".join(functions)
266 result += "]\n"
267 result += f"{self._nindent}{self._indent})\n{self._nindent}])\n"
268 return result

292 def binaryoperation_node(self, node):
293 '''This method is called when a BinaryOperation instance is found in
294 the PSyIR tree.
295
296 :param node: a BinaryOperation PSyIR node.
297 :type node: :py:class:`psyclone.psyir.nodes.BinaryOperation`
298
299 :returns: the SIR Python code.
300 :rtype: str
301
302 :raises VisitorError: if there is no mapping from the PSyIR \
303 operator to SIR.
304
305 '''
306 # Specify the mapping of PSyIR binary operators to SIR binary
307 # operators.
308 binary_operators = {
309 BinaryOperation.Operator.ADD: '+',
310 BinaryOperation.Operator.SUB: '-',
311 BinaryOperation.Operator.MUL: '*',
312 BinaryOperation.Operator.DIV: '/',
313 BinaryOperation.Operator.POW: '**',
314 BinaryOperation.Operator.EQ: '==',
315 BinaryOperation.Operator.NE: '!=',
316 BinaryOperation.Operator.LE: '<=',
317 BinaryOperation.Operator.LT: '<',
318 BinaryOperation.Operator.GE: '>=',
319 BinaryOperation.Operator.GT: '>',
320 BinaryOperation.Operator.AND: '&&',
321 BinaryOperation.Operator.OR: '||'}
322
323 self._depth += 1
324 lhs = self._visit(node.children[0])
325 try:
326 oper = binary_operators[node.operator]
327 except KeyError as err:
328 raise VisitorError(
329 f"Method binaryoperation_node in class SIRWriter, unsupported "
330 f"operator '{node.operator}' found.") from err
331 rhs = self._visit(node.children[1])
332 self._depth -= 1
333
334 result = f"{self._nindent}make_binary_operator(\n{lhs}"
335 # For better formatting, remove the newline if one exists.
336 result = result.rstrip("\n") + ",\n"
337 result += f"{self._nindent}{self._indent}\"{oper}\",\n{rhs}\n"\
338 f"{self._nindent}{self._indent})\n"
339 return result

341 def reference_node(self, node):
342 '''This method is called when a Reference instance is found in the
343 PSyIR tree.
344
345 :param node: a Reference PSyIR node.
346 :type node: :py:class:`psyclone.psyir.nodes.Reference`
347
348 :returns: the SIR Python code.
349 :rtype: str
350
351 :raises VisitorError: if this node has children.
352
353 '''
354 if node.children:
355 raise VisitorError(
356 "Method reference_node in class SIRWriter: SIR Reference "
357 "node is not expected to have any children.")
358 # _scalar_names is a set so duplicates will be ignored. It
359 # captures all unique scalar names as scalars are currently
360 # treated as temporaries (#521 captures this). The simplest
361 # way to declare a scalar temporary in Dawn is to treat it as
362 # a field temporary (as the Dawn backend works out if a scalar
363 # is required).
364 self._scalar_names_scalar_names.add(node.name)
365
366 return f"{self._nindent}make_field_access_expr(\"{node.name}\")"

410 def unaryoperation_node(self, node):
411 '''This method is called when a UnaryOperation instance is found in
412 the PSyIR tree.
413
414 :param node: a UnaryOperation PSyIR node.
415 :type node: :py:class:`psyclone.psyir.nodes.UnaryOperation`
416
417 :returns: the SIR Python code.
418 :rtype: str
419
420 :raises VisitorError: if there is no mapping from the PSyIR \
421 operator to SIR, or if the child of the PSyIR operator is not \
422 a literal (as only -<literal> is currently supported).
423
424 '''
425 # Currently only '-' and intrinsics are supported in the SIR mapping.
426 unary_operators = {
427 UnaryOperation.Operator.MINUS: '-',
428 }
429 try:
430 oper = unary_operators[node.operator]
431 except KeyError as err:
432 raise VisitorError(
433 f"Method unaryoperation_node in class SIRWriter, unsupported "
434 f"operator '{node.operator}' found.") from err
435
436 if isinstance(node.children[0], Literal):
437 # The unary minus operator is being applied to a
438 # literal. This is a special case as the literal value can
439 # be negative in SIR.
440 literal = node.children[0]
441 if literal.datatype.intrinsic not in [
442 ScalarType.Intrinsic.REAL, ScalarType.Intrinsic.INTEGER]:
443 # The '-' operator can only be applied to REAL and INTEGER
444 # datatypes.
445 raise VisitorError(
446 f"PSyIR type '{literal.datatype}' does not work with the "
447 f"'-' operator.")
448 result = literal.value
449 datatype = TYPE_MAP_TO_SIR[literal.datatype.intrinsic]
450 return f"{self._nindent}make_literal_access_expr(\"{oper}{result}"\
451 f"\", {datatype})"
452
453 # The unary minus operator is being applied to something that
454 # is not a literal. Default to REAL as we currently have no
455 # way of finding out the type, see issue #658. Replace -x with
456 # -1.0 * x.
457 datatype = TYPE_MAP_TO_SIR[ScalarType.Intrinsic.REAL]
458 self._depth += 1
459 lhs = f"{self._nindent}make_literal_access_expr(\"-1.0\", {datatype})"
460 operator = f"{self._nindent}\"*\""
461 rhs = self._visit(node.children[0])
462 self._depth -= 1
463 result = f"{self._nindent}make_binary_operator"\
464 f"(\n{lhs},\n{operator},\n{rhs})\n"
465 return result

467 def ifblock_node(self, node):
468 '''This method is called when an IfBlock instance is found in
469 the PSyIR tree.
470
471 :param node: an IfBlock PSyIR node.
472 :type node: :py:class:`psyclone.psyir.nodes.IfBlock`
473
474 :returns: SIR Python code.
475 :rtype: str
476
477 '''
478 cond_expr = self._visit(node.condition)
479 cond_part = "make_expr_stmt(" + cond_expr.lstrip().rstrip(",\n") + ")"
480
481 then_statements = self._visit(node.if_body).lstrip().rstrip(",\n")
482 then_part = "make_block_stmt([" + then_statements + "])"
483
484 if node.else_body:
485 else_statements = self._visit(node.else_body)
486 else_part = "make_block_stmt([" + \
487 else_statements.lstrip().rstrip(",\n") + "])"
488 else:
489 else_part = "None"
490
491 return f"{self._nindent}make_if_stmt({cond_part}, {then_part}, "\
492 f"{else_part}),\n"

494 def schedule_node(self, node):
495 '''This method is called when a Schedule instance is found in the
496 PSyIR tree. A Schedule instance captures an ordered sequence
497 of PSyIR nodes and is therefore found in places such as the
498 contents of the 'then' part of an 'if' statement and the
499 contents of the 'else' part of an 'if' statement. The schedule
500 has no content so simply calls its children and returns the
501 agregated result.
502
503 :param node: a Schedule PSyIR node.
504 :type node: :py:class:`psyclone.psyir.nodes.Schedule`
505
506 :returns: the SIR Python code.
507 :rtype: str
508
509 '''
510 result = ""
511 for child in node.children:
512 result += self._visit(child)
513 return result

515 def intrinsiccall_node(self, node):
516 '''This method is called when an Intrinsic node is found in
517 the PSyIR tree.
518
519 :param node: an IntrinsicCall PSyIR node.
520 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
521
522 :returns: the SIR Python code.
523 :rtype: str
524
525 :raises VisitorError: if there is no mapping from the PSyIR \
526 operator to SIR.
527
528 '''
529 # Specify the mapping of PSyIR nary operators to SIR nary
530 # operators. The assumption here is that the nary operators
531 # are SIR intrinsics.
532 supported_intrinsics = {
533 IntrinsicCall.Intrinsic.ABS: 'math::fabs',
534 IntrinsicCall.Intrinsic.MIN: 'math::min',
535 IntrinsicCall.Intrinsic.MAX: 'math::max',
536 IntrinsicCall.Intrinsic.SIGN: 'math::sign'}
537
538 try:
539 intrinsic = supported_intrinsics[node.intrinsic]
540 except KeyError as err:
541 intr_names = [intr.name for intr in supported_intrinsics]
542 raise VisitorError(
543 f"Method intrinsiccall_node in class SIRWriter, unsupported "
544 f"intrinsic '{node.intrinsic.name}' found. Expected one of "
545 f"'{intr_names}'.") from err
546
547 self._depth += 1
548
549 if node.intrinsic is IntrinsicCall.Intrinsic.SIGN:
550 # This is a special case as the implementation of SIGN
551 # in the PSyIR (which uses the Fortran implementation)
552 # is different to that in SIR (which uses the C
553 # implementation).
554 # [F] SIGN(A,B) == [C] FABS(A)*SIGN(B)
555 c_abs_fun = (f"make_fun_call_expr(\"math::fabs\", "
556 f"[{self._visit(node.arguments[0]).strip()}])")
557 c_sign_fun = (f"make_fun_call_expr(\"math::sign\", "
558 f"[{self._visit(node.arguments[1]).strip()}])")
559 result = (f"make_binary_operator({c_abs_fun}, "
560 f"\"*\", {c_sign_fun})")
561 else:
562 # Everything else it is a SIR intrinsics
563 arg_list = []
564 for child in node.arguments:
565 arg_list.append(f"[{self._visit(child).strip()}]")
566 arg_str = ", ".join(arg_list)
567 result = (f"{self._nindent}{self._indent}make_fun_call_expr("
568 f"\"{intrinsic}\", {arg_str})")
569
570 self._depth -= 1
571 return result
