58class ModuleInfoError(PSycloneError):
59 """
60 PSyclone-specific exception for use when an error with the module manager
61 happens - typically indicating that some module information cannot be
62 found.
63
64 :param str value: the message associated with the error.
65
66 """
67 def __init__(self, value):
68 PSycloneError.__init__(self, value)
69 self.valuevalue = "ModuleInfoError: " + str(value)

72class ModuleInfo:
73 # pylint: disable=too-many-instance-attributes
74 '''This class stores mostly memory-cached information about a Fortran
75 module.
76 It stores a FileInfo object holding details on the original source file,
77 the fparser2 parse tree and the PSyIR.
78 Memory-caching is always used in FileInfo with an opt-in feature of
79 caching the fparser parse tree to the file system.
80
81 :param module_name: Name of the module
82 :param file_info: FileInfo object. The fparser tree, PSyIR
83 and module information will be loaded from this file if it's
84 not yet available.
85
86 '''
87 def __init__(
88 self,
89 module_name: str,
90 file_info: FileInfo
91 ):
92 if not isinstance(module_name, str):
93 raise TypeError("Expected type 'str' for argument 'module_name'")
94
95 if not isinstance(file_info, FileInfo):
96 raise TypeError("Expected type 'FileInfo' for"
97 " argument 'file_info'")
98
99 self._name = module_name.lower()
100
101 # File handler including fparser and psyir representation
102 self._file_info: FileInfo = file_info
103
104 # The PSyIR representation
105 self._psyir_container_node_psyir_container_node: Container = None
106
107 # A cache for the module dependencies: this is just a set
108 # of all modules USEd by this module.
109 # We use an (ordered) list to preserve the order of the
110 # used modules
111 self._used_module_names: List[str] = None
112
113 # This is a dictionary containing the sets of symbols imported from
114 # each module, indexed by the module names: dict[str, set[str]].
115 self._map_module_name_to_used_symbols_map_module_name_to_used_symbols: Dict[str, set[str]] = None
116
117 # ------------------------------------------------------------------------
118 @property
119 def name(self) -> str:
120 ''':returns: the name of this module.
121
122 '''
123 return self._name
124
125 # ------------------------------------------------------------------------
126 @property
127 def filename(self) -> str:
128 '''
129
130 :returns: the filepath that contains the source code for this
131 module.
132
133 '''
134 return self._file_info.filename

136 # ------------------------------------------------------------------------
137 def get_source_code(self):
138 '''Returns the source code for the module using the associated
139 FileInfo instance (which caches it).
140
141 :returns: the source code.
142 :rtype: str
143
144 :raises ModuleInfoError: when the file cannot be read.
145
146 '''
147 try:
148 return self._file_info.get_source_code()
149 except FileNotFoundError as err:
150 raise ModuleInfoError(
151 f"Could not find file '{self._file_info.filename}' when trying"
152 f" to read source code for module '{self._name}'") from err
153
154 # ------------------------------------------------------------------------
155 def get_fparser_tree(self) -> Union[Program, None]:
156 '''Returns the fparser AST for this module.
157
158 :returns: The fparser AST for this module.
159
160 :raises ModuleInfoError: When errors have been triggered while
161 creating the fparser tree.
162 '''
163 try:
164 return self._file_info.get_fparser_tree()
165
166 except FileInfoFParserError as err:
167 raise ModuleInfoError(
168 f"Error(s) getting fparser tree of file '{self.filename}'"
169 f" for module '{self.name}':\n"
170 + str(err)) from err

173 def _extract_import_information(self):
174 '''This internal function analyses a given module source file and
175 caches which modules are imported (in self._used_modules), and which
176 symbol is imported from each of these modules (in
177 self._used_symbols_from_module_name).
178
179 '''
180
181 # Initialise the caches
182 self._used_module_names = []
183 self._map_module_name_to_used_symbols_map_module_name_to_used_symbols = {}
184
185 parse_tree = self.get_fparser_tree()
186
187 for use in walk(parse_tree, Fortran2003.Use_Stmt):
188 # Ignore intrinsic modules:
189 if str(use.items[0]) == "INTRINSIC":
190 continue
191
192 mod_name = str(use.items[2])
193 if mod_name not in self._used_module_names:
194 self._used_module_names.append(mod_name)
195 all_symbols = set()
196
197 only_list = use.items[4]
198 # If there is no only_list, then the set of symbols
199 # will stay empty
200 if only_list:
201 # Parse the only list:
202 for symbol in only_list.children:
203 all_symbols.add(str(symbol))
204
205 self._map_module_name_to_used_symbols_map_module_name_to_used_symbols[mod_name] = all_symbols
206
207 # ------------------------------------------------------------------------
208 def get_used_modules(self) -> List[str]:
209 '''This function returns a set of all modules `used` in this
210 module. Fortran `intrinsic` modules will be ignored. The information
211 is based on the fparser parse tree of the module (since fparser can
212 handle more files than PSyir, like LFRic's `constants_mod` which has
213 pre-processor directives).
214
215 :returns: a set with all imported module names.
216 :rtype: set[str]
217
218 '''
219 if self._used_module_names is None:
220 self._extract_import_information()
221
222 return self._used_module_names
223
224 # ------------------------------------------------------------------------
225 def get_used_symbols_from_modules(self):
226 '''This function returns information about which modules are used by
227 this module, and also which symbols are imported. The return value is
228 a dictionary with the used module name as key, and a set of all
229 imported symbol names as value.
230
231 :returns: a dictionary that gives for each module name the set \
232 of symbols imported from it.
233 :rtype: dict[str, set[str]]
234
235 '''
236 if self._map_module_name_to_used_symbols_map_module_name_to_used_symbols is None:
237 self._extract_import_information()
238
239 return self._map_module_name_to_used_symbols_map_module_name_to_used_symbols

241 def get_psyir(self):
242 '''Returns the PSyIR representation of this module. This is based
243 on the fparser tree (see get_parse_tree), and the information is
244 cached. If the PSyIR must be modified, it needs to be copied,
245 otherwise the modified tree will be returned from the cache in the
246 future.
247
248 If the conversion to PSyIR fails then None is returned.
249
250 :returns: PSyIR representing this module.
251 :rtype: :py:class:`psyclone.psyir.nodes.Container` | NoneType
252
253 :raises InternalError: if the named Container (module) does not
254 exist in the PSyIR.
255
256 '''
257 if self._psyir_container_node_psyir_container_node is None:
258 try:
259 self._file_info.get_fparser_tree()
260 except FileInfoFParserError as err:
261 # TODO #11: Add proper logging
262 print(f"Error parsing '{self.filename}': '{err}'")
263 return None
264
265 try:
266 self._psyir_container_node_psyir_container_node = self._file_info.get_psyir()
267 except (
268 GenerationError,
269 FileInfoFParserError) as err:
270 # TODO #11: Add proper logging
271 print(f"Error trying to create PSyIR for '{self.filename}': "
272 f"'{err}'")
273 return None
274
275 # Return the Container with the correct name.
276 for container_node in self._psyir_container_node_psyir_container_node.walk(Container):
277 if container_node.name.lower() == self.name:
278 return container_node
279
280 # We failed to find the Container - double-check the parse tree
281 for mod_stmt in walk(self.get_fparser_tree(), Fortran2003.Module_Stmt):
282 if mod_stmt.children[1].string.lower() == self.name:
283 # The module exists but we couldn't create PSyIR for it.
284 # TODO #11: Add proper logging
285 print(f"File '{self.filename}' does contain module "
286 f"'{self.name}' but PSyclone is unable to create the "
287 f"PSyIR of it.")
288 return None
289
290 raise InternalError(f"File '{self.filename}' does not contain a "
291 f"module named '{self.name}'")

294 def get_symbol(self, name: str) -> Union[Symbol, None]:
295 '''
296 Gets the PSyIR Symbol with the supplied name from the Container
297 representing this Fortran module (if available).
298
299 This utility mainly exists to insulate the user from having to check
300 that a Container has been successfully created for this module
301 (it might not have been if the source of the module cannot be found
302 or cannot be parsed) and that it contains the specified Symbol.
303
304 :param name: the name of the symbol to get from this module.
305
306 :returns: the Symbol with the supplied name if the Container has
307 been successfully created and contains such a symbol and None
308 otherwise.
309
310 '''
311 try:
312 container: Container = self.get_psyir()
313
314 # Handle "alternative" implementation of get_psyir_container_mode"
315 if container is None:
316 # PSyIR could not be obtained so cannot search for Symbol
317 return None
318
319 except (FileNotFoundError, FileInfoFParserError, GenerationError):
320 return None
321
322 try:
323 return container.symbol_table.lookup(name)
324 except KeyError:
325 return None
