56class ParallelLoopTrans(LoopTrans, metaclass=abc.ABCMeta):
57 '''
58 Adds an abstract directive (it needs to be specified by sub-classing this
59 transformation) to a loop indicating that it should be parallelised. It
60 performs some data dependency checks to guarantee that the loop can be
61 parallelised without changing the semantics of it.
62
63 '''
64 # The types of node that must be excluded from the section of PSyIR
65 # being transformed.
66 excluded_node_types = (nodes.Return, psyGen.HaloExchange, nodes.CodeBlock)
67
68 @abc.abstractmethod
69 def _directive(self, children, collapse=None):
70 '''
71 Returns the directive object to insert into the Schedule.
72 Must be implemented by sub-class.
73
74 :param children: list of nodes that will be children of this Directive.
75 :type children: list of :py:class:`psyclone.psyir.nodes.Node`
76 :param int collapse: the number of tightly-nested loops to which
77 this directive applies or None.
78
79 :returns: the new Directive node.
80 :rtype: sub-class of :py:class:`psyclone.psyir.nodes.Directive`.
81 '''
82
83 @staticmethod
84 def _attempt_privatisation(node, symbol_name, dry_run=False):
85 ''' Check and (if dry_run is False) perform symbol privatisation
86 for the given symbol_name in the given node.
87
88 :param node: the loop that will be parallelised.
89 :type node: :py:class:`psyclone.psyir.nodes.Loop`
90 :param str symbol_name: the symbol that we want to privatise.
91 :param bool dry_run: whether to perform the actual privatisation.
92
93 :returns: whether the symbol_name can be privatised.
94 :rtype: bool
95 '''
96 try:
97 sym = node.scope.symbol_table.lookup(symbol_name)
98 except KeyError:
99 # Structures are reported with the full expression:
100 # "mystruct%myfield" by the DA var_name, we purposely avoid
101 # privatising these
102 return False
103
104 # If it's not a local symbol, we cannot safely analyse its lifetime
105 if not isinstance(sym.interface, AutomaticInterface):
106 return False
107
108 # Check that the symbol is not referenced after this loop (before
109 # the loop is fine because we can use OpenMP/OpenACC first-private or
110 # Fortran do concurrent local_init())
111 if any(ref.symbol is sym
112 for ref in node.following(include_children=False)
113 if isinstance(ref, Reference)):
114 return False
115
116 if not dry_run:
117 node.explicitly_private_symbols.add(sym)
118
119 return True

121 def validate(self, node, options=None):
122 '''
123 Perform validation checks before applying the transformation
124
125 :param node: the node we are checking.
126 :type node: :py:class:`psyclone.psyir.nodes.Node`
127 :param options: a dictionary with options for transformations.
128 :type options: Optional[Dict[str, Any]]
129 :param bool|int options["collapse"]: if it's a bool and is False
130 (default), it won't collapse. If it's a bool and is True, it will
131 collapse as much as possible. If it's an integer, it will attempt
132 to collapse until the specified number of loops (if they exist and
133 are safe to collapse). The options 'ignore_dependencies_for'
134 and 'force' also affect the collapse applicability.
135 :param bool options["force"]: whether to force parallelisation of the
136 target loop (i.e. ignore any dependence analysis).
137 :param list[str] options["ignore_dependencies_for"]: whether to ignore
138 some symbol names from the dependence analysis checks.
139 :param bool options["sequential"]: whether this is a sequential loop.
140 :param bool options["verbose"]: whether to report the reasons the
141 validate and collapse steps have failed.
142
143 :raises TypeError: if 'collapse' is not an int or a bool.
144 :raises TypeError: if 'ignore_dependencies_for' is not a list of str.
145 :raises TransformationError: if the given loop iterates over a
146 colours (LFRic domain) iteration space.
147 :raises TransformationError: if the given loops calls a procedure that
148 is not guaranteed to be pure (and therefore could have dependencies
149 beyond the specified by the arguments intent)
150 :raises TransformationError: if the given loop is inside a pure routine
151 as these do not allow parallel constructs.
152 :raises TransformationError: if there is a data dependency that
153 prevents the parallelisation of the loop and the provided
154 options don't disregard them.
155
156 '''
157 # Check that the supplied node is a Loop and does not contain any
158 # unsupported nodes.
159 super().validate(node, options=options)
160
161 if not options:
162 options = {}
163 verbose = options.get("verbose", False)
164 collapse = options.get("collapse", False)
165 force = options.get("force", False)
166 ignore_dependencies_for = options.get("ignore_dependencies_for", [])
167 sequential = options.get("sequential", False)
168 privatise_arrays = options.get("privatise_arrays", False)
169
170 # Check we are not a sequential loop
171 if (not sequential and isinstance(node, PSyLoop) and
172 node.loop_type == 'colours'):
173 raise TransformationError(f"Error in {self.name} transformation. "
174 f"The target loop is over colours and "
175 f"must be computed serially.")
176
177 # If 'collapse' is specified, check that it is an int and that the
178 # loop nest has at least that number of loops in it
179 if collapse:
180 if not isinstance(collapse, (int, bool)):
181 raise TypeError(
182 f"The 'collapse' argument must be an integer or a bool but"
183 f" got an object of type {type(collapse)}")
184
185 if not isinstance(privatise_arrays, bool):
186 raise TypeError(
187 f"The 'privatise_arrays' option must be a bool "
188 f"but got an object of type {type(privatise_arrays).__name__}")
189
190 routine = node.ancestor(Routine)
191 if routine is not None and routine.parent is not None:
192 rsym = routine.symbol
193 if rsym.is_pure or rsym.is_elemental:
194 raise TransformationError(
195 f"Loops inside a pure (or elemental) routine cannot be"
196 f" parallelised, but attempted to parallelise loop "
197 f"inside '{routine.name}'"
198 )
199
200 # If it's sequential or we 'force' the transformation, the validations
201 # below this point are skipped
202 if sequential or force:
203 return
204
205 # Check that all calls inside the loop are pure, and therefore all its
206 # dependencies given by the intent of its arguments
207 not_pure = [call.routine.name for call in node.walk(Call)
208 if not call.is_pure]
209 if not_pure:
210 message = (
211 f"Loop cannot be parallelised because it cannot "
212 f"guarantee that the following calls are pure: "
213 f"{sorted(set(not_pure))}")
214 if verbose:
215 node.append_preceding_comment(message)
216 raise TransformationError(message)
217
218 if ignore_dependencies_for:
219 if (not isinstance(ignore_dependencies_for, Iterable) or
220 isinstance(ignore_dependencies_for, str) or not all(
221 isinstance(v, str) for v in ignore_dependencies_for)):
222 raise TypeError(
223 f"The 'ignore_dependencies_for' option must be an Iterable"
224 f" object containing str representing the "
225 f"symbols to ignore, but got '{ignore_dependencies_for}'.")
226
227 dep_tools = DependencyTools()
228
229 signatures = [Signature(name) for name in ignore_dependencies_for]
230
231 if not node.independent_iterations(
232 dep_tools=dep_tools,
233 test_all_variables=True,
234 signatures_to_ignore=signatures):
235 # The DependencyTools also returns False for things that are
236 # not an issue, so we ignore specific messages.
237 errors = []
238 for message in dep_tools.get_all_messages():
239 if message.code == DTCode.WARN_SCALAR_WRITTEN_ONCE:
240 continue
241 if (privatise_arrays and
242 message.code == DTCode.ERROR_WRITE_WRITE_RACE):
243 for var_name in message.var_names:
244 if not self._attempt_privatisation(node, var_name,
245 dry_run=True):
246 errors.append(
247 f"The write-write dependency in '{var_name}'"
248 f" cannot be solved by array privatisation "
249 f"because it is not a plain local array or "
250 f"it is used after the loop.")
251 continue
252 errors.append(str(message))
253
254 if errors:
255 error_lines = "\n".join(errors)
256 messages = (f"Loop cannot be parallelised because:\n"
257 f"{error_lines}\n"
258 f"Consider using the \"ignore_dependencies_for\""
259 f" transformation option if this is a false "
260 f"dependency\nConsider using the \"array_"
261 f"privatisation\" transformation option if "
262 f"this is a write-write dependency")
263 if verbose:
264 # This message can get quite long, we will skip it if an
265 # ancestor loop already has the exact same message
266 cursor = node.ancestor(Loop)
267 while cursor:
268 if messages in cursor.preceding_comment:
269 break
270 cursor = cursor.ancestor(Loop)
271 if not cursor:
272 node.append_preceding_comment(f"PSyclone: {messages}")
273 raise TransformationError(messages)

275 def apply(self, node, options=None):
276 '''
277 Apply the Loop transformation to the specified node in a
278 Schedule. This node must be a Loop since this transformation
279 corresponds to wrapping the generated code with directives,
280 e.g. for OpenMP:
281
282 .. code-block:: fortran
283
284 !$OMP DO
285 do ...
286 ...
287 end do
288 !$OMP END DO
289
290 At code-generation time (when gen_code()` is called), this node must be
291 within (i.e. a child of) a PARALLEL region.
292
293 :param node: the supplied node to which we will apply the
294 loop parallelisation transformation.
295 :type node: :py:class:`psyclone.psyir.nodes.Node`
296 :param options: a dictionary with options for transformations.
297 :type options: Optional[Dict[str, Any]]
298 :param bool|int options["collapse"]: if it's a bool and is False
299 (default), it won't collapse. If it's a bool and is True, it will
300 collapse as much as possible. If it's an integer, it will attempt
301 to collapse until the specified number of loops (if they exist and
302 are safe to collapse them). The options 'ignore_dependencies_for'
303 and 'force' also affect the collapse applicabilty analysis.
304 :param bool options["force"]: whether to force parallelisation of the
305 target loop (i.e. ignore any dependence analysis).
306 :param list[str] options["ignore_dependencies_for"]: whether to ignore
307 some symbol names from the dependence analysis checks.
308 :param bool options["sequential"]: whether this is a sequential loop.
309 :param bool options["verbose"]: whether to report the reasons the
310 validate and collapse steps have failed.
311
312 '''
313 if not options:
314 options = {}
315 self.validatevalidatevalidate(node, options=options)
316
317 verbose = options.get("verbose", False)
318 collapse = options.get("collapse", False)
319 ignore_dep_analysis = options.get("force", False)
320 list_of_names = options.get("ignore_dependencies_for", [])
321 privatise_arrays = options.get("privatise_arrays", False)
322 list_of_signatures = [Signature(name) for name in list_of_names]
323 dtools = DependencyTools()
324
325 # keep a reference to the node's original parent and its index as these
326 # are required and will change when we change the node's location
327 node_parent = node.parent
328 node_position = node.position
329
330 # If 'privatise_arrays' is specified, make the write-write symbols
331 # private (we know this succeeds because the validate did a dry_run)
332 if privatise_arrays and not node.independent_iterations(
333 dep_tools=dtools,
334 test_all_variables=True,
335 signatures_to_ignore=list_of_signatures):
336 for message in dtools.get_all_messages():
337 if message.code == DTCode.ERROR_WRITE_WRITE_RACE:
338 for var_name in message.var_names:
339 self._attempt_privatisation(node, var_name)
340
341 # If 'collapse' is specified, check that it is an int and that the
342 # loop nest has at least that number of loops in it
343 if collapse:
344 # Count the number of perfectly nested loops that can be collapsed
345 num_collapsable_loops = 0
346 next_loop = node
347 previous_iteration_variables = []
348 while isinstance(next_loop, Loop):
349 previous_iteration_variables.append(next_loop.variable)
350 num_collapsable_loops += 1
351 if not isinstance(collapse, bool):
352 if num_collapsable_loops >= collapse:
353 break
354
355 # If it has more than one child, the next loop will not be
356 # perfectly nested, so stop searching. If there is no child,
357 # we have an empty loop and we also stop here.
358 if len(next_loop.loop_body.children) != 1:
359 if (next_loop.loop_body.children and
360 isinstance(next_loop.loop_body[0], Loop)):
361 next_loop.loop_body[0].append_preceding_comment(
362 "Loop cannot be collapsed because it has siblings")
363 break
364
365 next_loop = next_loop.loop_body[0]
366 if not isinstance(next_loop, Loop):
367 break
368
369 # If it is a loop dependent on a previous iteration variable
370 # (e.g. a triangular iteration space), it can not be collapsed
371 dependent_on_previous_variable = False
372 for bound in (next_loop.start_expr, next_loop.stop_expr,
373 next_loop.step_expr):
374 for ref in bound.walk(Reference):
375 if ref.symbol in previous_iteration_variables:
376 dependent_on_previous_variable = ref.symbol
377 break
378 if dependent_on_previous_variable:
379 if verbose:
380 next_loop.append_preceding_comment(
381 f"Loop cannot be collapsed because one of the "
382 f"bounds depends on the previous iteration variab"
383 f"le '{dependent_on_previous_variable.name}'")
384 break
385
386 # Check that the next loop has no loop-carried dependencies
387 if not ignore_dep_analysis:
388 if not next_loop.independent_iterations(
389 dep_tools=dtools,
390 signatures_to_ignore=list_of_signatures):
391 if verbose:
392 msgs = dtools.get_all_messages()
393 next_loop.preceding_comment = (
394 "\n".join([str(m) for m in msgs]) +
395 " Consider using the \"ignore_dependencies_"
396 "for\" transformation option if this is a "
397 "false dependency.")
398 break
399 else:
400 num_collapsable_loops = None
401
402 # Add our orphan loop directive setting its parent to the node's
403 # parent and its children to the node. This calls down to the sub-class
404 # to get the type of directive we require.
405 directive = self._directive([node.detach()], num_collapsable_loops)
406
407 # Add the loop directive as a child of the node's parent
408 node_parent.addchild(directive, index=node_position)
