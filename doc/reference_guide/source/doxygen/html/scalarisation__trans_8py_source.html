50class ScalarisationTrans(LoopTrans):
51 '''This transformation takes a Loop and converts any array accesses
52 to scalar if the results of the loop are unused, and the initial value
53 is unused. For example in the following snippet the value of a(i)
54 is only used inside the loop, so can be turned into a scalar, wheras
55 the values of b(i) are used in the following loop so are kept as an array:
56
57 >>> from psyclone.psyir.backend.fortran import FortranWriter
58 >>> from psyclone.psyir.frontend.fortran import FortranReader
59 >>> from psyclone.psyir.transformations import ScalarisationTrans
60 >>> from psyclone.psyir.nodes import Loop
61 >>> code = """program test
62 ... integer :: i,j
63 ... real :: a(100), b(100)
64 ... do i = 1,100
65 ... a(i) = i
66 ... b(i) = a(i) * a(i)
67 ... end do
68 ... do j = 1, 100
69 ... if(b(i) > 200) then
70 ... print *, b(i)
71 ... end if
72 ... end do
73 ... end program"""
74 >>> psyir = FortranReader().psyir_from_source(code)
75 >>> scalarise = ScalarisationTrans()
76 >>> scalarise.apply(psyir.walk(Loop)[0])
77 >>> print(FortranWriter()(psyir))
78 program test
79 integer :: i
80 integer :: j
81 real, dimension(100) :: a
82 real, dimension(100) :: b
83 real :: a_scalar
84 <BLANKLINE>
85 do i = 1, 100, 1
86 a_scalar = i
87 b(i) = a_scalar * a_scalar
88 enddo
89 do j = 1, 100, 1
90 if (b(i) > 200) then
91 ! PSyclone CodeBlock (unsupported code) reason:
92 ! - Unsupported statement: Print_Stmt
93 PRINT *, b(i)
94 end if
95 enddo
96 <BLANKLINE>
97 end program test
98 <BLANKLINE>
99 '''

101 @staticmethod
102 def _is_local_array(signature: Signature,
103 var_accesses: VariablesAccessInfo) -> bool:
104 '''
105 :param signature: The signature to check if it is a local array symbol
106 or not.
107 :param var_accesses: The VariableAccessesInfo object containing
108 signature.
109 :returns: whether the symbol corresponding to signature is a
110 local array symbol or not.
111 '''
112 if not var_accesses[signature].is_array():
113 return False
114 # If any of the accesses are to a CodeBlock then we stop. This can
115 # happen if there is a string access inside a string concatenation,
116 # e.g. NEMO4.
117 for access in var_accesses[signature].all_accesses:
118 if isinstance(access.node, CodeBlock):
119 return False
120 base_symbol = var_accesses[signature].all_accesses[0].node.symbol
121 if not base_symbol.is_automatic:
122 return False
123 # If its a derived type then we don't scalarise.
124 if isinstance(var_accesses[signature].all_accesses[0].node,
125 StructureReference):
126 return False
127 # Find the containing routine
128 rout = var_accesses[signature].all_accesses[0].node.ancestor(Routine)
129 # If the array is the return symbol then its not a local
130 # array symbol
131 if base_symbol is rout.return_symbol:
132 return False
133
134 return True

136 @staticmethod
137 def _have_same_unmodified_index(
138 signature: Signature,
139 var_accesses: VariablesAccessInfo) -> bool:
140 '''
141 :param signature: The signature to check.
142 :param var_accesses: The VariableAccessesInfo object containing
143 signature.
144 :returns: whether all the array accesses to signature use the
145 same index, and whether the index is unmodified in
146 the code region.
147 '''
148 array_indices = None
149 scalarisable = True
150 for access in var_accesses[signature].all_accesses:
151 if array_indices is None:
152 array_indices = access.component_indices
153 # For some reason using == on the component_lists doesn't work
154 # so we use [:] notation.
155 elif array_indices[:] != access.component_indices[:]:
156 scalarisable = False
157 break
158 # For each index, we need to check they're not written to in
159 # the loop.
160 flattened_indices = list(itertools.chain.from_iterable(
161 array_indices))
162 for index in flattened_indices:
163 # Index may not be a Reference, so we need to loop over the
164 # References
165 for ref in index.walk(Reference):
166 # This Reference could be the symbol for a Call or
167 # IntrinsicCall, which we don't allow to scalarise
168 if isinstance(ref.symbol, RoutineSymbol):
169 scalarisable = False
170 break
171 sig, _ = ref.get_signature_and_indices()
172 if var_accesses[sig].is_written():
173 scalarisable = False
174 break
175
176 return scalarisable

178 @staticmethod
179 def _check_first_access_is_write(signature: Signature,
180 loop: Loop,
181 var_accesses: VariablesAccessInfo) \
182 -> bool:
183 '''
184 :param signature: The signature to check.
185 :param loop: The Loop object being transformed.
186 :param var_accesses: The VariableAccessesInfo object containing
187 signature.
188 :returns: whether the first access to signature is a write.
189 '''
190 if not var_accesses[signature].is_written_first():
191 return False
192 # Need to find the first access and check if its in a conditional.
193 accesses = var_accesses[signature].all_accesses
194 first_node = accesses[0].node
195 ifblock = first_node.ancestor(IfBlock)
196 # If the depth of the ifblock is larger than loop then the write
197 # is in a conditional
198 if ifblock and ifblock.depth > loop.depth:
199 return False
200 return True

202 @staticmethod
203 def _get_index_values_from_indices(
204 node: ArrayMixin, indices: List[Node]) -> Tuple[bool, List[Node]]:
205 '''
206 Compute a list of index values for a given node. Looks at loop bounds
207 and range declarations to attempt to convert loop variables to an
208 explicit range, i.e. an access like
209 .. code-block:: fortran
210 do i = 1, 100
211 array(i) = ...
212 end do
213
214 the returned list would contain a range object for [1:100].
215
216 If the computed indexes contains a non-unit stride, or an index is
217 not a Range, Reference or Literal then this function will return
218 True as the first element of the returned tuple, and the list of
219 indices will be incomplete.
220
221 :param node: The node to compute index values for.
222 :param indices: the list of indexes to have values computed.
223
224 :returns: a tuple containing a bool value set to True if any of the
225 index values are not computed, and a list of the computed
226 index values.
227 '''
228 index_values = []
229 has_complex_index = False
230 for index in indices:
231 # If the index is an array or structure and there are any more
232 # accesses to the signature we're trying to scalarise, then we
233 # should not scalarise.
234 if (type(index) is not Range and type(index) is not Reference and
235 type(index) is not Literal):
236 has_complex_index = True
237 index_values.append(None)
238
239 one_literal = Literal("1", INTEGER_TYPE)
240 ancestor_loop = node.ancestor(Loop)
241 # For Range or Literal array indices this is easy.
242 for i, index in enumerate(indices):
243 if isinstance(index, (Range, Literal)):
244 index_values[i] = index
245
246 while ancestor_loop is not None and not has_complex_index:
247 for i, index in enumerate(indices):
248 # Skip over indices we already set.
249 if index_values[i] is not None:
250 continue
251 if ancestor_loop.variable == index.symbol:
252 start_val = ancestor_loop.start_expr
253 stop_val = ancestor_loop.stop_expr
254 step_val = ancestor_loop.step_expr
255 # If the step value is not exactly 1 then we treat
256 # this as a complex index, as we can't currently
257 # do precise comparisons on non-unit stride accesses.
258 if step_val != one_literal:
259 has_complex_index = True
260 # Create a range for this and add it to the index values.
261 index_range = Range.create(
262 start_val.copy(),
263 stop_val.copy()
264 )
265 index_values[i] = index_range
266 ancestor_loop = ancestor_loop.ancestor(Loop)
267
268 # If we couldn't work out any of the index_values, then we treat this
269 # as a complex index
270 for index in index_values:
271 if index is None:
272 has_complex_index = True
273
274 return has_complex_index, index_values

276 @staticmethod
277 def _value_unused_after_loop(sig: Signature,
278 loop: Loop,
279 var_accesses: VariablesAccessInfo) -> bool:
280 '''
281 :param sig: The signature to check.
282 :param loop: The loop the transformation is operating on.
283 :param var_accesses: The VariableAccessesInfo object containing
284 signature.
285 :returns: whether the value computed in the loop containing
286 sig is read from after the loop.
287 '''
288 # Find the last access of the signature
289 last_access = var_accesses[sig].all_accesses[-1].node
290 # Compute the indices used in this loop. We know that all of the
291 # indices used in this loop must be the same.
292 indices = last_access.indices
293
294 # Find the next accesses to this symbol
295 next_accesses = last_access.next_accesses()
296
297 # Compute the indices ranges.
298 has_complex_index, index_values = \
299 ScalarisationTrans._get_index_values_from_indices(
300 last_access, indices
301 )
302
303 for next_access in next_accesses:
304 # next_accesses looks backwards to the start of the loop,
305 # but we don't care about those accesses here.
306 if next_access.is_descendent_of(loop):
307 continue
308
309 # If we have a next_access outside of the loop and have a complex
310 # index then we do not scalarise this at the moment.
311 if has_complex_index:
312 return False
313
314 # If next access is a Call or CodeBlock or Kern then
315 # we have to assume the value is used. These nodes don't
316 # have the is_read property that Reference has, so we need
317 # to be explicit.
318 if isinstance(next_access, (CodeBlock, Call, Kern)):
319 return False
320
321 # If the access is a read, then return False
322 if next_access.is_read:
323 return False
324
325 # If the next access is a Reference then we had a full range
326 # access described without any range, which means a full
327 # range access so we can skip the followup checks.
328 if type(next_access) is Reference:
329 continue
330
331 # We need to ensure that the following write accesses the same
332 # or more of the array.
333 next_indices = next_access.indices
334 next_complex_index, next_values = \
335 ScalarisationTrans._get_index_values_from_indices(
336 next_access, next_indices
337 )
338 # If we can't compute the indices of the next access then we
339 # cannot scalarise
340 if next_complex_index:
341 return False
342 # Check the indices of next_access are greater than or equal to
343 # that of the potential scalarisation.
344 for i in range(len(next_values)):
345 # If the next index is a full range we can skip it as it must
346 # cover the previous access
347 if next_access.is_full_range(i):
348 continue
349 # Convert both to ranges if either was a literal
350 next_index = next_values[i]
351 orig_index = index_values[i]
352 if not isinstance(next_index, Range):
353 next_index = Range.create(next_index.copy(),
354 next_index.copy())
355 if not isinstance(orig_index, Range):
356 orig_index = Range.create(orig_index.copy(),
357 orig_index.copy())
358 sm = SymbolicMaths.get()
359 # Need to check that next_index stop point is >= orig_index.
360 # If its not then this can't cover the full range so we can
361 # return False to not Scalarise this.
362 if not (sm.greater_than(next_index.stop, orig_index.stop)
363 == SymbolicMaths.Fuzzy.TRUE or
364 sm.equal(next_index.stop, orig_index.stop)):
365 return False
366 # Need to check the next_index start point is <= orig_index
367 if not (sm.less_than(next_index.start, orig_index.start)
368 == SymbolicMaths.Fuzzy.TRUE or
369 sm.equal(next_index.start, orig_index.start)):
370 return False
371
372 return True

374 def apply(self, node: Loop, options: Optional[Dict[str, Any]] = None) \
375 -> None:
376 '''
377 Apply the scalarisation transformation to a loop.
378 All of the array accesses that are identified as being able to be
379 scalarised will be transformed by this transformation.
380
381 An array access will be scalarised if:
382 1. All accesses to the array use the same indexing statement.
383 2. All References contained in the indexing statement are not modified
384 inside of the loop (loop variables are ok).
385 3. The array symbol is either not accessed again or is written to
386 as its next access. If the next access is inside a conditional
387 that is not an ancestor of the input loop, then PSyclone will
388 assume that we cannot scalarise that value instead of attempting to
389 understand the control flow.
390 4. The array symbol is a local variable.
391
392 :param node: the supplied loop to apply scalarisation to.
393 :param options: a dictionary with options for transformations.
394
395 '''
396 # For each array reference in the Loop:
397 # Find every access to the same symbol in the loop
398 # They all have to be accessed with the same index statement, and
399 # that index needs to not be written to inside the loop body.
400 # For each symbol that meets this criteria, we then need to check the
401 # first access is a write
402 # Then, for each symbol still meeting this criteria, we need to find
403 # the next access outside of this loop. If its inside an ifblock that
404 # is not an ancestor of this loop then we refuse to scalarise for
405 # simplicity. Otherwise if its a read we can't scalarise safely.
406 # If its a write then this symbol can be scalarised.
407
408 var_accesses = VariablesAccessInfo(nodes=node.loop_body)
409
410 # Find all the arrays that are only accessed by a single index, and
411 # that index is only read inside the loop.
412 potential_targets = filter(
413 lambda sig:
414 ScalarisationTrans._is_local_array(sig, var_accesses),
415 var_accesses)
416 potential_targets = filter(
417 lambda sig:
418 ScalarisationTrans._have_same_unmodified_index(sig,
419 var_accesses),
420 potential_targets)
421
422 # Now we need to check the first access is a write and remove those
423 # that aren't.
424 potential_targets = filter(
425 lambda sig:
426 ScalarisationTrans._check_first_access_is_write(sig,
427 node,
428 var_accesses),
429 potential_targets)
430
431 # Check the values written to these arrays are not used after this loop
432 finalised_targets = filter(
433 lambda sig:
434 ScalarisationTrans._value_unused_after_loop(sig,
435 node,
436 var_accesses),
437 potential_targets)
438
439 routine_table = node.ancestor(Routine).symbol_table
440 # For each finalised target we can replace them with a scalarised
441 # symbol
442 for target in finalised_targets:
443 target_accesses = var_accesses[target].all_accesses
444 first_access = target_accesses[0].node
445 symbol_type = first_access.symbol.datatype.datatype
446 symbol_name = first_access.symbol.name
447 scalar_symbol = routine_table.new_symbol(
448 root_name=f"{symbol_name}_scalar",
449 symbol_type=DataSymbol,
450 datatype=symbol_type)
451 ref_to_copy = Reference(scalar_symbol)
452 for access in target_accesses:
453 node = access.node
454 node.replace_with(ref_to_copy.copy())
