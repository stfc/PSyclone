48class LFRicKernCallFactory():
49 ''' Create the necessary framework for an LFRic kernel call.
50 This consists of a Loop over cells containing a call to the
51 user-supplied kernel routine.
52
53 '''
54 # pylint: disable=too-few-public-methods
55 @staticmethod
56 def create(call, parent=None):
57 '''
58 Create the objects needed for a call to the kernel
59 described in the call object.
60
61 :param call: information on the kernel call as obtained from the \
62 Algorithm layer.
63 :type call: :py:class:`psyclone.parse.algorithm.KernelCall`
64 :param parent: the parent of this kernel call in the PSyIR.
65 :type parent: :py:class:`psyclone.psyir.nodes.Schedule`
66
67 '''
68 if call.ktype.iterates_over == "domain":
69 # Kernel operates on whole domain so there is no loop.
70 # We still need a loop object though as that is where the logic
71 # for handling halo exchanges is currently implemented.
72 loop_type = "null"
73 elif call.ktype.iterates_over == "dof":
74 # Loop over dofs within a field.
75 loop_type = "dof"
76 else:
77 # Loop over cells, indicated by an empty string.
78 loop_type = ""
79 cloop = LFRicLoop(parent=parent, loop_type=loop_type)
80
81 # The kernel itself
82 kern = LFRicKern()
83 kern.load(call, cloop.loop_body)
84
85 # Add the kernel as a child of the loop
86 cloop.loop_body.addchild(kern)
87
88 # Set-up the loop now we have the kernel object
89 cloop.load(kern)
90 return cloop
91
