68 def create_from_kernel(self, name, kernel_path):
69 '''
70 Generates LFRic algorithm PSyIR that calls the supplied kernel through
71 an 'invoke'. All of the arguments required by the kernel are
72 constructed and intialised appropriately. Fields and scalars are all
73 set to unity.
74
75 :param str name: name to use for the algorithm subroutine.
76 :param str kernel_path: location of Kernel source code.
77
78 :returns: LFRic algorithm PSyIR.
79 :rtype: :py:class:`psyclone.psyir.nodes.Container`
80
81 :raises NotImplementedError: if the specified kernel file does not \
82 follow the LFRic naming convention by having a module with a name \
83 ending in '_mod'.
84
85 '''
86 # pylint: disable=too-many-locals
87
88 # Create PSyIR for an algorithm routine.
89 cont = self.create_alg_routine(name)
90 sub = cont.walk(Routine)[0]
91 table = sub.symbol_table
92
93 # Parse the kernel metadata. Currently this uses fparser1 as that's
94 # what the existing meta-data handling is based upon. Ultimately, this
95 # will be replaced by the new, fparser2-based functionality being
96 # implemented in #1631.
97 parse_tree = get_kernel_parse_tree(kernel_path)
98
99 # Get the name of the module that contains the kernel and create a
100 # ContainerSymbol for it.
101 kernel_mod_name = parse_tree.content[0].name
102 # TODO #1806. The current meta-data parsing requires that we specify
103 # the name of the kernel. It would be much better if we could query the
104 # meta-data for the name of the kernel. For now we require that the
105 # LFRic naming scheme is strictly adhered to (since this is simpler
106 # than trying to walk through the deprecated fparser1 parse tree).
107 if not kernel_mod_name.endswith("_mod"):
108 raise NotImplementedError(
109 f"The supplied kernel ({kernel_path}) contains a module named "
110 f"'{kernel_mod_name}' which does not follow the LFRic naming "
111 f"convention of ending in '_mod'.")
112 kernel_name = kernel_mod_name[:-4] + "_type"
113
114 kernel_mod = table.new_symbol(kernel_mod_name,
115 symbol_type=ContainerSymbol)
116 kernel_routine = table.new_symbol(
117 kernel_name,
118 symbol_type=DataTypeSymbol,
119 datatype=UnresolvedType(),
120 interface=ImportInterface(kernel_mod))
121
122 kern = self.kernel_from_metadata(parse_tree, kernel_name)
123
124 # Declare and initialise the data structures required by the kernel
125 # arguments. Appropriate symbols are added to the symbol table
126 # associated with the routine we are constructing.
127 kern_args = self.construct_kernel_args(sub, kern)
128
129 # Initialise argument values to unity. Since we are using this somewhat
130 # arbitrary value, we use an *integer* literal for this, irrespective
131 # of the actual type of the scalar argument. The compiler/run-time will
132 # take care of appropriate type casting.
133 table.add_lfric_precision_symbol("i_def")
134 for sym in kern_args.scalars:
135 sub.addchild(Assignment.create(
136 Reference(sym),
137 Literal("1", LFRicTypes("LFRicIntegerScalarDataType")())))
138
139 # We use the setval_c builtin to initialise all fields to unity.
140 # As with the scalar initialisation, we don't worry about precision
141 # here since we are just setting the field values to unity. If the
142 # field itself is of a precision other than r_def (or is perhaps
143 # integer rather than real) we rely on type casting by the
144 # compiler/run-time.
145 factory = LFRicBuiltinFunctorFactory.get()
146 table.add_lfric_precision_symbol("r_def")
147 kernel_list = []
148 for sym, _ in kern_args.fields:
149 kernel_list.append(
150 factory.create(
151 "setval_c", table,
152 [Reference(sym),
153 Literal("1.0",
154 LFRicTypes("LFRicRealScalarDataType")())]))
155
156 # Finally, add the kernel itself to the list for the invoke().
157 arg_nodes = []
158 for arg in kern_args.arglist:
159 arg_nodes.append(Reference(table.lookup(arg)))
160 kern = LFRicKernelFunctor.create(kernel_routine, arg_nodes)
161 kernel_list.append(kern)
162
163 # Create the 'call invoke(...)' for the list of kernels.
164 invoke_sym = table.new_symbol("invoke", symbol_type=RoutineSymbol)
165 sub.addchild(LFRicAlgorithmInvokeCall.create(invoke_sym,
166 kernel_list, 0))
167 return cont

169 @staticmethod
170 def create_alg_routine(name):
171 '''
172 Creates an LFRic algorithm subroutine within a module. The
173 generated subroutine has three arguments:
174
175 * mesh: pointer to the LFRic mesh object.
176 * chi: coordinate field (optional).
177 * panel_id: field mapping cells to panel IDs (optional).
178
179 :param str name: the name to give the created routine. The associated \
180 container will have "_mod" appended to this name.
181
182 :returns: a container.
183 :rtype: :py:class:`psyclone.psyir.nodes.Container`
184
185 :raises TypeError: if the 'name' argument is of the wrong type.
186
187 '''
188 if not isinstance(name, str):
189 raise TypeError(f"Supplied routine name must be a str but got "
190 f"'{type(name).__name__}'")
191 # Make sure the scoping node creates LFRicSymbolTables
192 # pylint: disable=protected-access
193 # TODO #1954 Remove the protected access using a factory
194 ScopingNode._symbol_table_class = LFRicSymbolTable
195 alg_sub = Routine.create(name)
196 table = alg_sub.symbol_table
197
198 # Create Container and Type Symbols for each of the modules/types that
199 # we will need.
200 for root in ["field", "function_space", "mesh"]:
201 csym = table.new_symbol(root + "_mod", symbol_type=ContainerSymbol)
202
203 table.new_symbol(root + "_type", symbol_type=DataTypeSymbol,
204 datatype=UnresolvedType(),
205 interface=ImportInterface(csym))
206
207 fsc_mod = table.new_symbol("function_space_collection_mod",
208 symbol_type=ContainerSymbol)
209 table.new_symbol("function_space_collection",
210 symbol_type=DataSymbol,
211 datatype=UnresolvedType(),
212 interface=ImportInterface(fsc_mod))
213
214 # Declare the three arguments to the subroutine. All of them have to be
215 # of UnsupportedFortranType - the mesh because it is a pointer and chi
216 # and panel_id because they are optional.
217 mesh_ptr_type = UnsupportedFortranType(
218 "type(mesh_type), pointer, intent(in) :: mesh")
219 mesh_ptr = DataSymbol("mesh", mesh_ptr_type,
220 interface=ArgumentInterface())
221 table.add(mesh_ptr)
222
223 chi_type = UnsupportedFortranType(
224 "type(field_type), dimension(3), intent(in), optional :: chi")
225 chi = DataSymbol("chi", chi_type, interface=ArgumentInterface())
226 table.add(chi, tag="coord_field")
227
228 pid_type = UnsupportedFortranType(
229 "type(field_type), intent(in), optional :: panel_id")
230 pid = DataSymbol("panel_id", pid_type, interface=ArgumentInterface())
231 table.add(pid, tag="panel_id_field")
232 table.specify_argument_list([mesh_ptr, chi, pid])
233
234 # Create top-level Container and put the new Subroutine inside it.
235 container = Container(name+"_mod")
236 container.addchild(alg_sub)
237
238 return container

240 def _create_function_spaces(self, prog, fspaces):
241 '''
242 Adds PSyIR to the supplied Routine that declares and intialises
243 the specified function spaces. The order of these spaces is
244 set by the element_order_<h,v> variables which are provided by the
245 LFRic finite_element_config_mod module.
246
247 :param prog: the routine to which to add declarations and \
248 initialisation.
249 :type prog: :py:class:`psyclone.psyir.nodes.Routine`
250 :param fspaces: the names of the required function spaces.
251 :type fspaces: list[str]
252
253 :raises InternalError: if a function space is supplied that is not a \
254 recognised LFRic function space.
255
256 '''
257 table = prog.symbol_table
258
259 reader = FortranReader()
260
261 # The order of the finite-element scheme.
262 fe_config_mod = table.new_symbol(
263 "finite_element_config_mod", symbol_type=ContainerSymbol)
264 order_h = table.new_symbol(
265 "element_order_h", tag="element_order_h",
266 symbol_type=DataSymbol, datatype=UnresolvedType(),
267 interface=ImportInterface(fe_config_mod))
268 order_v = table.new_symbol(
269 "element_order_v", tag="element_order_v",
270 symbol_type=DataSymbol, datatype=UnresolvedType(),
271 interface=ImportInterface(fe_config_mod))
272
273 fs_cont_mod = table.new_symbol("fs_continuity_mod",
274 symbol_type=ContainerSymbol)
275
276 # Initialise the function spaces required by the kernel arguments.
277 const = LFRicConstants()
278
279 for space in sorted(fspaces):
280
281 if space.lower() not in const.VALID_FUNCTION_SPACE_NAMES:
282 raise InternalError(
283 f"Function space '{space}' is not a valid LFRic function "
284 f"space (one of {const.VALID_FUNCTION_SPACE_NAMES})")
285
286 table.new_symbol(f"{space}", tag=f"{space}",
287 symbol_type=DataSymbol,
288 datatype=UnresolvedType(),
289 interface=ImportInterface(fs_cont_mod))
290
291 vsym_ptr = table.new_symbol(
292 f"vector_space_{space}_ptr", symbol_type=DataSymbol,
293 tag=f"{space}_ptr",
294 datatype=UnsupportedFortranType(
295 f"TYPE(function_space_type), POINTER :: "
296 f"vector_space_{space}_ptr"))
297
298 cblock = reader.psyir_from_statement(
299 f"{vsym_ptr.name} => function_space_collection%get_fs( mesh, "
300 f"{order_h.name}, {order_v.name}, {space})", table)
301
302 prog.addchild(cblock)

304 @staticmethod
305 def initialise_field(prog, sym, space):
306 '''
307 Creates the PSyIR for initialisation of the field or field vector
308 represented by the supplied symbol and adds it to the supplied
309 routine.
310
311 :param prog: the routine to which to add initialisation code.
312 :type prog: :py:class:`psyclone.psyir.nodes.Routine`
313 :param sym: the symbol representing the LFRic field.
314 :type sym: :py:class:`psyclone.psyir.symbols.DataSymbol`
315 :param str space: the function space of the field.
316
317 :raises InternalError: if the supplied symbol is of the wrong type.
318
319 '''
320 reader = FortranReader()
321
322 prog.symbol_table.add_lfric_precision_symbol("i_def")
323
324 if isinstance(sym.datatype, DataTypeSymbol):
325 # Single field argument.
326 prog.addchild(
327 reader.psyir_from_statement(
328 f"CALL {sym.name} % initialise(vector_space = "
329 f"vector_space_{space}_ptr, name = '{sym.name}')",
330 prog.symbol_table))
331
332 elif isinstance(sym.datatype, ArrayType):
333 # Field vector argument.
334 for dim in range(int(sym.datatype.shape[0].lower.value),
335 int(sym.datatype.shape[0].upper.value)+1):
336 prog.addchild(
337 reader.psyir_from_statement(
338 f"CALL {sym.name}({dim}_i_def) % initialise("
339 f"vector_space = vector_space_{space}_ptr, "
340 f"name = '{sym.name}')", prog.symbol_table))
341 else:
342 raise InternalError(
343 f"Expected a field symbol to either be of ArrayType or have "
344 f"a type specified by a DataTypeSymbol but found "
345 f"{sym.datatype} for field '{sym.name}'")

372 @staticmethod
373 def initialise_quadrature(prog, qr_sym, shape):
374 '''
375 Adds the necessary declarations and intialisation for the supplied
376 quadrature to the supplied routine.
377
378 :param prog: the routine to which to add suitable declarations etc.
379 :type prog: :py:class:`psyclone.psyir.nodes.Routine`
380 :param qr_sym: the symbol representing a quadrature object.
381 :type qr_sym: :py:class:`psyclone.psyir.symbols.DataSymbol`
382 :param str shape: the shape of the quadrature.
383
384 :raises NotImplementedError: if the quadrature shape is anything \
385 other than gh_quadrature_xyoz.
386 '''
387 reader = FortranReader()
388 table = prog.symbol_table
389 try:
390 qr_rule_sym = table.lookup("quadrature_rule")
391 except KeyError:
392 qr_gaussian_mod = table.new_symbol(
393 "quadrature_rule_gaussian_mod", symbol_type=ContainerSymbol)
394 qr_gaussian_type = table.new_symbol(
395 "quadrature_rule_gaussian_type", symbol_type=DataTypeSymbol,
396 datatype=UnresolvedType(),
397 interface=ImportInterface(qr_gaussian_mod))
398 qr_rule_sym = table.new_symbol("quadrature_rule",
399 symbol_type=DataSymbol,
400 datatype=qr_gaussian_type)
401
402 if shape == "gh_quadrature_xyoz":
403 order_h = table.lookup_with_tag("element_order_h")
404 order_v = table.lookup_with_tag("element_order_v")
405 expr = reader.psyir_from_expression(
406 f"quadrature_xyoz_type({order_h.name}+3,{order_h.name}+3,"
407 f"{order_v.name}+3,{qr_rule_sym.name})",
408 table)
409 prog.addchild(Assignment.create(Reference(qr_sym), expr))
410
411 else:
412 raise NotImplementedError(f"Initialisation for quadrature of type "
413 f"'{shape}' is not yet implemented.")

415 @staticmethod
416 def kernel_from_metadata(parse_tree, kernel_name):
417 '''
418 Given an fparser1 parse tree for an LFRic kernel, creates and returns
419 a LFRicKern object.
420
421 :param parse_tree: the fparser1 parse tree for the LFRic kernel.
422 :type parse_tree: :py:class:`fparser.one.block_statements.BeginSource`
423 :param str kernel_name: the name of the kernel contained in the \
424 supplied parse tree for which a LFRicKern is to be created.
425
426 :returns: a LFRicKern object describing the LFRic kernel.
427 :rtype: :py:class:`psyclone.domain.lfric.LFRicKern`
428
429 :raises ValueError: if an LFRic kernel with the specified name cannot \
430 be found in the supplied parse tree.
431 '''
432 try:
433 ktype = KernelTypeFactory(api="lfric").create(parse_tree,
434 name=kernel_name)
435 except ParseError as err:
436 raise ValueError(
437 f"Failed to find kernel '{kernel_name}' in supplied "
438 f"code: '{parse_tree}'. Is it a valid LFRic kernel? Original "
439 f"error was '{err}'.") from err
440 # Construct a LFRicKern using the metadata.
441 kern = LFRicKern()
442 kern.load_meta(ktype)
443 return kern

445 def construct_kernel_args(self, prog, kern):
446 '''
447 Extends the supplied routine with all the declarations and
448 initialisation required for the arguments of the supplied kernel.
449
450 :param prog: the routine to which to add the declarations etc.
451 :type prog: :py:class:`psyclone.psyir.nodes.Routine`
452 :param kern: the kernel for which we are to create arguments.
453 :type kern: :py:class:`psyclone.domain.lfric.LFRicKern`
454
455 :returns: object capturing all of the kernel arguments.
456 :rtype: :py:class:`psyclone.domain.lfric.KernCallInvokeArgList`
457
458 '''
459 const = LFRicConstants()
460 # Construct a list of the names of the function spaces that the field
461 # argument(s) are on and any operators map between. We use
462 # LFRicConstants.specific_function_space()
463 # to ensure that any 'wildcard' names in the meta-data are converted to
464 # an appropriate, specific function space.
465 function_spaces = []
466 for fspace in kern.arguments.unique_fss:
467 name = fspace.orig_name.lower()
468 function_spaces.append(const.specific_function_space(name))
469 self._create_function_spaces(prog, set(function_spaces))
470
471 # Construct the argument list and add suitable symbols to the table.
472 kern_args = KernCallInvokeArgList(kern, prog.symbol_table)
473 kern_args.generate()
474
475 # Initialise field datastructures using the symbols added to the table
476 # when setting up the kernel arguments and the information on their
477 # respective function spaces extracted from the kernel metadata.
478 for sym, space in kern_args.fields:
479 self.initialise_field(prog, sym, space)
480
481 for sym, from_space, to_space in kern_args.operators:
482 self.initialise_operator(prog, sym, from_space, to_space)
483
484 for qr_sym, shape in kern_args.quadrature_objects:
485 self.initialise_quadrature(prog, qr_sym, shape)
486
487 return kern_args
