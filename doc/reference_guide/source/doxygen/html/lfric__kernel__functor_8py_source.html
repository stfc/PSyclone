62class LFRicBuiltinFunctor(LFRicFunctor):
63 ''' Base class which all LFRic builtins subclass. Contains a builtin call,
64 a description of its required interface and the arguments to be passed
65 to it.
66
67 '''
68 _text_name = "LFRicBuiltinFunctor"
69 _builtin_name = ""
70
71 @classmethod
72 def create(cls, table, arguments):
73 '''
74 An appropriate DataTypeSymbol is created and added to the supplied
75 symbol table (if it does not already contain one). This is then
76 passed to the create() method in the base class to create an instance
77 of an LFRic builtin call with the supplied list of arguments.
78
79 :param table: the symbol table to which to add the symbol for this \
80 functor.
81 :type table: :py:class:`psyclone.psyir.symbols.SymbolTable`
82 :param arguments: the arguments to this routine. These are \
83 added as child nodes.
84 :type arguments: List[:py:class:`psyclone.psyir.nodes.DataNode`]
85
86 :returns: a functor object describing an LFRic builtin.
87 :rtype: :py:class:`psyclone.domain.lfric.algorithm.LFRicBuiltinFunctor`
88
89 :raises InternalError: if a symbol is found that has the same name as \
90 a built-in but does not have the correct properties.
91 '''
92 # We can't use find_or_create() here as that raises an Exception if
93 # the symbol that is found is not of the correct type and we want to
94 # allow for the case where an unresolved Symbol of the right name
95 # has already been added to the table.
96 try:
97 sym = table.lookup(cls._builtin_name)
98 # pylint: disable=unidiomatic-typecheck
99 if type(sym) is Symbol:
100 if not sym.is_unresolved:
101 raise InternalError(
102 f"A symbol with the same name as builtin '{sym.name}' "
103 f"exists but has an interface of '{sym.interface}' "
104 f"instead of being unresolved.")
105 sym.specialise(DataTypeSymbol)
106 sym.datatype = StructureType()
107 elif not isinstance(sym, DataTypeSymbol):
108 raise InternalError(
109 f"A symbol with the same name as builtin '{sym.name}' "
110 f"exists but it is a '{type(sym).__name__}' and not a "
111 f"DataTypeSymbol.")
112 except KeyError:
113 sym = table.new_symbol(cls._builtin_name,
114 symbol_type=DataTypeSymbol,
115 datatype=StructureType())
116
117 return super().create(sym, arguments)

119 def lower_to_language_level(self):
120 ''' Removes the symbol representing this BuiltIn as it only
121 exists in the DSL.
122
123 :returns: the lowered version of this node.
124 :rtype: :py:class:`psyclone.psyir.node.Node`
125
126 '''
127 table = self.scope.symbol_table
128 try:
129 sym = table.lookup(self._builtin_name)
130 table = sym.find_symbol_table(self)
131 # TODO #898 SymbolTable.remove() does not yet support
132 # DataTypeSymbols.
133 # pylint: disable=protected-access
134 del table._symbols[self._builtin_name]
135 except KeyError:
136 # The symbol has already been removed by a previous lowering
137 # of the same builtin.
138 pass
139 return self

142class LFRicBuiltinFunctorFactory():
143 '''
144 This class is a singleton which generates and stores a Functor class for
145 each built-in supported by LFRic. An instance of one of these classes may
146 be obtained by using the `create()` method.
147
148 :raises ValueError: if an attempt is made to construct a second instance \
149 of this class.
150 '''
151 #: The singleton instance of this class.
152 _instance = None
153
154 @staticmethod
155 def get():
156 '''
157 :returns: the singleton instance of this class.
158 :rtype: :py:class:\
159 `psyclone.domain.lfric.algorithm.psyir.LFRicBuiltinFunctorFactory`
160 '''
161 if not LFRicBuiltinFunctorFactory._instance:
162 LFRicBuiltinFunctorFactory._instance = LFRicBuiltinFunctorFactory()
163 return LFRicBuiltinFunctorFactory._instance
164
165 def __init__(self):
166 if LFRicBuiltinFunctorFactory._instance:
167 raise ValueError(
168 "Only one instance of LFRicBuiltinFunctorFactory "
169 "is permitted. Use the get() method to access it.")
170 # Classes representing LFRic built-in functors, indexed by name.
171 self._builtin_functor_map = {}
172
173 def _create_classes(self):
174 '''
175 Generate classes representing LFRic BuiltIn Functors by using
176 the type() function. The classes are stored in a class variable
177 so that this construction is only ever performed once.
178
179 '''
180 if self._builtin_functor_map:
181 return
182
183 for name in BUILTIN_MAP_CAPITALISED:
184 lname = name.lower()
185 self._builtin_functor_map[lname] = type(
186 f"LFRic_{name}_Functor",
187 (LFRicBuiltinFunctor,),
188 {"_builtin_name": name.lower()})
189
190 def _get_builtin_class(self, name):
191 '''
192 Look-up the class associated with the named built-in.
193
194 :param str name: name of an LFRic built-in kernel (not case sensitive).
195
196 :returns: the class representing the functor for the named built-in.
197 :rtype: type
198
199 '''
200 if not self._builtin_functor_map:
201 self._create_classes()
202 return self._builtin_functor_map[name.lower()]
203
204 def create(self, name, table, arguments):
205 '''
206 Create a BuiltinFunctor for the named LFRic builtin.
207
208 :param str name: the built-in for which a functor is required.
209 :param table: the symbol table to which to add a corresponding symbol.
210 :type table: :py:class:`psyclone.psyir.symbols.SymbolTable`
211 :param arguments: the arguments to give to the functor.
212 :type arguments: List[:py:class:`psyclone.psyir.nodes.DataNode`]
213
214 '''
215 cls = self._get_builtin_class(name)
216 return cls.create(table, arguments)
