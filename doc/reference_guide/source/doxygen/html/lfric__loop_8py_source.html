58class LFRicLoop(PSyLoop):
59 '''
60 The LFRic-specific PSyLoop class. This passes the LFRic-specific
61 loop information to the base class so it creates the one
62 we require. Creates LFRic-specific loop bounds when the code is
63 being generated.
64
65 :param str loop_type: the type (iteration space) of this loop.
66 :param kwargs: additional keyword arguments provided to the PSyIR node.
67 :type kwargs: unwrapped dict.
68
69 :raises InternalError: if an unrecognised loop_type is specified.
70
71 '''
72 # pylint: disable=too-many-instance-attributes
73 def __init__(self, loop_type="", **kwargs):
74 const = LFRicConstants()
75 super().__init__(valid_loop_types=const.VALID_LOOP_TYPES, **kwargs)
76 self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type = loop_type
77
78 # Set our variable at initialisation as it might be required
79 # by other classes before code generation. A 'null' loop does not
80 # have an associated variable.
81 if self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type != "null":
82
83 if self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type == "colours":
84 tag = "colours_loop_idx"
85 suggested_name = "colour"
86 elif self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type == "colour":
87 tag = "cell_loop_idx"
88 suggested_name = "cell"
89 elif self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type == "dof":
90 tag = "dof_loop_idx"
91 suggested_name = "df"
92 elif self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type == "":
93 tag = "cell_loop_idx"
94 suggested_name = "cell"
95 else:
96 raise InternalError(
97 f"Unsupported loop type '{self.loop_type}' found when "
98 f"creating loop variable. Supported values are 'colours', "
99 f"'colour', 'dof' or '' (for cell-columns).")
100
101 self.variablevariablevariablevariable = self.scope.symbol_table.find_or_create_tag(
102 tag, root_name=suggested_name, symbol_type=DataSymbol,
103 datatype=LFRicTypes("LFRicIntegerScalarDataType")())
104
105 # Pre-initialise the Loop children # TODO: See issue #440
106 self.addchild(Literal("NOT_INITIALISED", INTEGER_TYPE,
107 parent=self)) # start
108 self.addchild(Literal("NOT_INITIALISED", INTEGER_TYPE,
109 parent=self)) # stop
110 self.addchild(Literal("1", INTEGER_TYPE, parent=self)) # step
111 self.addchild(Schedule(parent=self)) # loop body
112
113 # At this stage we don't know what our loop bounds are
114 self._lower_bound_name = None
115 self._lower_bound_index = None
116 self._upper_bound_name_upper_bound_name = None
117 self._upper_bound_halo_depth_upper_bound_halo_depth = None

119 def lower_to_language_level(self):
120 '''In-place replacement of DSL or high-level concepts into generic
121 PSyIR constructs. This function replaces a LFRicLoop with a PSyLoop
122 and inserts the loop boundaries into the new PSyLoop, or removes
123 the loop node in case of a domain kernel. Once TODO #1731 is done
124 (which should fix the loop boundaries, which atm rely on index of
125 the loop in the schedule, i.e. can change when transformations are
126 applied), this function can likely be removed.
127
128 :returns: the lowered version of this node.
129 :rtype: :py:class:`psyclone.psyir.node.Node`
130
131 '''
132 if self._loop_type_loop_type != "null":
133
134 # This is not a 'domain' loop (i.e. there is a real loop). First
135 # check that there isn't any validation issues with the node.
136 for child in self.loop_body.children:
137 child.validate_global_constraints()
138
139 # Then generate the loop bounds, this needs to be done BEFORE
140 # lowering the loop body because it needs kernel information.
141 start = self.start_exprstart_exprstart_expr.copy()
142 stop = self.stop_exprstop_exprstop_expr.copy()
143 step = self.step_exprstep_expr.copy()
144
145 # Now we can lower the nodes in the loop body
146 for child in self.loop_body.children:
147 child.lower_to_language_level()
148
149 # Finally create the new lowered Loop and replace the domain one
150 loop = Loop.create(self._variable_variable, start, stop, step, [])
151 loop.loop_body._symbol_table = \
152 self.loop_body.symbol_table.shallow_copy()
153 loop.children[3] = self.loop_body.copy()
154 self.replace_with(loop)
155 lowered_node = loop
156 else:
157 # If loop_type is "null" we do not need a loop at all, just the
158 # kernel in its loop_body
159 for child in self.loop_body.children:
160 child.lower_to_language_level()
161 # TODO #1010: This restriction can be removed when also lowering
162 # the parent InvokeSchedule
163 if len(self.loop_body.children) > 1:
164 raise NotImplementedError(
165 f"Lowering LFRic domain loops that produce more than one "
166 f"children is not yet supported, but found:\n "
167 f"{self.view()}")
168 lowered_node = self.loop_body[0].detach()
169 self.replace_with(lowered_node)
170
171 return lowered_node

174 ''' Creates a text summary of this loop node. We override this
175 method from the Loop class because, in LFRic, the function
176 space is now an object and we need to call orig_name on it. We
177 also include the upper loop bound as this can now be modified.
178
179 :param bool colour: whether or not to include control codes for colour.
180
181 :returns: text summary of this node, optionally with control codes \
182 for colour highlighting.
183 :rtype: str
184
185 '''
186 if self._loop_type_loop_type == "null":
187 return f"{self.coloured_name(colour)}[type='null']"
188
189 if self._upper_bound_halo_depth_upper_bound_halo_depth:
190 upper_bound = (f"{self._upper_bound_name}"
191 f"({self._upper_bound_halo_depth})")
192 else:
193 upper_bound = self._upper_bound_name_upper_bound_name
194 return (f"{self.coloured_name(colour)}[type='{self._loop_type}', "
195 f"field_space='{self._field_space.orig_name}', "
196 f"it_space='{self.iteration_space}', "
197 f"upper_bound='{upper_bound}']")

199 def load(self, kern):
200 '''
201 Load the state of this Loop using the supplied Kernel
202 object. This method is provided so that we can individually
203 construct Loop objects for a given kernel call.
204
205 :param kern: Kernel object to use to populate state of Loop
206 :type kern: :py:class:`psyclone.domain.lfric.LFRicKern`
207
208 :raises GenerationError: if the field updated by the kernel has an \
209 unexpected function space or if the kernel's 'operates-on' is \
210 not consistent with the loop type.
211
212 '''
213 self._kern_kern = kern
214
215 self._field_field = kern.arguments.iteration_space_arg()
216 self._field_name_field_name = self._field_field.name
217 self._field_space_field_space = self._field_field.function_space
218
219 if self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type == "null" and kern.iterates_over != "domain":
220 raise GenerationError(
221 f"A LFRicLoop of type 'null' can only contain a kernel that "
222 f"operates on the 'domain' but kernel '{kern.name}' operates "
223 f"on '{kern.iterates_over}'.")
224 self._iteration_space_iteration_space = kern.iterates_over # cell_columns etc.
225
226 # Loop bounds
227 self.set_lower_bound("start")
228 const = LFRicConstants()
229 if kern.iterates_over == "dof":
230 # This loop must be over DoFs
231 if Config.get().api_conf("lfric").compute_annexed_dofs \
232 and Config.get().distributed_memory \
233 and not kern.is_reduction:
234 self.set_upper_bound("nannexed")
235 else:
236 self.set_upper_bound("ndofs")
237 return
238
239 if "halo" in kern.iterates_over:
240 if Config.get().distributed_memory:
241 if kern.iterates_over == "halo_cell_column":
242 # In LFRic, the local cell-indexing scheme is set up such
243 # that owned cells have lower indices than halo cells, the
244 # first halo cell starts immediately after the last owned
245 # cell, and the cell indices are contiguous.
246 self.set_lower_bound("cell_halo_start")
247 self.set_upper_bound("cell_halo", halo_depth=kern.halo_depth)
248 return
249
250 if not Config.get().distributed_memory:
251 # Sequential
252 self.set_upper_bound("ncells")
253 return
254
255 # Otherwise, distributed memory is enabled.
256 if self._field_field.is_operator:
257 # We always compute operators redundantly out to the L1
258 # halo
259 self.set_upper_bound("cell_halo", halo_depth=1)
260 return
261 if (self.field_spacefield_space.orig_name in
262 const.VALID_DISCONTINUOUS_NAMES):
263 # Iterate to ncells for all discontinuous quantities,
264 # including any_discontinuous_space
265 self.set_upper_bound("ncells")
266 return
267 if (self.field_spacefield_space.orig_name in
268 const.CONTINUOUS_FUNCTION_SPACES):
269 # Must iterate out to L1 halo for continuous quantities
270 # unless the only arguments that are updated all have
271 # 'GH_WRITE' access. The only time such an access is
272 # permitted for a field on a continuous space is when the
273 # kernel is implemented such that any writes to a given
274 # shared dof are guaranteed to write the same value. There
275 # is therefore no need to iterate into the L1 halo in order
276 # to get correct values for annexed dofs.
277 if not kern.all_updates_are_writes:
278 self.set_upper_bound("cell_halo", halo_depth=1)
279 return
280 self.set_upper_bound("ncells")
281 return
282 if (self.field_spacefield_space.orig_name in
283 const.VALID_ANY_SPACE_NAMES):
284 # We don't know whether any_space is continuous or not
285 # so we have to err on the side of caution and assume that
286 # it is. Again, if the only arguments that are updated have
287 # 'GH_WRITE' access then we can relax this condition.
288 if not kern.all_updates_are_writes:
289 self.set_upper_bound("cell_halo", halo_depth=1)
290 return
291 self.set_upper_bound("ncells")
292 return
293
294 raise GenerationError(
295 f"Unexpected function space found. Expecting one of "
296 f"{const.VALID_FUNCTION_SPACES} but found "
297 f"'{self.field_space.orig_name}'")

299 def set_lower_bound(self, name, index=None):
300 ''' Set the lower bounds of this loop '''
301 const = LFRicConstants()
302 if name not in const.VALID_LOOP_BOUNDS_NAMES:
303 raise GenerationError(
304 "The specified lower bound loop name is invalid")
305 if name in ["inner"] + const.HALO_ACCESS_LOOP_BOUNDS and index < 1:
306 raise GenerationError(
307 "The specified index '{index}' for this lower loop bound is "
308 "invalid")
309 self._lower_bound_name = name
310 self._lower_bound_index = index
311
312 def set_upper_bound(self, name, halo_depth=None):
313 '''Set the upper bound of this loop.
314
315 :param str name: Loop upper-bound name. Must be a supported name.
316 :param halo_depth: An optional argument indicating the depth of halo
317 that this loop accesses.
318 :type halo_depth: Optional[:py:class:`psyclone.psyir.nodes.DataNode` |
319 int]
320
321 :raises GenerationError: if supplied with an invalid upper-bound name.
322 :raises GenerationError: if supplied with a halo depth < 1.
323 :raises TypeError: if the supplied halo_depth value is neither an int
324 or DataNode.
325 '''
326 const = LFRicConstants()
327 if name not in const.VALID_LOOP_BOUNDS_NAMES:
328 raise GenerationError(
329 f"The specified upper loop bound name is invalid. Expected "
330 f"one of {const.VALID_LOOP_BOUNDS_NAMES} but found '{name}'")
331 if name == "start":
332 raise GenerationError("'start' is not a valid upper bound")
333 # Only halo bounds and inner may have an index. We could just
334 # test for index here and assume that index is None for other
335 # types of bounds, but checking the type of bound as well is a
336 # safer option.
337 if (name in (["inner"] + const.HALO_ACCESS_LOOP_BOUNDS) and
338 isinstance(halo_depth, int)):
339 if halo_depth < 1:
340 raise GenerationError(
341 f"The specified halo depth '{halo_depth}' for this loop "
342 f"upper bound is < 1 which is invalid.")
343 self._upper_bound_name_upper_bound_name = name
344 if halo_depth and isinstance(halo_depth, int):
345 # We support specifying depth as an int as a convenience but we
346 # now convert it to a PSyIR literal.
347 psyir = Literal(f"{halo_depth}", INTEGER_TYPE)
348 self._upper_bound_halo_depth_upper_bound_halo_depth = psyir
349 else:
350 if halo_depth is not None and not isinstance(halo_depth, DataNode):
351 raise TypeError(f"When setting the upper bound of a loop, any "
352 f"halo depth must be supplied as an int or "
353 f"PSyIR DataNode but got {type(halo_depth)}")
354 self._upper_bound_halo_depth_upper_bound_halo_depth = halo_depth
355
356 @property
357 def upper_bound_name(self):
358 ''' Returns the name of the upper loop bound '''
359 return self._upper_bound_name_upper_bound_name

361 @property

363 '''Returns the index of the upper loop bound. This is None if the upper
364 bound name is not in HALO_ACCESS_LOOP_BOUNDS.

366 :returns: the depth of the halo for a loops upper bound. If it \
367 is None then a depth has not been provided. The depth value is \
368 only valid when the upper-bound name is associated with a halo \
369 e.g. 'cell_halo'.
370 :rtype: int
371
372 '''
373 return self._upper_bound_halo_depth_upper_bound_halo_depth

375 def _lower_bound_fortran(self):
376 '''
377 Create the associated Fortran code for the type of lower bound.
378
379 TODO: Issue #440. lower_bound_fortran should generate PSyIR.
380
381 :returns: the Fortran code for the lower bound.
382 :rtype: str
383
384 :raises GenerationError: if self._lower_bound_name is not "start"
385 for sequential code.
386 :raises GenerationError: if self._lower_bound_name is unrecognised.
387
388 '''
389 if (not Config.get().distributed_memory and
390 self._lower_bound_name != "start"):
391 raise GenerationError(
392 f"The lower bound must be 'start' if we are sequential but "
393 f"found '{self._upper_bound_name}'")
394 if self._lower_bound_name == "start":
395 return "1"
396
397 # the start of our space is the end of the previous space +1
398 if self._lower_bound_name == "inner":
399 prev_space_name = self._lower_bound_name
400 prev_space_index_str = str(self._lower_bound_index + 1)
401 elif self._lower_bound_name == "ncells":
402 prev_space_name = "inner"
403 prev_space_index_str = "1"
404 elif (self._lower_bound_name == "cell_halo" and
405 self._lower_bound_index == 1):
406 prev_space_name = "ncells"
407 prev_space_index_str = ""
408 elif self._lower_bound_name == "cell_halo_start":
409 prev_space_name = "edge"
410 prev_space_index_str = ""
411 elif (self._lower_bound_name == "cell_halo" and
412 self._lower_bound_index > 1):
413 prev_space_name = self._lower_bound_name
414 prev_space_index_str = str(self._lower_bound_index - 1)
415 else:
416 raise GenerationError(
417 f"Unsupported lower bound name '{self._lower_bound_name}' "
418 f"found")
419 # Use InvokeSchedule SymbolTable to share the same symbol for all
420 # Loops in the Invoke.
421 mesh_obj_name = self.ancestor(InvokeSchedule).symbol_table.\
422 find_or_create_tag("mesh").name
423 return mesh_obj_name + "%get_last_" + prev_space_name + "_cell(" \
424 + prev_space_index_str + ")+1"

426 @property
427 def _mesh_name(self):
428 '''
429 :returns: the name of the mesh variable from which to get the bounds \
430 for this loop.
431 :rtype: str
432 '''
433 # We must allow for self._kern being None (as it will be for
434 # a built-in).
435 if self._kern_kern and self._kern_kern.is_intergrid:
436 # We have more than one mesh object to choose from and we
437 # want the coarse one because that determines the iteration
438 # space. _field_name holds the name of the argument that
439 # determines the iteration space of this kernel and that
440 # is set-up to be the one on the coarse mesh (in
441 # DynKernelArguments.iteration_space_arg()).
442 tag_name = "mesh_" + self._field_name_field_name
443 else:
444 # It's not an inter-grid kernel so there's only one mesh
445 tag_name = "mesh"
446
447 # The symbol for the mesh will already have been added to the
448 # symbol table associated with the InvokeSchedule.
449 return self.ancestor(InvokeSchedule).symbol_table.\
450 lookup_with_tag(tag_name).name

452 def _upper_bound_fortran(self):
453 ''' Create the Fortran code that gives the appropriate upper bound
454 value for this type of loop.
455
456 TODO: Issue #440. upper_bound_fortran should generate PSyIR.
457
458 :returns: Fortran code for the upper bound of this loop.
459 :rtype: str
460
461 '''
462 # pylint: disable=too-many-branches, too-many-return-statements
463 # precompute halo_index as a string as we use it in more than
464 # one of the if clauses
465 halo_index = ""
466 if self._upper_bound_halo_depth_upper_bound_halo_depth:
467 halo_index = FortranWriter()(self._upper_bound_halo_depth_upper_bound_halo_depth)
468
469 if self._upper_bound_name_upper_bound_name == "ncolours":
470 # Loop over colours
471 kernels = self.walk(LFRicKern)
472 if not kernels:
473 raise InternalError(
474 "Failed to find a kernel within a loop over colours.")
475 # Check that all kernels have been coloured. We can't check the
476 # number of colours since that is only known at runtime.
477 for kern in kernels:
478 if not kern.ncolours_var:
479 raise InternalError(
480 f"All kernels within a loop over colours must have "
481 f"been coloured but kernel '{kern.name}' has not")
482 return kernels[0].ncolours_var
483
484 if self._upper_bound_name_upper_bound_name == "ncolour":
485 # Loop over cells of a particular colour when DM is disabled.
486 # We use the same, DM API as that returns sensible values even
487 # when running without MPI.
488 root_name = "last_edge_cell_all_colours"
489 if self._kern_kern.is_intergrid:
490 root_name += "_" + self._field_name_field_name
491 sym = self.ancestor(
492 InvokeSchedule).symbol_table.find_or_create_tag(root_name)
493 return f"{sym.name}(colour)"
494 if self._upper_bound_name_upper_bound_name == "colour_halo":
495 # Loop over cells of a particular colour when DM is enabled. The
496 # LFRic API used here allows for colouring with redundant
497 # computation.
498 sym_tab = self.ancestor(InvokeSchedule).symbol_table
499 if halo_index:
500 # The colouring API provides a 2D array that holds the last
501 # halo cell for a given colour and halo depth.
502 depth = halo_index
503 else:
504 # If no depth is specified then we go to the full halo depth
505 depth = sym_tab.find_or_create_tag(
506 f"max_halo_depth_{self._mesh_name}").name
507 root_name = "last_halo_cell_all_colours"
508 if self._kern_kern.is_intergrid:
509 root_name += "_" + self._field_name_field_name
510 sym = sym_tab.find_or_create_tag(root_name)
511 return f"{sym.name}(colour, {depth})"
512 if self._upper_bound_name_upper_bound_name in ["ndofs", "nannexed"]:
513 if Config.get().distributed_memory:
514 if self._upper_bound_name_upper_bound_name == "ndofs":
515 result = (
516 f"{self.field.proxy_name_indexed}%"
517 f"{self.field.ref_name()}%get_last_dof_owned()")
518 else: # nannexed
519 result = (
520 f"{self.field.proxy_name_indexed}%"
521 f"{self.field.ref_name()}%get_last_dof_annexed()")
522 else:
523 result = self._kern_kern.undf_name
524 return result
525 if self._upper_bound_name_upper_bound_name == "ncells":
526 if Config.get().distributed_memory:
527 result = f"{self._mesh_name}%get_last_edge_cell()"
528 else:
529 result = (f"{self.field.proxy_name_indexed}%"
530 f"{self.field.ref_name()}%get_ncell()")
531 return result
532 if self._upper_bound_name_upper_bound_name == "cell_halo":
533 if Config.get().distributed_memory:
534 return f"{self._mesh_name}%get_last_halo_cell({halo_index})"
535 raise GenerationError(
536 "'cell_halo' is not a valid loop upper bound for "
537 "sequential/shared-memory code")
538 if self._upper_bound_name_upper_bound_name == "dof_halo":
539 if Config.get().distributed_memory:
540 return (f"{self.field.proxy_name_indexed}%"
541 f"{self.field.ref_name()}%get_last_dof_halo("
542 f"{halo_index})")
543 raise GenerationError(
544 "'dof_halo' is not a valid loop upper bound for "
545 "sequential/shared-memory code")
546 if self._upper_bound_name_upper_bound_name == "inner":
547 if Config.get().distributed_memory:
548 return f"{self._mesh_name}%get_last_inner_cell({halo_index})"
549 raise GenerationError(
550 "'inner' is not a valid loop upper bound for "
551 "sequential/shared-memory code")
552 raise GenerationError(
553 f"Unsupported upper bound name '{self._upper_bound_name}' found "
554 f"in lfricloop.upper_bound_fortran()")

556 def _halo_read_access(self, arg):
557 '''
558 Determines whether the supplied argument has (or might have) its
559 halo data read within this loop. Returns True if it does, or if
560 it might and False if it definitely does not.
561
562 :param arg: an argument contained within this loop.
563 :type arg: :py:class:`psyclone.dynamo0p3.DynArgument`
564
565 :returns: True if the argument reads, or might read from the \
566 halo and False otherwise.
567 :rtype: bool
568
569 :raises GenerationError: if an unsupported upper loop bound name is \
570 provided for kernels with stencil access.
571 :raises InternalError: if an unsupported field access is found.
572 :raises InternalError: if an unsupported argument type is found.
573
574 '''
575 const = LFRicConstants()
576 if arg.is_scalar or arg.is_operator:
577 # Scalars and operators do not have halos
578 return False
579 if arg.is_field:
580 # This is a field so might read from a halo
581 if arg.access in [AccessType.WRITE]:
582 # This is not a read access
583 return False
584 if arg.access in AccessType.all_read_accesses():
585 # This is a read access
586 if arg.descriptor.stencil:
587 if self._upper_bound_name_upper_bound_name not in ["cell_halo", "ncells"]:
588 raise GenerationError(

628 def _add_field_component_halo_exchange(self, halo_field, idx=None):
629 '''An internal helper method to add the halo exchange call immediately
630 before this loop using the halo_field argument for the
631 associated field information and the optional idx argument if
632 the field is a vector field.
633
634 In certain situations the halo exchange will not be
635 required. This is dealt with by adding the halo exchange,
636 asking it if it is required and then removing it if it is
637 not. This may seem strange but the logic for determining
638 whether a halo exchange is required is within the halo
639 exchange class so it is simplest to do it this way
640
641 :param halo_field: the argument requiring a halo exchange
642 :type halo_field: :py:class:`psyclone.dynamo0p3.DynArgument`
643 :param index: optional argument providing the vector index.
644 :type index: Optional[int]
645
646 :raises InternalError: if there are two forward write
647 dependencies and they are both associated with halo
648 exchanges.
649
650 '''
651 # Avoid circular import
652 # pylint: disable=import-outside-toplevel
653 from psyclone.dynamo0p3 import LFRicHaloExchange
654 exchange = LFRicHaloExchange(halo_field,
655 parent=self.parentparent,
656 vector_index=idx)
657 self.parentparent.children.insert(self.positionpositionposition,
658 exchange)
659
660 # Is this halo exchange required? The halo exchange being
661 # added may replace an existing halo exchange, which would
662 # then be returned as a halo exchange dependence and an
663 # exception raised (as a halo exchange should not have another
664 # halo exchange as a dependence). Therefore, halo exchange
665 # dependencies are ignored here by setting the ignore_hex_dep
666 # optional argument.
667 required, _ = exchange.required(ignore_hex_dep=True)
668 if not required:
669 exchange.detach()
670 else:
671 # The halo exchange we have added may be replacing an
672 # existing one. If so, the one being replaced will be the
673 # first and only write dependence encountered and must be
674 # removed.
675 results = exchange.field.forward_write_dependencies()
676 if results:
677 first_dep_call = results[0].call
678 if isinstance(first_dep_call, HaloExchange):
679 # Sanity check. If the first dependence is a field
680 # accessed within a halo exchange then the
681 # subsequent one must not be.
682 next_results = results[0].forward_write_dependencies()
683 if next_results and any(tmp for tmp in next_results
684 if isinstance(tmp.call,
685 HaloExchange)):
686 raise InternalError(
687 f"When replacing a halo exchange with another one "
688 f"for field {exchange.field.name}, a subsequent "
689 f"dependent halo exchange was found. This should "
690 f"never happen.")
691 first_dep_call.detach()

712 def update_halo_exchanges(self):
713 '''add and/or remove halo exchanges due to changes in the loops
714 bounds'''
715 # this call adds any new halo exchanges that are
716 # required. This is done by adding halo exchanges before this
717 # loop for any fields in the loop that require a halo exchange
718 # and don't already have one
719 self.create_halo_exchanges()
720 # Now remove any existing halo exchanges that are no longer
721 # required. This is done by removing halo exchanges after this
722 # loop where a field in this loop previously had a forward
723 # dependence on a halo exchange but no longer does

739 halo_exchange.detach()

741 def create_halo_exchanges(self):
742 '''Add halo exchanges before this loop as required by fields within
743 this loop. To keep the logic simple we assume that any field
744 that accesses the halo will require a halo exchange and then
745 remove the halo exchange if this is not the case (when
746 previous writers perform sufficient redundant computation). It
747 is implemented this way as the halo exchange class determines
748 whether it is required or not so a halo exchange needs to
749 exist in order to find out. The appropriate logic is coded in
750 the _add_halo_exchange helper method. In some cases a new halo
751 exchange will replace an existing one. In this situation that
752 routine also removes the old one.
753
754 '''
755 for halo_field in self.unique_fields_with_halo_reads():
756 # for each unique field in this loop that has its halo
757 # read (including annexed dofs), find the previous update
758 # of this field
759 prev_arg_list = halo_field.backward_write_dependencies()
760 if not prev_arg_list:
761 # field has no previous dependence so create new halo
762 # exchange(s) as we don't know the state of the fields
763 # halo on entry to the invoke
764 self._add_halo_exchange(halo_field)
765 else:
766 # field has one or more previous dependencies
767 if len(prev_arg_list) > 1:
768 # field has more than one previous dependencies so
769 # should be a vector
770 if halo_field.vector_size <= 1:
771 raise GenerationError(
772 f"Error in create_halo_exchanges. Expecting field "
773 f"'{halo_field.name}' to be a vector as it has "
774 f"multiple previous dependencies")
775 if len(prev_arg_list) != halo_field.vector_size:
776 raise GenerationError(
777 f"Error in create_halo_exchanges. Expecting a "
778 f"dependence for each vector index for field "
779 f"'{halo_field.name}' but the number of "
780 f"dependencies is '{halo_field.vector_size}' and "
781 f"the vector size is '{len(prev_arg_list)}'.")
782 for arg in prev_arg_list:
783 # Avoid circular import
784 # pylint: disable=import-outside-toplevel
785 from psyclone.dynamo0p3 import LFRicHaloExchange
786 if not isinstance(arg.call, LFRicHaloExchange):
787 raise GenerationError(
788 "Error in create_halo_exchanges. Expecting "
789 "all dependent nodes to be halo exchanges")
790 prev_node = prev_arg_list[0].call
791 # Avoid circular import
792 # pylint: disable=import-outside-toplevel
793 from psyclone.dynamo0p3 import LFRicHaloExchange
794 if not isinstance(prev_node, LFRicHaloExchange):
795 # previous dependence is not a halo exchange so
796 # call the add halo exchange logic which
797 # determines whether a halo exchange is required
798 # or not
799 self._add_halo_exchange(halo_field)

985 def independent_iterations(self,
986 test_all_variables=False,
987 signatures_to_ignore=None,
988 dep_tools=None):
989 '''
990 This function is an LFRic-specific override of the default method
991 in the Loop class. It allows domain-specific rules to be applied when
992 determining whether or not loop iterations are independent.
993
994 :param bool test_all_variables: if True, it will test if all variable
995 accesses are independent, otherwise it will stop after the first
996 variable access is found that isn't.
997 :param signatures_to_ignore: list of signatures for which to skip
998 the access checks.
999 :type signatures_to_ignore: Optional[
1000 List[:py:class:`psyclone.core.Signature`]]
1001 :param dep_tools: an optional instance of DependencyTools so that the
1002 caller can access any diagnostic messages detailing why the loop
1003 iterations are not independent.
1004 :type dep_tools: Optional[
1005 :py:class:`psyclone.psyir.tools.DependencyTools`]
1006
1007 :returns: True if the loop iterations are independent, False otherwise.
1008 :rtype: bool
1009
1010 '''
1011 # pylint: disable=import-outside-toplevel
1012 from psyclone.psyir.tools import DependencyTools, DTCode
1013 if not dep_tools:
1014 dtools = DependencyTools()
1015 else:
1016 dtools = dep_tools
1017
1018 if self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type in ["null", "colours"]:
1019 # We know we can't parallelise these loops. ("null" means there
1020 # is no actual loop and "colours" is the *outer* loop over the
1021 # different colours used - it is the inner, "colour" loop over
1022 # cells of a single colour which can be parallelised.)
1023 return False
1024
1025 try:
1026 stat = dtools.can_loop_be_parallelised(
1027 self, test_all_variables=test_all_variables,
1028 signatures_to_ignore=signatures_to_ignore)
1029 if stat:
1030 return True
1031 except (InternalError, KeyError):
1032 # LFRic still has symbols that don't exist in the symbol_table
1033 # until the gen_code() step, so the dependency analysis raises
1034 # errors in some cases.
1035 # TODO #1648 - when a transformation colours a loop we must
1036 # ensure "last_[halo]_cell_all_colours" is added to the symbol
1037 # table.
1038 return True
1039
1040 # The generic DA says that this loop cannot be parallelised. However,
1041 # we use domain-specific information to qualify this.
1042 if self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type == "colour":
1043 # This loop is either over cells of a single colour.
1044 # According to LFRic rules this is safe to parallelise.
1045 return True
1046
1047 if self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type == "dof":
1048 # The generic DA can't see the PSyIR of this Builtin (because it
1049 # hasn't been lowered to language level) so we use
1050 # domain-specific knowledge about its properties.
1051 if self.kernelkernel.is_reduction:
1052 dtools._add_message(
1053 f"Builtin '{self.kernel.name}' performs a reduction",
1054 DTCode.WARN_SCALAR_REDUCTION)
1055 return False
1056 return True
1057
1058 if self.loop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_typeloop_type == "":
1059 # We can parallelise a non-coloured loop if it only updates
1060 # quantities on discontinuous function spaces. If an LFRic kernel
1061 # updates quantities on a continuous function space then it must
1062 # have at least one argument with GH_INC access. Therefore, we
1063 # can simply check whether or not it has such an argument in order
1064 # to infer the continuity of the space.
1065 if self.has_inc_arg():
1066 dtools._add_message(
1067 f"Kernel '{self.kernel.name}' performs an INC update",
1068 DTCode.ERROR_WRITE_WRITE_RACE)
1069 return False
1070 return True
1071
1072 raise InternalError(f"independent_iterations: loop of type "
1073 f"'{self.loop_type}' is not supported.")
