67def _compute_lfric_inner_products(prog, scalars, field_sums, sum_sym):
68 '''
69 Adds PSyIR to a supplied Routine to compute the sum of the inner products
70 of the supplied scalars and fields.
71
72 :param prog: the Routine to which to add PSyIR.
73 :type prog: :py:class:`psyclone.psyir.nodes.Routine`
74 :param scalars: list of pairs of scalars to multiply and sum.
75 :type scalars: List[Tuple[:py:class:`psyclone.psyir.symbols.DataSymbol`,
76 :py:class:`psyclone.psyir.symbols.DataSymbol`]]
77 :param field_sums: the results of all of the inner products of the \
78 various field arguments.
79 :type field_sums: List[:py:class:`psyclone.psyir.symbols.DataSymbol`]
80 :param sum_sym: the symbol into which to accumulate the final sum.
81 :type sum_sym: :py:class:`psyclone.psyir.symbols.DataSymbol`
82
83 '''
84 table = prog.symbol_table
85 idef_sym = table.add_lfric_precision_symbol("i_def")
86 idef_type = ScalarType(ScalarType.Intrinsic.INTEGER, idef_sym)
87
88 # Initialise the sum to zero: sum = 0.0
89 prog.addchild(Assignment.create(Reference(sum_sym),
90 Literal("0.0", sum_sym.datatype)))
91 for scalar in scalars:
92 # Compute the product of the pair of scalars: scalar[0]*scalar[1]
93 # (unless they are boolean).
94 if scalar[0].datatype.intrinsic == ScalarType.Intrinsic.BOOLEAN:
95 continue
96 prod = BinaryOperation.create(BinaryOperation.Operator.MUL,
97 Reference(scalar[0]),
98 Reference(scalar[1]))
99 # Add this product to the sum:
100 # sum = sum + scalar[0]*scalar[1]
101 prog.addchild(
102 Assignment.create(
103 Reference(sum_sym),
104 BinaryOperation.create(BinaryOperation.Operator.ADD,
105 Reference(sum_sym), prod)))
106 for sym in field_sums:
107 if sym.is_scalar:
108 # Add this result of a field inner product to the sum:
109 # sum = sum + field_sum
110 prog.addchild(
111 Assignment.create(
112 Reference(sum_sym),
113 BinaryOperation.create(BinaryOperation.Operator.ADD,
114 Reference(sum_sym),
115 Reference(sym))))
116 else:
117 # For a field vector we have an array of inner-product values that
118 # must be summed.
119 for dim in range(int(sym.datatype.shape[0].lower.value),
120 int(sym.datatype.shape[0].upper.value)+1):
121 # sum = sum + field_sum(dim)
122 add_op = BinaryOperation.create(
123 BinaryOperation.Operator.ADD,
124 Reference(sum_sym),
125 ArrayReference.create(sym,
126 [Literal(str(dim), idef_type)]))
127 prog.addchild(Assignment.create(Reference(sum_sym), add_op))

130def _compute_field_inner_products(routine, field_pairs):
131 '''
132 Constructs the assignments and kernel functors needed to compute the
133 inner products of the supplied list of fields.
134
135 :param routine: the routine to which to add the assignments.
136 :type routine: :py:class:`psyclone.psyir.nodes.Routine`
137 :param field_pairs: the pairs of fields for which we need to compute the \
138 inner products.
139 :type field_pairs: List[ \
140 Tuple[:py:class:`psyclone.psyir.symbols.DataSymbol`, \
141 :py:class:`psyclone.psyir.symbols.DataSymbol`]]
142
143 :returns: the symbols containing the inner products and the list of \
144 kernel functors that computes them.
145 :rtype: Tuple[List[:py:class:`psyclone.psyir.symbols.DataSymbol`], \
146 List[:py:class:`psyclone.domain.lfric.algorithm.KernelFunctor`]]
147
148 :raises TypeError: if any of the supplied symbols are not DataSymbols.
149 :raises InternalError: if any of the supplied pairs of symbols do not \
150 have matching types.
151 :raises InternalError: if any of the supplied symbols are not of the \
152 correct type for an LFRic field.
153
154 '''
155 # pylint: disable=too-many-branches, too-many-locals
156 table = routine.symbol_table
157 rdef_sym = table.add_lfric_precision_symbol("r_def")
158 rdef_type = ScalarType(ScalarType.Intrinsic.REAL, rdef_sym)
159 idef_sym = table.add_lfric_precision_symbol("i_def")
160 idef_type = ScalarType(ScalarType.Intrinsic.INTEGER, idef_sym)
161
162 builtin_factory = LFRicBuiltinFunctorFactory.get()
163
164 field_ip_symbols = []
165 kernel_list = []
166 for sym1, sym2 in field_pairs:
167
168 if not (isinstance(sym1, DataSymbol) and isinstance(sym2, DataSymbol)):
169 raise TypeError(
170 f"Each pair of fields/field-vectors must be supplied as "
171 f"DataSymbols but got: {type(sym1)}, {type(sym2)}")
172
173 if sym1.datatype != sym2.datatype:
174 raise InternalError(
175 f"Cannot compute the inner product of fields '{sym1.name}' "
176 f"and '{sym2.name}' because they are of different types: "
177 f"{sym1.datatype} and {sym2.datatype}")
178
179 if sym1 is sym2:
180 inner_prod_name = f"{sym1.name}_inner_prod"
181 else:
182 inner_prod_name = f"{sym1.name}_{sym2.name}_inner_prod"
183
184 if isinstance(sym1.datatype, DataTypeSymbol):
185 # Create and initialise a variable to hold the result of the
186 # inner product of this pair of fields.
187 ip_sym = table.new_symbol(inner_prod_name,
188 symbol_type=DataSymbol,
189 datatype=rdef_type)
190 # name_inner_prod = 0.0
191 routine.addchild(Assignment.create(Reference(ip_sym),
192 Literal("0.0", rdef_type)))
193 if sym2 is sym1:
194 # Inner product of field with itself.
195 kernel_list.append(
196 builtin_factory.create("x_innerproduct_x", table,
197 [Reference(ip_sym),
198 Reference(sym1)]))
199 else:
200 # Inner product of two different fields.
201 kernel_list.append(
202 builtin_factory.create(
203 "x_innerproduct_y", table,
204 [Reference(ip_sym), Reference(sym1), Reference(sym2)]))
205
206 elif isinstance(sym1.datatype, ArrayType):
207 # This is a pair of field vectors. We compute the inner product of
208 # each component and store the results in an array of the same
209 # length as the field vector.
210
211 # Create the array in which the results will be stored.
212 dtype = ArrayType(rdef_type, sym1.datatype.shape)
213 ip_sym = table.new_symbol(inner_prod_name,
214 symbol_type=DataSymbol,
215 datatype=dtype)
216 for dim in range(int(sym1.datatype.shape[0].lower.value),
217 int(sym1.datatype.shape[0].upper.value)+1):
218 lit = Literal(str(dim), idef_type)
219 # Zero the inner product for this component pair.
220 routine.addchild(Assignment.create(
221 ArrayReference.create(ip_sym,
222 [lit]), Literal("0.0", rdef_type)))
223 if sym2 is sym1:
224 # Inner product of field with itself.
225 kernel_list.append(
226 builtin_factory.create(
227 "x_innerproduct_x", table,
228 [ArrayReference.create(ip_sym, [lit.copy()]),
229 ArrayReference.create(sym1, [lit.copy()])]))
230 else:
231 # Inner product of two different fields.
232 kernel_list.append(
233 builtin_factory.create(
234 "x_innerproduct_y", table,
235 [ArrayReference.create(ip_sym, [lit.copy()]),
236 ArrayReference.create(sym1, [lit.copy()]),
237 ArrayReference.create(sym2, [lit.copy()])]))
238 else:
239 raise InternalError(
240 f"Expected a field symbol to either be of ArrayType or have "
241 f"a type specified by a DataTypeSymbol but found "
242 f"{sym1.datatype} for field '{sym1.name}'")
243
244 # Store the list of symbols holding the various inner products.
245 field_ip_symbols.append(ip_sym)
246
247 return field_ip_symbols, kernel_list

250def _init_fields_random(fields, input_symbols, table):
251 '''
252 Creates a suitable kernel functor for each field that requires
253 initialising with pseudo-random data.
254
255 :param fields: those fields requiring initialisation.
256 :type fields: List[:py:class:`psyclone.psyir.symbols.DataSymbol`]
257 :param input_symbols: map from field name to corresponding field copy.
258 :type input_symbols: Dict[str, \
259 :py:class:`psyclone.psyir.symbols.DataTypeSymbol` | \
260 :py:class:`psyclone.psyir.symbols.ArrayType`]
261 :param table: the symbol table to which to add new symbols.
262 :type table: :py:class:`psyclone.psyir.symbols.SymbolTable`
263
264 :returns: the required kernel calls.
265 :rtype: List[:py:class:`psyclone.domain.common.algorithm.Functor`]
266
267 '''
268 idef_sym = table.add_lfric_precision_symbol("i_def")
269 idef_type = ScalarType(ScalarType.Intrinsic.INTEGER, idef_sym)
270 # We use the setval_random builtin to initialise all fields.
271 kernel_list = []
272 builtin_factory = LFRicBuiltinFunctorFactory.get()
273 for sym in fields:
274 input_sym = input_symbols[sym.name]
275 if isinstance(sym.datatype, DataTypeSymbol):
276 # Initialise the field with pseudo-random numbers.
277 kernel_list.append(
278 builtin_factory.create("setval_random", table,
279 [Reference(sym)]))
280 # Keep a copy of the values in the associated 'input' field.
281 kernel_list.append(builtin_factory.create(
282 "setval_x", table, [Reference(input_sym), Reference(sym)]))
283 elif isinstance(sym.datatype, ArrayType):
284 # Initialise each member of the field vector with pseudo-random
285 # numbers.
286 for dim in range(int(sym.datatype.shape[0].lower.value),
287 int(sym.datatype.shape[0].upper.value)+1):
288 lit = Literal(str(dim), idef_type)
289 # Initialise this component with pseudo-random numbers.
290 kernel_list.append(
291 builtin_factory.create("setval_random", table,
292 [ArrayReference.create(sym,
293 [lit])]))
294 # Keep a copy of the values in the associated 'input' field.
295 kernel_list.append(
296 builtin_factory.create(
297 "setval_x", table,
298 [ArrayReference.create(input_sym, [lit.copy()]),
299 ArrayReference.create(sym, [lit.copy()])]))
300 else:
301 raise InternalError(
302 f"Expected a field symbol to either be of ArrayType or have "
303 f"a type specified by a DataTypeSymbol but found "
304 f"{sym.datatype} for field '{sym.name}'")
305
306 # Return the list of kernel functors.
307 return kernel_list

310def _init_operators_random(operators, table):
311 '''
312 Creates a suitable kernel functor for each operator that requires
313 initialising with pseudo-random data.
314
315 :param fields: those operators requiring initialisation.
316 :type fields: List[:py:class:`psyclone.psyir.symbols.DataSymbol`]
317 :param table: the symbol table to which to add new symbols.
318 :type table: :py:class:`psyclone.psyir.symbols.SymbolTable`
319
320 :returns: the required kernel calls.
321 :rtype: List[:py:class:`psyclone.domain.common.algorithm.Functor`]
322
323 '''
324 # We use the 'setop_random' kernel to do the initialisation. This
325 # is a general-purpose kernel (not a built-in) that is part of the
326 # PSyclone distribution.
327 kname = "setop_random_kernel"
328 kernel_mod = table.new_symbol(kname+"_mod",
329 symbol_type=ContainerSymbol)
330 kernel_sym = table.new_symbol(kname+"_type", symbol_type=DataTypeSymbol,
331 datatype=UnresolvedType(),
332 interface=ImportInterface(kernel_mod))
333 # Create a functor to initialise each operator
334 kernel_list = [LFRicKernelFunctor.create(kernel_sym, [Reference(sym)])
335 for sym in operators]
336
337 # Return the list of kernel functors.
338 return kernel_list

341def _init_scalar_value(scalar_arg, routine, input_symbols):
342 '''
343 Extends the supplied Routine with the necessary statements to initialise
344 the supplied scalar argument. If that argument appears in the
345 `input_symbols` dict then its initial value is also assigned to the Symbol
346 in the dict entry.
347
348 :param scalar_arg: the scalar kernel argument to initialise.
349 :type scalar_arg: :py:class:`psyclone.psyir.symbols.DataSymbol`
350 :param routine: the routine to which to add assignments.
351 :type routine: :py:class:`psyclone.psyir.nodes.Routine`
352 :param input_symbols: dict containing those kernel arguments for which we \
353 need to keep copies of their input values.
354 :type input_symbols: Dict[str, \
355 :py:class:`psyclone.psyir.symbols.DataSymbol`]
356
357 :raises InternalError: if the type of the scalar argument is not supported.
358
359 '''
360 if scalar_arg.datatype.intrinsic == ScalarType.Intrinsic.REAL:
361 # TODO #2087 - we use the RANDOM_NUMBER intrinsic but make no attempt
362 # to scale the resulting value.
363 routine.addchild(
364 IntrinsicCall.create(IntrinsicCall.Intrinsic.RANDOM_NUMBER,
365 [Reference(scalar_arg)]))
366 elif scalar_arg.datatype.intrinsic == ScalarType.Intrinsic.BOOLEAN:
367 # TODO #2087 - just set the variable to False for the moment.
368 routine.addchild(Assignment.create(
369 Reference(scalar_arg),
370 Literal("false", LFRicTypes("LFRicLogicalScalarDataType")())))
371 elif scalar_arg.datatype.intrinsic == ScalarType.Intrinsic.INTEGER:
372 # TODO #2087 - just set the variable to 1 for the moment.
373 routine.addchild(Assignment.create(
374 Reference(scalar_arg),
375 Literal("1", LFRicTypes("LFRicIntegerScalarDataType")())))
376 else:
377 raise InternalError(
378 f"_init_scalar_value: only scalars of REAL, INTEGER or BOOLEAN "
379 f"type are supported but got symbol '{scalar_arg.name}' of type "
380 f"'{scalar_arg.datatype}'.")
381
382 if scalar_arg.name in input_symbols:
383 # We need to keep a copy of the input value of this argument.
384 input_sym = input_symbols[scalar_arg.name]
385 routine.addchild(Assignment.create(Reference(input_sym),
386 Reference(scalar_arg)))

389def _validate_geom_arg(kern, arg_idx, name, valid_spaces, vec_len):
390 '''
391 Check that the argument at the supplied index is consistent with the
392 properties of the field that it is supposed to represent.
393
394 :param kern: the kernel under consideration.
395 :type kern: :py:class:`psyclone.domain.lfric.LFRicKern`
396 :param in arg_idx: the 1-indexed position of the argument in the list \
397 defined in the kernel metadata.
398 :param str name: the name of the argument that we are expecting.
399 :param List[str] valid_spaces: the function spaces that the argument is \
400 permitted to be on.
401 :param int vec_len: the expected vector length of the argument.
402
403 :raises ValueError: if any of the properties of the specified kernel \
404 argument are inconsistent with the supplied values.
405 '''
406 num_metadata_args = len(kern.arg_descriptors)
407 if arg_idx < 1 or arg_idx > num_metadata_args:
408 raise ValueError(
409 f"The supplied LFRic TL kernel '{kern.name}' has "
410 f"{num_metadata_args} arguments specified in its metadata. "
411 f"Therefore, the index of the argument containing the "
412 f"'{name}' field must be between 1 and {num_metadata_args} "
413 f"(inclusive) but got {arg_idx}.")
414 # Check that the specified argument is of the correct type.
415 descriptor = kern.arg_descriptors[arg_idx-1]
416 if descriptor.argument_type != 'gh_field':
417 raise ValueError(
418 f"The '{name}' argument is expected to be a field but argument "
419 f"{arg_idx} to kernel '{kern.name}' is a "
420 f"'{descriptor.argument_type}'")
421 if descriptor.function_space not in valid_spaces:
422 raise ValueError(
423 f"The '{name}' field argument to kernel '{kern.name}' is expected "
424 f"to be on one of the {valid_spaces} spaces but the argument at "
425 f"the specified position ({arg_idx}) is on the "
426 f"'{descriptor.function_space}' space.")
427 if descriptor.vector_size != vec_len:
428 if vec_len > 1:
429 raise ValueError(
430 f"The '{name}' field argument to kernel '{kern.name}' is "
431 f"expected to be a field vector of length {vec_len} but the "
432 f"argument at the specified position ({arg_idx}) has a length "
433 f"of {descriptor.vector_size}.")
434 raise ValueError(
435 f"The '{name}' field argument to kernel '{kern.name}' is expected "
436 f"to be a field but the argument at the specified position "
437 f"({arg_idx}) is a field vector of length "
438 f"{descriptor.vector_size}.")

441def _lfric_create_real_comparison(sym_table, kernel, var1, var2):
442 '''Creates PSyIR that checks the values held by Symbols var1 and var2
443 for equality, allowing for machine precision and writes the
444 success or failure of the checks to the LFRic-specific logging
445 API. The generic version that writes to stdout can be found in
446 psyad/domain/common/adjoint_utils.py
447
448 :param sym_table: the SymbolTable in which to put new Symbols.
449 :type sym_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
450 :param kernel: the routine for which this adjoint test is being performed.
451 :type kernel: :py:class:`psyclone.psyir.nodes.Routine`
452 :param var1: the symbol holding the first value for the comparison.
453 :type var1: :py:class:`psyclone.psyir.symbols.DataSymbol`
454 :param var2: the symbol holding the second value for the comparison.
455 :type var2: :py:class:`psyclone.psyir.symbols.DataSymbol`
456
457 :returns: the PSyIR nodes that perform the check and write its
458 success or failure to the LFRic-specific logging API.
459 :rtype: List[:py:class:`psyclone.psyir.nodes.Node`]
460
461 '''
462 statements = []
463 statements.extend(common_real_comparison(sym_table, var1, var2))
464 statements.extend(_lfric_log_write(sym_table, kernel, var1, var2))
465 return statements

468def _lfric_log_write(sym_table, kernel, var1, var2):
469 '''Creates PSyIR that writes whether the precision test passed or
470 failed to the LFRic-specific logging API.
471
472 :param sym_table: the SymbolTable in which to read existing Symbols.
473 :type sym_table: :py:class:`psyclone.psyir.symbols.SymbolTable`
474 :param kernel: the routine for which this adjoint test is being performed.
475 :type kernel: :py:class:`psyclone.psyir.nodes.Routine`
476 :param var1: the symbol holding the first value for the comparison.
477 :type var1: :py:class:`psyclone.psyir.symbols.DataSymbol`
478 :param var2: the symbol holding the second value for the comparison.
479 :type var2: :py:class:`psyclone.psyir.symbols.DataSymbol`
480
481 :returns: PSyIR nodes that write out test success or failure to
482 the LFRic logging API.
483 :rtype: List[:py:class:`psyclone.psyir.nodes.Node`]
484
485 '''
486 freader = FortranReader()
487 statements = []
488 rel_diff = sym_table.lookup_with_tag("relative_diff")
489 overall_tol = sym_table.lookup_with_tag("overall_tolerance")
490 log_mod = sym_table.find_or_create("log_mod", symbol_type=ContainerSymbol)
491 log_scratch = sym_table.find_or_create(
492 "log_scratch_space", symbol_type=DataSymbol, datatype=UnresolvedType(),
493 interface=ImportInterface(log_mod))
494 log_level_info = sym_table.new_symbol(
495 "log_level_info", symbol_type=DataSymbol, datatype=UnresolvedType(),
496 interface=ImportInterface(log_mod))
497 log_level_error = sym_table.new_symbol(
498 "log_level_error", symbol_type=DataSymbol, datatype=UnresolvedType(),
499 interface=ImportInterface(log_mod))
500 log_event = sym_table.new_symbol(
501 "log_event", symbol_type=RoutineSymbol,
502 interface=ImportInterface(log_mod))
503 write_pass = freader.psyir_from_statement(
504 f"WRITE({log_scratch.name}, *) \"PASSED {kernel.name}:\", "
505 f"{var1.name}, {var2.name}, {rel_diff.name}", sym_table)
506 write_fail = freader.psyir_from_statement(
507 f"WRITE({log_scratch.name}, *) \"FAILED {kernel.name}:\", "
508 f"{var1.name}, {var2.name}, {rel_diff.name}", sym_table)
509 log_pass = freader.psyir_from_statement(
510 f"call {log_event.name}( {log_scratch.name}, {log_level_info.name} )",
511 sym_table)
512 log_fail = freader.psyir_from_statement(
513 f"call {log_event.name}( {log_scratch.name}, {log_level_error.name} )",
514 sym_table)
515
516 statements.append(
517 IfBlock.create(BinaryOperation.create(BinaryOperation.Operator.LT,
518 Reference(rel_diff),
519 Reference(overall_tol)),
520 [write_pass, log_pass], [write_fail, log_fail]))
521
522 return statements

525def generate_lfric_adjoint_harness(tl_psyir, coord_arg_idx=None,
526 panel_id_arg_idx=None,
527 test_name="adjoint_test"):
528 '''
529 Constructs and returns the PSyIR for a Container and Routine that
530 implements a test harness for the adjoint of the supplied tangent-linear
531 kernel. The base name to use for the Container and Routine is given by
532 the test_name argument.
533
534 :param tl_psyir: the PSyIR of an LFRic module defining a \
535 tangent-linear kernel.
536 :type tl_psyir: :py:class:`psyclone.psyir.nodes.Container`
537 :param Optional[int] coord_arg_idx: 1-indexed position of the coordinate \
538 field in the list of arguments in the kernel metadata (if present).
539 :param Optional[int] panel_id_arg_idx: 1-indexed position of the panel-id \
540 field in the list of arguments in the kernel metadata (if present).
541 :param Optional[str] test_name: Name of the adjoint test algorithm \
542 (if present).
543
544 :returns: PSyIR of an Algorithm that tests the adjoint of the supplied \
545 LFRic TL kernel.
546 :rtype: :py:class:`psyclone.psyir.nodes.Container`
547
548 :raises ValueError: if the supplied PSyIR does not have a Container (that \
549 is *not* a FileContainer).
550 :raises ValueError: if the name of the Container (module) in the supplied \
551 PSyIR does not follow the LFRic naming convention of ending in '_mod'.
552
553 '''
554 # pylint: disable=too-many-statements, too-many-locals, too-many-branches
555 tl_container = find_container(tl_psyir)
556 if not tl_container:
557 raise ValueError(
558 f"Test harness code can only be generated if the supplied TL "
559 f"kernel is within a module (Container) but the supplied PSyIR "
560 f"does not have a Container node:\n{tl_psyir.view(colour=False)}")
561
562 lfalg = LFRicAlg()
563 # Variable test_name is validated inside create_alg_routine.
564 container = lfalg.create_alg_routine(test_name)
565 routine = container.walk(Routine)[0]
566 table = routine.symbol_table
567
568 tl_subroutine = tl_container.children[0]
569 tl_subroutine_table = tl_subroutine.symbol_table
570 tl_argument_list = tl_subroutine_table.argument_list
571
572 # Parse the kernel metadata. This still uses fparser1 as that's what
573 # the meta-data handling is currently based upon. We therefore have to
574 # convert back from PSyIR to Fortran for the moment.
575 # TODO #1806 - replace this with the new PSyIR-based metadata handling.
576 # pylint: disable=import-outside-toplevel
577 from psyclone.psyir.backend.fortran import FortranWriter
578 writer = FortranWriter()
579 tl_source = writer(tl_container)
580 parse_tree = fpapi.parse(tl_source)
581
582 # Get the name of the module that contains the kernel and create a
583 # ContainerSymbol for it.
584 kernel_mod_name = tl_container.name.lower()
585 if not kernel_mod_name.endswith("_mod"):
586 raise ValueError(
587 f"The supplied LFRic TL kernel is contained within a module named "
588 f"'{kernel_mod_name}'. This does not end in '_mod' and as such "
589 f"does not comply with the LFRic naming convention.")
590
591 kernel_mod = table.new_symbol(kernel_mod_name, symbol_type=ContainerSymbol)
592 # Assume the LFRic naming convention is followed in order to infer the name
593 # of the TL kernel. (If this convention isn't followed in the supplied code
594 # then the call to `kernel_from_metadata` below will raise an appropriate
595 # exception.)
596 kernel_name = kernel_mod_name.replace("_mod", "_type")
597
598 adj_mod = table.new_symbol(create_adjoint_name(kernel_mod_name),
599 symbol_type=ContainerSymbol)
600 kernel_routine = table.new_symbol(kernel_name,
601 symbol_type=DataTypeSymbol,
602 datatype=UnresolvedType(),
603 interface=ImportInterface(kernel_mod))
604 adj_routine = table.new_symbol(create_adjoint_name(kernel_name),
605 symbol_type=DataTypeSymbol,
606 datatype=UnresolvedType(),
607 interface=ImportInterface(adj_mod))
608
609 # Construct a LFRicKern using the metadata and then use it to construct
610 # the kernel argument list.
611 # TODO #1806 - once we have the new PSyIR-based metadata handling then
612 # we can pass PSyIR to this routine rather than an fparser1 parse tree.
613 kern = lfalg.kernel_from_metadata(parse_tree, kernel_name)
614
615 # Replace generic names for fields. operators etc generated in
616 # LFRicKern with the scientific names used by the tangent-linear
617 # kernel. This makes the harness code more readable. Changing the
618 # names in-place within LFRicKern is the neatest solution given that
619 # this is a legacy structure.
620
621 # First raise the tangent-linear kernel PSyIR to LFRic PSyIR. This
622 # gives us access to the kernel metadata.
623 kern_trans = RaisePSyIR2LFRicKernTrans()
624 kern_trans.apply(tl_psyir, options={"metadata_name": kernel_name})
625 metadata = tl_psyir.children[0].metadata
626 # Use the metadata to determine the mapping from a metadata
627 # meta_arg index to the kernel argument index. Note, the meta_arg
628 # index corresponds to the order of the arguments stored in
629 # LFRicKern.
630 index_map = ArgIndexToMetadataIndex.mapping(metadata)
631 inv_index_map = {value: key for key, value in index_map.items()}
632
633 # For each kernel argument, replace the generic name with the
634 # scientific name used in the tangent-linear code.
635 for idx, arg in enumerate(kern.arguments.args):
636 tl_arg_idx = inv_index_map[idx]
637 # pylint: disable=protected-access
638 arg._name = tl_argument_list[tl_arg_idx].name
639 # pylint: enable=protected-access
640
641 kern_args = lfalg.construct_kernel_args(routine, kern)
642
643 # Validate the index values for the coordinate and face_id fields if
644 # supplied.
645 geometry_arg_indices = []
646
647 #: Properties of the various 'geometry' arguments that an LFRic kernel can
648 #: accept. Used to validate that the arguments specified by the user have
649 #: the expected properties.
650 _geometry_arg_mapping = {
651 "coordinate": {
652 "valid_spaces": ["wchi"],
653 "vector_len": 3
654 },
655 "panel-id": {
656 "valid_spaces": LFRicConstants().VALID_DISCONTINUOUS_NAMES,
657 "vector_len": 1
658 }
659 }
660
661 if coord_arg_idx is not None:
662 _validate_geom_arg(kern, coord_arg_idx, "coordinate",
663 _geometry_arg_mapping["coordinate"]["valid_spaces"],
664 _geometry_arg_mapping["coordinate"]["vector_len"])
665 # Convert to 0-indexed
666 coord_arg_idx -= 1
667 geometry_arg_indices.append(coord_arg_idx)
668 if panel_id_arg_idx is not None:
669 # Check that the specified argument is of the correct type.
670 _validate_geom_arg(kern, panel_id_arg_idx, "panel-id",
671 _geometry_arg_mapping["panel-id"]["valid_spaces"],
672 _geometry_arg_mapping["panel-id"]["vector_len"])
673 # Convert to 0-indexed
674 panel_id_arg_idx -= 1
675 geometry_arg_indices.append(panel_id_arg_idx)
676
677 # Create symbols that will store copies of the inputs to the TL kernel.
678 # We don't need to do this for operators since they are never 'active'.
679 field_args = [fsym for fsym, _ in kern_args.fields]
680 scalar_and_field_args = kern_args.scalars + field_args
681 # Double check that there aren't any operator arguments that are written to
682 for op_sym, _, _ in kern_args.operators:
683 idx = kern_args.arglist.index(op_sym.name)
684 if kern.arguments.args[idx].access != AccessType.READ:
685 raise GenerationError(
686 f"Operator argument '{op_sym.name}' to TL kernel "
687 f"'{kernel_name}' is written to. This is not supported.")
688
689 input_symbols = {}
690 for sym in scalar_and_field_args:
691 idx = kern_args.arglist.index(sym.name)
692 if (kern_args.metadata_index_from_actual_index(idx) in
693 geometry_arg_indices):
694 # This kernel argument is not modified by the test harness so we
695 # don't need to keep a copy of it.
696 continue
697 input_sym = table.new_symbol(
698 f"{sym.name}_input", symbol_type=DataSymbol,
699 datatype=UnresolvedType())
700 input_sym.copy_properties(sym)
701 input_symbols[sym.name] = input_sym
702
703 # Initialise all input field objects unless they contain geometric
704 # information (as specified by the user via command-line arguments).
705 kernel_input_arg_list = []
706 for sym, space in kern_args.fields:
707 idx = kern_args.arglist.index(sym.name)
708 if (kern_args.metadata_index_from_actual_index(idx) in
709 geometry_arg_indices):
710 continue
711 # This kernel argument is not one that is passed through from the
712 # Algorithm layer.
713 lfalg.initialise_field(routine, input_symbols[sym.name], space)
714 kernel_input_arg_list.append(sym)
715
716 # Initialise argument values and keep copies.
717
718 # Scalars.
719 for sym in kern_args.scalars:
720 idx = kern_args.arglist.index(sym.name)
721 if (kern_args.metadata_index_from_actual_index(idx) in
722 geometry_arg_indices):
723 # This kernel argument is not modified by the test harness
724 # because it contains geometry information.
725 continue
726 _init_scalar_value(sym, routine, input_symbols)
727
728 # Fields.
729 kernel_list = _init_fields_random(kernel_input_arg_list, input_symbols,
730 table)
731
732 # Operators.
733 kernel_list.extend(_init_operators_random(
734 [sym for sym, _, _ in kern_args.operators], table))
735
736 # Finally, add the kernel itself to the list for the invoke().
737 arg_nodes = []
738 for idx, arg in enumerate(kern_args.arglist):
739 # Check whether this argument contains geometric information that
740 # is passed through from the Algorithm layer.
741 mdata_idx = kern_args.metadata_index_from_actual_index(idx)
742 if mdata_idx is not None and mdata_idx == coord_arg_idx:
743 # Replace the argument with that containing the
744 # coordinate field.
745 sym = table.lookup_with_tag("coord_field")
746 elif mdata_idx is not None and mdata_idx == panel_id_arg_idx:
747 # Replace the argument with that containing the panel-id field.
748 sym = table.lookup_with_tag("panel_id_field")
749 else:
750 # This argument isn't special so use the existing symbol.
751 sym = table.lookup(arg)
752 # Create the reference to the selected symbol and add it to the list
753 # of arguments for the kernel functor.
754 arg_nodes.append(Reference(sym))
755 kern = LFRicKernelFunctor.create(kernel_routine, arg_nodes)
756 kernel_list.append(kern)
757
758 # Compute the inner products of the results of the TL kernel. We exclude
759 # any fields passed through (unmodified) from the Algorithm layer as well
760 # as any operators.
761 fld_pairs = []
762 for sym, _ in kern_args.fields:
763 if sym in kernel_input_arg_list:
764 fld_pairs.append((sym, sym))
765 field_ip_symbols, ip_kernels = _compute_field_inner_products(routine,
766 fld_pairs)
767 kernel_list.extend(ip_kernels)
768
769 # Create the 'call invoke(...)' for the list of kernels.
770 invoke_sym = table.new_symbol("invoke", symbol_type=RoutineSymbol)
771 inv_call = LFRicAlgorithmInvokeCall.create(invoke_sym,
772 kernel_list, 0)
773 inv_call.preceding_comment = (
774 "Initialise arguments and call the tangent-linear kernel.")
775 routine.addchild(inv_call)
776
777 rdef_sym = table.add_lfric_precision_symbol("r_def")
778 rdef_type = ScalarType(ScalarType.Intrinsic.REAL, rdef_sym)
779
780 # Compute the first inner products.
781 inner1_sym = table.new_symbol("inner1", symbol_type=DataSymbol,
782 datatype=rdef_type)
783
784 scalars = zip(kern_args.scalars, kern_args.scalars)
785 _compute_lfric_inner_products(routine, scalars, field_ip_symbols,
786 inner1_sym)
787
788 # Construct the functor for the adjoint kernel. We pass it the same
789 # arguments as the TL kernel.
790 adj_kern = LFRicKernelFunctor.create(adj_routine,
791 [arg.copy() for arg in arg_nodes])
792 # Compute the inner product of its outputs with the original inputs.
793 fld_pairs = []
794 for sym, _ in kern_args.fields:
795 if sym in kernel_input_arg_list:
796 fld_pairs.append((sym, input_symbols[sym.name]))
797 field_ip_symbols, kernel_list = _compute_field_inner_products(routine,
798 fld_pairs)
799 kernel_list = [adj_kern] + kernel_list
800
801 # Create the 'call invoke(...)' for the list of kernels.
802 routine.addchild(LFRicAlgorithmInvokeCall.create(
803 invoke_sym, kernel_list, 1))
804
805 # Sum up the second set of inner products
806 inner2_sym = table.new_symbol("inner2", symbol_type=DataSymbol,
807 datatype=rdef_type)
808 scalars = []
809 for sym in kern_args.scalars:
810 if sym.name in input_symbols:
811 # This scalar is an input so include it in the inner product.
812 scalars.append((sym, input_symbols[sym.name]))
813 _compute_lfric_inner_products(routine, scalars, field_ip_symbols,
814 inner2_sym)
815
816 # Finally, compare the two inner products with an LFRic-specific routine.
817 stmts = _lfric_create_real_comparison(table, kern, inner1_sym, inner2_sym)
818 for stmt in stmts:
819 routine.addchild(stmt)
820
821 return container
