47class ArgIndexToMetadataIndex(MetadataToArgumentsRules):
48 '''Provides a mapping from the index of a kernel argument to the index
49 of its associated meta_arg kernel metadata. Mappings are not
50 provided to arguments that do not have associated meta_arg kernel
51 metadata.
52
53 '''
54
55 @classmethod
56 def _initialise(cls, _):
57 '''Initial state (for the _info variable) is not used by this subclass
58 i.e. the optional info argument to the mapping method is not
59 used. Instead, initial state is always set to an empty
60 dictionary. Also initialises any additional state for this
61 class.
62
63 '''
64 # We could use super()._initialise({}) here but it is simpler
65 # to just set the value of _info directly and it avoids pylint
66 # complaining.
67 cls._info_info = {}
68 # Initialise the local _index variable. This is used to keep
69 # track of the current argument index.
70 cls._index = 0
71
72 @classmethod
73 def _scalar(cls, meta_arg):
74 '''Argument providing an LFRic scalar value.
75
76 :param meta_arg: the metadata associated with this scalar argument.
77 :type meta_arg: \
78 :py:class:`psyclone.domain.lfric.kernel.ScalarArgMetadata`
79
80 '''
81 cls._add_arg(meta_arg)
82
83 @classmethod
84 def _field(cls, meta_arg):
85 '''Argument providing an LFRic field.
86
87 :param meta_arg: the metadata associated with this field argument.
88 :type meta_arg: \
89 :py:class:`psyclone.domain.lfric.kernel.FieldArgMetadata`
90
91 '''
92 cls._add_arg(meta_arg)
93
94 @classmethod
95 def _field_vector(cls, meta_arg):
96 '''Arguments providing the components of an LFRic field vector.
97
98 :param meta_arg: the metadata associated with this field \
99 vector argument.
100 :type meta_arg: \
101 :py:class:`psyclone.domain.lfric.kernel.FieldVectorArgMetadata`
102
103 '''
104 for _ in range(int(meta_arg.vector_length)):
105 cls._add_arg(meta_arg)
106
107 @classmethod
108 def _operator(cls, meta_arg):
109 '''Arguments providing an LMA operator.
110
111 :param meta_arg: the metadata associated with the operator \
112 arguments.
113 :type meta_arg: \
114 :py:class:`psyclone.domain.lfric.kernel.OperatorArgMetadata`
115
116 '''
117 cls._index += 1 # ncell_3d
118 cls._add_arg(meta_arg)
119
120 @classmethod
121 def _cma_operator(cls, meta_arg):
122 '''Arguments providing a columnwise operator.
123
124 :param meta_arg: the metadata associated with the CMA operator \
125 arguments.
126 :type meta_arg: :py:class:`psyclone.domain.lfric.kernel.\
127 ColumnwiseOperatorArgMetadata`
128
129 '''
130 cls._add_arg(meta_arg)
131 cls._index += 1 # nrow
132 if meta_arg.function_space_to != meta_arg.function_space_from:
133 cls._index += 1 # ncol
134 cls._index += 5 # bandwidth, alpha, beta, gamma_m, gamma_p
135
136 @classmethod
137 def _add_arg(cls, meta_arg):
138 '''Utility method to add a mapping (to the _info dictionary) from the
139 current argument index (_index) to the index of the supplied
140 meta_arg metadata argument.
141
142 :param meta_arg: the current meta_arg metadata.
143 :type meta_arg: \
144 :py:class:`psyclone.domain.lfric.kernel.CommonMetaArgMetadata`
145
146 '''
147 meta_arg_index = cls._metadata.meta_args.index(meta_arg)
148 cls._info_info[cls._index] = meta_arg_index
149 cls._index += 1
150
151 @classmethod
152 def _cell_position(cls):
153 '''A cell position argument.'''
154 cls._index += 1
155
156 @classmethod
157 def _mesh_height(cls):
158 '''A mesh height argument.'''
159 cls._index += 1
160
161 @classmethod
162 def _mesh_ncell2d_no_halos(cls):

164 halos.

167 cls._index += 1

169 @classmethod
170 def _mesh_ncell2d(cls):
171 '''Argument providing the number of columns in the mesh including
172 halos.
173
174 '''
175 cls._index += 1
176
177 @classmethod
178 def _cell_map(cls):
179 '''Arguments providing a mapping from coarse to fine mesh for the
180 current column.
181
182 '''
183 cls._index += 4

266 cls._index += 1
