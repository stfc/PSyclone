38'''
39This module provides the ACCUpdateTrans transformation that, on programs that
40use accelerator devices without managed memory support, ensures that data is
41kept up-to-date on the host for the execution of host code and that data
42is returned to the device in time for the execution of compute regions.
43'''

58class ACCUpdateTrans(Transformation):
59 '''
60 Examines the supplied Schedule and adds OpenACC "update" directives
61 for any data accessed outside of a kernels or parallel region.
62 For example:
63
64 >>> from psyclone.psyir.backend.fortran import FortranWriter
65 >>> from psyclone.psyir.frontend.fortran import FortranReader
66 >>> from psyclone.psyir.nodes import Routine
67 >>> from psyclone.psyir.transformations import ACCUpdateTrans
68 >>>
69 >>> code = """
70 ... subroutine run_it()
71 ... real :: a(10)
72 ... a(:) = 7.0
73 ... end subroutine run_it"""
74 >>>
75 >>> psyir = FortranReader().psyir_from_source(code)
76 >>> routine = psyir.walk(Routine)[0]
77 >>>
78 >>> # Add update directives
79 >>> uptrans = ACCUpdateTrans()
80 >>> uptrans.apply(routine)
81 >>>
82 >>> # Uncomment to see a text view of the new routine schedule
83 >>> # print(routine.view())
84 >>> print(FortranWriter()(routine))
85 subroutine run_it()
86 real, dimension(10) :: a
87 <BLANKLINE>
88 !$acc update if_present host(a)
89 a(:) = 7.0
90 !$acc update if_present device(a)
91 <BLANKLINE>
92 end subroutine run_it
93 <BLANKLINE>
94
95 '''
96 # Tuple of OpenACC directives we ignore when traversing a schedule.
97 _ACC_IGNORE = (ACCEnterDataDirective, )
98 # Tuple of OpenACC compute directives delimiting possible device execution.
99 _ACC_COMPUTE = (ACCParallelDirective, ACCKernelsDirective)

101 def validate(self, node, options=None):
102 '''
103 Checks that it is valid to apply this transformation to the supplied
104 schedule.
105
106 :param node: the Schedule that is to be transformed.
107 :type node: :py:class:`psyclone.psyir.nodes.Schedule`
108 :param options: any options to this transformation.
109 :type options: Optional[Dict[str, Any]]
110
111 :raises TransformationError: if the supplied node is not a Schedule.
112 :raises TransformationError: if the supplied node is within \
113 an OpenACC region.
114 '''
115 if not isinstance(node, Schedule):
116 raise TransformationError(f"Expected a Schedule but got a node of "
117 f"type '{type(node).__name__}'")
118
119 if node.ancestor(self._ACC_COMPUTE):
120 raise TransformationError(
121 "Cannot apply the ACCUpdateTrans to nodes that are within "
122 "an OpenACC compute region.")
123
124 super().validate(node, options)

126 def apply(self, node, options=None):
127 '''
128 Applies this transformation to the supplied Schedule. Identifies any
129 regions of code outside of OpenACC regions and adds the necessary
130 OpenACC update directives to ensure that the host and device copies of
131 any variables are kept up-to-date.
132
133 :param node: the Schedule that is to be transformed.
134 :type node: :py:class:`psyclone.psyir.nodes.Schedule`
135 :param options: any options to this transformation.
136 :type options: Optional[Dict[str, Any]]
137
138 '''
139 self.validatevalidate(node, options)
140
141 # Call the routine that recursively adds updates to all Schedules
142 # within the supplied Schedule.
143 self._add_updates_to_schedule(node)

145 @staticmethod
146 def _may_compute(node):
147 '''This method returns True if the supplied node is a Call node. This
148 is because a Call node may call a routine whose body (or part
149 of its body) executes on the device. For consistency with
150 earlier implementations it also returns True if the supplied
151 node is an allocate or deallocate intrinsic.
152
153 :param node: a PSyIR node.
154 :type node: :py:class:`psyclone.psyir.nodes.Node`
155
156 :returns: True if the supplied node is a Call node or an \
157 allocate or deallocate intrinsic.
158 :rtype: bool
159
160 '''
161 # pylint: disable=unidiomatic-typecheck
162 if type(node) is Call or (
163 isinstance(node, IntrinsicCall) and node.routine.name in
164 ["ALLOCATE", "DEALLOCATE"]):
165 return True
166 return False

168 def _add_updates_to_schedule(self, sched):
169 '''
170 Recursively identify those statements that are not being executed
171 on the device and add any required OpenACC update directives.
172
173 :param sched: the schedule of statements to process.
174 :type sched: :py:class:`psyclone.psyir.nodes.Schedule`
175
176 '''
177 # We must walk through the Schedule and find those nodes
178 # representing contiguous regions of code that are not
179 # executed on the device. Any Call nodes are taken as
180 # boundaries of such regions because it may be that (part of)
181 # their bodies are executed on the device. Allocate and
182 # deallocate intrinsics are also taken as boundaries of such
183 # regions.
184 node_list = []
185 for child in sched[:]:
186 if isinstance(child, self._ACC_IGNORE):
187 continue
188 found = False
189 for node in child.walk(Node):
190 if isinstance(node, self._ACC_COMPUTE) or \
191 self._may_compute(node):
192 found = True
193 break
194 if not found:
195 node_list.append(child)
196 else:
197 self._add_update_directives(node_list)
198 node_list.clear()
199 if self._may_compute(child):
200 # Conservatively add an update host statement just
201 # before the node.
202 self._add_update_directives([child])
203 elif isinstance(child, IfBlock):
204 # Add any update host statements that are required due to
205 # (read) accesses within the condition of the If.
206 self._add_update_directives([child.condition])
207 # Recurse down into the if body
208 self._add_updates_to_schedule(child.if_body)
209 if child.else_body:
210 self._add_updates_to_schedule(child.else_body)
211 elif isinstance(child, Loop):
212 # A loop on the host with code that may be on the device.
213 # The loop start, stop and step values could all
214 # potentially have been written to on the device.
215 self._add_update_directives([child.start_expr,
216 child.stop_expr,
217 child.step_expr])
218 # Recurse down into the loop body
219 self._add_updates_to_schedule(child.loop_body)
220
221 # We've reached the end of the list of children - are there any
222 # last nodes that represent computation on the host?
223 self._add_update_directives(node_list)

226 # pylint: disable=too-many-branches
227 def _add_update_directives(self, node_list):
228 # pylint: disable=too-many-locals, too-many-branches
229 '''
230 Adds the required OpenACC update directives before and after the nodes
231 in the supplied list.
232
233 :param node_list: the PSyIR nodes representing code executed on the \
234 host rather than the device.
235 :type node_list: List[:py:class:`psyclone.psyir.nodes.Node`]
236
237 '''
238 if not node_list:
239 return
240
241 # TODO #1872: the lack of precise array access descriptions might
242 # unnecessarily increase the data transfer volume.
243 read_write_info = CallTreeUtils().get_in_out_parameters(node_list)
244 inputs = set(read_write_info.signatures_read)
245 outputs = set(read_write_info.signatures_written)
246
247 # TODO #1872: as a workaround for the lack of precise array access
248 # descriptions, we currently overapproximate dependencies by adding any
249 # variables being written to the input set (in addition to the output
250 # set). The aim is to avoid situations where the host only writes an
251 # array slice and the respective update device directive would then
252 # incorrectly overwrite the entire array on the device, including the
253 # data outside the slice. If or when this is no longer needed, the step
254 # of the algorithm below dealing with if stmts will need updating.
255 inputs.update(outputs)
256
257 # TODO #1732: for CodeBlock nodes, until we have semantic information
258 # on the statements involved, we conservatively assume every referenced
259 # variable is both an input and an output.
260 # TODO #1872: CodeBlock.get_symbol_names might include function names
261 # mistaken for array names. However, as these do not correspond to any
262 # arrays, the compiler should be able to ignore them in the directive.
263 for node in node_list:
264 if isinstance(node, CodeBlock):
265 for symbol_name in node.get_symbol_names():
266 inputs.add(Signature(symbol_name))
267 outputs.add(Signature(symbol_name))
268
269 # Copy any data that is accessed by this host region to the host
270 # (resp. device) if it is on the device (resp. host).
271 # In this function, IN and OUT are used to determine the access mode of
272 # data references on the host, as per the value of mode.
273 for mode, host_sig in enumerate((inputs, outputs)):
274 if mode == IN:
275 node_index, node_offset = 0, 0
276 elif mode == OUT:
277 node_index, node_offset = -1, 1
278
279 child = node_list[node_index]
280 sched = child.parent
281 # Since the supplied nodes may be the children of an IfBlock or a
282 # Loop, we get the deepest ancestor that is a child of a Schedule.
283 while not isinstance(sched, Schedule):
284 child, sched = child.parent, sched.parent
285
286 # While there are variables from this host region which have not
287 # been placed in an update directive, we check if they need to be
288 # placed at the current schedule, in which case they are, or
289 # if we can bring them up to the next deepest ancestor schedule.
290 # The main goal is to avoid placing update directives inside loop
291 # statements, but the algorithm also tries to minimise the total
292 # number of update directives by promoting the coalescence of
293 # neighbouring directives.
294 while host_sig:
295 # Most conservative position for a new update directive at this
296 # schedule, i.e. that closest to the host region. This is used
297 # as a boundary to compute textual dependencies, i.e. between
298 # a pair of statements not within a loop or the same iteration
299 # of a loop, and loop-carried dependencies, i.e. between a pair
300 # of statements on different iterations of a loop.
301 # It is also the position for the update directive used by
302 # those variables with textual dependencies.
303 # Perhaps, eventually, specially if we adopt the async clause,
304 # it may be beneficial to textually move update host directives
305 # earlier within the schedule as much as legally possible to
306 # leverage overlapping communication and computation.
307 update_pos = sched.children.index(child) + node_offset
308
309 if mode == IN:
310 beg, end = None, update_pos
311 elif mode == OUT:
312 beg, end = update_pos, None
313
314 # The inability to place variables in update directives in
315 # outer schedules is determined by their involvement in textual
316 # or loop-carried dependencies in the current schedule. First,
317 # we find all statements which may incur data dependencies.
318 text_dep_stmts = sched[beg:end]
319 loop_dep_stmts = sched[end:beg]
320 # Second, we find which of the variables in the host region
321 # require synchronisation at the current schedule, i.e. those
322 # that are used in the compute regions within those statements.
323 text_sync = self._sig_to_sync(text_dep_stmts, host_sig, mode)
324 loop_sync = self._sig_to_sync(loop_dep_stmts, host_sig, mode)
325
326 # Those variables that require both textual and, if within a
327 # loop, also loop-carried synchronisation, need only be updated
328 # once and can be dropped from the latter set.
329 loop_sync.difference_update(text_sync)
330
331 # If textual synchronisation is required, those variables need
332 # to be updated straight away. Indeed, for update device
333 # directives, but not quite for update host directives, where
334 # placing the directive earlier and adding the async clause
335 # might perform better.
336 self._place_update(sched, update_pos, text_sync, mode)
337 host_sig.difference_update(text_sync)
338
339 # All the remaining variables that need synchronisation at this
340 # schedule can be placed in an update directive at the edge
341 # of the schedule, promoting coalescence with other directives.
342 update_pos = None
343
344 # If within loop body, we must cover loop-carried dependencies.
345 # If within if stmt, update device directives can only be moved
346 # out, i.e. executed unconditionally, if an update host
347 # directive for the same variable - which always exists as all
348 # outputs are inputs - has already been moved out as well. This
349 # happens if and only if there's no textually preceding kernel
350 # within the if stmt which writes that variable. Since that
351 # variable would appear in loop_sync as part of the outputs of
352 # any textually preceding kernels, we may simply place an
353 # update device directive inside the if statement for those
354 # variables in loop_sync. loop_sync also includes the outputs
355 # of this host region which are inputs of any textually
356 # preceding kernels but, even though these might not need to
357 # be updated here, i.e. conditionally, they can safely be.
358 if isinstance(sched.parent, Loop) or \
359 isinstance(sched.parent, IfBlock) and mode == OUT:
360 self._place_update(sched, update_pos, loop_sync, mode)
361 host_sig.difference_update(loop_sync)
362
363 # This schedule is the body of a routine and, at least until we
364 # can do interprocedural analysis, this is the end of the road.
365 if isinstance(sched, Routine):
366 self._place_update(sched, update_pos, host_sig, mode)
367 break
368
369 # Move up the code tree to the next deepest ancestor schedule.
370 child, sched = child.parent, sched.parent
371 while not isinstance(sched, Schedule):
372 child, sched = child.parent, sched.parent

374 def _sig_to_sync(self, dep_stmts, host_sig, mode):
375 '''
376 Retrieves, amongst the host region signatures in host_sig, those that
377 require synchronisation because of accesses in the compute regions
378 within the statements in dep_stmts.
379
380 :param dep_stmts: list of statements which may include compute regions.
381 :type dep_stmts: List[:py:class:`psyclone.psyir.nodes.Statement`]
382 :param host_sig: access signature(s) that need to be synchronised \
383 with the accelerator device.
384 :type host_sig: Set[:py:class:`psyclone.core.Signature`]
385 :param int mode: the access mode from the point of view of the host,
386 either IN (read) or OUT (write).
387
388 '''
389 # If there is a statement (e.g. a call) among the dependent
390 # statements that may launch device kernels, we conservatively
391 # assume a dependency for all variables in the host region
392 # regardless of access mode. For the performance of the Python
393 # code there is an assumption here that _may_compute only
394 # returns True for Call nodes.
395 for stmt in dep_stmts:
396 if any(self._may_compute(call) for call in stmt.walk(Call)):
397 return host_sig.copy()
398
399 # Set of all signatures in compute kernels that may require syncing.
400 kern_sig = set()
401
402 for stmt in dep_stmts:
403 for acc in stmt.walk(self._ACC_COMPUTE):
404 # Kernel outputs can be both input and output dependencies.
405 # The latter is since we must guarantee no kernel write is
406 # overwritten by an earlier host write whose update device
407 # directive could appear later.
408 kern_sig.update(acc.signatures[OUT])
409 if mode == OUT:
410 kern_sig.update(acc.signatures[IN])
411
412 return host_sig.intersection(kern_sig)

414 def _place_update(self, sched, update_pos, host_sig, mode):
415 '''
416 Places, avoiding redundancy where possible, an update directive in the
417 provided schedule, at the requested position, for the requested
418 variables and access mode.
419
420 :param sched: the schedule in which to add the directive.
421 :type sched: :py:class:`psyclone.psyir.nodes.Schedule`
422 :param int update_pos: where to place the directive in the schedule.
423 :param host_sig: the access signature(s) that need to be synchronised \
424 with the accelerator device.
425 :type host_sig: Set[:py:class:`psyclone.core.Signature`]
426 :param int mode: the access mode from the point of view of the host,
427 either IN (read) or OUT (write).
428
429 '''
430 # Avoid rewriting the set of signatures on the caller.
431 host_sig = host_sig.copy()
432
433 # Specify the data movement direction with the right clause depending
434 # on the access mode. When the update directive can be placed at the
435 # edge of the schedule, the position of the directive, update_pos, is
436 # undefined until this point.
437 if mode == IN:
438 direction = "host"
439 if update_pos is None:
440 update_pos = 0
441 elif mode == OUT:
442 direction = "device"
443 if update_pos is None:
444 update_pos = len(sched.children)
445
446 # Check preceding and succeeding schedule positions for existing update
447 # directives, coalescing with existing directives if possible.
448 # If at the beginning or at the end of the schedule, we may go out of
449 # bounds, thus the try statement.
450 for update_offset in (-1, 0):
451 try:
452 neighbour_node = sched[update_pos + update_offset]
453 if isinstance(neighbour_node, ACCUpdateDirective):
454 # Merge neighbour update directives in the same direction.
455 if neighbour_node.direction == direction:
456 neighbour_node.sig_set.update(host_sig)
457 host_sig.clear()
458 # Avoid updating the host just after updating the device.
459 else:
460 host_sig.difference_update(neighbour_node.sig_set)
461 except IndexError:
462 pass
463
464 # Add the update directive to the schedule.
465 if host_sig:
466 update_dir = ACCUpdateDirective(host_sig, direction)
467 sched.addchild(update_dir, update_pos)
