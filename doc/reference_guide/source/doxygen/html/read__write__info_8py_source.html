37''' This module provides the class to store information about which variables
38are written, and which ones are read.'''
39
40
41class ReadWriteInfo:
42 '''This class stores signature and container name of variables read or
43 written. The container name is optional, it will default to "" if the
44 signature belongs to a symbol declared in the local scope, otherwise it
45 is the name of the container from which it must be imported.
46
47 The information is stored in lists of tuples, the first element being
48 the container name, the second the signature. When accessing any of these
49 two lists, the getter will make sure that the lists are sorted. This
50 will guarantee that, for example, the kernel extraction and driver creation
51 read the variables in the same order.
52 '''
53
54 def __init__(self):
55 self._read_list = []
56 self._write_list = []
57 self._sorted = True
58
59 # -------------------------------------------------------------------------
60 @property
61 def read_list(self):
62 ''':returns the sorted list of container_name,signature pairs that \
63 are read.
64 :rtype: List[Tuple[str,:py:class:`psyclone.core.Signature`]]
65
66 '''
67 if not self._sorted:
68 self._read_list.sort()
69 self._write_list.sort()
70 self._sorted = True
71 return self._read_list
72
73 # -------------------------------------------------------------------------
74 @property
75 def signatures_read(self):
76 '''Convenience function to return only the signatures read.
77
78 :returns the list of all signatures read.
79 :rtype: List[:py:class:`psyclone.core.Signature`]
80
81 '''
82 return [sig for _, sig in self.read_list]
83
84 # -------------------------------------------------------------------------
85 @property
86 def write_list(self):
87 ''':returns the sorted list of container_name,signature pairs that \
88 are written.
89 :rtype: List[Tuple[str,:py:class:`psyclone.core.Signature`]]
90
91 '''
92 if not self._sorted:
93 self._read_list.sort()
94 self._write_list.sort()
95 self._sorted = True
96 return self._write_list
97
98 # -------------------------------------------------------------------------
99 @property
100 def signatures_written(self):
101 '''Convenience function to return only the signatures written.
102
103 :returns the list of all signatures written.
104 :rtype: List[:py:class:`psyclone.core.Signature`]
105
106 '''
107 return [sig for _, sig in self.write_list]
108
109 # -------------------------------------------------------------------------
110 @property
111 def set_of_all_used_vars(self):
112 '''This property returns a set with all (container_name, Signature)
113 tuples. Since it is a set this guarantees that each tuple is only
114 listed once.
115
116 :returns: set with all (container_name, Signature) pairs.
117 :rtype: Set[Tuple[str, :py:class:`psyclone.core.Signature`]]
118
119 '''
120 return set(self._read_list) | set(self._write_list)
121
122 # -------------------------------------------------------------------------
123 def add_read(self, signature, container_name=None):
124 '''This function adds a read access to the specified signature and
125 container name. The container_name is optional and defaults to "",
126 indicating that this signature is not based on importing a symbol
127 from an external container (i.e. a module in Fortran).
128
129 :param signature: the signature of the access.
130 :type signature: :py:class:`psyclone.core.Signature`
131 :param container_name: the container name (optional)
132 :type container_name: Optional[str]
133
134 '''
135 if container_name:
136 self._read_list.append((container_name, signature))
137 else:
138 self._read_list.append(("", signature))
139 self._sorted = False
140
141 # -------------------------------------------------------------------------
142 def add_write(self, signature, container_name=None):
143 '''This function adds a write access to the specified signature and
144 container name. The container_name is optional and defaults to "",
145 indicating that this signature is not based on importing a symbol
146 from an external container (i.e. a module in Fortran).
147
148 :param signature: the signature of the access.
149 :type signature: :py:class:`psyclone.core.Signature`
150 :param container_name: the container name (optional)
151 :type container_name: Optional[str]
152
153 '''
154 if container_name:
155 self._write_list.append((container_name, signature))
156 else:
157 self._write_list.append(("", signature))
158 self._sorted = False
159
160 # -------------------------------------------------------------------------
161 def is_read(self, signature):
162 ''':returns: whether the signature is in the read list (independent \
163 of the container name).
164 :rtype: bool
165
166 '''
167 return any(signature == sig for _, sig in self._read_list)
