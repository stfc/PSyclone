55class FortranReader():
56 ''' PSyIR Fortran frontend. This frontend translates Fortran from a string
57 or a file into PSyIR using the fparser2 utilities.
58
59 :param free_form: If parsing free-form code or not (default True).
60 :param ignore_comments: If comments should be ignored or not
61 (default True).
62 :param ignore_directives: If directives should be ignored or not
63 (default True). Only has an effect
64 if ignore_comments is False.
65 :param last_comments_as_codeblocks: If the last comments in the
66 a given block (e.g. subroutine,
67 do, if-then body, etc.) should
68 be kept as code blocks or lost
69 (default False).
70 Only has an effect if ignore_comments
71 is False.
72 :param resolve_modules: Whether to resolve modules while parsing a file,
73 for more precise control it also accepts a list of module names.
74 Defaults to False.
75
76 :raises ValueError: If ignore_directives is set to False but
77 ignore_comments is set to True.
78
79 '''
80 # Save parser object across instances to reduce the initialisation time
81 _parser = None
82
83 def __init__(self, free_form: bool = True, ignore_comments: bool = True,
84 ignore_directives: bool = True,
85 last_comments_as_codeblocks: bool = False,
86 resolve_modules: Union[bool, List[str]] = False):
87 if not self._parser:
88 std = Config.get().fortran_standard
89 self._parser = ParserFactory().create(std=std)
90 self._free_form = free_form
91 if ignore_comments and not ignore_directives:
92 raise ValueError(
93 "Setting ignore_directives to False in the FortranReader will"
94 " only have an effect if ignore_comments is also set to False."
95 )
96 self._ignore_comments = ignore_comments
97 self._processor = Fparser2Reader(ignore_directives,
98 last_comments_as_codeblocks,
99 resolve_modules)
100 SYMBOL_TABLES.clear()

104 '''
105 Utility method that checks that the supplied name is a valid
106 Fortran name.
107
108 :param name: the name to check.
109
110 :raises TypeError: if the name is not a string.
111 :raises ValueError: if this is not a valid name.
112

115 raise TypeError(
116 f"A name should be a string, but found "
117 f"'{type(name).__name__}'.")
118 if not pattern_tools.abs_name.match(name):

122 def psyir_from_source(self, source_code: str):
123 ''' Generate the PSyIR tree representing the given Fortran source code.
124
125 :param str source_code: text representation of the code to be parsed.
126
127 :returns: PSyIR representing the provided Fortran source code.
128 :rtype: :py:class:`psyclone.psyir.nodes.Node`
129
130 '''
131 SYMBOL_TABLES.clear()
132 string_reader = FortranStringReader(
133 source_code, include_dirs=Config.get().include_paths,
134 ignore_comments=self._ignore_comments)
135 # Set reader to free format.
136 string_reader.set_format(FortranFormat(self._free_form, False))
137 parse_tree = self._parser(string_reader)
138
139 psyir = self._processor.generate_psyir(parse_tree)
140 return psyir

142 def psyir_from_expression(self, source_code: str,
143 symbol_table: Optional[SymbolTable] = None):
144 '''Generate the PSyIR tree for the supplied Fortran statement. The
145 symbol table is expected to provide all symbols found in the
146 expression.
147
148 :param source_code: text of the expression to be parsed.
149 :param symbol_table: the SymbolTable in which to search for any
150 symbols that are encountered.
151
152 :returns: PSyIR representing the provided Fortran expression.
153 :rtype: :py:class:`psyclone.psyir.nodes.Node`
154
155 :raises TypeError: if no valid SymbolTable is supplied.
156 :raises ValueError: if the supplied source does not represent a
157 Fortran expression.
158
159 '''
160 if symbol_table is None:
161 symbol_table = SymbolTable()
162 elif not isinstance(symbol_table, SymbolTable):
163 raise TypeError(f"Must be supplied with a valid SymbolTable but "
164 f"got '{type(symbol_table).__name__}'")
165
166 try:
167 parse_tree = Fortran2003.Expr(source_code)
168 except NoMatchError as err:
169 raise ValueError(
170 f"Supplied source does not represent a Fortran "
171 f"expression: '{source_code}'") from err
172
173 # Create a fake sub-tree connected to the supplied symbol table so
174 # that we can process the expression and lookup any symbols that it
175 # references. We provide the SymbolTable directly to the private
176 # attribute to avoid the symbol table's node link to connect to the
177 # new Schedule and therefore stealing it from its original scope.
178 # pylint: disable=protected-access
179 fake_parent = Schedule()
180 # pylint: disable=protected-access
181 fake_parent._symbol_table = symbol_table
182 fake_parent.addchild(Assignment())
183 self._processor.process_nodes(fake_parent[0], [parse_tree])
184 return fake_parent[0].children[0].detach()

186 def psyir_from_statement(self, source_code: str,
187 symbol_table: Optional[SymbolTable] = None):
188 '''Generate the PSyIR tree for the supplied Fortran statement. The
189 symbolt table is expected to provide all symbols found in the
190 statement.
191
192 :param source_code: text of the statement to be parsed.
193 :param symbol_table: the SymbolTable in which to search for any
194 symbols that are encountered.
195
196 :returns: PSyIR representing the provided Fortran statement.
197 :rtype: :py:class:`psyclone.psyir.nodes.Node`
198
199 :raises TypeError: if no valid SymbolTable is supplied.
200 :raises ValueError: if the supplied source does not represent a
201 Fortran statement.
202
203 '''
204 if symbol_table is None:
205 symbol_table = SymbolTable()
206 elif not isinstance(symbol_table, SymbolTable):
207 raise TypeError(f"Must be supplied with a valid SymbolTable but "
208 f"got '{type(symbol_table).__name__}'")
209 string_reader = FortranStringReader(source_code)
210 # Set reader to free format.
211 string_reader.set_format(FortranFormat(True, False))
212 try:
213 exec_part = Fortran2003.Execution_Part(string_reader)
214 except NoMatchError as err:
215 raise ValueError(f"Supplied source does not represent a Fortran "
216 f"statement: '{source_code}'") from err
217
218 # Create a fake sub-tree connected to the supplied symbol table so
219 # that we can process the statement and lookup any symbols that it
220 # references.
221 routine_name = "dummy"
222 fake_parent = Routine.create(
223 routine_name, SymbolTable(), [])
224 # pylint: disable=protected-access
225 fake_parent._symbol_table = symbol_table
226
227 # Process the statement, giving the Routine we've just
228 # created as the parent.
229 self._processor.process_nodes(fake_parent, exec_part.children)
230 return fake_parent[0].detach()

232 def psyir_from_file(self, file_path):
233 ''' Generate the PSyIR tree representing the given Fortran file.
234
235 :param file_path: path of the file to be read and parsed.
236 :type file_path: str or any Python Path format.
237
238 :returns: PSyIR representing the provided Fortran file.
239 :rtype: :py:class:`psyclone.psyir.nodes.Node`
240
241 '''
242 SYMBOL_TABLES.clear()
243
244 # Note that this is the main performance hotspot in PSyclone, taking
245 # more than 90% of the runtime in some cases. Therefore this is a good
246 # place to implement caching in order to avoid repeating parsing steps
247 # that have already been done before.
248
249 # Using the FortranFileReader instead of manually open the file allows
250 # fparser to keep the filename information in the tree
251 reader = FortranFileReader(file_path,
252 include_dirs=Config.get().include_paths,
253 ignore_comments=self._ignore_comments)
254 reader.set_format(FortranFormat(self._free_form, False))
255 parse_tree = self._parser(reader)
256 _, filename = os.path.split(file_path)
257
258 psyir = self._processor.generate_psyir(parse_tree, filename)
259 return psyir
