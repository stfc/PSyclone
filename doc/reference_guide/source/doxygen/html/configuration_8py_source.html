60# The different naming schemes supported when transforming kernels:
61# multiple = Every transformed kernel is given a unique name. This permits
62# multiple versions of the same kernel to be created.
63# single = If any given kernel (within a single Application) is transformed
64# more than once then the same transformation must always be
65# applied and only one version of the transformed kernel is created.

73class ConfigurationError(PSycloneError):
74 '''
75 Class for all configuration-related errors.
76
77 :param str value: error message.
78 :param config: the Config object associated with the error.
79 :type config: :py:class:`psyclone.configuration.Config`.
80 '''
81 def __init__(self, value, config=None):
82 PSycloneError.__init__(self, value)
83 self.valuevalue = "PSyclone configuration error"
84 if config:
85 self.valuevalue = f"{self.value} (file={config.filename})"
86 self.valuevalue += f": {value}"

92 '''
93 Handles all configuration management. It is implemented as a singleton
94 using a class _instance variable and a get() function.
95 '''
96 # Class variable to store the singleton instance
97 _instance = None
98
99 # A consistency flag that is set to true the moment the proper config
100 # file is loaded. If an instance of this class should be created before
101 # the loading of the config file, an exception will be raised.
102 _HAS_CONFIG_BEEN_INITIALISED = False
103
104 # List of supported API by PSyclone
105 _supported_api_list = LFRIC_API_NAMES + GOCEAN_API_NAMES
106 # List for printing purposes (remove duplicates and use prefered names)
107 _curated_api_list = [LFRIC_API_NAMES[0], GOCEAN_API_NAMES[0]]

109 # List of supported stub API by PSyclone
110 _supported_stub_api_list = LFRIC_API_NAMES
111

114 # kernel by ensuring that each transformed kernel is given a
115 # unique name (within the specified kernel-output directory).
116 # N.B. the default location to which to write transformed kernels is
117 # the current working directory. Since this may change between the
118 # importing of this module and the actual generation of code (e.g. as

124 # PSyIR symbol table.

233 def load(self, config_file=None):
234 '''Loads a configuration file.
235
236 :param str config_file: Override default configuration file to read.
237 :raises ConfigurationError: if there are errors or inconsistencies in \
238 the specified config file.
239 '''
240 # pylint: disable=too-many-branches, too-many-statements
241 if config_file:
242 # Caller has explicitly provided the full path to the config
243 # file to read
244 if not os.path.isfile(config_file):
245 raise ConfigurationError(
246 f"File {config_file} does not exist")
247 self._config_file = config_file[:]
248 else:
249 # Search for the config file in various default locations
250 self._config_file = Config.find_file()
251 # Add a getlist method to the ConfigParser instance using the
252 # converters argument. The lambda functions also handles the
253 # case of an empty specification ('xx = ''), returning an
254 # empty list instead of a list containing the empty string:
255 self._config = ConfigParser(
256 converters={'list': lambda x: [] if not x else
257 [i.strip() for i in x.split(',')]})
258 try:
259 self._config.read(self._config_file)
260 # Check for missing section headers and general parsing errors
261 # (e.g. incomplete or incorrect key-value mapping)
262 except (MissingSectionHeaderError, ParsingError) as err:
263 raise ConfigurationError(
264 f"ConfigParser failed to read the configuration file. Is it "
265 f"formatted correctly? (Error was: {err})",
266 config=self) from err
267
268 # Check that the configuration file has a [DEFAULT] section. Even
269 # if there isn't one in the file, ConfigParser creates an (empty)
270 # dictionary entry for it.
271 if 'DEFAULT' not in self._config or \
272 not self._config['DEFAULT'].keys():
273 raise ConfigurationError(
274 "Configuration file has no [DEFAULT] section", config=self)
275
276 # The call to the 'read' method above populates a dictionary.
277 # All of the entries in that dict are unicode strings so here
278 # we pull out the values we want and deal with any type
279 # conversion. We protect each of these with a try block so as
280 # to catch any conversion errors due to incorrect entries in
281 # the config file.
282 try:
283 self._distributed_mem = self._config['DEFAULT'].getboolean(
284 'DISTRIBUTED_MEMORY')
285 except ValueError as err:
286 raise ConfigurationError(
287 f"Error while parsing DISTRIBUTED_MEMORY: {err}",
288 config=self) from err
289
290 try:
291 self._reproducible_reductions = self._config['DEFAULT'].getboolean(
292 'REPRODUCIBLE_REDUCTIONS')
293 except ValueError as err:
294 raise ConfigurationError(
295 f"Error while parsing REPRODUCIBLE_REDUCTIONS: {err}",
296 config=self) from err
297
298 try:
299 self._reprod_pad_size = self._config['DEFAULT'].getint(
300 'REPROD_PAD_SIZE')
301 except ValueError as err:
302 raise ConfigurationError(
303 f"error while parsing REPROD_PAD_SIZE: {err}",
304 config=self) from err
305
306 if 'PSYIR_ROOT_NAME' not in self._config['DEFAULT']:
307 # Use the default name if no default is specified for the
308 # root name.
309 self._psyir_root_name = Config._default_psyir_root_name
310 else:
311 self._psyir_root_name = self._config['DEFAULT']['PSYIR_ROOT_NAME']
312
313 # Read the valid PSyData class prefixes. If the keyword does
314 # not exist then return an empty list.
315 self._valid_psy_data_prefixes = \
316 self._config["DEFAULT"].getlist("VALID_PSY_DATA_PREFIXES", [])
317 try:
318 self._ocl_devices_per_node = self._config['DEFAULT'].getint(
319 'OCL_DEVICES_PER_NODE')
320 except ValueError as err:
321 raise ConfigurationError(
322 f"error while parsing OCL_DEVICES_PER_NODE: {err}",
323 config=self) from err
324
325 # Verify that the prefixes will result in valid Fortran names:
326 valid_var = re.compile(r"[A-Z][A-Z0-9_]*$", re.I)
327 for prefix in self._valid_psy_data_prefixes:
328 if not valid_var.match(prefix):
329 raise ConfigurationError(
330 f"Invalid PsyData-prefix '{prefix}' in config file. The "
331 f"prefix must be valid for use as the start of a Fortran "
332 f"variable name.", config=self)
333
334 # Whether validation is performed in the PSyIR backends.
335 if 'BACKEND_CHECKS_ENABLED' in self._config['DEFAULT']:
336 try:
337 self._backend_checks_enabled_backend_checks_enabled = (
338 self._config['DEFAULT'].getboolean(
339 'BACKEND_CHECKS_ENABLED'))
340 except ValueError as err:
341 raise ConfigurationError(
342 f"Error while parsing BACKEND_CHECKS_ENABLED: {err}",
343 config=self) from err
344
345 # Now we deal with the API-specific sections of the config file. We
346 # create a dictionary to hold the API-specific Config objects.
347 self._api_conf = {}
348 for api in Config._supported_api_list:
349 if api in self._config:
350 if api in LFRIC_API_NAMES:
351 key = LFRIC_API_NAMES[0] # Use the first name internally
352 self._api_conf[key] = LFRicConfig(self, self._config[api])
353 elif api in GOCEAN_API_NAMES:
354 key = GOCEAN_API_NAMES[0] # Use the first name internally
355 self._api_conf[key] = GOceanConfig(self, self._config[api])
356 else:
357 raise NotImplementedError(
358 f"Configuration file '{self._config_file}' contains a "
359 f"'{api}' section but no Config sub-class has "
360 f"been implemented for this API")
361
362 # The scheme to use when re-naming transformed kernels.
363 # By default we ensure that each transformed kernel is given a
364 # unique name (within the specified kernel-output directory).
365 self._kernel_naming = Config._default_kernel_naming
366
367 ignore_modules = self._config['DEFAULT'].getlist("IGNORE_MODULES", [])
368 # Avoid circular import
369 # pylint: disable=import-outside-toplevel
370 from psyclone.parse import ModuleManager
371 mod_manager = ModuleManager.get()
372 for module_name in ignore_modules:
373 mod_manager.add_ignore_module(module_name)
374
375 # Get the Fortran standard to use:
376 self._fortran_standard_fortran_standard = \
377 self._config['DEFAULT'].get("FORTRAN_STANDARD", "f2008").lower()
378 valid_standard = ["f2003", "f2008"]
379 if self._fortran_standard_fortran_standard not in valid_standard:
380 raise ConfigurationError(f"Invalid Fortran standard "
381 f"'{self._fortran_standard}' specified "
382 f"in config file. Must be one of"
383 f"{valid_standard}")
384
385 # Set the flag that the config file has been loaded now.
386 Config._HAS_CONFIG_BEEN_INITIALISED = True
387
388 def api_conf(self, api=None):
389 '''
390 Getter for the object holding API-specific configuration options.
391
392 :param str api: Optional, the API for which configuration details are
393 required. If none is specified, returns the config for the
394 default API.
395 :returns: object containing API-specific configuration
396 :rtype: One of :py:class:`psyclone.configuration.LFRicConfig`,
397 :py:class:`psyclone.configuration.GOceanConfig` or None.
398
399 :raises ConfigurationError: if api is not in the list of supported \
400 APIs.
401 :raises ConfigurationError: if the config file did not contain a \
402 section for the requested API.
403 '''
404 if not api:
405 api = self._api
406
407 if api not in self.supported_apissupported_apis:
408 raise ConfigurationError(
409 f"API '{api}' is not in the list '{self.supported_apis}'' of "
410 f"supported APIs.")
411 if api not in self._api_conf:
412 raise ConfigurationError(
413 f"Configuration file did not contain a section for the "
414 f"'{api}' API", config=self)
415 return self._api_conf[api]

417 @staticmethod
418 def find_file():
419 '''
420 Static method that searches various locations for a configuration
421 file. If the full path to an existing file has been provided in
422 the PSYCLONE_CONFIG environment variable then that is returned.
423 Otherwise, we search the following locations, in order:
424
425 - ${PWD}/.psyclone/
426 - if inside-a-virtual-environment:
427 <base-dir-of-virtual-env>/share/psyclone/
428 - ${HOME}/.local/share/psyclone/
429 - <system-install-prefix>/share/psyclone/
430 - <psyclone-installation-base>/share/psyclone/
431 - <psyclone-src-base>/config/
432
433 :returns: the fully-qualified path to the configuration file
434 :rtype: str
435
436 :raises ConfigurationError: if no config file is found
437
438 '''
439 # Moving this to the top causes test failures
440 # pylint: disable=import-outside-toplevel
441 from psyclone.utils import within_virtual_env
442
443 # If $PSYCLONE_CONFIG is set then we use that unless the
444 # file it points to does not exist
445 _psy_config = os.environ.get('PSYCLONE_CONFIG')
446 if _psy_config and os.path.isfile(_psy_config):
447 return _psy_config
448
449 # Set up list of locations to search
450 share_dir = os.path.join(sys.prefix, "share", "psyclone")
451 pkg_share_dir = [
452 os.path.join(os.path.dirname(psyclone_path), "share", "psyclone")
453 for psyclone_path in psyclone.__path__]
454
455 # 1. .psyclone/ in the CWD
456 _file_paths = [os.path.join(os.getcwd(), ".psyclone")]
457 if within_virtual_env():
458 # 2. <virtual-env-base>/share/psyclone/
459 _file_paths.append(share_dir)
460 # 3. ~/.local/share/psyclone/
461 _file_paths.append(os.path.join(os.path.expanduser("~"),
462 ".local", "share", "psyclone"))
463 if not within_virtual_env():
464 # 4. <python-installation-base>/share/psyclone/
465 _file_paths.append(share_dir)
466
467 # 5. <psyclone-installation-base>/share/psyclone/
468 _file_paths.extend(pkg_share_dir)
469
470 # 6. <psyclone-src-base>/config/
471 # Search for configuration file relative to this source file
472 dev_dir_tmp = os.path.dirname(__file__)
473 dev_dir_tmp = os.path.split(dev_dir_tmp)[0] # Go down one level
474 dev_dir_tmp = os.path.split(dev_dir_tmp)[0] # Go down another level
475 dev_path = os.path.join(dev_dir_tmp, "config")
476 _file_paths.append(dev_path)
477
478 for cfile in [os.path.join(cdir, _FILE_NAME) for cdir in _file_paths]:
479 if os.path.isfile(cfile):
480 return cfile
481
482 # If we get to here then we have failed to find a config file
483 raise ConfigurationError(f"{_FILE_NAME} not found in any of "
484 f"{_file_paths}")

736 '''A base class for functions that each API-specific class must provide.

738 API-specific access-descriptor strings and the PSyclone internal
739 AccessType.

741 :raises ConfigurationError: if an access-mapping is provided that \

746 def __init__(self, section):

748 # having to specify those mappings.
749 self._access_mapping = {"read": "read", "write": "write",
750 "readwrite": "readwrite", "inc": "inc",
751 "sum": "sum"}
752 # Get the mapping if one exists and convert it into a
753 # dictionary. The input is in the format: key1:value1,
754 # key2=value2, ...
755 if "ACCESS_MAPPING" in section:
756 mapping_list = section.getlist("ACCESS_MAPPING")
757 if mapping_list is not None:

761 # TODO (issue #710): Add checks for duplicate or missing access

763 # Avoid circular import
764 # pylint: disable=import-outside-toplevel
765 from psyclone.core.access_type import AccessType

775 f"'{api_access_name}'") from err

781 @staticmethod
782 def create_dict_from_list(input_list):
783 '''Takes a list of strings each with the format: key:value and creates
784 a dictionary with the key,value pairs. Any leading or trailing
785 white space is removed.
786
787 :param input_list: the input list.
788 :type input_list: list of str
789
790 :returns: a dictionary with the key,value pairs from the input list.
791 :rtype: dict[str, Any]
792

794 does not contain a ":".
795
796 '''
797 return_dict = {}
798 for entry in input_list:
799 try:
800 key, value = entry.split(":", 1)
801 except ValueError as err:
802 # Raised when split does not return two elements:
803 raise ConfigurationError(
804 f"Invalid format for mapping: {entry.strip()}") from err
805 # Remove spaces and convert unicode to normal strings in Python2
806 return_dict[str(key.strip())] = str(value.strip())
807 return return_dict

1116class GOceanConfig(BaseConfig):
1117 '''Gocean1.0-specific Config sub-class. Holds configuration options
1118 specific to the GOcean 1.0 API.
1119
1120 :param config: The 'parent' Config object.
1121 :type config: :py:class:`psyclone.configuration.Config`

1123 the configuration file, as produced by ConfigParser.
1124 :type section: :py:class:`configparser.SectionProxy`
1125
1126 '''
1127 # pylint: disable=too-few-public-methods, too-many-branches
1128 def __init__(self, config, section):
1129 # pylint: disable=too-many-locals
1130 super().__init__(section)
1131 # Setup the mapping for the grid properties. This dictionary stores
1132 # the name of the grid property as key (e.g. ``go_grid_dx_t``),
1133 # with the value being a named tuple with an entry for 'property'
1134 # and 'type'. The 'property' is a format string to dereference
1135 # a property, and 'type' is a string.
1136 # These values are taken from the psyclone config file.
1137 self._grid_properties = {}
1138 # Initialise debug_mode settings (default to False if it doesn't exist)
1139 self._debug_mode = False
1140 for key in section.keys():
1141 # Do not handle any keys from the DEFAULT section
1142 # since they are handled by Config(), not this class.
1143 if key in config.get_default_keys():
1144 continue
1145 if key == "iteration-spaces":
1146 # The value associated with the iteration-spaces key is a
1147 # set of lines, each line defining one new iteration space.
1148 # Each individual iteration space added is checked
1149 # in add_bounds for correctness.
1150 value_as_str = str(section[key])
1151 new_iteration_spaces = value_as_str.split("\n")
1152 # Avoid circular import
1153 # pylint: disable=import-outside-toplevel
1154 from psyclone.gocean1p0 import GOLoop
1155 for it_space in new_iteration_spaces:
1156 GOLoop.add_bounds(it_space)
1157 elif key == "access_mapping":
1158 # Handled in the base class BaseConfig
1159 pass
1160 elif key == "debug_mode":
1161 # Boolean that specifies if debug mode is enabled
1162 try:
1163 self._debug_mode = section.getboolean("debug_mode")
1164 except ValueError as err:
1165 raise ConfigurationError(
1166 f"error while parsing DEBUG_MODE in the [gocean1p0] "
1167 f"section of the config file: {err}") from err
1168 elif key == "grid-properties":
1169 # Grid properties have the format:
1170 # go_grid_area_u: {0}%%grid%%area_u: array: real,
1171 # First the name, then the Fortran code to access the property,
1172 # followed by the type ("array" or "scalar") and then the
1173 # intrinsic type ("integer" or "real")
1174 all_props = self.create_dict_from_list(section.getlist(key))
1175 for grid_property, property_str in all_props.items():
1176 try:
1177 fortran, variable_type, intrinsic_type = \
1178 property_str.split(":")
1179 except ValueError as err:
1180 # Raised when the string does not contain exactly
1181 # three values separated by ":"
1182 error = (f"Invalid property '{grid_property}' found "
1183 f"with value '{property_str}' in "
1184 f"'{config.filename}'. It must have exactly "
1185 f"three ':'-delimited separated values: the "
1186 f"property, whether it is a scalar or an "
1187 f"array, and the intrinsic type (real or "
1188 f"integer).")
1189 raise ConfigurationError(error) from err
1190 # Make sure to remove the spaces which the config
1191 # file might contain
1192 self._grid_properties[grid_property] = \
1193 GOceanConfig.make_property(fortran.strip(),
1194 variable_type.strip(),
1195 intrinsic_type.strip())
1196 # Check that the required values for xstop and ystop
1197 # are defined:
1198 for required in ["go_grid_xstop", "go_grid_ystop",
1199 "go_grid_data",
1200 "go_grid_internal_inner_start",
1201 "go_grid_internal_inner_stop",
1202 "go_grid_internal_outer_start",
1203 "go_grid_internal_outer_stop",
1204 "go_grid_whole_inner_start",
1205 "go_grid_whole_inner_stop",
1206 "go_grid_whole_outer_start",
1207 "go_grid_whole_outer_stop"]:
1208 if required not in self._grid_properties:
1209 error = (f"The config file {config.filename} does not "
1210 f"contain values for the following, mandatory"
1211 f" grid property: '{required}'.")
1212 raise ConfigurationError(error)
1213 else:
1214 raise ConfigurationError(f"Invalid key '{key}' found in "
1215 f"'{config.filename}'.")
