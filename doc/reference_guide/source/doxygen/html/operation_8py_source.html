52class Operation(DataNode, metaclass=ABCMeta):
53 '''
54 Abstract base class for PSyIR nodes representing operators.
55
56 :param operator: the operator used in the operation.
57 :type operator: Union[
58 :py:class:`psyclone.psyir.nodes.UnaryOperation.Operator`,
59 :py:class:`psyclone.psyir.nodes.BinaryOperation.Operator`]
60 :param parent: the parent node of this Operation in the PSyIR.
61 :type parent: Optional[:py:class:`psyclone.psyir.nodes.Node`]
62
63 :raises TypeError: if the supplied operator is not an instance of
64 self.Operator.
65
66 '''
67 #: Must be overridden in sub-class to hold an Enumeration of the Operators
68 #: that it can represent.
69 Operator = object
70 #: Colour of the node in a view tree.
71 _colour = "blue"
72
73 def __init__(self, operator, parent=None):
74 super().__init__(parent=parent)
75
76 if not isinstance(operator, self.Operator):
77 raise TypeError(
78 f"{type(self).__name__} operator argument must be of type "
79 f"{type(self).__name__}.Operator but found "
80 f"{type(operator).__name__}.")
81 self._operator = operator

83 def __eq__(self, other):
84 '''Checks whether two Operations are equal. Operations are equal
85 if they are the same type, have the same operator and if the inherited
86 equality is True.
87
88 :param object other: the object to check equality to.
89
90 :returns: whether other is equal to self.
91 :rtype: bool
92 '''
93 is_eq = super().__eq__(other)
94 is_eq = is_eq and self.operator == other.operator
95
96 return is_eq

318 def _get_result_precision(self, precisions):
319 '''
320 Compares the two precisions to determine the precision of the result
321 of the operation.
322
323 If the two precisions are the same, then that value is returned.
324 Otherwise, Section 7.1.9.3 of the Fortran2008 standard says that in
325 this case, the precision of the result is the greater of the two.
326 If the precision cannot be determined then
327 `ScalarType.Precision.UNDEFINED` is returned.
328
329 :param precisions: the precision of the two operands.
330 :type precisions: list[int |
331 :py:class:`psyclone.psyir.symbols.ScalarType.Precision |
332 :py:class:`psyclone.psyir.nodes.Reference`]
333
334 :returns: the precision of the result of the operation.
335 :rtype: int | :py:class:`psyclone.psyir.symbols.ScalarType.Precision
336
337 :raises InternalError: if an unsupported Precision value is encountered
338 (this is to defend against any future extension of
339 ScalarType.Precision).
340
341 '''
342 if precisions[0] == precisions[1]:
343 return precisions[0]
344
345 # Operands have different precisions.
346 if all(isinstance(prec, int) for prec in precisions):
347 # Both precisions are integer.
348 return max(precisions)
349
350 if all(isinstance(prec, ScalarType.Precision) for
351 prec in precisions):
352 # Both precisions are of ScalarType.Precision type.
353 if ScalarType.Precision.UNDEFINED in precisions:
354 return ScalarType.Precision.UNDEFINED
355 if ScalarType.Precision.DOUBLE in precisions:
356 return ScalarType.Precision.DOUBLE
357 raise InternalError(
358 f"Operation._get_result_precision: got unsupported Precision "
359 f"value(s) '{precisions[0]}' and '{precisions[1]}' for "
360 f"operands '{self.children[0].debug_string()}' and "
361 f"'{self.children[1].debug_string()}'")
362
363 # We can't reason about the precision of the result.
364 return ScalarType.Precision.UNDEFINED

366 def _get_result_scalar_type(self, argtypes):
367 '''
368 Examines the two operand types to determine the base type of the
369 operation using the rules in Section 7.2 of the Fortran2008 standard.
370 If the type cannot be determined then an instance of `UnresolvedType`
371 is returned.
372
373 :param argtypes: the types of the two operands.
374 :type argtypes: list[:py:class:`psyclone.psyir.symbols.DataType`,
375 :py:class:`psyclone.psyir.symbols.DataType`]
376
377 :returns: the base type of the result of the operation.
378 :rtype: :py:class:`psyclone.psyir.symbols.DataType`
379
380 :raises TypeError: if an unexpected intrinsic type is found for
381 either of the operands to a numeric operation.
382
383 '''
384 if self.operator not in self._numeric_ops:
385 # Must be a relational or logical operator. Intrinsic type of
386 # result will be boolean.
387 return BOOLEAN_TYPE
388
389 # We have a numerical operation.
390 if any(isinstance(atype.intrinsic, UnresolvedType)
391 for atype in argtypes):
392 # datatype of a numerical operation on a UnresolvedType is a
393 # UnresolvedType.
394 return UnresolvedType()
395
396 base_type = None
397
398 # If either of the operands has REAL intrinsic type then the result
399 # must also be REAL.
400 if argtypes[0].intrinsic == argtypes[1].intrinsic:
401 # Operands are of the same intrinsic type.
402 precision = self._get_result_precision([argtypes[0].precision,
403 argtypes[1].precision])
404 base_type = ScalarType(argtypes[0].intrinsic, precision)
405 elif argtypes[0].intrinsic == ScalarType.Intrinsic.REAL:
406 base_type = argtypes[0]
407 elif argtypes[1].intrinsic == ScalarType.Intrinsic.REAL:
408 base_type = argtypes[1]
409
410 # Check that the type of the result is consistent with a numerical
411 # operation.
412 if not base_type or base_type.intrinsic not in (
413 ScalarType.Intrinsic.INTEGER,
414 ScalarType.Intrinsic.REAL):
415 for atype in argtypes:
416 if atype.intrinsic not in (ScalarType.Intrinsic.INTEGER,
417 ScalarType.Intrinsic.REAL):
418 raise TypeError(
419 f"Invalid argument of type '{atype.intrinsic}' to "
420 f"numerical operation '{self.operator}' in "
421 f"'{self.debug_string()}'. Currently only "
422 f"ScalarType.Intrinsic.REAL/INTEGER are "
423 f"supported (TODO #1590)")
424 return base_type

426 @property
427 def datatype(self):
428 '''
429 Determines the datatype of this operation. If it cannot be determined
430 for any reason then an instance of UnresolvedType is returned.
431
432 :returns: the datatype of the result of this BinaryOperation.
433 :rtype: :py:class:`psyclone.psyir.symbols.DataType`
434
435 :raises InternalError: if the operands are both arrays but are of
436 different shapes.
437 '''
438 # Get the types of the operands.
439 argtypes = []
440 for child in self.childrenchildrenchildrenchildren:
441 # If the operand is itself an operation this will recurse.
442 dtype = child.datatype
443 if isinstance(dtype, UnsupportedFortranType):
444 if dtype.partial_datatype:
445 # We are still OK provided we have partial type information
446 # since that means the intrinsic type can be handled in the
447 # PSyIR.
448 dtype = dtype.partial_datatype
449 else:
450 return UnresolvedType()
451 if isinstance(dtype, UnresolvedType):
452 # If either operand is of UnresolvedType then we can't do
453 # any better.
454 return UnresolvedType()
455 if isinstance(dtype, ArrayType):
456 # We know this is an array but do we know its intrinsic type?
457 if isinstance(dtype.intrinsic, UnresolvedType):
458 dtype = ArrayType(UnresolvedType(), shape=dtype.shape)
459 if isinstance(dtype.intrinsic, UnsupportedType):
460 if (isinstance(dtype.intrinsic, UnsupportedFortranType) and
461 dtype.intrinsic.partial_datatype):
462 dtype = ArrayType(dtype.intrinsic.partial_datatype,
463 shape=dtype.shape)
464 else:
465 dtype = ArrayType(UnresolvedType(), shape=dtype.shape)
466
467 argtypes.append(dtype)
468
469 # Determine the base (scalar) type of the result.
470 base_type = self._get_result_scalar_type(argtypes)
471 if (isinstance(base_type, UnresolvedType) or
472 all(isinstance(atype, ScalarType) for atype in argtypes)):
473 # Both operands are of scalar type.
474 return base_type
475
476 if all(isinstance(atype, ArrayType) for atype in argtypes):
477 # Both operands are of array type.
478 if len(argtypes[0].shape) != len(argtypes[1].shape):
479 raise InternalError(
480 f"Binary operation '{self.debug_string()}' has operands "
481 f"of different shape: '{self.children[0].debug_string()}' "
482 f"has rank {len(argtypes[0].shape)} and "
483 f"'{self.children[1].debug_string()}' has rank "
484 f"{len(argtypes[1].shape)}")
485 # In general there is no way we can check that the extents of each
486 # dimension match so we have to assume that they do.
487 shape = (argtypes[0].shape if isinstance(argtypes[0], ArrayType) else
488 argtypes[1].shape)
489 return ArrayType(base_type, shape=shape)
