43'''

60class ArrayAssignment2LoopsTrans(Transformation):
61 '''Provides a transformation from a PSyIR Array Range to a PSyIR
62 Loop. For example:
63
64 >>> from psyclone.psyir.frontend.fortran import FortranReader
65 >>> from psyclone.psyir.nodes import Assignment
66 >>> from psyclone.psyir.transformations import ArrayAssignment2LoopsTrans
67 >>> code = """
68 ... subroutine sub()
69 ... real :: tmp(10)
70 ... tmp(:) = tmp(:) + 3
71 ... end subroutine sub"""
72 >>> psyir = FortranReader().psyir_from_source(code)
73 >>> assignment = psyir.walk(Assignment)[0]
74 >>> trans = ArrayAssignment2LoopsTrans()
75 >>> trans.apply(assignment)
76 >>> print(psyir.debug_string())
77 subroutine sub()
78 real, dimension(10) :: tmp
79 integer :: idx
80 <BLANKLINE>
81 do idx = LBOUND(tmp, dim=1), UBOUND(tmp, dim=1), 1
82 tmp(idx) = tmp(idx) + 3
83 enddo
84 <BLANKLINE>
85 end subroutine sub
86 <BLANKLINE>
87
88 By default the transformation will reject character arrays, though this
89 can be overriden by setting the 'allow_string' option to True. Note that
90 PSyclone expresses syntax such as `character(LEN=100)` as
91 UnsupportedFortranType, and this transformation will convert unknown or
92 unsupported types to loops.
93 '''
94 def apply(self, node, options=None):
95 '''Apply the transformation to the specified array assignment node.
96 Each range node within the assignment is replaced with an explicit
97 loop. The bounds of the loop are determined from the bounds of the
98 array range on the left hand side of the assignment.
99
100 :param node: an Assignment node.
101 :type node: :py:class:`psyclone.psyir.nodes.Assignment`
102 :type options: Optional[Dict[str, Any]]
103 :param bool options["allow_string"]: whether to allow the
104 transformation on a character type array range. Defaults to False.
105 :param bool options["verbose"]: log the reason the validation failed,
106 at the moment with a comment in the provided PSyIR node.
107
108 '''
109 self.validatevalidate(node, options)
110
111 # If possible use the routine-level symbol table for nicer names
112 if node.ancestor(Routine):
113 symbol_table = node.ancestor(Routine).symbol_table
114 else:
115 # We checked in the validate that at least there is a scope
116 symbol_table = node.scope.symbol_table
117
118 # If there is any array reference without the accessor syntax,
119 # we need to add it first.
120 for reference in node.walk(Reference):
121 try:
122 Reference2ArrayRangeTrans().apply(reference)
123 except TransformationError:
124 pass
125
126 # Start by the rightmost array range
127 for lhs_range in reversed(node.lhs.walk(Range)):
128 lhs_array = lhs_range.parent
129 lhs_index = lhs_array.index_of(lhs_range)
130
131 # Create a new, unique, iteration variable for the new loop
132 loop_variable_symbol = symbol_table.new_symbol(
133 root_name="idx",
134 symbol_type=DataSymbol,
135 datatype=INTEGER_TYPE)
136
137 # Replace one range for each top-level array expression in the
138 # assignment
139 for expr in reversed(node.walk(ArrayMixin, stop_type=ArrayMixin)):
140 # We use a "for" to capture the value of the last range or
141 # continue if there is none, but we don't iterate all Ranges
142 for range_to_replace in reversed(expr.walk(Range)):
143 array = range_to_replace.parent
144 index = array.index_of(range_to_replace)
145 if lhs_array.same_range(lhs_index, array, index):
146 # If they iterate over the same bounds we just need a
147 # reference to the iteration index
148 index_expr = Reference(loop_variable_symbol)
149 else:
150 # If we can not prove that both ranges iterate over the
151 # exact same bounds we need to provide an offset like:
152 # array2(idx1 + (lbound_array2 - lbound_array1)
153 offset = BinaryOperation.create(
154 BinaryOperation.Operator.SUB,
155 range_to_replace.start.copy(),
156 lhs_range.start.copy())
157 index_expr = BinaryOperation.create(
158 BinaryOperation.Operator.ADD,
159 Reference(loop_variable_symbol),
160 offset)
161 range_to_replace.replace_with(index_expr)
162 break # We just substitue one per top-level array
163
164 # Replace the assignment with the new explicit loop structure
165 start, stop, step = lhs_range.pop_all_children()
166 loop = Loop.create(loop_variable_symbol, start, stop, step, [])
167 node.replace_with(loop)
168 loop.loop_body.addchild(node)
169
170 def __str__(self):
171 return ("Convert a PSyIR assignment with array Ranges into explicit "
172 "PSyIR Loops.")

174 def validate(self, node, options=None):
175 '''Perform various checks to ensure that it is valid to apply the
176 ArrayAssignment2LoopsTrans transformation to the supplied PSyIR Node.
177
178 By default the validate function will throw an TransofmrationError
179 on character arrays, though this can be overriden by setting the
180 allow_string option to True. Note that PSyclone expresses syntax such
181 as `character(LEN=100)` as UnsupportedFortranType, and this
182 transformation will convert unknown or unsupported types to loops.
183
184 :param node: the node that is being checked.
185 :type node: :py:class:`psyclone.psyir.nodes.Assignment`
186 :param options: a dictionary with options for transformations
187 :type options: Optional[Dict[str, Any]]
188 :param bool options["allow_string"]: whether to allow the
189 transformation on a character type array range. Defaults to False.
190 :param bool options["verbose"]: log the reason the validation failed,
191 at the moment with a comment in the provided PSyIR node.
192
193 :raises TransformationError: if the node argument is not an
194 Assignment.
195 :raises TransformationError: if the node argument is an
196 Assignment whose left hand side is not an ArrayReference.
197 :raises TransformationError: if the node argument is an
198 Assignment whose left hand side is an ArrayReference that does
199 not have Range specifying the access to at least one of its
200 dimensions.
201 :raises TransformationError: if two or more of the loop ranges
202 in the assignment are different or are not known to be the
203 same.
204 :raises TransformationError: if node contains a character type
205 child and the allow_strings option is
206 not set.
207
208 '''
209 if not options:
210 options = {}
211
212 if not isinstance(node, Assignment):
213 raise TransformationError(
214 f"Error in {self.name} transformation. The supplied node "
215 f"should be a PSyIR Assignment, but found "
216 f"'{type(node).__name__}'.")
217
218 try:
219 node.scope
220 except SymbolError:
221 # pylint: disable=raise-missing-from
222 raise TransformationError(LazyString(
223 lambda: f"Error in {self.name} transformation. The "
224 f"assignment should be in a scope to create the necessary new "
225 f"symbols, but '{node.debug_string()}' is not."))
226
227 array_accessors = node.lhs.walk(ArrayMixin)
228 if not (isinstance(node.lhs, Reference) and array_accessors):
229 raise TransformationError(LazyString(
230 lambda: f"Error in {self.name} transformation. The LHS of the"
231 f" supplied Assignment node should be a Reference that "
232 f"contains an array accessor somewhere in the expression, "
233 f"but found '{node.lhs.debug_string()}'."))
234
235 # There should be at least one Range in the LHS
236 if not node.lhs.walk(Range):
237 raise TransformationError(LazyString(
238 lambda: f"Error in {self.name} transformation. The LHS of"
239 f" the supplied Assignment node should contain an array "
240 f"accessor with at least one of its dimensions being a "
241 f"Range, but none were found in '{node.debug_string()}'."))
242
243 # All the ArrayMixins must have the same number of Ranges to expand
244 num_of_ranges = None
245 for accessor in node.walk(ArrayMixin):
246 count = len([x for x in accessor.indices if isinstance(x, Range)])
247 if count > 0:
248 if not num_of_ranges:
249 # If it's the first value we find, we store it
250 num_of_ranges = count
251 else:
252 # Otherwise we compare it against the previous found value
253 if count != num_of_ranges:
254 raise TransformationError(LazyString(
255 lambda: f"{self.name} does not support statements"
256 f" containing array accesses that have varying "
257 f"numbers of ranges in their accessors, but found:"
258 f"\n{node.debug_string()}"))
259
260 # Any errors below this point will optionally log the resason, which
261 # at the moment means adding a comment in the output code
262 verbose = options.get("verbose", False)
263
264 # Do not allow to transform expressions with CodeBlocks
265 if node.walk(CodeBlock):
266 message = (f"{self.name} does not support array assignments that"
267 f" contain a CodeBlock anywhere in the expression")
268 if verbose:
269 node.append_preceding_comment(message)
270 raise TransformationError(LazyString(
271 lambda: f"{message}, but found:\n{node.debug_string()}"))
272
273 # We don't support nested range expressions anywhere in the assignment
274 for range_expr in node.walk(Range):
275 ancestor_array = range_expr.parent.ancestor(ArrayMixin)
276 if ancestor_array and any(index.walk(Range) for index
277 in ancestor_array.indices):
278 message = (f"{self.name} does not support array assignments "
279 f"that contain nested Range expressions")
280 if verbose:
281 node.append_preceding_comment(message)
282 # pylint: disable=cell-var-from-loop
283 raise TransformationError(LazyString(
284 lambda: f"{message}, but found:\n{node.debug_string()}"))
285
286 # If we allow string arrays then we can skip the check.
287 if not options.get("allow_string", False):
288 message = (f"{self.name} does not expand ranges "
289 f"on character arrays by default (use the"
290 f"'allow_string' option to expand them)")
291 self.validate_no_char(node, message, options)
292
293 # We don't accept calls that are not guaranteed to be elemental
294 for call in node.rhs.walk(Call):
295 if isinstance(call, IntrinsicCall):
296 # Intrinsics that return scalars are also fine.
297 if call.intrinsic in (IntrinsicCall.Intrinsic.LBOUND,
298 IntrinsicCall.Intrinsic.UBOUND,
299 IntrinsicCall.Intrinsic.SIZE):
300 continue
301 name = call.intrinsic.name
302 else:
303 name = call.routine.symbol.name
304 if not call.is_elemental:
305 message = (f"{self.name} does not accept calls which are not"
306 f" guaranteed to be elemental, but found: "
307 f"{name}")
308 if verbose:
309 node.append_preceding_comment(message)
310 # pylint: disable=cell-var-from-loop
311 raise TransformationError(LazyString(
312 lambda: f"{message} in:\n{node.debug_string()}."))
313
314 # For each top-level reference (because we don't support nesting), the
315 # apply() will have to determine whether it's an Array (and access it
316 # with the index) or a Scalar (and leave it as it is). We cannot
317 # transform references where this is unclear.
318 for reference in node.walk(Reference, stop_type=Reference):
319 if isinstance(reference.parent, Call):
320 # The call routine references are fine
321 if reference.parent.routine is reference:
322 continue
323 # Arguments of inquiry intrinsics are also fine
324 if isinstance(reference.parent, IntrinsicCall):
325 if reference.parent.is_inquiry:
326 continue
327
328 # In some cases we can infer that they are scalar without needing
329 # the datatype by the fact that they come from valid Fortran
330 # assignments, for instance, if the expression already has ranges:
331 # e.g. A%B(:)%C
332 # C can not be further expanded into an array because
333 # e.g. A%B(:)%C(:) is not valid Fortran
334 # Also, if we find explicit array syntax in the inner element,
335 # e.g. A%B%C(3), C doesn't have more dimensions because it would
336 # cause a rank mismatch
337 if reference.walk(Range):
338 continue
339
340 # Otherwise we need the datatype to ensure that it represents a
341 # scalar.
342 if not isinstance(reference.datatype, (ScalarType, ArrayType)):
343 if isinstance(reference.symbol, DataSymbol):
344 typestr = f"an {reference.datatype}"
345 else:
346 typestr = "not a DataSymbol"
347 message = (
348 f"{self.name} cannot expand expression because it "
349 f"contains the access '{reference.debug_string()}' "
350 f"which is {typestr} and therefore cannot be guaranteed"
351 f" to be ScalarType.")
352 if not isinstance(reference.symbol, DataSymbol) or \
353 isinstance(reference.symbol.datatype, UnresolvedType):
354 message += (" Resolving the import that brings this"
355 " variable into scope may help.")
356 if verbose:
357 node.append_preceding_comment(message)
358 # pylint: disable=cell-var-from-loop
359 raise TransformationError(LazyString(
360 lambda: f"{message} In:\n{node.debug_string()}"))
361
362 @staticmethod
363 def validate_no_char(node: Node, message: str, options: dict) -> None:
364 '''
365 Check that there is no character variable accessed in the sub-tree with
366 the supplied node at its root.
367
368 :param node: the root node to check for character assignments.
369 :param message: the message to use if a character assignment is found.
370 :param options: any options that apply to this check.
371 :param bool options["verbose"]: log the reason the validation failed,
372 at the moment with a comment in the provided PSyIR node.
373
374 :raises TransformationError: if the supplied node contains a
375 child of character type.
376
377 '''
378 # Whether or not to log the resason for raising an error. At the moment
379 # "logging" means adding a comment in the output code.
380 verbose = options.get("verbose", False)
381
382 for child in node.walk((Literal, Reference)):
383 try:
384 forbidden = ScalarType.Intrinsic.CHARACTER
385 if (child.is_character(unknown_as=False) or
386 (child.symbol.datatype.intrinsic == forbidden)):
387 if verbose:
388 node.append_preceding_comment(message)
389 # pylint: disable=cell-var-from-loop
390 raise TransformationError(LazyString(
391 lambda: f"{message}, but found:"
392 f"\n{node.debug_string()}"))
393 except (NotImplementedError, AttributeError):
394 # We cannot always get the datatype, we ignore this for now
395 pass
