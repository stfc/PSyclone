76class ChildrenList(list):
77 '''
78 Customized list to keep track of the children nodes. It is initialised with
79 a callback function that allows the validation of the inserted children.
80 Since this is a subclass of the standard list, all operations (e.g. append,
81 insert, extend, comparisons, list arithmetic operations) are conserved and
82 make use of the validation. They also trigger an update of all ancestor
83 nodes so that action can be taken in order to keep the tree consistent when
84 necessary (e.g. to update the data-movement clauses on an OpenACC data
85 region).
86
87 :param node: reference to the node where the list belongs.
88 :type node: :py:class:`psyclone.psyir.nodes.Node`
89 :param validation_function: callback function to the validation method.
90 :type validation_function: \
91 function(int, :py:class:`psyclone.psyir.nodes.Node`)
92 :param str validation_text: textual representation of the valid children.
93
94 '''
95 def __init__(self, node, validation_function, validation_text):
96 super().__init__()
97 self._node_reference = node
98 self._validation_function = validation_function
99 self._validation_text_validation_text = validation_text
100
101 def _validate_item(self, index, item):
102 '''
103 Validates the provided index and item before continuing inserting the
104 item into the list.
105
106 :param int index: position where the item is inserted into.
107 :param item: object that needs to be validated in the given position.
108 :type item: :py:class:`psyclone.psyir.nodes.Node`
109
110 :raises GenerationError: if the given index and item are not valid \
111 children for this list.
112 '''
113 if not self._validation_function(index, item):
114 errmsg = (f"Item '{item.__class__.__name__}' can't be child "
115 f"{index} of "
116 f"'{self._node_reference.coloured_name(False)}'.")
117
118 if self._validation_text_validation_text == "<LeafNode>":
119 errmsg = (f"{errmsg} "
120 f"{self._node_reference.coloured_name(False)} is a "
121 f"LeafNode and doesn't accept children.")
122 else:
123 errmsg = (f"{errmsg} The valid format is: "
124 f"'{self._validation_text}'.")
125
126 raise GenerationError(errmsg)

128 def _check_is_orphan(self, item):
129 '''
130 Checks that the provided item is an orphan (has no parent or the parent
131 was predefined in the constructor but the other end of connection has
132 not finalised until now).
133
134 :param item: object that needs to be validated.
135 :type item: :py:class:`psyclone.psyir.nodes.Node`
136
137 :raises GenerationError: if the given item is not an orphan.
138 :raises GenerationError: if the item had been provided with a parent \
139 argument on its constructor and this operation is trying to \
140 make its parent a different node.
141
142 '''
143 if item.parent and not item.has_constructor_parent:
144 raise GenerationError(
145 f"Item '{item.coloured_name(False)}' can't be added as child "
146 f"of '{self._node_reference.coloured_name(False)}' because "
147 f"it is not an orphan. It already has a "
148 f"'{item.parent.coloured_name(False)}' as a parent.")
149
150 if item.parent and item.has_constructor_parent:
151 if item.parent is not self._node_reference:
152 raise GenerationError(
153 f"'{self._node_reference.coloured_name(False)}' cannot be "
154 f"set as parent of '{item.coloured_name(False)}' because "
155 f"its constructor predefined the parent reference to a "
156 f"different '{item.parent.coloured_name(False)}' node.")

215 def insert(self, index, item):
216 ''' Extends list insert method with children node validation.
217
218 :param int index: position where to insert the item.
219 :param item: item to be inserted to the list.
220 :type item: :py:class:`psyclone.psyir.nodes.Node`
221
222 '''
223 positiveindex = index if index >= 0 else len(self) - index
224 self._validate_item(positiveindex, item)
225 self._check_is_orphan(item)
226 # Check that all displaced items will still in valid positions
227 for position in range(positiveindex, len(self)):
228 self._validate_item(position + 1, self[position])
229 super().insert(index, item)
230 self._set_parent_link(item)
231 self._node_reference.update_signal()

233 def extend(self, items):
234 ''' Extends list extend method with children node validation.
235
236 :param items: list of items to be appened to the list.
237 :type items: list of :py:class:`psyclone.psyir.nodes.Node`
238
239 '''
240 for index, item in enumerate(items):
241 self._validate_item(len(self) + index, item)
242 self._check_is_orphan(item)
243 super().extend(items)
244 for item in items:
245 self._set_parent_link(item)

276 def pop(self, index=-1):
277 ''' Extends list pop method with children node validation.
278
279 :param int index: position of the item that is popped out, if not \
280 given, the last element is popped out.
281
282 :returns: the last value or the given index value from the list.
283 :rtype: :py:class:`psyclone.psyir.nodes.Node`
284
285 '''
286 positiveindex = index if index >= 0 else len(self) - index
287 # Check if displaced items after 'positiveindex' will still be valid
288 for position in range(positiveindex + 1, len(self)):
289 self._validate_item(position - 1, self[position])
290 self._del_parent_link(self[index])
291 obj = super().pop(index)
292 self._node_reference.update_signal()
293 return obj

296 ''' Extends list reverse method with children node validation. '''

323class Node():
324 '''
325 Base class for a PSyIR node.
326
327 :param ast: reference into the fparser2 AST corresponding to this node.
328 :type ast: sub-class of :py:class:`fparser.two.Fortran2003.Base`
329 :param children: the PSyIR nodes that are children of this node.
330 :type children: list of :py:class:`psyclone.psyir.nodes.Node`
331 :param parent: that parent of this node in the PSyIR tree.
332 :type parent: :py:class:`psyclone.psyir.nodes.Node`
333 :param annotations: Tags that provide additional information about \
334 the node. The node should still be functionally correct when \
335 ignoring these tags.
336 :type annotations: list of str
337
338 :raises TypeError: if a parent is supplied that is not an instance of Node.
339 :raises InternalError: if an invalid annotation tag is supplied.
340
341 '''
342 # pylint: disable=too-many-public-methods
343 # Define two class constants: START_DEPTH and START_POSITION
344 # START_DEPTH is used to calculate depth of all Nodes in the tree
345 # (1 for main Nodes and increasing for their descendants).
346 START_DEPTH = 0
347 # START_POSITION is used to to calculate position of all Nodes in
348 # the tree (absolute or relative to a parent).
349 START_POSITION = 0
350 # The list of valid annotations for this Node. Populated by sub-class.
351 valid_annotations = tuple()
352 # Textual description of the node. (Set up with None since this is
353 # an abstract class, subclasses need to initialize them with strings.)
354 # In python >= 3 this can be better implemented by creating @classmethod
355 # properties for each of them and chain the ABC @abstractmethod annotation.
356 _children_valid_format = None
357 _text_name = None
358 _colour = None
359
360 def __init__(self, ast=None, children=None, parent=None, annotations=None):
361 if parent and not isinstance(parent, Node):
362 raise TypeError(f"The parent of a Node must also be a Node but "
363 f"got '{type(parent).__name__}'")
364 self._disable_tree_update = True
365 # Keep a record of whether a parent node was supplied when constructing
366 # this object. In this case it still won't appear in the parent's
367 # children list. When both ends of the reference are connected this
368 # will become False.
369 self._has_constructor_parent = parent is not None
370 self._parent = parent
371 self._children = ChildrenList(self, self._validate_child,
372 self._children_valid_format)
373 if children:
374 self._children.extend(children)
375 # Reference into fparser2 AST (if any)
376 self._ast = ast
377 # Ref. to last fparser2 parse tree node associated with this Node.
378 # This is required when adding directives.
379 self._ast_end = None
380 # List of tags that provide additional information about this Node.
381 self._annotations = []
382 if annotations:
383 for annotation in annotations:
384 if annotation in self.valid_annotations:
385 self._annotations.append(annotation)
386 else:
387 raise InternalError(
388 f"{self.__class__.__name__} with unrecognised "
389 f"annotation '{annotation}', valid "
390 f"annotations are: {self.valid_annotations}.")
391 self._disable_tree_update = False
392 self.update_signal()

394 def __eq__(self, other):
395 '''
396 Checks whether two nodes are equal. The basic implementation of this
397 checks whether the nodes are the same type, and whether all children
398 of the nodes are equal, and if so then
399 they are considered equal.
400
401 :param object other: the object to check equality to.
402
403 :returns: whether other is equal to self.
404 :rtype: bool
405 '''
406 super().__eq__(other)
407 is_eq = type(self) is type(other)
408 is_eq = is_eq and (len(self.childrenchildrenchildren) == len(other.children))
409 for index, child in enumerate(self.childrenchildrenchildren):
410 is_eq = is_eq and child == other.children[index]
411
412 return is_eq

563 def dag_gen(self, graph):
564 '''Output my node's graph (dag) information and call any
565 children. Nodes with children are represented as two vertices,
566 a start and an end. Forward dependencies are represented as
567 green edges, backward dependencies are represented as red
568 edges (but their direction is reversed so the layout looks
569 reasonable) and parent child dependencies are represented as
570 blue edges.'''
571 # Import outside top-level to avoid circular dependencies.
572 # pylint: disable=too-many-branches, import-outside-toplevel
573 from psyclone.psyir.nodes.loop import Loop
574 from psyclone.psyir.nodes.routine import Routine
575
576 # names to append to my default name to create start and end vertices
577 start_postfix = "_start"
578 end_postfix = "_end"
579 if self.childrenchildrenchildren:
580 # I am represented by two vertices, a start and an end
581 graph.node(self.dag_namedag_name+start_postfix)
582 graph.node(self.dag_namedag_name+end_postfix)
583 else:
584 # I am represented by a single vertex
585 graph.node(self.dag_namedag_name)
586 # first deal with forward dependencies
587 remote_node = self.forward_dependence()
588 local_name = self.dag_namedag_name
589 if self.childrenchildrenchildren:
590 # edge will come from my end vertex as I am a forward dependence
591 local_name += end_postfix
592 if remote_node:
593 # this node has a forward dependence
594 remote_name = remote_node.dag_name
595 if remote_node.children:
596 # the remote node has children so I will connect to
597 # its start vertex
598 remote_name += start_postfix
599 # Create the forward dependence edge in green
600 graph.edge(local_name, remote_name, color="green")
601 elif not isinstance(self, Routine):
602 # If this node is not a Routine (where the DAG context finishes)
603 # and has no forward dependence, connect it to the end vertex
604 # of its parent. Use blue to indicate a parent child
605 # relationship.
606 remote_name = self.parentparent.dag_name + end_postfix
607 graph.edge(local_name, remote_name, color="blue")
608 # now deal with backward dependencies. When creating the edges
609 # we reverse the direction of the dependence (place
610 # remote_node before local_node) to help with the graph
611 # layout
612 remote_node = self.backward_dependence()
613 local_name = self.dag_namedag_name
614 if self.childrenchildrenchildren:
615 # the edge will come from my start vertex as I am a
616 # backward dependence
617 local_name += start_postfix
618 if remote_node:
619 # this node has a backward dependence.
620 remote_name = remote_node.dag_name
621 if remote_node.children:
622 # the remote node has children so I will connect to
623 # its end vertex
624 remote_name += end_postfix
625 # Create the backward dependence edge in red.
626 graph.edge(remote_name, local_name, color="red")
627 elif not isinstance(self, Routine):
628 # If this node is not a Routine (where the DAG context finishes)
629 # and has no backward dependence, connect it to the start vertex
630 # of its parent. Use blue to indicate a parent child
631 # relationship.
632 remote_name = self.parentparent.dag_name + start_postfix
633 graph.edge(remote_name, local_name, color="blue")
634 # now call any children so they can add their information to
635 # the graph
636 if isinstance(self, Loop):
637 # In case of a loop only look at the body (the other part
638 # of the tree contain start, stop, step values):
639 self.loop_body.dag_gen(graph)
640 else:
641 for child in self.childrenchildrenchildren:
642 child.dag_gen(graph)

676 def backward_dependence(self):
677 '''Returns the closest preceding Node that this Node has a direct
678 dependence with or None if there is not one. Only Nodes with
679 the same parent as self are returned. Nodes inherit their
680 descendants' dependencies. The reason for this is that for
681 correctness a node must maintain its parent if it is
682 moved. For example a halo exchange and a kernel call may have
683 a dependence between them but it is the loop body containing
684 the kernel call that the halo exchange must not move beyond
685 i.e. the loop body inherits the dependencies of the routines
686 within it.'''
687 dependence = None
688 # look through all the backward dependencies of my arguments
689 for arg in self.argsargs:
690 dependent_arg = arg.backward_dependence()
691 if dependent_arg:
692 # this argument has a backward dependence
693 node = dependent_arg.call
694 # if the remote node is deeper in the tree than me
695 # then find the ancestor that is at the same level of
696 # the tree as me.
697 while node.depth > self.depthdepth:
698 node = node.parent
699 if self.sameParent(node) and node is not self:
700 # The remote node (or one of its ancestors) shares
701 # the same parent as me (but its not me)
702 if not dependence:
703 # this is the first dependence found so keep it
704 dependence = node
705 else:
706 # we have already found a dependence
707 if dependence.position < node.position:
708 # the new dependence is closer to me than
709 # the previous dependence so keep it
710 dependence = node
711 return dependence

713 def forward_dependence(self):
714 '''Returns the closest following Node that this Node has a direct
715 dependence with or None if there is not one. Only Nodes with
716 the same parent as self are returned. Nodes inherit their
717 descendants' dependencies. The reason for this is that for
718 correctness a node must maintain its parent if it is
719 moved. For example a halo exchange and a kernel call may have
720 a dependence between them but it is the loop body containing
721 the kernel call that the halo exchange must not move beyond
722 i.e. the loop body inherits the dependencies of the routines
723 within it.'''
724 dependence = None
725 # look through all the forward dependencies of my arguments
726 for arg in self.argsargs:
727 dependent_arg = arg.forward_dependence()
728 if dependent_arg:
729 # this argument has a forward dependence
730 node = dependent_arg.call
731 # if the remote node is deeper in the tree than me
732 # then find the ancestor that is at the same level of
733 # the tree as me.
734 while node.depth > self.depthdepth:
735 node = node.parent
736 if self.sameParent(node) and node is not self:
737 # The remote node (or one of its ancestors) shares
738 # the same parent as me (but its not me)
739 if not dependence:
740 # this is the first dependence found so keep it
741 dependence = node
742 else:
743 if dependence.position > node.position:
744 # the new dependence is closer to me than
745 # the previous dependence so keep it
746 dependence = node
747 return dependence

749 def is_valid_location(self, new_node, position="before"):
750 '''If this Node can be moved to the new_node
751 (where position determines whether it is before of after the
752 new_node) without breaking any data dependencies then return True,
753 otherwise return False.
754
755 :param new_node: Node to which this node should be moved.
756 :type new_node: :py:class:`psyclone.psyir.nodes.Node`
757 :param str position: either 'before' or 'after'.
758
759 :raises GenerationError: if new_node is not an\
760 instance of :py:class:`psyclone.psyir.nodes.Node`.
761 :raises GenerationError: if position is not 'before' or 'after'.
762 :raises GenerationError: if self and new_node do not have the same\
763 parent.
764 :raises GenerationError: self and new_node are the same Node.
765
766 :returns: whether or not the specified location is valid for this node.
767 :rtype: bool
768
769 '''
770 # First perform correctness checks
771 # 1: check new_node is a Node
772 if not isinstance(new_node, Node):
773 raise GenerationError(
774 f"In the psyir.nodes.Node.is_valid_location() method the "
775 f"supplied argument is not a Node, it is a "
776 f"'{type(new_node).__name__}'.")
777
778 # 2: check position has a valid value
779 valid_positions = ["before", "after"]
780 if position not in valid_positions:
781 raise GenerationError(
782 f"The position argument in the psyGenNode.is_valid_location() "
783 f"method must be one of {valid_positions} but found "
784 f"'{position}'")
785
786 # 3: check self and new_node have the same parent
787 if not self.sameParent(new_node):
788 raise GenerationError(
789 "In the psyir.nodes.Node.is_valid_location() method "
790 "the node and the location do not have the same parent")
791
792 # 4: check proposed new position is not the same as current position
793 new_position = new_node.position
794 if new_position < self.positionposition and position == "after":
795 new_position += 1
796 elif new_position > self.positionposition and position == "before":
797 new_position -= 1
798
799 if self.positionposition == new_position:
800 raise GenerationError(
801 "In the psyir.nodes.Node.is_valid_location() method, the "
802 "node and the location are the same so this transformation "
803 "would have no effect.")
804
805 # Now determine whether the new location is valid in terms of
806 # data dependencies
807 # Treat forward and backward dependencies separately
808 if new_position < self.positionposition:
809 # the new_node is before this node in the schedule
810 prev_dep_node = self.backward_dependence()
811 if not prev_dep_node:
812 # There are no backward dependencies so the move is valid
813 return True
814 # return (is the dependent node before the new_position?)
815 return prev_dep_node.position < new_position
816 # new_node.position > self.position
817 # the new_node is after this node in the schedule
818 next_dep_node = self.forward_dependence()
819 if not next_dep_node:
820 # There are no forward dependencies so the move is valid
821 return True
822 # return (is the dependent node after the new_position?)
823 return next_dep_node.position > new_position

1061 def walk(self, my_type, stop_type=None, depth=None):
1062 ''' Recurse through the PSyIR tree and return all objects that are
1063 an instance of 'my_type', which is either a single class or a tuple
1064 of classes. In the latter case all nodes are returned that are
1065 instances of any classes in the tuple. The recursion into the tree
1066 is stopped if an instance of 'stop_type' (which is either a single
1067 class or a tuple of classes) is found. This can be used to avoid
1068 analysing e.g. inlined kernels, or as performance optimisation to
1069 reduce the number of recursive calls. The recursion into the tree is
1070 also stopped if the (optional) 'depth' level is reached.
1071
1072 :param my_type: the class(es) for which the instances are collected.
1073 :type my_type: type | Tuple[type, ...]
1074 :param stop_type: class(es) at which recursion is halted (optional).
1075 :type stop_type: Optional[type | Tuple[type, ...]]
1076 :param depth: the depth value the instances must have (optional).
1077 :type depth: Optional[int]
1078
1079 :returns: list with all nodes that are instances of my_type \
1080 starting at and including this node.
1081 :rtype: List[:py:class:`psyclone.psyir.nodes.Node`]
1082
1083 '''
1084 local_list = []
1085 if isinstance(self, my_type) and depth in [None, self.depthdepth]:
1086 local_list.append(self)
1087
1088 # Stop recursion further into the tree if an instance of a class
1089 # listed in stop_type is found.
1090 if stop_type and isinstance(self, stop_type):
1091 return local_list
1092
1093 # Stop recursion further into the tree if a depth level has been
1094 # specified and it is reached.
1095 if depth is not None and self.depthdepth >= depth:
1096 return local_list
1097
1098 for child in self.childrenchildrenchildren:
1099 local_list += child.walk(my_type, stop_type, depth=depth)
1100 return local_list

1102 def get_sibling_lists(self, my_type, stop_type=None):
1103 '''
1104 Recurse through the PSyIR tree and return lists of Nodes that are
1105 instances of 'my_type' and are immediate siblings. Here 'my_type' is
1106 either a single class or a tuple of classes. In the latter case all
1107 nodes are returned that are instances of any classes in the tuple. The
1108 recursion into the tree is stopped if an instance of 'stop_type' (which
1109 is either a single class or a tuple of classes) is found.
1110
1111 :param my_type: the class(es) for which the instances are collected.
1112 :type my_type: type | Tuple[type, ...]
1113 :param stop_type: class(es) at which recursion is halted (optional).
1114 :type stop_type: Optional[type | Tuple[type, ...]]
1115
1116 :returns: list of lists, each of which containing nodes that are
1117 instances of my_type and are immediate siblings, starting at
1118 and including this node.
1119 :rtype: List[List[:py:class:`psyclone.psyir.nodes.Node`]]
1120
1121 '''
1122 # Separate nodes by depth
1123 by_depth = {}
1124 for node in self.walk(my_type, stop_type=stop_type):
1125 depth = node.depth
1126 if depth not in by_depth:
1127 by_depth[depth] = []
1128 by_depth[depth].append(node)
1129
1130 # Determine lists of consecutive nodes
1131 global_list = []
1132 for depth, local_list in sorted(by_depth.items()):
1133 block = []
1134 for node in local_list:
1135 if not block or node.immediately_follows(block[-1]):
1136 block.append(node)
1137 else:
1138 # Current node does not immediately follow the previous one
1139 # so this marks the end of the current block.
1140 global_list.append(block)
1141 block = [node]
1142 if block:
1143 global_list.append(block)
1144 return global_list

1146 def ancestor(self, my_type, excluding=None, include_self=False,
1147 limit=None, shared_with=None):
1148 '''
1149 Search back up the tree and check whether this node has an ancestor
1150 that is an instance of the supplied type. If it does then we return
1151 it otherwise we return None. An individual (or tuple of) (sub-)
1152 class(es) to ignore may be provided via the `excluding` argument. If
1153 `include_self` is True then the current node is included in the search.
1154 If `limit` is provided then the search ceases if/when the supplied
1155 node is encountered.
1156 If `shared_with` is provided, then the ancestor search will find an
1157 ancestor of both this node and the node provided as `shared_with` if
1158 such an ancestor exists.
1159
1160 :param my_type: class(es) to search for.
1161 :type my_type: type | Tuple[type, ...]
1162 :param excluding: (sub-)class(es) to ignore or None.
1163 :type excluding: Optional[type | Tuple[type, ...]]
1164 :param bool include_self: whether or not to include this node in the \
1165 search.
1166 :param limit: an optional node at which to stop the search.
1167 :type limit: Optional[:py:class:`psyclone.psyir.nodes.Node`]
1168 :param shared_with: an optional node which must also have the
1169 found node as an ancestor.
1170 :type shared_with: Optional[:py:class:`psyclone.psyir.nodes.Node`]
1171
1172 :returns: First ancestor Node that is an instance of any of the \
1173 requested classes or None if not found.
1174 :rtype: Optional[:py:class:`psyclone.psyir.nodes.Node`]
1175
1176 :raises TypeError: if `excluding` is provided but is not a type or \
1177 tuple of types.
1178 :raises TypeError: if `limit` is provided but is not an instance \
1179 of Node.
1180 '''
1181 if include_self:
1182 myparent = self
1183 else:
1184 myparent = self.parentparent
1185
1186 if excluding is not None:
1187 if isinstance(excluding, type):
1188 excludes = (excluding, )
1189 elif isinstance(excluding, tuple):
1190 excludes = excluding
1191 else:
1192 raise TypeError(
1193 f"The 'excluding' argument to ancestor() must be a type or"
1194 f" a tuple of types but got: '{type(excluding).__name__}'")
1195
1196 if limit and not isinstance(limit, Node):
1197 raise TypeError(
1198 f"The 'limit' argument to ancestor() must be an instance of "
1199 f"Node but got '{type(limit).__name__}'")
1200
1201 # If we need to find a shared ancestor, then find a starting ancestor
1202 # for the sharing node.
1203 shared_ancestor = None
1204 if shared_with is not None:
1205 shared_ancestor = shared_with.ancestor(
1206 my_type, excluding=excluding,
1207 include_self=include_self, limit=limit)
1208
1209 while myparent is not None:
1210 if isinstance(myparent, my_type):
1211 if not (excluding and isinstance(myparent, excludes)):
1212 # If this is a valid instance but not the same as for
1213 # the shared_with node, we do logic afterwards to continue
1214 # searching upwards, as we could be either higher or
1215 # lower than the shared_ancestor found previously.
1216 if shared_ancestor and shared_ancestor is not myparent:
1217 break
1218 # This parent node is not an instance of an excluded
1219 # sub-class so return it
1220 return myparent
1221 if myparent is limit:
1222 break
1223 myparent = myparent.parent
1224
1225 # We search up the tree until we find an ancestor of the requested
1226 # type(s) shared by the shared_with node.
1227 while (myparent is not shared_ancestor and myparent and
1228 shared_ancestor):
1229 if myparent is limit:
1230 break
1231 if myparent.depth > shared_ancestor.depth:
1232 # If myparent is deeper in the tree than the current
1233 # potential shared ancestor, search upwards to find
1234 # the next valid ancestor of this node.
1235 myparent = myparent.ancestor(
1236 my_type, excluding=excluding,
1237 include_self=False, limit=limit)
1238 else:
1239 # shared_ancestor is equal or deeper in the tree than
1240 # myparent, so search upwards for the next valid ancestor
1241 # of shared_ancestor.
1242 shared_ancestor = shared_ancestor.ancestor(
1243 my_type, excluding=excluding, include_self=False,
1244 limit=limit)
1245 # If myparent is shared ancestor then return myparent.
1246 if myparent is shared_ancestor:
1247 return myparent
1248
1249 # Otherwise we didn't find an ancestor that was valid.
1250 return None
