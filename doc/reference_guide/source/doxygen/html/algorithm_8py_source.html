70def parse(alg_filename, api="", invoke_name="invoke", kernel_paths=None,
71 line_length=False):
72 '''Takes a PSyclone conformant algorithm file as input and outputs a
73 parse tree of the code contained therein and an object containing
74 information about the 'invoke' calls in the algorithm file and any
75 associated kernels within the invoke calls.
76
77 :param str alg_filename: the file containing the algorithm \
78 specification.
79 :param str api: the PSyclone API to use when parsing the \
80 code. Defaults to empty string.
81 :param str invoke_name: the expected name of the invocation calls \
82 in the algorithm code. Defaults to "invoke".
83 :param kernel_paths: the paths to search for kernel source files \
84 (if different from the location of the algorithm source). \
85 Defaults to None.
86 :type kernel_paths: list of str or NoneType
87 :param bool line_length: a logical flag specifying whether we care \
88 about line lengths being longer than 132 characters. If so, \
89 the input (algorithm and kernel) code is checked to make sure \
90 that it conforms and an error raised if not. The default is \
91 False.
92
93 :returns: 2-tuple consisting of the fparser2 parse tree of the \
94 Algorithm file and an object holding details of the invokes \
95 found.
96 :rtype: (:py:class:`fparser.two.Fortran2003.Program`, \
97 :py:class:`psyclone.parse.FileInfo`)
98
99 For example:
100
101 >>> from psyclone.parse.algorithm import parse
102 >>> ast, info = parse(SOURCE_FILE)
103
104 '''
105 if kernel_paths is None:
106 kernel_paths = []
107 # Parsing is encapsulated in the Parser class. We keep this
108 # function for compatibility.
109 my_parser = Parser(api, invoke_name, kernel_paths, line_length)
110 return my_parser.parse(alg_filename)

113# pylint: disable=too-many-instance-attributes
114class Parser():
115 '''Supports the parsing of PSyclone conformant algorithm code within a
116 file and extraction of relevant information for any 'invoke' calls
117 contained within the code.
118
119 :param str api: the PSyclone API to use when parsing the code.
120 :param str invoke_name: the expected name of the invocation calls in the \
121 algorithm code.
122 :param kernel_paths: the paths to search for kernel source files \
123 (if different from the location of the algorithm source). \
124 Defaults to None.
125 :type kernel_paths: list of str or NoneType
126 :param bool line_length: a logical flag specifying whether we care \
127 about line lengths being longer than 132 characters. If so, \
128 the input (algorithm and kernel) code is checked to make sure \
129 that it conforms and an error raised if not. The default is \
130 False.
131
132 For example:
133
134 >>> from psyclone.parse.algorithm import Parser
135 >>> parser = Parser(api="gocean")
136 >>> ast, info = parser.parse(SOURCE_FILE)
137
138 '''
139
140 def __init__(self, api="", invoke_name="invoke", kernel_paths=None,
141 line_length=False):
142
143 self._invoke_name = invoke_name
144 if kernel_paths is None:
145 self._kernel_paths = []
146 else:
147 self._kernel_paths = kernel_paths
148 self._line_length = line_length
149
150 check_api(api)
151 self._api = api
152 Config.get().api = api
153
154 self._arg_name_to_module_name = {}
155 # Dict holding a 2-tuple consisting of type and precision
156 # information for each variable declared in the algorithm
157 # file, indexed by variable name.
158 self._arg_type_defns_arg_type_defns = {}
159 self._unique_invoke_labels = []
160
161 # Use the get_builtin_defs helper function to access
162 # information about the builtins supported by this API. The
163 # first argument contains the names of the builtins and the
164 # second is the file where these names are defined.
165 self._builtin_name_map, \
166 self._builtin_defs_file = get_builtin_defs(self._api)
167
168 self._alg_filename = None
169
170 def parse(self, alg_filename):
171 '''Takes a PSyclone conformant algorithm file as input and outputs a
172 parse tree of the code contained therein and an object
173 containing information about the 'invoke' calls in the
174 algorithm file and any associated kernels within the invoke
175 calls.
176
177 :param str alg_filename: The file containing the algorithm code.
178
179 :returns: 2-tuple consisting of the fparser2 parse tree of the
180 algorithm code and an object holding details of the
181 algorithm code and the invokes found within it.
182 :rtype: Tuple[:py:class:`fparser.two.Fortran2003.Program`,
183 :py:class:`psyclone.parse.FileInfo`]
184
185 '''
186 self._alg_filename = alg_filename
187 if self._line_length:
188 # Make sure the code conforms to the line length limit.
189 check_line_length(alg_filename)
190 alg_parse_tree = parse_fp2(alg_filename)
191
192 return alg_parse_tree, self.invoke_info(alg_parse_tree)

194 def invoke_info(self, alg_parse_tree):
195 '''Takes an fparser2 representation of a PSyclone-conformant algorithm
196 code as input and returns an object containing information
197 about the 'invoke' calls in the algorithm file and any
198 associated kernels within the invoke calls. Also captures the
199 type and precision of every variable declaration within the
200 parse tree.
201
202 :param alg_parse_tree: the fparser2 representation of the \
203 algorithm code.
204 :type: :py:class:`fparser.two.Fortran2003.Program`
205
206 :returns: an object holding details of the algorithm \
207 code and the invokes found within it.
208 :rtype: :py:class:`psyclone.parse.FileInfo`
209
210 :raises ParseError: if a program, module, subroutine or \
211 function is not found in the fparser2 tree.
212 :raises InternalError: if the fparser2 tree representing the \
213 type declaration statements is not in the expected form.
214 :raises NotImplementedError: if the algorithm code contains \
215 two different datatypes with the same name.
216
217 '''
218 # Find the first program, module, subroutine or function in the
219 # parse tree. The assumption here is that the first is the one
220 # that is required. See issue #307.
221 container_name = None
222 for child in alg_parse_tree.content:
223 if isinstance(child, (Main_Program, Module, Subroutine_Subprogram,
224 Function_Subprogram)):
225 container_name = str(child.content[0].items[1])
226 break
227
228 if not container_name:
229 # Nothing relevant found.
230 raise ParseError(
231 "algorithm.py:parser:parse: Program, module, function or "
232 "subroutine not found in fparser2 parse tree.")
233
234 self._unique_invoke_labels = []
235 self._arg_name_to_module_name = OrderedDict()
236 # Dict holding a 2-tuple consisting of type and precision
237 # information for each variable declared in the algorithm
238 # file, indexed by variable name.
239 self._arg_type_defns_arg_type_defns = {}
240 invoke_calls = []
241
242 # Find all invoke calls and capture information about
243 # them. Also find information about use statements and find
244 # all declarations within the supplied parse
245 # tree. Declarations will include the definitions of any
246 # components of derived types that are defined within the
247 # code.
248 for statement in walk(alg_parse_tree.content,
249 types=(Type_Declaration_Stmt,
250 Data_Component_Def_Stmt,
251 Use_Stmt, Call_Stmt)):
252 if isinstance(statement,
253 (Type_Declaration_Stmt, Data_Component_Def_Stmt)):
254 # Capture datatype information for the variable
255 spec = statement.children[0]
256 if isinstance(spec, Declaration_Type_Spec):
257 # This is a type declaration
258 my_type = spec.children[1].string.lower()
259 my_precision = None
260 elif isinstance(spec, Intrinsic_Type_Spec):
261 # This is an intrinsic declaration
262 my_type = spec.children[0].lower()
263 my_precision = None
264 if isinstance(spec.children[1], Kind_Selector):
265 my_precision = \
266 spec.children[1].children[1].string.lower()
267 else:
268 raise InternalError(
269 f"Expected first child of Type_Declaration_Stmt or "
270 f"Data_Component_Def_Stmt to be Declaration_Type_Spec "
271 f"or Intrinsic_Type_Spec but found "
272 f"'{type(spec).__name__}'")
273 for decl in walk(statement.children[2], (
274 Entity_Decl, Component_Decl)):
275 # Determine the variables names. Note that if a
276 # variable declaration is a component of a derived
277 # type, its name is stored 'as is'. This means
278 # that e.g. real :: a will clash with a
279 # derived-type definition if the latter has a
280 # component named 'a' and their datatypes differ.
281 my_var_name = decl.children[0].string.lower()
282 if my_var_name in self._arg_type_defns_arg_type_defns and (
283 self._arg_type_defns_arg_type_defns[my_var_name][0] != my_type or
284 self._arg_type_defns_arg_type_defns[my_var_name][1] !=
285 my_precision):
286 raise NotImplementedError(
287 f"The same symbol '{my_var_name}' is used for "
288 f"different datatypes, "
289 f"'{self._arg_type_defns[my_var_name][0]}, "
290 f"{self._arg_type_defns[my_var_name][1]}' and "
291 f"'{my_type}, {my_precision}'. This is not "
292 f"currently supported.")
293 # Store the variable name and information about its type
294 self._arg_type_defns_arg_type_defns[my_var_name] = (my_type, my_precision)
295
296 if isinstance(statement, Use_Stmt):
297 # found a Fortran use statement
298 self.update_arg_to_module_map(statement)
299
300 if isinstance(statement, Call_Stmt):
301 # found a Fortran call statement
302 call_name = str(statement.items[0])
303 if call_name.lower() == self._invoke_name.lower():
304 # The call statement is an invoke
305 invoke_call = self.create_invoke_call(statement)
306 invoke_calls.append(invoke_call)
307
308 return AlgFileInfo(container_name, invoke_calls)

310 def create_invoke_call(self, statement):
311 '''Takes the part of a parse tree containing an invoke call and
312 returns an InvokeCall object which captures the required
313 information about the invoke.
314
315 :param statement: Parse tree of the invoke call.
316 :type statement: :py:class:`fparser.two.Fortran2003.Call_Stmt`
317 :returns: An InvokeCall object which contains relevant \
318 information about the invoke call.
319 :rtype: :py:class:`psyclone.parse.algorithm.InvokeCall`
320 :raises ParseError: if more than one invoke argument contains \
321 'name=xxx'.
322 :raises ParseError: if an unknown or unsupported invoke \
323 argument is found.
324
325 '''
326 # Extract argument list.
327 argument_list = statement.items[1].items
328
329 invoke_label = None
330 kernel_calls = []
331
332 for argument in argument_list:
333
334 if isinstance(argument, Actual_Arg_Spec):
335 # This should be the invoke label.
336 if invoke_label:
337 raise ParseError(
338 f"algorithm.py:Parser():create_invoke_call: An invoke "
339 f"must contain one or zero 'name=xxx' arguments but "
340 f"found more than one in: {statement} in file "
341 f"{self._alg_filename}")
342 invoke_label = self.check_invoke_label(argument)
343
344 elif isinstance(
345 argument, (Data_Ref, Part_Ref, Structure_Constructor)):
346 # This should be a kernel call.
347 kernel_call = self.create_kernel_call(argument)
348 kernel_calls.append(kernel_call)
349
350 else:
351 # Unknown and/or unsupported argument type
352 raise ParseError(
353 f"algorithm.py:Parser():create_invoke_call: Expecting "
354 f"argument to be of the form 'name=xxx' or a Kernel call "
355 f"but found '{argument}' in file '{self._alg_filename}'.")
356
357 return InvokeCall(kernel_calls, name=invoke_label)
358
359 def create_kernel_call(self, argument):
360 '''Takes the parse tree of an invoke argument containing a
361 reference to a kernel or a builtin and returns the kernel or
362 builtin object respectively which contains the required
363 information.
364
365 :param argument: Parse tree of an invoke argument. This \
366 should contain a kernel name and associated arguments.
367 :type argument: :py:class:`fparser.two.Fortran2003.Part_Ref` or \
368 :py:class:`fparser.two.Fortran2003.Structure_Constructor`
369
370 :returns: A builtin or coded kernel call object which contains \
371 relevant information about the Kernel.
372 :rtype: :py:class:`psyclone.parse.algorithm.KernelCall` or \
373 :py:class:`psyclone.parse.algorithm.BuiltInCall`
374
375 '''
376 kernel_name, args = get_kernel(argument, self._alg_filename,
377 self._arg_type_defns_arg_type_defns)
378 if kernel_name.lower() in self._builtin_name_map:
379 # This is a builtin kernel
380 kernel_call = self.create_builtin_kernel_call(
381 kernel_name, args)
382 else:
383 # This is a coded kernel
384 kernel_call = self.create_coded_kernel_call(
385 kernel_name, args)
386 return kernel_call

388 def create_builtin_kernel_call(self, kernel_name, args):
389 '''Takes the builtin kernel name and a list of Arg objects which
390 capture information about the builtin call arguments and
391 returns a BuiltinCall instance with content specific to the
392 particular API (as specified in self._api).
393
394 :param str kernel_name: the name of the builtin kernel being \
395 called
396 :param args: a list of 'Arg' instances containing the required \
397 information for the arguments being passed from the algorithm \
398 layer. The list order is the same as the argument order.
399 :type arg: list of :py:class:`psyclone.parse.algorithm.Arg`
400 :returns: a BuiltInCall instance with information specific to \
401 the API.
402 :rtype: :py:class:`psyclone.parse.algorithm.BuiltInCall`
403 :raises ParseError: if the builtin is specified in a use \
404 statement in the algorithm layer
405
406 '''
407 if kernel_name.lower() in self._arg_name_to_module_name:
408 raise ParseError(
409 f"A built-in cannot be named in a use statement but "
410 f"'{kernel_name}' is used from module "
411 f"'{self._arg_name_to_module_name[kernel_name.lower()]}' in "
412 f"file {self._alg_filename}")
413
414 return BuiltInCall(BuiltInKernelTypeFactory(api=self._api).create(
415 self._builtin_name_map.keys(), self._builtin_defs_file,
416 name=kernel_name.lower()), args)
417
418 def create_coded_kernel_call(self, kernel_name, args):
419 '''Takes a coded kernel name and a list of Arg objects which
420 capture information about the coded call arguments and
421 returns a KernelCall instance with content specific to the
422 particular API (as specified in self._api).
423
424 :param str kernel_name: the name of the coded kernel being \
425 called
426 :param args: a list of 'Arg' instances containing the required \
427 information for the arguments being passed from the algorithm \
428 layer. The list order is the same as the argument order.
429 :type arg: list of :py:class:`psyclone.parse.algorithm.Arg`
430 :returns: a KernelCall instance with information specific to \
431 the API.
432 :rtype: :py:class:`psyclone.parse.algorithm.KernelCall`
433 :raises ParseError: if the kernel is not specified in a use \
434 statement in the algorithm layer
435
436 '''
437 try:
438 module_name = self._arg_name_to_module_name[kernel_name.lower()]
439 except KeyError as info:
440 message = (
441 f"kernel call '{kernel_name.lower()}' must either be named in "
442 f"a use statement (found "
443 f"{list(self._arg_name_to_module_name.values())}) or be a "
444 f"recognised built-in (one of "
445 f"'{list(self._builtin_name_map.keys())}' for "
446 f"this API)")
447 raise ParseError(message) from info
448
449 modast = get_kernel_ast(module_name, self._alg_filename,
450 self._kernel_paths, self._line_length)
451 return KernelCall(module_name,
452 KernelTypeFactory(api=self._api).create(
453 modast, name=kernel_name), args)

455 def update_arg_to_module_map(self, statement):
456 '''Takes a use statement and adds its contents to the internal
457 arg_name_to_module_name map. This map associates names
458 specified in the 'only' list with the corresponding use name.
459
460 :param statement: A use statement
461 :type statement: :py:class:`fparser.two.Fortran2003.Use_Stmt`
462 :raises InternalError: if the statement being passed is not an \
463 fparser use statement.
464
465 '''
466 # make sure statement is a use
467 if not isinstance(statement, Use_Stmt):
468 raise InternalError(
469 f"algorithm.py:Parser:update_arg_to_module_map: Expected "
470 f"a use statement but found instance of "
471 f"'{type(statement)}'.")
472
473 use_name = str(statement.items[2])
474
475 # Extract 'only' list.
476 if statement.items[4]:
477 only_list = statement.items[4].items
478 else:
479 only_list = []
480
481 for item in only_list:
482 self._arg_name_to_module_name[str(item).lower()] = use_name
483
484 def check_invoke_label(self, argument):
485 '''Takes the parse tree of an invoke argument containing an invoke
486 label. Raises an exception if this label has already been used
487 by another invoke in the same algorithm code. If all is well
488 it returns the label as a string.
489
490 :param argument: Parse tree of an invoke argument. This \
491 should contain a "name=xxx" argument.
492 :type argument: :py:class:`fparser.two.Actual_Arg_Spec`
493 :returns: the label as a string.
494 :rtype: str
495 :raises ParseError: if this label has already been used by \
496 another invoke in this algorithm code.
497
498 '''
499 invoke_label = get_invoke_label(argument, self._alg_filename)
500 if invoke_label in self._unique_invoke_labels:
501 raise ParseError(
502 f"Found multiple named invoke()'s with the same "
503 f"label ('{invoke_label}') when parsing {self._alg_filename}")
504 self._unique_invoke_labels.append(invoke_label)
505 return invoke_label

513def get_builtin_defs(api):
514 '''
515 Get the names of the supported built-in operations and the file
516 containing the associated meta-data for the supplied API
517
518 :param str api: the specified PSyclone API.
519 :returns: a 2-tuple containing a dictionary of the supported \
520 built-ins and the filename where these built-ins are specified.
521 :rtype: (dict, str)
522
523 '''
524
525 # Check that the supplied API is valid
526 check_api(api)
527
528 # pylint: disable=import-outside-toplevel
529 if api in LFRIC_API_NAMES:
530 from psyclone.domain.lfric.lfric_builtins import BUILTIN_MAP \
531 as builtins
532 from psyclone.domain.lfric.lfric_builtins import \
533 BUILTIN_DEFINITIONS_FILE as fname
534 else:
535 # We don't support any built-ins for this API
536 builtins = {}
537 fname = None
538 return builtins, fname

541def get_invoke_label(parse_tree, alg_filename, identifier="name"):
542 '''Takes an invoke argument contained in the parse_tree argument and
543 returns the label specified within it.
544
545 :param parse_tree: Parse tree of an invoke argument. This should \
546 contain a "name=xxx" argument.
547 :type parse_tree: :py:class:`fparser.two.Actual_Arg_Spec`
548 :param str alg_filename: The file containing the algorithm code.
549 :param str identifier: An optional name used to specify a named argument. \
550 Defaults to 'name'.
551
552 :returns: the label as a lower-cased string.
553 :rtype: str
554
555 :except InternalError: if the form of the argument is not what was \
556 expected.
557 :except InternalError: if the number of items contained in the \
558 argument is not what was expected.
559 :except ParseError: if the name used for the named argument does \
560 not match what was expected.
561 :except ParseError: if the label is not specified as a string.
562 :except ParseError: if the label is not a valid Fortran name.
563
564 '''
565 if not isinstance(parse_tree, Actual_Arg_Spec):
566 raise InternalError(
567 f"algorithm.py:Parser:get_invoke_label: Expected a Fortran "
568 f"argument of the form name=xxx but found instance of "
569 f"'{type(parse_tree)}'.")
570
571 if len(parse_tree.items) != 2:
572 raise InternalError(
573 f"algorithm.py:Parser:get_invoke_label: Expected the Fortran "
574 f"argument to have two items but found "
575 f"'{len(parse_tree.items)}'.")
576
577 ident = str(parse_tree.items[0])
578 if ident.lower() != identifier.lower():
579 raise ParseError(
580 f"algorithm.py:Parser:get_invoke_label: Expected named identifier "
581 f"to be '{identifier.lower()}' but found '{ident.lower()}'")
582
583 if not isinstance(parse_tree.items[1], Char_Literal_Constant):
584 raise ParseError(
585 f"algorithm.py:Parser:get_invoke_label: The (optional) name of an "
586 f"invoke must be specified as a string, but found "
587 f"{parse_tree.items[1]} in {alg_filename}")
588
589 invoke_label = parse_tree.items[1].items[0]
590 invoke_label = invoke_label.lower()
591 invoke_label = invoke_label.strip()
592 if invoke_label[0] == '"' and invoke_label[-1] == '"' or \
593 invoke_label[0] == "'" and invoke_label[-1] == "'":
594 invoke_label = invoke_label[1:-1]
595
596 try:
597 if invoke_label:
598 FortranReader.validate_name(invoke_label)
599 # We store any name as lowercase.
600 invoke_label = invoke_label.lower()
601 except (TypeError, ValueError) as err:
602 raise (
603 ParseError(
604 f"algorithm.py:Parser:get_invoke_label the (optional) name of "
605 f"an invoke must be a string containing a valid Fortran name "
606 f"(with no whitespace) but got '{invoke_label}' in file "
607 f"{alg_filename}")) from err
608
609 return invoke_label

612def get_kernel(parse_tree, alg_filename, arg_type_defns):
613 '''Takes the parse tree of an invoke kernel argument and returns the
614 name of the kernel and a list of Arg instances which capture the
615 relevant information about the arguments associated with the
616 kernel.
617
618 :param parse_tree: parse tree of an invoke argument. This \
619 should contain a kernel name and associated arguments.
620 :type parse_tree: :py:class:`fparser.two.Fortran2003.Part_Ref` or \
621 :py:class:`fparser.two.Fortran2003.Structure_Constructor`
622 :param str alg_filename: The file containing the algorithm code.
623
624 :param arg_type_defns: dictionary holding a 2-tuple consisting of \
625 type and precision information for each variable declared in \
626 the algorithm layer, indexed by variable name.
627 :type arg_type_defns: dict[str] = (str, str or NoneType)
628
629 :returns: a 2-tuple with the name of the kernel being called and a \
630 list of 'Arg' instances containing the required information for \
631 the arguments being passed from the algorithm layer. The list \
632 order is the same as the argument order.
633 :rtype: (str, list of :py:class:`psyclone.parse.algorithm.Arg`)
634
635 :raises InternalError: if the parse tree is of the wrong type.
636 :raises InternalError: if Part_Ref or Structure_Constructor do not \
637 have two children.
638 :raises InternalError: if Proc_Component_Ref has a child with an \
639 unexpected type.
640 :raises InternalError: if Data_Ref has a child with an unexpected \
641 type.
642 :raises NotImplementedError: if an expression contains a variable.
643 :raises InternalError: if an unsupported argument format is found.
644
645 '''
646 # pylint: disable=too-many-branches
647 if not isinstance(parse_tree, (Part_Ref, Structure_Constructor)):
648 raise InternalError(
649 f"algorithm.py:get_kernel: Expected a parse tree (type Part_Ref "
650 f"or Structure_Constructor) but found instance of "
651 f"'{type(parse_tree)}'.")
652
653 if len(parse_tree.items) != 2:
654 raise InternalError(
655 f"algorithm.py:get_kernel: Expected Part_Ref or "
656 f"Structure_Constructor to have 2 children "
657 f"but found {len(parse_tree.items)}.")
658
659 kernel_name = str(parse_tree.items[0])
660
661 # Extract argument list. This can be removed when fparser#211 is fixed.
662 argument_list = []
663 if isinstance(parse_tree.items[1],
664 (Section_Subscript_List, Component_Spec_List)):
665 argument_list = parse_tree.items[1].items
666 else:
667 # Expecting a single entry rather than a list
668 argument_list = [parse_tree.items[1]]
669
670 arguments = []
671 for argument in argument_list:
672 if isinstance(argument, (Real_Literal_Constant, Int_Literal_Constant)):
673 # A simple constant e.g. 1.0, or 1_i_def
674 precision = argument.children[1]
675 if precision:
676 precision = str(precision)
677 if isinstance(argument, Real_Literal_Constant):
678 datatype = ("real", precision)
679 else:
680 datatype = ("integer", precision)
681 arguments.append(Arg('literal', argument.tostr().lower(),
682 datatype=datatype))
683 elif isinstance(argument, Name):
684 # A simple variable e.g. arg
685 full_text = str(argument).lower()
686 var_name = full_text
687 datatype = arg_type_defns.get(var_name)
688 arguments.append(Arg('variable', full_text, varname=var_name,
689 datatype=datatype))
690 elif isinstance(argument, (Part_Ref, Structure_Constructor)):
691 # An indexed variable e.g. arg(n). (Sometimes fparser matches
692 # these as structure constructors - fparser/#384.)
693 full_text = argument.tostr().lower()
694 var_name = str(argument.items[0]).lower()
695 datatype = arg_type_defns.get(var_name)
696 arguments.append(Arg('indexed_variable', full_text,
697 varname=var_name, datatype=datatype))
698 elif isinstance(argument, Function_Reference):
699 # A function reference e.g. func(). The datatype of this
700 # function is not determined so datatype in Arg is None.
701 full_text = argument.tostr().lower()
702 designator = argument.items[0]
703 lhs = designator.items[0]
704 lhs = create_var_name(lhs)
705 rhs = str(designator.items[2])
706 var_name = f"{lhs}_{rhs}"
707 var_name = var_name.lower()
708 arguments.append(Arg('indexed_variable', full_text,
709 varname=var_name))
710 elif isinstance(argument, (Data_Ref, Proc_Component_Ref)):
711 # A structure access e.g. a % b or self % c
712 if isinstance(argument, Proc_Component_Ref):
713 if isinstance(argument.children[2], Name):
714 arg = argument.children[2].string.lower()
715 else:
716 # It does not appear to be possible to get to here
717 # as an array (e.g. self%a(10)) is not treated as
718 # being a Proc_Component_Ref by fparser2 and
719 # Data_Ref otherwise always has a Name on the rhs
720 # (3rd argument).
721 raise InternalError(
722 f"The third argument to to a Proc_Component_Ref is "
723 f"expected to be a Name, but found "
724 f"'{type(argument.children[2]).__name__}'.")
725 elif isinstance(argument, Data_Ref):
726 rhs_node = argument.children[-1]
727 if isinstance(rhs_node, Part_Ref):
728 rhs_node = rhs_node.children[0]
729 if not isinstance(rhs_node, Name):
730 raise InternalError(
731 f"The last child of a Data_Ref is expected to be "
732 f"a Name or a Part_Ref whose first child is a "
733 f"Name, but found '{type(rhs_node).__name__}'.")
734 arg = rhs_node.string.lower()
735 datatype = arg_type_defns.get(arg)
736 full_text = argument.tostr().lower()
737 var_name = create_var_name(argument).lower()
738 collection_type = None
739 if not datatype and isinstance(argument, Data_Ref):
740 # This could be a collection of some sort.
741 # Find the name of the parent type.
742 collection = argument.children[-2]
743 if isinstance(collection, Part_Ref):
744 collection = collection.children[0]
745 if isinstance(collection, Name):
746 collection_type = arg_type_defns.get(
747 collection.tostr().lower())
748 if collection_type:
749 arguments.append(
750 Arg('collection', full_text,
751 varname=var_name, datatype=collection_type))
752 else:
753 arguments.append(Arg('variable', full_text,
754 varname=var_name, datatype=datatype))
755 elif isinstance(argument, (Level_2_Unary_Expr, Add_Operand,
756 Parenthesis)):
757 # An expression e.g. -1, 1*n, ((1*n)/m). Note, for some
758 # reason Add_Operation represents binary expressions in
759 # fparser2. Walk the tree to look for an argument.
760 if walk(argument, Name):
761 raise NotImplementedError(
762 f"algorithm.py:get_kernel: Expressions containing "
763 f"variables are not yet supported '{type(argument)}', "
764 f"value '{str(argument)}', kernel '{parse_tree}' in "
765 f"file '{alg_filename}'.")
766 # This is a literal so store the full expression as a
767 # string.
768 candidate_datatype = None
769 for literal in walk(
770 # Determine datatype and precision.
771 argument, (Real_Literal_Constant,
772 Int_Literal_Constant)):
773 precision = literal.children[1]
774 if precision:
775 precision = str(precision)
776 if isinstance(literal, Real_Literal_Constant):
777 datatype = ("real", precision)
778 else: # it's an Int_Literal_Constant
779 datatype = ("integer", precision)
780
781 if not candidate_datatype:
782 # This is the first candidate
783 candidate_datatype = datatype
784 elif candidate_datatype != datatype:
785 raise NotImplementedError(
786 f"Found two non-matching literals within an "
787 f"expression ('{str(argument)}') passed into an "
788 f"invoke from the algorithm layer. '{datatype}' and "
789 f"'{candidate_datatype}' do not match. This is not "
790 f"supported in PSyclone.")
791 arguments.append(Arg('literal', argument.tostr().lower(),
792 datatype=datatype))
793 else:
794 raise InternalError(
795 f"algorithm.py:get_kernel: Unsupported argument structure "
796 f"'{type(argument)}', value '{str(argument)}', kernel "
797 f"'{parse_tree}' in file '{alg_filename}'.")
798
799 return kernel_name, arguments

802def create_var_name(arg_parse_tree):
803 '''Creates a valid variable name from an argument that optionally
804 includes brackets and potentially dereferences using '%'.
805
806 :param arg_parse_tree: the input argument. Contains braces and \
807 potentially dereferencing. e.g. a%b(c).
808 :type arg_parse_tree: :py:class:`fparser.two.Fortran2003.Name` or \
809 :py:class:`fparser.two.Fortran2003.Data_Ref` or \
810 :py:class:`fparser.two.Fortran2003.Part_Ref` or \
811 :py:class:`fparser.two.Fortran2003.Proc_Component_Ref`
812
813 :returns: a valid variable name.
814 :rtype: str
815
816 :raises InternalError: if unrecognised fparser content is found.
817
818 '''
819 tree = arg_parse_tree
820 if isinstance(tree, Name):
821 return str(tree)
822 if isinstance(tree, Part_Ref):
823 return str(tree.items[0])
824 if isinstance(tree, Proc_Component_Ref):
825 # Proc_Component_Ref is of the form 'variable %
826 # proc-name'. Its RHS (proc-name) is always a Name but its
827 # LHS (variable) could be more complex, so call the function
828 # again for the LHS.
829 return f"{create_var_name(tree.items[0])}_{tree.items[2]}"
830 if isinstance(tree, Data_Ref):
831 component_names = []
832 for item in tree.items:
833 if isinstance(item, (Data_Ref, Part_Ref)):
834 component_names.append(str(item.items[0]))
835 elif isinstance(item, Name):
836 component_names.append(str(item))
837 else:
838 raise InternalError(
839 f"algorithm.py:create_var_name unrecognised structure "
840 f"'{type(item)}' in '{type(tree)}'.")
841 return "_".join(component_names)
842 raise InternalError(
843 f"algorithm.py:create_var_name unrecognised structure '{type(tree)}'")
