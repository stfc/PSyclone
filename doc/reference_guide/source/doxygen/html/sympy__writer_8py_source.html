54class SymPyWriter(FortranWriter):
55 '''Implements a PSyIR-to-SymPy writer, which is used to create a
56 representation of the PSyIR tree that can be understood by SymPy. Most
57 Fortran expressions work as expected without modification. This class
58 implements special handling for constants (which can have a precision
59 attached, e.g. 2_4) and some intrinsic functions (e.g. ``MAX``, which SymPy
60 expects to be ``Max``). Array accesses are converted into functions (while
61 SymPy supports indexed expression, they cannot be used as expected when
62 solving, SymPy does not solve component-wise - ``M[x]-M[1]`` would not
63 result in ``x=1``, while it does for SymPy unknown functions).
64 Array expressions are supported by the writer: it will convert any array
65 expression like ``a(i:j:k)`` by using three arguments: ``a(i, j, k)``.
66 Then simple array accesses like ``b(i,j)`` are converted to
67 ``b(i,i,1,j,j,1)``. Similarly, if ``a`` is known to be an array, then the
68 writer will use ``a(sympy_lower,sympy_upper,1)``. This makes sure all SymPy
69 unknown functions that represent an array use the same number of
70 arguments.
71
72 The simple use case of converting a (list of) PSyIR expressions to SymPy
73 expressions is as follows::
74
75 symp_expr_list = SymPyWriter(exp1, exp2, ...)
76
77 If additional functionality is required (access to the type map or
78 to convert a potentially modified SymPy expression back to PSyIR), an
79 instance of SymPy writer must be created::
80
81 writer = SymPyWriter()
82 symp_expr_list = writer([exp1, exp2, ...])
83
84 It additionally supports accesses to structure types. A full description
85 can be found in the manual:
86 https://psyclone-dev.readthedocs.io/en/latest/sympy.html#sympy
87
88 '''
89 # This option will disable the lowering of abstract nodes into language
90 # level nodes, and as a consequence the backend does not need to deep-copy
91 # the tree and is much faster to execute.
92 # Be careful not to modify anything from the input tree when this option
93 # is set to True as the modifications will persist after the Writer!
94 _DISABLE_LOWERING = True
95
96 # A list of all reserved Python keywords (Fortran variables that are the
97 # same as a reserved name must be renamed, otherwise parsing will fail).
98 # This class attribute will get initialised in __init__:
99 _RESERVED_NAMES = set()
100
101 def __init__(self):
102 super().__init__()
103
104 # The symbol table is used to create unique names for structure
105 # members that are being accessed (these need to be defined as
106 # SymPy functions or symbols, which could clash with other
107 # references in the expression).
108 self._symbol_table_symbol_table = None
109
110 # The writer will use special names in array expressions to indicate
111 # the lower and upper bound (e.g. ``a(::)`` becomes
112 # ``a(sympy_lower, sympy_upper, 1)``). The symbol table will be used
113 # to resolve a potential name clash with a user variable.
114 self._lower_bound = "sympy_lower"
115 self._upper_bound = "sympy_upper"
116
117 if not SymPyWriter._RESERVED_NAMES:
118 # Get the list of all reserved Python words from the
119 # keyword module:
120 for reserved in keyword.kwlist:
121 # Some Python keywords are capitalised (True, False, None).
122 # Since all symbols in PSyclone are lowercase, these can
123 # never clash when using SymPy. So only take keywords that
124 # are in lowercase:
125 if reserved.lower() == reserved:
126 SymPyWriter._RESERVED_NAMES.add(reserved)
127
128 # This dictionary will be supplied when parsing a string by SymPy
129 # and defines which symbols in the parsed expressions are scalars
130 # (SymPy symbols) or arrays (SymPy functions).
131 self._sympy_type_map = {}
132
133 # The set of intrinsic Fortran operations that need a rename or
134 # are case sensitive in SymPy:
135 self._intrinsic = set()
136 self._intrinsic_to_str = {}
137
138 # Create the mapping of intrinsics to the name SymPy expects.
139 for intr, intr_str in [(IntrinsicCall.Intrinsic.MAX, "Max"),
140 (IntrinsicCall.Intrinsic.MIN, "Min"),
141 (IntrinsicCall.Intrinsic.FLOOR, "floor"),
142 (IntrinsicCall.Intrinsic.TRANSPOSE,
143 "transpose"),
144 (IntrinsicCall.Intrinsic.MOD, "Mod"),
145 # exp is needed for a test case only, in
146 # general the maths functions can just be
147 # handled as unknown sympy functions.
148 (IntrinsicCall.Intrinsic.EXP, "exp"),
149 ]:
150 self._intrinsic.add(intr_str)
151 self._intrinsic_to_str[intr] = intr_str

200 def _create_sympy_array_function(self, name, sig=None, num_dims=None):
201 '''Creates a Function class with the given name to be used for SymPy
202 parsing. This Function overwrites the conversion to string, and will
203 replace the triplicated array indices back to the normal Fortran
204 syntax. If the signature Sig and number of dimensions for each
205 component of the signature are given, it will add this information
206 to the object, so that the SymPyReader can recreate the proper
207 access to a user-defined type.
208
209 :param str name: name of the function class to create.
210 :param sig: the signature of the variable, which is required
211 to convert user defined types back properly. Only defined for
212 user-defined types.
213 :type sig: Optional[:py:class:`psyclone.core.Signature`]
214 :param num_dims: the number of dimensions for each component of a
215 user defined type.
216 :type num_dims: Optional[List[int]]
217
218 :returns: a SymPy function, which has a special ``_sympystr`` function
219 defined as attribute to print user-defined types..
220 :rtype: :py:class:`sympy.Function`
221 '''
222
223 # Now a new Fortran array is used. Create a new function
224 # instance, and overwrite how this function is converted back
225 # into a string by defining the ``_sympystr`` attribute,
226 # which points to a function that controls how this object
227 # is converted into a string. Use the ``print_fortran_array``
228 # function from the SymPyReader for this. Note that we cannot
229 # create a derived class based on ``Function`` and define
230 # this function there: SymPy tests internally if the type is a
231 # Function (not if it is an instance), therefore, SymPy's
232 # behaviour would change if we used a derived class:
233 # https://docs.sympy.org/latest/modules/functions/index.html:
234 # "It [Function class] also serves as a constructor for undefined
235 # function classes."
236 new_func = sympy.Function(name)
237 # pylint: disable=protected-access
238 new_func._sympystr = SymPyReader.print_fortran_array
239
240 # Store the signature and the number of dimensions of each
241 # component, so that SymPyReader.print_fortran_array can match
242 # the indices back to the user defined types.
243 new_func._sig = sig
244 new_func._num_dims = num_dims
245 # pylint: enable=protected-access
246 return new_func

249 def _create_type_map(self, list_of_expressions, identical_variables=None,
250 all_variables_positive=None):
251 '''This function creates a dictionary mapping each Reference in any
252 of the expressions to either a SymPy Function (if the reference
253 is an array reference) or a Symbol (if the reference is not an
254 array reference). It defines a new SymPy function for each array,
255 which has a special write method implemented that automatically
256 converts array indices back by combining each three arguments into
257 one expression (i. e. ``a(1,9,2)`` would become ``a(1:9:2)``).
258
259 A new symbol table is created any time this function is called, so
260 it is important to provide all expressions at once for the symbol
261 table to avoid name clashes in any expression.
262
263 This function also handles reserved names like 'lambda' which might
264 occur in one of the Fortran expressions. These reserved names must be
265 renamed (otherwise SymPy parsing, which uses ``eval`` internally,
266 fails). A symbol table is used which is initially filled with all
267 reserved names. Then for each reference accounted, a unique name is
268 generated using this symbol table - so if the reference is a reserved
269 name, a new name will be created (e.g. ``lambda`` might become
270 ``lambda_1``). The SymPyWriter (e.g. in ``reference_node``) will
271 use the renamed value when creating the string representation.
272
273 The optional identical_variables dictionary can contain information
274 about variables which are known to be the same. For example, if
275 `identical_variables={'i': 'j'}`, then 'i+1' and 'j+1' will be
276 considered equal.
277
278 The optional `all_variables_positive` flag can be used to indicate that
279 all variables are positive definite. This means that, e.g. 'i+j' will
280 be considered greater than 'i'.
281
282 :param list_of_expressions: the list of expressions from which all
283 references are taken and added to a symbol table to avoid
284 renaming any symbols (so that only member names will be renamed).
285 :type list_of_expressions: List[:py:class:`psyclone.psyir.nodes.Node`]
286 :param identical_variables: which variable names are known to represent
287 identical quantities.
288 :type identical_variables: Optional[dict[str, str]]
289 :param Optional[bool] all_variables_positive: whether or not (the
290 default) to assume that all variables are positive definite
291 quantities.
292
293 '''
294 # Create a new symbol table, so previous symbol will not affect this
295 # new conversion (i.e. this avoids name clashes with a previous
296 # conversion). First add all reserved names so that these names will
297 # automatically be renamed. The symbol table is used later to also
298 # create guaranteed unique names for lower and upper bounds.
299 self._symbol_table_symbol_table = SymbolTable()
300 for reserved in SymPyWriter._RESERVED_NAMES:
301 self._symbol_table_symbol_table.new_symbol(reserved)
302
303 # Set-up whether we should assume all Symbols are positive.
304 assumptions = {}
305 if all_variables_positive:
306 assumptions["positive"] = True
307
308 # Find each reference in each of the expression, and declare this name
309 # as either a SymPy Symbol (scalar reference), or a SymPy Function
310 # (an array).
311 for expr in list_of_expressions:
312 # TODO #2542. References should be iterated with the
313 # reference_acess method when its issues are fixed.
314 for ref in expr.walk(Reference):
315 if (isinstance(ref.parent, Call) and
316 ref.parent.children[0] is ref):
317 continue
318
319 name = ref.name
320 # The reserved Python keywords do not have tags, so they
321 # will not be found.
322 if name in self._symbol_table_symbol_table.tags_dict:
323 continue
324
325 # Any symbol from the list of expressions to be handled
326 # will be created with a tag, so if the same symbol is
327 # used more than once, the previous test will prevent
328 # calling new_symbol again. If the name is a Python
329 # reserved symbol, a new unique name will be created by
330 # the symbol table.
331 unique_sym = self._symbol_table_symbol_table.new_symbol(name, tag=name)
332 # Test if an array or an array expression is used:
333 if not ref.is_array:
334 self._sympy_type_map[unique_sym.name] = sympy.Symbol(
335 name, **assumptions)
336 continue
337
338 # A Fortran array is used which has not been seen before.
339 # Declare a new SymPy function for it. This SymPy function
340 # will convert array expressions back into the original
341 # Fortran code.
342 self._sympy_type_map[unique_sym.name] = \
343 self._create_sympy_array_function(name)
344
345 if not identical_variables:
346 identical_variables = {}
347 # For all variables that are the same, set the symbols to be
348 # identical. This means if e.g. identical_variables={'i': 'j'},
349 # the expression i-j becomes j-j = 0
350
351 for var1, var2 in identical_variables.items():
352 if var1 in self._sympy_type_map and var2 in self._sympy_type_map:
353 self._sympy_type_map[var1] = self._sympy_type_map[var2]
354
355 # Now all symbols have been added to the symbol table, create
356 # unique names for the lower- and upper-bounds using special tags:
357 self._lower_bound = \
358 self._symbol_table_symbol_table.new_symbol("sympy_lower",
359 tag="sympy!lower_bound").name
360 self._upper_bound = \
361 self._symbol_table_symbol_table.new_symbol("sympy_upper",
362 tag="sympy!upper_bound").name
363
364 # -------------------------------------------------------------------------

366 def lower_bound(self):
367 ''':returns: the name to be used for an unspecified lower bound.
368 :rtype: str

370 '''
371 return self._lower_bound

375 def upper_bound(self):

393 def _to_str(self, list_of_expressions, identical_variables=None,
394 all_variables_positive=False):
395 '''Converts PSyIR expressions to strings. It will replace Fortran-
396 specific expressions with code that can be parsed by SymPy. The
397 argument can either be a single element (in which case a single string
398 is returned) or a list/tuple, in which case a list is returned.
399 The optional identical_variables dictionary can contain information
400 about variables which are known to be the same. For example, if
401 `identical_variables={'i': 'j'}`, then 'i+1' and 'j+1' will be
402 considered equal.
403
404 :param identical_variables: which variable names are known to be
405 identical
406 :type identical_variables: Optional[dict[str, str]]
407
408 :param list_of_expressions: the list of expressions which are to be
409 converted into SymPy-parsable strings.
410 :type list_of_expressions: Union[:py:class:`psyclone.psyir.nodes.Node`,
411 List[:py:class:`psyclone.psyir.nodes.Node`]]
412 :param Optional[bool] all_variables_positive: whether or not (the
413 default) to assume that all variables are positive definite
414 quantities.
415
416 :returns: the converted strings(s).
417 :rtype: Union[str, List[str]]
418
419 '''
420 is_list = isinstance(list_of_expressions, (tuple, list))
421 if not is_list:
422 list_of_expressions = [list_of_expressions]
423
424 # Create the type map in `self._sympy_type_map`, which is required
425 # when converting these strings to SymPy expressions
426 self._create_type_map(list_of_expressions,
427 identical_variables=identical_variables,
428 all_variables_positive=all_variables_positive)
429
430 expression_str_list = []
431 for expr in list_of_expressions:
432 expression_str_list.append(super().__call__(expr))
433
434 # If the argument was a single expression, only return a single
435 # expression, otherwise return a list
436 if not is_list:
437 return expression_str_list[0]
438 return expression_str_list

440 # -------------------------------------------------------------------------
441 def __call__(self, list_of_expressions, identical_variables=None,
442 all_variables_positive=False):
443 '''
444 This function takes a list of PSyIR expressions, and converts
445 them all into Sympy expressions using the SymPy parser.
446 It takes care of all Fortran specific conversion required (e.g.
447 constants with kind specification, ...), including the renaming of
448 member accesses, as described in
449 https://psyclone-dev.readthedocs.io/en/latest/sympy.html#sympy
450 The optional identical_variables dictionary can contain information
451 about variables which are known to be the same. For example, if
452 `identical_variables={'i': 'j'}`, then 'i+1' and 'j+1' will be
453 considered equal.
454
455 :param list_of_expressions: the list of expressions which are to be
456 converted into SymPy-parsable strings.
457 :type list_of_expressions: list of
458 :py:class:`psyclone.psyir.nodes.Node`
459 :param identical_variables: which variable names are known to be
460 identical
461 :type identical_variables: Optional[dict[str, str]]
462 :param Optional[bool] all_variables_positive: whether or not (the
463 default) to assume that all variables are positive definite
464 quantities.
465
466 :returns: a 2-tuple consisting of the the converted PSyIR
467 expressions, followed by a dictionary mapping the symbol names
468 to SymPy Symbols.
469 :rtype: Union[:py:class:`sympy.core.basic.Basic`,
470 List[:py:class:`sympy.core.basic.Basic`]]
471
472 :raises VisitorError: if an invalid SymPy expression is found.
473 :raises TypeError: if the identical_variables parameter is not
474 a dict, or does contain a key or value that is not a string.
475
476 '''
477 if identical_variables:
478 if not isinstance(identical_variables, dict):
479 raise TypeError(f"Expected identical_variables to be "
480 f"a dictionary, but got "
481 f"{type(identical_variables)}.")
482 if any(not isinstance(key, str) or not isinstance(value, str)
483 for key, value in identical_variables.items()):
484 raise TypeError("Dictionary identical_variables "
485 "contains a non-string key or value.")
486
487 is_list = isinstance(list_of_expressions, (tuple, list))
488 if not is_list:
489 list_of_expressions = [list_of_expressions]
490 expression_str_list = self._to_str(
491 list_of_expressions, identical_variables=identical_variables,
492 all_variables_positive=all_variables_positive)
493
494 result = []
495 for expr in expression_str_list:
496 try:
497 result.append(parse_expr(expr, self.type_maptype_map))
498 except SyntaxError as err:
499 raise VisitorError(f"Invalid SymPy expression: '{expr}'.") \
500 from err
501
502 if is_list:
503 return result
504 # We had no list initially, so only convert the one and only
505 # list member
506 return result[0]

539 def arrayofstructuresreference_node(self, node):
540 '''This handles ArrayOfStructureReferences (and also simple
541 StructureReferences). An access like ``a(i)%b(j)`` is converted to
542 the string ``a_b(i,i,1,j,j,1)`` (also handling name clashes in case
543 that the user code already contains a symbol ``a_b``). The SymPy
544 function created for this new symbol will store the original signature
545 and the number of indices for each member (so in the example above
546 that would be ``Signature("a%b")`` and ``(1,1)``. This information
547 is sufficient to convert the SymPy symbol back to the correct Fortran
548 representation
549
550 :param node: a StructureReference PSyIR node.
551 :type node: :py:class:`psyclone.psyir.nodes.StructureReference`
552
553 :returns: the code as string.
554 :rtype: str
555
556 '''
557 sig, indices = node.get_signature_and_indices()
558
559 out = []
560 num_dims = []
561 all_dims = []
562 is_array = False
563 for i, name in enumerate(sig):
564 num_dims.append(len(indices[i]))
565 for index in indices[i]:
566 all_dims.append(index)
567 is_array = True
568 out.append(name)
569 flat_name = "_".join(out)
570
571 # Find (or create) a unique variable name:
572 try:
573 unique_name = self._symbol_table_symbol_table.lookup_with_tag(str(sig)).name
574 except KeyError:
575 unique_name = self._symbol_table_symbol_table.new_symbol(flat_name,
576 tag=str(sig)).name
577 if is_array:
578 indices_str = self.gen_indicesgen_indicesgen_indices(all_dims)
579 # Create the corresponding SymPy function, which will store
580 # the signature and num_dims, so that the correct Fortran
581 # representation can be recreated later.
582 self._sympy_type_map[unique_name] = \
583 self._create_sympy_array_function(unique_name, sig, num_dims)
584 return f"{unique_name}({','.join(indices_str)})"
585
586 # Not an array access. We use the unique name for the string,
587 # but the required symbol is mapped to the original name, which means
588 # if the SymPy expression is converted to a string (in order to be
589 # parsed), it will use the original structure reference syntax:
590 self._sympy_type_map[unique_name] = sympy.Symbol(sig.to_language())
591 return unique_name

594 def literal_node(self, node):
595 '''This method is called when a Literal instance is found in the PSyIR
596 tree. For SymPy we need to handle booleans (which are expected to
597 be capitalised: True). Real values work by just ignoring any precision
598 information (e.g. 2_4, 3.1_wp). Character constants are not supported
599 and will raise an exception.
600
601 :param node: a Literal PSyIR node.
602 :type node: :py:class:`psyclone.psyir.nodes.Literal`
603
604 :returns: the SymPy representation for the literal.
605 :rtype: str
606
607 :raises TypeError: if a character constant is found, which
608 is not supported with SymPy.
609
610 '''
611 if node.datatype.intrinsic == ScalarType.Intrinsic.BOOLEAN:
612 # Booleans need to be converted to SymPy format
613 return node.value.capitalize()
614
615 if node.datatype.intrinsic == ScalarType.Intrinsic.CHARACTER:
616 raise TypeError(f"SymPy cannot handle strings "
617 f"like '{node.value}'.")
618 # All real (single, double precision) and integer work by just
619 # using the node value. Single and double precision both use
620 # 'e' as specification, which SymPy accepts, and precision
621 # information can be ignored.
622 return node.value

624 def intrinsiccall_node(self, node):
625 ''' This method is called when an IntrinsicCall instance is found in
626 the PSyIR tree. The Sympy backend will use the exact sympy name for
627 some math intrinsics (listed in _intrinsic_to_str) and will remove
628 named arguments.
629
630 :param node: an IntrinsicCall PSyIR node.
631 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
632
633 :returns: the SymPy representation for the Intrinsic.
634 :rtype: str
635
636 '''
637 # Sympy does not support argument names, remove them for now
638 if any(node.argument_names):
639 # TODO #2302: This is not totally right without canonical intrinsic
640 # positions for arguments. One alternative is to refuse it with:
641 # raise VisitorError(
642 # f"Named arguments are not supported by SymPy but found: "
643 # f"'{node.debug_string()}'.")
644 # but this leaves sympy comparisons almost always giving false when
645 # out of order arguments are rare, so instead we ignore it for now.
646
647 # It makes a copy (of the parent because if matters to the call
648 # visitor) because we don't want to delete the original arg names
649 parent = node.parent.copy()
650 node = parent.children[node.position]
651 for idx in range(len(node.argument_names)):
652 # pylint: disable=protected-access
653 node._argument_names[idx] = (node._argument_names[idx][0],
654 None)
655 try:
656 name = self._intrinsic_to_str[node.intrinsic]
657 args = self._gen_arguments(node)
658 return f"{self._nindent}{name}({args})"
659 except KeyError:
660 return super().call_node(node)

663 def reference_node(self, node):
664 '''This method is called when a Reference instance is found in the
665 PSyIR tree. It handles the case that this normal reference might
666 be an array expression, which in the SymPy writer needs to have
667 indices added explicitly: it basically converts the array expression
668 ``a`` to ``a(sympy_lower, sympy_upper, 1)``.
669
670 :param node: a Reference PSyIR node.
671 :type node: :py:class:`psyclone.psyir.nodes.Reference`
672
673 :returns: the text representation of this reference.
674 :rtype: str
675
676 '''
677 # Support renaming a symbol (e.g. if it is a reserved Python name).
678 # Look up with the name as tag, which will return the symbol with
679 # a unique name (e.g. lambda --> lambda_1):
680 try:
681 name = self._symbol_table_symbol_table.lookup_with_tag(node.name).name
682 except KeyError:
683 # If the tag did not exist it means that this symbol has not
684 # been re-named, and we can use it as is.
685 name = node.name
686 if not node.is_array:
687 # This reference is not an array, just return the name
688 return name
689
690 # Now this must be an array expression without parenthesis. Add
691 # the triple-array indices to represent `lower:upper:1` for each
692 # dimension:
693 shape = node.symbol.shape
694 result = [f"{self.lower_bound},"
695 f"{self.upper_bound},1"]*len(shape)
696
697 return (f"{name}{self.array_parenthesis[0]}"
698 f"{','.join(result)}{self.array_parenthesis[1]}")

701 def gen_indices(self, indices, var_name=None):
702 '''Given a list of PSyIR nodes representing the dimensions of an
703 array, return a list of strings representing those array dimensions.
704 This is used both for array references and array declarations. Note
705 that 'indices' can also be a shape in case of Fortran. The
706 implementation here overwrites the one in the base class to convert
707 each array index into three parameters to support array expressions.
708
709 :param indices: list of PSyIR nodes.
710 :type indices: List[:py:class:`psyclone.psyir.symbols.Node`]
711 :param str var_name: name of the variable for which the dimensions
712 are created. Not used in this implementation.
713
714 :returns: the Fortran representation of the dimensions.
715 :rtype: List[str]
716
717 :raises NotImplementedError: if the format of the dimension is not
718 supported.
719
720 '''
721 dims = []
722 for index in indices:
723 if isinstance(index, DataNode):
724 # literal constant, symbol reference, or computed
725 # dimension
726 expression = self._visit(index)
727 dims.extend([expression, expression, "1"])
728 elif isinstance(index, Range):
729 # literal constant, symbol reference, or computed
730 # dimension
731 expression = self._visit(index)
732 dims.append(expression)
733 elif isinstance(index, ArrayType.ArrayBounds):
734 # Lower and upper bounds of an array declaration specified
735 # by literal constant, symbol reference, or computed dimension
736 lower_expression = self._visit(index.lower)
737 upper_expression = self._visit(index.upper)
738 dims.extend([lower_expression, upper_expression, "1"])
739 elif isinstance(index, ArrayType.Extent):
740 # unknown extent
741 dims.extend([self.lower_boundlower_bound, self.upper_boundupper_bound, "1"])
742 else:
743 raise NotImplementedError(
744 f"unsupported gen_indices index '{index}'")
745 return dims

748 def range_node(self, node):
749 '''This method is called when a Range instance is found in the PSyIR
750 tree. This implementation converts a range into three parameters
751 for the corresponding SymPy function.
752
753 :param node: a Range PSyIR node.
754 :type node: :py:class:`psyclone.psyir.nodes.Range`
755
756 :returns: the Fortran code as a string.
757 :rtype: str
758
759 '''
760 if node.parent and node.parent.is_lower_bound(
761 node.parent.index_of(node)):
762 # The range starts for the first element in this
763 # dimension, so use the generic name for lower bound:
764 start = self.lower_boundlower_bound
765 else:
766 start = self._visit(node.start)
767
768 if node.parent and node.parent.is_upper_bound(
769 node.parent.index_of(node)):
770 # The range ends with the last element in this
771 # dimension, so use the generic name for the upper bound:
772 stop = self.upper_boundupper_bound
773 else:
774 stop = self._visit(node.stop)
775 result = f"{start},{stop}"
776
777 step = self._visit(node.step)
778 result += f",{step}"
779
780 return result
