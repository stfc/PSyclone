39''' This module contians the LFRicRunTimeChecks class which handles
40declarations and code generation for run-time checks. The methods
41check fields' function spaces and read-only fields against kernel
42function-space metadata on initialisation. The class inherits from
43the LFRicCollection class.'''

78 def _check_field_fs(self, parent):
79 '''
80 Internal method that adds run-time checks to make sure that the
81 field's function space is consistent with the appropriate
82 kernel metadata function spaces.
83
84 :param parent: the node in the f2pygen AST representing the PSy- \
85 layer routine.
86 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
87
88 '''
89 parent.add(CommentGen(
90 parent, " Check field function space and kernel metadata "
91 "function spaces are compatible"))
92
93 # When issue #30 is addressed (with issue #79 helping further)
94 # we may know some or all field function spaces statically. If
95 # so, we should remove these from the fields to check at run
96 # time (as they will have already been checked at code
97 # generation time).
98
99 const = LFRicConstants()
100 existing_checks = []
101 for kern_call in self._invoke.schedule.kernels():
102 for arg in kern_call.arguments.args:
103 if not arg.is_field:
104 # This check is limited to fields
105 continue
106 fs_name = arg.function_space.orig_name
107 field_name = arg.name_indexed
108 if fs_name in const.VALID_ANY_SPACE_NAMES:
109 # We don't need to check validity of a field's
110 # function space if the metadata specifies
111 # any_space as this means that all spaces are
112 # valid.
113 continue
114 if (fs_name, field_name) in existing_checks:
115 # This particular combination has already been
116 # checked.
117 continue
118 existing_checks.append((fs_name, field_name))
119
120 if fs_name in const.VALID_ANY_DISCONTINUOUS_SPACE_NAMES:
121 # We need to check against all discontinuous
122 # function spaces
123 function_space_names = const.DISCONTINUOUS_FUNCTION_SPACES
124 elif fs_name == "any_w2":
125 # We need to check against all any_w2 function
126 # spaces
127 function_space_names = const.ANY_W2_FUNCTION_SPACES
128 else:
129 # We need to check against a specific function space
130 function_space_names = [fs_name]
131
132 if_condition = " .and. ".join(
133 [f"{field_name}%which_function_space() /= {name.upper()}"
134 for name in function_space_names])
135 if_then = IfThenGen(parent, if_condition)
136 call_abort = CallGen(
137 if_then, "log_event(\"In alg "
138 f"'{self._invoke.invokes.psy.orig_name}' invoke "
139 f"'{self._invoke.name}', the field '{arg.name}' is passed "
140 f"to kernel '{kern_call.name}' but its function space is "
141 f"not compatible with the function space specified in the "
142 f"kernel metadata '{fs_name}'.\", LOG_LEVEL_ERROR)")
143 if_then.add(call_abort)
144 parent.add(if_then)

146 def _check_field_ro(self, parent):
147 '''
148 Internal method that adds runtime checks to make sure that if the
149 field is on a read-only function space then the associated
150 kernel metadata does not specify that the field is modified.
151
152 As we make use of the LFRic infrastructure halo exchange
153 function, there is no need to check whether the halo of a
154 read-only field is clean (which it should always be) as the
155 LFric halo-exchange will raise an exception if it is called
156 with a read-only field.
157
158 Whilst the LFRic infrastructure halo exchange would also
159 indirectly pick up a readonly field being modified, it would
160 not be picked up where the error occured. Therefore adding
161 checks here is still useful.
162
163 :param parent: the node in the f2pygen AST representing the PSy- \
164 layer routine.
165 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
166
167 '''
168 # When issue #30 is addressed (with issue #79 helping further)
169 # we may know some or all field function spaces statically. If
170 # so, we should remove these from the fields to check at run
171 # time (as they will have already been checked at code
172 # generation time).
173
174 # Create a list of modified fields
175 modified_fields = []
176 for call in self._invoke.schedule.kernels():
177 for arg in call.arguments.args:
178 if (arg.text and arg.is_field and
179 arg.access != AccessType.READ and
180 not [entry for entry in modified_fields if
181 entry[0].name == arg.name]):
182 modified_fields.append((arg, call))
183 if modified_fields:
184 parent.add(CommentGen(
185 parent, " Check that read-only fields are not modified"))
186 for field, call in modified_fields:
187 if_then = IfThenGen(
188 parent, f"{field.proxy_name_indexed}%vspace%is_readonly()")
189 call_abort = CallGen(
190 if_then, "log_event(\"In alg "
191 f"'{self._invoke.invokes.psy.orig_name}' invoke "
192 f"'{self._invoke.name}', field '{field.name}' is on a "
193 f"read-only function space but is modified by kernel "
194 f"'{call.name}'.\", LOG_LEVEL_ERROR)")
195 if_then.add(call_abort)
196 parent.add(if_then)

198 def initialise(self, parent):
199 '''Add runtime checks to make sure that the arguments being passed
200 from the algorithm layer are consistent with the metadata
201 specified in the associated kernels. Currently checks are
202 limited to ensuring that field function spaces are consistent
203 with the associated kernel function-space metadata.
204
205 :param parent: the node in the f2pygen AST representing the PSy- \
206 layer routine.
207 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
208
209 '''
210 if not Config.get().api_conf("lfric").run_time_checks:
211 # Run-time checks are not requested.
212 return
213
214 parent.add(CommentGen(parent, ""))
215 parent.add(CommentGen(parent, " Perform run-time checks"))
216 parent.add(CommentGen(parent, ""))
217
218 # Check that field function spaces are compatible with the
219 # function spaces specified in the kernel metadata.
220 self._check_field_fs(parent)
221
222 # Check that fields on read-only function spaces are not
223 # passed into a kernel where the kernel metadata specifies
224 # that the field will be modified.
225 self._check_field_ro(parent)
226
227 # These checks should be expanded. Issue #768 suggests
228 # extending function space checks to operators.
