105 def cma_operator(self, arg, var_accesses=None):
106 '''Add the CMA operator and associated scalars to the argument
107 list and optionally add them to the variable access
108 information.
109
110 :param arg: the CMA operator argument.
111 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
112 :param var_accesses: optional VariablesAccessInfo instance to store \
113 the information about variable accesses.
114 :type var_accesses: \
115 :py:class:`psyclone.core.VariablesAccessInfo`
116
117 '''
118 # The CMA operator itself
119 self.append(arg.name, var_accesses)
120 # Associated scalar parameters
121 nrow = arg.name + "_nrow"
122 _local_args = [nrow]
123 if arg.function_space_to.orig_name != \
124 arg.function_space_from.orig_name:
125 # If the to- and from-spaces are different then so are ncol and
126 # nrow so we pass both of them. If they are the same then we
127 # could pass either but choose to pass nrow and not ncol.
128 ncol = arg.name + "_ncol"
129 _local_args.append(ncol)
130 bandwidth = arg.name + "_bandwidth"
131 alpha = arg.name + "_alpha"
132 beta = arg.name + "_beta"
133 gamma_m = arg.name + "_gamma_m"
134 gamma_p = arg.name + "_gamma_p"
135 _local_args += [bandwidth, alpha, beta, gamma_m, gamma_p]
136 self.extend(_local_args, var_accesses)

138 def field_vector(self, argvect, var_accesses=None):
139 '''Add the field vector associated with the argument 'argvect' to the
140 argument list. If supplied it also stores these accesses to the
141 var_access object.
142
143 :param argvect: the field vector to add.
144 :type argvect: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
145 :param var_accesses: optional VariablesAccessInfo instance to store \
146 the information about variable accesses.
147 :type var_accesses: \
148 :py:class:`psyclone.core.VariablesAccessInfo`
149
150 '''
151 # the range function below returns values from
152 # 1 to the vector size which is what we
153 # require in our Fortran code
154 for idx in range(1, argvect.vector_size+1):
155 text = (argvect.name + "_" +
156 argvect.function_space.mangled_name +
157 "_v" + str(idx))
158 self.append(text, var_accesses)

333 def basis(self, function_space, var_accesses=None):
334 '''Add basis function information for this function space to the
335 argument list and optionally to the variable access information.
336 There can be more than one if this is an evaluator and/or multiple
337 'gh_shape's have been requested in the kernel metadata. If supplied
338 it also stores these accesses in var_accesses.
339
340 :param function_space: the function space for which to provide \
341 the basis functions
342 :type function_space: :py:class:`psyclone.domain.lfric.FunctionSpace`
343 :param var_accesses: optional VariablesAccessInfo instance to store \
344 the information about variable accesses.
345 :type var_accesses: \
346 :py:class:`psyclone.core.VariablesAccessInfo`
347
348 :raises InternalError: if the evaluator shape is not recognised.
349
350 '''
351 const = LFRicConstants()
352 for shape in self._kern_kern.eval_shapes:
353 if shape in const.VALID_QUADRATURE_SHAPES:
354 # A kernel stub won't have a name for the corresponding
355 # quadrature argument so we create one by appending the last
356 # part of the shape name to "qr_".
357 basis_name = function_space.get_basis_name(
358 qr_var="qr_"+shape.split("_")[-1])
359 self.append(basis_name, var_accesses)
360
361 elif shape in const.VALID_EVALUATOR_SHAPES:
362 # Need a basis array for each target space upon which the basis
363 # functions have been evaluated. _kern.eval_targets is a dict
364 # where the values are 2-tuples of (FunctionSpace, argument).
365 for _, target in self._kern_kern.eval_targets.items():
366 basis_name = \
367 function_space.get_basis_name(on_space=target[0])
368 self.append(basis_name, var_accesses)
369 else:
370 raise InternalError(
371 f"Unrecognised evaluator shape ('{shape}'). Expected one "
372 f"of: {const.VALID_EVALUATOR_SHAPES}")

374 def diff_basis(self, function_space, var_accesses=None):
375 '''Add differential basis information for the function space to the
376 argument list. If supplied it also stores this access in
377 var_accesses.
378
379 :param function_space: the function space for which the differential \
380 basis functions are required.
381 :type function_space: :py:class:`psyclone.domain.lfric.FunctionSpace`
382 :param var_accesses: optional VariablesAccessInfo instance to store \
383 the information about variable accesses.
384 :type var_accesses: \
385 :py:class:`psyclone.core.VariablesAccessInfo`
386
387 :raises InternalError: if the evaluator shape is not recognised.
388
389 '''
390 const = LFRicConstants()
391 for shape in self._kern_kern.eval_shapes:
392 if shape in const.VALID_QUADRATURE_SHAPES:
393 # We need differential basis functions for quadrature. A
394 # kernel stub won't have a name for the corresponding
395 # quadrature argument so we create one by appending the
396 # last part of the shape name to "qr_".
397 diff_basis_name = function_space.get_diff_basis_name(
398 qr_var="qr_"+shape.split("_")[-1])
399 self.append(diff_basis_name, var_accesses)
400
401 elif shape in const.VALID_EVALUATOR_SHAPES:
402 # We need differential basis functions for an evaluator,
403 # potentially for multiple target spaces. _kern.eval_targets is
404 # a dict where the values are 2-tuples of
405 # (FunctionSpace, argument).
406 for _, target in self._kern_kern.eval_targets.items():
407 diff_basis_name = function_space.get_diff_basis_name(
408 on_space=target[0])
409 self.append(diff_basis_name, var_accesses)
410 else:
411 raise InternalError(f"Unrecognised evaluator shape "
412 f"('{shape}'). Expected one of: "
413 f"{const.VALID_EVALUATOR_SHAPES}")

479 def indirection_dofmap(self, function_space, operator=None,
480 var_accesses=None):
481 '''Add indirection dofmap required when applying a CMA operator. If
482 supplied it also stores this access in var_accesses.
483
484 :param function_space: the function space for which the indirect \
485 dofmap is required.
486 :type function_space: :py:class:`psyclone.domain.lfric.FunctionSpace`
487 :param operator: the CMA operator.
488 :type operator: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
489 :param var_accesses: optional VariablesAccessInfo instance to store \
490 the information about variable accesses.
491 :type var_accesses: \
492 :py:class:`psyclone.core.VariablesAccessInfo`
493
494 :raises InternalError: if no kernel argument is supplied.
495 :raises InternalError: if the supplied kernel argument is not a \
496 CMA operator.
497
498 '''
499 if not operator:
500 raise InternalError("No CMA operator supplied.")
501 if operator.argument_type != "gh_columnwise_operator":
502 raise InternalError(
503 f"A CMA operator (gh_columnwise_operator) must "
504 f"be supplied but got '{operator.argument_type}'.")
505 super().indirection_dofmap(function_space, operator, var_accesses)
