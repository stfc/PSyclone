55class AdjointVisitor(PSyIRVisitor):
56 '''An Adjoint Visitor that translates the PSyIR of a tangent-linear
57 code into its adjoint form.
58
59 :param active_variable_names: a list of the active variables.
60 :type active_variable_names: list of str
61
62 :raises ValueError: if no active variables are supplied.
63
64 '''
65 def __init__(self, active_variable_names):
66 super().__init__()
67 if not active_variable_names:
68 raise ValueError(
69 "There should be at least one active variable supplied to "
70 "an AdjointVisitor.")
71 self._active_variable_names = active_variable_names
72 self._active_variables_active_variables = None
73 self._logger = logging.getLogger(__name__)

75 def container_node(self, node):
76 '''This method is called if the visitor finds a Container node. A copy
77 of the container is returned (as this does not change when
78 converting from tangent linear to adjoint) containing
79 processed descendants.
80
81 :param node: a Container PSyIR node.
82 :type node: :py:class:`psyclone.psyir.nodes.Container`
83
84 :returns: a new PSyIR tree containing the adjoint of this node \
85 and its descendant nodes.
86 :rtype: :py:class:`psyclone.psyir.nodes.Node`
87
88 '''
89 self._logger.debug("Copying Container")
90 return self._copy_and_process(node)

92 def schedule_node(self, node):
93 '''This method is called if the visitor finds a Schedule node. A copy
94 of the schedule is returned, as this does not change when
95 converting from tangent linear to adjoint and its children are
96 re-ordered and sorted dependending on whether they are active
97 or passive nodes.
98
99 As a schedule contains variable scoping information, i.e. a
100 symbol table, the symbols representing the active variable
101 names supplied to the visitor are found and added to an
102 internal list so they are available when processing any
103 descendants.
104
105 :param node: a Schedule PSyIR node.
106 :type node: :py:class:`psyclone.psyir.nodes.Schedule`
107
108 :returns: a new PSyIR tree containing the adjoint equivalent \
109 of this node and its descendants.
110 :rtype: :py:class:`psyclone.psyir.nodes.Schedule`
111
112 '''
113 # pylint: disable=too-many-locals, too-many-branches
114 # pylint: disable=too-many-statements
115 self._logger.debug("Transforming Schedule")
116
117 # A schedule has a scope so determine and store active variables
118 symbol_table = node.scope.symbol_table
119 self._active_variables_active_variables = []
120 for variable_name in self._active_variable_names:
121 self._active_variables_active_variables.append(symbol_table.lookup(variable_name))
122
123 # We only need to copy this node. Issue #1440 will address
124 # this.
125 node_copy = node.copy()
126 node_copy.children = []
127
128 if isinstance(node, Routine):
129 # Zero local active variables.
130 self._logger.debug("Zero-ing any local active variables")
131 for active_variable in self._active_variables_active_variables:
132 if active_variable.is_automatic:
133 if not (active_variable.is_scalar or
134 active_variable.is_array):
135 # Issue #1627 structures are not allowed.
136 raise NotImplementedError(
137 f"Active local variables can only be scalars and "
138 f"arrays, but found '{active_variable}'.")
139 datatype = active_variable.datatype.intrinsic.name
140 if datatype == "REAL":
141 value = "0.0"
142 elif datatype == "INTEGER":
143 value = "0"
144 else:
145 raise NotImplementedError(
146 f"Datatype '{datatype}' is not supported (for "
147 f"active local variable "
148 f"'{active_variable.name}'). Supported types are "
149 f"'REAL' and 'INTEGER'.")
150 node_copy.children.append(
151 Assignment.create(
152 Reference(active_variable),
153 Literal(value, active_variable.datatype)))
154
155 # Split active and passive nodes.
156 self._logger.debug("Adding passive code into new schedule")
157 active_nodes = []
158 for child in node.children:
159 if node_is_passive(child, self._active_variables_active_variables):
160 # Add passive nodes back into the schedule as they do
161 # not change.
162 node_copy.children.append(child.copy())
163 else:
164 # Store active nodes for further processing.
165 active_nodes.append(child)
166
167 # Reverse active nodes.
168 self._logger.debug("Reversing order of active code")
169 active_nodes.reverse()
170
171 # Process active nodes.
172 self._logger.debug(
173 "Processing active code and adding results into new schedule")
174 for child in active_nodes:
175 result = self._visit(child)
176 if isinstance(result, list):
177 node_copy.children.extend(result)
178 else:
179 node_copy.children.append(result)
180
181 # Creating the adjoint may have altered the way variables are
182 # accessed within the code. If any of the active variables are
183 # subroutine arguments then we must update the access property of
184 # the associated ArgumentInterface.
185
186 # Since a piece of code could contain many Schedules, ensure we are
187 # currently handling the one representing the routine.
188 if isinstance(node, Routine):
189 ctu = CallTreeUtils()
190 # Input signatures ('in_sigs') are those whose first access is a
191 # read.
192 # Output signatures ('out_sigs') are those that are written to at
193 # some point.
194 read_write_info = ctu.get_in_out_parameters(node_copy.children)
195 # Get the variable name associated with each of these signatures.
196 in_names = [sig.var_name
197 for sig in read_write_info.signatures_read]
198 out_names = [sig.var_name
199 for sig in read_write_info.signatures_written]
200
201 # We must update the symbols in the table of the new tree
202 adj_table = node_copy.symbol_table
203
204 for vname in self._active_variable_names:
205 sym = adj_table.lookup(vname)
206 if not sym.is_argument:
207 continue
208 # Ensure that the interface we modify is private to this
209 # symbol.
210 # TODO #1544 how do we ensure that an interface is only
211 # referred to by one symbol?
212 new_interface = sym.interface.copy()
213 sym.interface = new_interface
214 if vname in in_names:
215 if vname in out_names:
216 sym.interface.access = \
217 ArgumentInterface.Access.READWRITE
218 else:
219 sym.interface.access = ArgumentInterface.Access.READ
220 else:
221 sym.interface.access = ArgumentInterface.Access.WRITE
222
223 return node_copy

225 def assignment_node(self, node):
226 '''This method is called if the visitor finds an Assignment node. The
227 adjoint of this tangent-linear assignment is returned via the
228 AssignmentTrans transformation. As the adjoint of a single
229 tangent-linear assignment can consist of multiple assignments,
230 a list of nodes is returned.
231
232 :param node: an Assignment PSyIR node.
233 :type node: :py:class:`psyclone.psyir.nodes.Assignment`
234
235 :returns: a list of PSyIR nodes containing the adjoint \
236 of this node.
237 :rtype: list of :py:class:`psyclone.psyir.nodes.Node`
238
239 :raises VisitorError: if the schedule_node method has not been \
240 called previously.
241
242 '''
243 self._logger.debug("Transforming active assignment")
244 if self._active_variables_active_variables is None:
245 raise VisitorError(
246 "An assignment node should not be visited before a schedule, "
247 "as the latter sets up the active variables.")
248 assign_trans = AssignmentTrans(self._active_variables_active_variables)
249 new_node = node.copy()
250 # Temporary parent schedule required by the transformation.
251 dummy_schedule = Schedule()
252 dummy_schedule.children.append(new_node)
253 assign_trans.apply(new_node)
254 return dummy_schedule.pop_all_children()

256 def loop_node(self, node):
257 '''This method is called if the visitor finds a Loop node. If the loop
258 (including any descendants) contains active variables then a
259 new loop is returned which iterates in the reverse order of
260 the original loop and the body of the new loop is the result
261 of processing the body of the original loop. If the loop does
262 not contain any active variables then an exception is raised
263 as this case should have been dealt with by the
264 schedule_node() method.
265
266 :param node: a Loop PSyIR node.
267 :type node: :py:class:`psyclone.psyir.nodes.Loop`
268
269 :returns: a new PSyIR tree containing the adjoint equivalent \
270 of this node and its descendants.
271 :rtype: :py:class:`psyclone.psyir.nodes.Loop`
272
273 :raises VisitorError: if the loop node is visited before a \
274 schedule.
275 :raises VisitorError: if the loop bounds contain any active \
276 variables.
277 :raises VisitorError: if a passive loop is found.
278
279 '''
280 if self._active_variables_active_variables is None:
281 raise VisitorError(
282 "A loop node should not be visited before a schedule, "
283 "as the latter sets up the active variables.")
284
285 # Check that variables in loop bounds and the iterator are
286 # passive, unless they are part of the 1st argument to an
287 # LBOUND or UBOUND function, as this is used to determine
288 # the size of the array, not modify its content.
289 for expr, description in [(node.start_expr, "lower bound"),
290 (node.stop_expr, "upper bound"),
291 (node.step_expr, "step")]:
292 # TODO #2542. References should be iterated with the
293 # reference_acess method when its issues are fixed.
294 for ref in expr.walk(Reference):
295 if (isinstance(ref.parent, Call) and
296 ref is ref.parent.children[0]):
297 continue
298 if ref.symbol in self._active_variables_active_variables:
299 # Ignore LBOUND and UBOUND
300 if not (isinstance(ref.parent, IntrinsicCall) and
301 ref.position == 1 and
302 ref.parent.intrinsic in [
303 IntrinsicCall.Intrinsic.LBOUND,
304 IntrinsicCall.Intrinsic.UBOUND]):
305 raise VisitorError(
306 f"The {description} of a loop should not contain "
307 f"active variables, but found '{ref.name}' in "
308 f"'{expr.debug_string()}'.")
309
310 if node_is_active(Reference(node.variable), self._active_variables_active_variables):
311 raise VisitorError(
312 f"The loop iterator '{node.variable.name}' should not be an "
313 f"active variable.")
314
315 if node_is_passive(node, self._active_variables_active_variables):
316 raise VisitorError(
317 "A passive loop node should not be processed by the "
318 "loop_node() method within the AdjointVisitor() class, as "
319 "it should have been dealt with by the schedule_node() "
320 "method.")
321
322 self._logger.debug("Transforming active loop")
323
324 # The approach taken is to swap the loop bounds and multiply
325 # the step by minus one. However, the loop step might not
326 # align with the loop stop and in this case an offset needs to
327 # be computed e.g. 1 to 4 step 2 gives 1,3, but 4 to 1 step -2
328 # gives 4,2, which is not correct. In this example the offset
329 # (hi-lo mod step) is 1, so we will have (4-1) to 1 step -2,
330 # giving the expected 3,1.
331
332 # Default to no offset required for the loop starting point
333 # (on the assumption that the step is unitary, which it is in
334 # most cases).
335 offset = None
336 if not (isinstance(node.step_expr, Literal) and
337 node.step_expr.value.strip() in ["1", "-1"]):
338 # The loop step might not be unitary so compute an offset:
339 # stop-start mod step
340 fortran_writer = FortranWriter()
341 hi_str = fortran_writer(node.stop_expr)
342 lo_str = fortran_writer(node.start_expr)
343 step_str = fortran_writer(node.step_expr)
344 # TODO: use language independent PSyIR, see issue #1345
345 ptree = Fortran2003.Intrinsic_Function_Reference(
346 f"mod({hi_str}-{lo_str},{step_str})")
347 offset = CodeBlock([ptree], CodeBlock.Structure.EXPRESSION)
348
349 # We only need to copy this node and its bounds. Issue #1440
350 # will address this.
351 new_node = node.copy()
352
353 # Reverse loop order
354 start_expr = new_node.start_expr.copy()
355 if offset:
356 new_node.start_expr = BinaryOperation.create(
357 BinaryOperation.Operator.SUB, new_node.stop_expr.copy(),
358 offset)
359 else:
360 new_node.start_expr = new_node.stop_expr.copy()
361 new_node.stop_expr = start_expr
362 new_node.step_expr = negate_expr(new_node.step_expr.copy())
363
364 # Determine the adjoint of the loop body
365 new_node.children[3] = self._visit(node.children[3])
366 return new_node

368 def ifblock_node(self, node):
369 '''This method is called if the visitor finds an ifblock node. An
370 exception is raised if the condition of the ifblock node
371 contains an active variable as this is not valid
372 tangent-linear code. Otherwise, the ifblock and its condition
373 are returned unchanged and the contents of the "then" and
374 "else" parts of the ifblock are returned after being processed
375 by PSyAD.
376
377 :param node: an IfBlock PSyIR node.
378 :type node: :py:class:`psyclone.psyir.nodes.IfBlock`
379
380 :returns: a new PSyIR tree containing the adjoint equivalent \
381 of this node and its descendants.
382 :rtype: :py:class:`psyclone.psyir.nodes.IfBlock`
383
384 :raises: VisitorError if the condition of the ifblock contains \
385 any active variables.
386 :raises VisitorError: if the ifblock node is visited before a \
387 schedule.
388 :raises VisitorError: if a passive ifblock node is found.
389
390 '''
391 if self._active_variables_active_variables is None:
392 raise VisitorError(
393 "An ifblock node should not be visited before a schedule, "
394 "as the latter sets up the active variables.")
395
396 if node_is_active(node.condition, self._active_variables_active_variables):
397 raise VisitorError(
398 f"The if condition '{node.condition.debug_string()}' of an "
399 f"ifblock node should not contain an active variable (one or "
400 f"more of {self._active_variable_names}).")
401
402 if node_is_passive(node, self._active_variables_active_variables):
403 raise VisitorError(
404 "A passive ifblock node should not be processed by the "
405 "ifblock_node() method within the AdjointVisitor() class, as "
406 "it should have been dealt with by the schedule_node() "
407 "method.")
408
409 self._logger.debug("Transforming active ifblock")
410
411 new_condition = node.condition.copy()
412 new_if_schedule = self._visit(node.if_body)
413 new_if_body = new_if_schedule.pop_all_children()
414
415 new_else_body = None
416 if node.else_body:
417 new_else_schedule = self._visit(node.else_body)
418 new_else_body = new_else_schedule.pop_all_children()
419
420 return IfBlock.create(new_condition, new_if_body, new_else_body)

422 def _copy_and_process(self, node):
423 '''Utility function to return a copy the current node containing the
424 result of processing all descendants.
425
426 :param node: a PSyIR node.
427 :type node: :py:class:`psyclone.psyir.nodes.Node`
428
429 :returns: a new PSyIR tree containing a copy of this node \
430 which contains the result of processing all of its \
431 descendants.
432 :rtype: :py:class:`psyclone.psyir.nodes.Node`
433
434 '''
435 # We only need to copy this node. Issue #1440 will address
436 # this.
437 node_copy = node.copy()
438 node_copy.children = []
439 for child in node.children:
440 result = self._visit(child)
441 # Result may be a node or a list of nodes
442 if isinstance(result, Node):
443 result = [result]
444 node_copy.children.extend(result)
445
446 # TODO #2596 Update any GenericInterfaceSymbols to the use the new
447 # routine symbols created during copying.
448 sym_tab = node_copy.symbol_table
449 for symbol in node_copy.symbol_table.get_symbols():
450 symbol_obj = sym_tab.lookup(symbol)
451 if isinstance(symbol_obj,
452 GenericInterfaceSymbol):
453 new_routines = []
454 for rinfo in symbol_obj.routines:
455 new_routines.append((
456 sym_tab.lookup(rinfo.symbol.name),
457 rinfo.from_container
458 ))
459 symbol_obj.routines = new_routines
460
461 return node_copy
