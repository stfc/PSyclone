54class LFRicAlgInvoke2PSyCallTrans(AlgInvoke2PSyCallTrans):
55 '''
56 Transforms an LFRicAlgorithmInvokeCall into a standard Call to a generated
57 PSy-layer routine.
58
59 This transformation would normally be written as a lowering method
60 on a LFRicAlgorithmInvokeCall. However, we don't always want to
61 lower the code as we want the flexibility to also be able to
62 output algorithm-layer code containing invoke's. We therefore need
63 to selectively apply the lowering, which is naturally written as a
64 transformation.
65
66 '''
67 def validate(self, node, options=None):
68 '''Validate the node argument.
69
70 :param node: a PSyIR node capturing an LFRicinvoke call.
71 :type node: \
72 :py:class:`psyclone.domain.lfric.algorithm.LFRicAlgorithmInvokeCall`
73 :param options: a dictionary with options for transformations.
74 :type options: Optional[Dict[str, Any]]
75 :param options["kernels"]: this option provides a list of \
76 LFRic kernels for this LFRic Algorithm Invoke call.
77 :type options["kernels"]: \
78 List[:py:class:`psyclone.psyir.nodes.Container`]
79
80 :raises TransformationError: if the supplied call argument is \
81 not a PSyIR AlgorithmInvokeCall node.
82 :raises TransformationError: if the 'kernels' option is not provided.
83
84 '''
85 if not isinstance(node, LFRicAlgorithmInvokeCall):
86 raise TransformationError(
87 f"Error in {self.name} transformation. The supplied call "
88 f"argument should be an `LFRicAlgorithmInvokeCall` node but "
89 f"found '{type(node).__name__}'.")
90 try:
91 kernels = options["kernels"]
92 except (KeyError, TypeError) as info:
93 raise TransformationError(
94 "A dictionary containing LFRic kernel PSyIR must be passed "
95 "into the LFRicAlgInvoke2PSyCallTrans transformation but "
96 "this was not found.") from info
97 if not isinstance(kernels, dict):
98 raise TransformationError(
99 f"The value of 'kernels' in the options argument must be a "
100 f"dictionary but found '{type(kernels).__name__}'.")
101 for kern_call in node.arguments:
102 if isinstance(kern_call, LFRicBuiltinFunctor):
103 # Skip builtins as their metadata is stored internally
104 continue
105 try:
106 _ = kernels[id(kern_call)]
107 except KeyError as info:
108 raise TransformationError(
109 f"The 'kernels' option must be a dictionary containing "
110 f"LFRic kernel PSyIR indexed by the id's of the "
111 f"associated kernel functors, but the id for "
112 f"kernel functor '{kern_call.name}' was not "
113 f"found.") from info
114
115 # Check the algorithm arguments and kernel metadata match.
116 self.get_arguments(node, options=options, check_args=True)

118 @staticmethod
119 def _get_metadata(kernel):
120 '''Utility method to extract the kernel metadata from an LFRic
121 kernel.
122
123 :param kernel: LFRic kernel PSyIR.
124 :type kernel: :py:class:`psyclone.psyir.nodes.Container`
125
126 :returns: LFRic kernel metadata.
127 :rtype: :py:class:`psyclone.domain.lfric.kernel.LFRicKernelMetadata`
128
129 :raises TransformationError: if the supplied kernel argument \
130 is not a PSyIR container.
131 :raises TransformationError: if the supplied kernel argument \
132 does not contain an LFRicKernelContainer as the root node \
133 or first child of the root node.
134
135 '''
136 if not isinstance(kernel, Container):
137 raise TransformationError(
138 f"A PSyIR Container (an LFRic kernel module) was expected, "
139 f"but found '{type(kernel).__name__}'.")
140 if isinstance(kernel, LFRicKernelContainer):
141 kernel_metadata = kernel.metadata
142 elif kernel.children and isinstance(
143 kernel.children[0], LFRicKernelContainer):
144 kernel_metadata = kernel.children[0].metadata
145 else:
146 raise TransformationError(
147 "LFRic kernel PSyIR should contain an "
148 "LFRicKernelContainer as the root or first child of the "
149 "root but this was not found.")
150 return kernel_metadata

154 def get_arguments(self, node, options=None, check_args=False):
155 '''By default this method creates the LFRic processed (lowered)
156 argument list from the argument lists of the kernel functors
157 within the invoke call and the kernel metadata.
158
159 If the check_args flag is set to True this method does not
160 create an argument list, but instead checks that the number of
161 arguments expected by the kernel metadata and the number of
162 arguments supplied in the algorithm layer match.
163
164 Check args does not create the argument list because a) it is
165 faster to not create the list and b) a mismatch in the number
166 of expected and actual arguments can cause an index error.
167
168 :param node: an LFRic algorithm invoke call.
169 :type node: :py:class:`psyclone.domain.lfric.algorithm.psyir.\
170 LFRicAlgorithmInvokeCall`
171 :param options: a dictionary with options for transformations.
172 :type options: Optional[Dict[str, Any]]
173 :param options["kernels"]: this option provides a list of \
174 LFRic kernels for this LFRic Algorithm Invoke call.
175 :type options["kernels"]: \
176 List[:py:class:`psyclone.psyir.nodes.Container`]
177 :param bool check_args: if True, checks the number of kernel \
178 functor arguments matches the number expected by the kernel \
179 metadata. Defaults to False.
180
181 :returns: the processed (lowered) argument list.
182 :rtype: List[:py:class:`psyclone.psyir.nodes.Node`]
183
184 :raises GenerationError: if the number of arguments in the \
185 invoke does not match the expected number of arguments \
186 specified by the metadata.
187
188 '''
189 # TODO #1618. This routine could/should make use of the new
190 # LFRicArgOrder class if and when it makes its way onto trunk.
191
192 const = LFRicConstants()
193 # No need to check the lookup of "kernels" in the options
194 # dictionary as it has already been validated by the validate()
195 # method.
196 kernels = options["kernels"]
197
198 # 4 separate lists are used below because the processed
199 # (lowered) argument list expects all scalar, field and
200 # operator arguments first, then all stencil arguments
201 # (separated into size arguments first followed by direction
202 # arguments) and finally all qr arguments.
203
204 # The processed (lowered) argument list for scalar, field and
205 # operator arguments.
206 arguments = []
207 # The processed (lowered) argument lists for any stencil
208 # arguments (separated into stencil size and direction
209 # arguments).
210 stencil_size_arguments = []
211 stencil_direction_arguments = []
212 # The processed (lowered) argument list for any quadrature
213 # arguments.
214 quad_arguments = []
215
216 # pylint: disable=too-many-nested-blocks
217 for kern_call in node.arguments:
218 if isinstance(kern_call, LFRicBuiltinFunctor):
219 # metadata is stored in PSyclone builtin classes
220 builtin_name = kern_call.name.lower()
221 kernel_metadata = BUILTIN_MAP[builtin_name].metadata()
222 else:
223 # metadata is stored in the user-written kernel
224 kernel = kernels[id(kern_call)]
225 kernel_metadata = self._get_metadata(kernel)
226 arg_idx = 0
227 for meta_arg in kernel_metadata.meta_args:
228 if not check_args:
229 arg = kern_call.children[arg_idx]
230 self._add_arg(arg, arguments)
231 if type(meta_arg) in [
232 FieldArgMetadata, FieldVectorArgMetadata,
233 InterGridArgMetadata, InterGridVectorArgMetadata]:
234 if meta_arg.stencil:
235 arg_idx += 1
236 if not check_args:
237 stencil_arg = kern_call.children[arg_idx]
238 self._add_arg(stencil_arg, stencil_size_arguments)
239 if meta_arg.stencil == "xory1d":
240 # An additional direction stencil argument
241 # is required in the algorithm layer.
242 arg_idx += 1
243 if not check_args:
244 stencil_arg = kern_call.children[arg_idx]
245 if isinstance(stencil_arg, Literal):
246 raise GenerationError(
247 f"A literal is not a valid value for "
248 f"a stencil direction, but found "
249 f"'{stencil_arg.value}' for field "
250 f"'{arg.name}'.")
251 if stencil_arg.name.lower() not in [
252 "x_direction", "y_direction"]:
253 self._add_arg(
254 stencil_arg,
255 stencil_direction_arguments)
256 arg_idx += 1
257 if kernel_metadata.shapes:
258 quad_args = [quad for quad in kernel_metadata.shapes
259 if quad in const.VALID_QUADRATURE_SHAPES]
260 for quad_arg in quad_args:
261 # There is an additional quadrature argument.
262 if not check_args:
263 quad_arg = kern_call.children[arg_idx]
264 self._add_arg(quad_arg, quad_arguments)
265 arg_idx += 1
266
267 # Incorrect number of kernel functor arguments
268 if check_args and len(kern_call.children) != arg_idx:
269 raise GenerationError(
270 f"The invoke kernel functor '{kern_call.name}' has "
271 f"{len(kern_call.children)} arguments, but the kernel "
272 f"metadata expects there to be {arg_idx} arguments.")
273
274 # Add the stencil and quadrature arguments in the order
275 # expected in the processed (lowered) argument list. (We
276 # expect all scalar, field and operator arguments first, then
277 # all stencil arguments (separated into size arguments first
278 # followed by direction arguments) and finally all qr
279 # arguments).
280 arguments.extend(stencil_size_arguments)
281 arguments.extend(stencil_direction_arguments)
282 arguments.extend(quad_arguments)
283
284 return arguments

286 def apply(self, node, options=None):
287 ''' Apply the transformation to the supplied LFRicAlgorithmInvokeCall
288 node. That node is replaced by a Call to the corresponding PSy-layer
289 routine with appropriate arguments. The symbols representing any
290 LFRic Builtins that are now no longer referred to are removed.
291
292 :param node: a PSyIR algorithm invoke call node.
293 :type node: \
294 :py:class:`psyclone.domain.lfric.algorithm.LFRicAlgorithmInvokeCall`
295 :param options: a dictionary with options for transformations.
296 :type options: Optional[Dict[str, Any]]
297 :param options["kernels"]: this option provides a list of \
298 LFRic kernels for this LFRic Algorithm Invoke call.
299 :type options["kernels"]: \
300 List[:py:class:`psyclone.psyir.nodes.Container`]
301
302 '''
303 self.validate(node, options=options)
304
305 # The generic class does not handle Builtins so we do that here. We
306 # have to record which Builtins are involved before the call is
307 # transformed.
308 builtin_symbols = set()
309 for kern in node.children:
310 if isinstance(kern, LFRicBuiltinFunctor):
311 builtin_symbols.add(kern.symbol)
312
313 # 'node' will get replaced with a new Call node so keep a record
314 # of where we are in the tree.
315 parent = node.parent
316
317 super().apply(node, options=options)
318
319 # Now that the transformation is done, check whether we can remove
320 # any of the symbols for the Builtins.
321 for sym in builtin_symbols:
322 table = sym.find_symbol_table(parent)
323 node = table.node
324 functors = node.walk(LFRicBuiltinFunctor)
325 for func in functors:
326 if sym is func.symbol:
327 break
328 else:
329 # We didn't find a Functor referring to this symbol so
330 # we can remove it.
331 # TODO #898 SymbolTable.remove() does not support
332 # DataTypeSymbol so remove it manually.
333 # pylint: disable=protected-access
334 del table._symbols[sym.name]
