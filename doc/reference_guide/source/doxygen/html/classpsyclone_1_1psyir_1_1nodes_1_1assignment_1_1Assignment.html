Detailed Description

Node representing an Assignment statement. As such it has a LHS and RHS
as children 0 and 1 respectively.

:param bool is_pointer: whether this represents a pointer assignment.
:param kwargs: additional keyword arguments provided to the Node.
:type kwargs: unwrapped dict.

214 def is_array_assignment(self):
215 '''
216 :returns: True if the lhs of the assignment is an array access with at
217 least one of its dimensions being a range and False otherwise.
218 :rtype: bool
219
220 '''
221 # It's not sufficient simply to check for a Range node as that may be
222 # part of an argument to an Operator or function that performs a
223 # reduction and thus returns a scalar result, e.g. a(SUM(b(:))) = 1.0
224 # TODO #658 this check for reductions needs extending to also support
225 # user-implemented functions.
226 if isinstance(self.lhs, (ArrayReference, StructureReference)):
227 ranges = self.lhs.walk(Range)
228 for array_range in ranges:
229 opn = array_range.ancestor(IntrinsicCall)
230 while opn:
231 if opn.intrinsic in REDUCTION_INTRINSICS:
232 # The current array range is in an argument to a
233 # reduction intrinsic so we assume that the result
234 # is a scalar.
235 # TODO #658 this could still be a reduction
236 # into an array (e.g. SUM(a(:,:), dim=1)) but
237 # we need to be able to interrogate the type
238 # of a PSyIR expression in order to be
239 # sure. e.g. SUM(a(:,:), mask=mask(:,:)) will
240 # return a scalar.
241 break
242 opn = opn.ancestor(IntrinsicCall)
243 else:
244 # We didn't find a reduction intrinsic so there is an
245 # array range on the LHS
246 return True
247 return False

175 def reference_accesses(self, var_accesses):
176 '''Get all variable access information from this node. The assigned-to
177 variable will be set to 'WRITE'.
178
179 :param var_accesses: VariablesAccessInfo instance that stores the \
180 information about variable accesses.
181 :type var_accesses: \
182 :py:class:`psyclone.core.VariablesAccessInfo`
183
184 '''
185 # It is important that a new instance is used to handle the LHS,
186 # since a check in 'change_read_to_write' makes sure that there
187 # is only one access to the variable! Also forward the options
188 # from the original object to the new object.
189 accesses_left = VariablesAccessInfo(options=var_accesses.options())
190 self.lhs.reference_accesses(accesses_left)
191 # Now change the (one) access to the assigned variable to be WRITE:
192 sig, _ = self.lhs.get_signature_and_indices()
193 var_info = accesses_left[sig]
194 try:
195 var_info.change_read_to_write()
196 except InternalError as err:
197 # An internal error typically indicates that the same variable
198 # is used twice on the LHS, e.g.: g(g(1)) = ... This is not
199 # supported in PSyclone.
200 raise NotImplementedError(f"The variable '{self.lhs.name}' appears"
201 f" more than once on the left-hand side "
202 f"of an assignment.") from err
203
204 # Merge the data (that shows now WRITE for the variable) with the
205 # parameter to this function. It is important that first the
206 # RHS is added, so that in statements like 'a=a+1' the read on
207 # the RHS comes before the write on the LHS (they have the same
208 # location otherwise, but the order is still important)
209 self.rhs.reference_accesses(var_accesses)
210 var_accesses.merge(accesses_left)
211 var_accesses.next_location()
212
