50class PSyDataTrans(RegionTrans):
51 ''' Create a PSyData region around a list of statements. For
52 example:
53
54 >>> from psyclone.parse.algorithm import parse
55 >>> from psyclone.parse.utils import ParseError
56 >>> from psyclone.psyGen import PSyFactory
57 >>> api = "gocean"
58 >>> ast, invoke_info = parse(SOURCE_FILE, api=api)
59 >>> psy = PSyFactory(api).create(invoke_info)
60 >>>
61 >>> from psyclone.psyir.transformations import PSyDataTrans
62 >>> data_trans = PSyDataTrans()
63 >>>
64 >>> schedule = psy.invokes.get('invoke_0').schedule
65 >>> # Uncomment the following line to see a text view of the schedule
66 >>> # print(schedule.view())
67 >>>
68 >>> # Enclose all children within a single PSyData region
69 >>> data_trans.apply(schedule.children)
70 >>> # Uncomment the following line to see a text view of the schedule
71 >>> # print(schedule.view())
72 >>> # Or to use custom region name:
73 >>> data_trans.apply(schedule.children,
74 ... {"region_name": ("module","region")})
75
76 :param node_class: The Node class of which an instance will be inserted \
77 into the tree (defaults to PSyDataNode).
78 :type node_class: :py:class:`psyclone.psyir.nodes.ExtractNode`
79
80 '''
81 # Unlike other transformations we can be fairly relaxed about the nodes
82 # that a region can contain as we don't have to understand them.
83 excluded_node_types = (Return,)
84
85 # This dictionary keeps track of region+module names that are already
86 # used. For each key (which is module_name+"|"+region_name) it contains
87 # how many regions with that name have been created. This number will
88 # then be added as an index to create unique region identifiers.
89 _used_kernel_names = {}
90
91 def __init__(self, node_class=PSyDataNode):
92 super().__init__()
93 self._node_class = node_class
94
95 # ------------------------------------------------------------------------
96 def __str__(self):
97 return (f"Create a sub-tree of the PSyIR that has a node of type "
98 f"{self._node_class.__name__} at its root.")

115 def get_unique_region_name(self, nodes, options):
116 '''This function returns the region and module name. If they are
117 specified in the user options, these names will just be returned (it
118 is then up to the user to guarantee uniqueness). Otherwise a name
119 based on the module and invoke will be created using indices to
120 make sure the name is unique.
121
122 :param nodes: a list of nodes.
123 :type nodes: list of :py:obj:`psyclone.psyir.nodes.Node`
124 :param options: a dictionary with options for transformations.
125 :type options: Dict[str, Any]
126 :param (str,str) options["region_name"]: an optional name to \
127 use for this PSyData area, provided as a 2-tuple containing a \
128 location name followed by a local name. The pair of strings \
129 should uniquely identify a region unless aggregate information \
130 is required (and is supported by the runtime library).
131
132 '''
133 # We don't use a static method here since it might be useful to
134 # overwrite this functions in derived classes
135 name = options.get("region_name", None)
136 if name:
137 # pylint: disable=too-many-boolean-expressions
138 if not isinstance(name, tuple) or not len(name) == 2 or \
139 not name[0] or not isinstance(name[0], str) or \
140 not name[1] or not isinstance(name[1], str):
141 raise InternalError(
142 "Error in PSyDataTrans. The name must be a "
143 "tuple containing two non-empty strings.")
144 # pylint: enable=too-many-boolean-expressions
145 # Valid PSyData names have been provided by the user.
146 return name
147
148 invoke = nodes[0].ancestor(InvokeSchedule).invoke
149 module_name = invoke.invokes.psy.name
150
151 # Use the invoke name as a starting point.
152 region_name = invoke.name
153 kerns = []
154 for node in nodes:
155 kerns.extend(node.walk(Kern))
156
157 if len(kerns) == 1:
158 # This PSyData region only has one kernel within it,
159 # so append the kernel name.
160 region_name += f"-{kerns[0].name}"
161
162 # Add a region index to ensure uniqueness when there are
163 # multiple regions in an invoke.
164 key = module_name + "|" + region_name
165 idx = PSyDataTrans._used_kernel_names.get(key, 0)
166 PSyDataTrans._used_kernel_names[key] = idx + 1
167 region_name += f"-r{idx}"
168 return (module_name, region_name)

171 def validate(self, nodes, options=None):
172 '''
173 Checks that the supplied list of nodes is valid, that the location
174 for this node is valid (not between a loop-directive and its loop),
175 that there aren't any name clashes with symbols that must be
176 imported from the appropriate PSyData library and finally, calls the
177 validate method of the base class.
178
179 :param nodes: a node or list of nodes to be instrumented with \
180 PSyData API calls.
181 :type nodes: (list of) :py:class:`psyclone.psyir.nodes.Loop`
182
183 :param options: a dictionary with options for transformations.
184 :type options: Optional[Dict[str, Any]]
185 :param str options["prefix"]: a prefix to use for the PSyData module \
186 name (``PREFIX_psy_data_mod``) and the PSyDataType \
187 (``PREFIX_PSYDATATYPE``) - a "_" will be added automatically. \
188 It defaults to "".
189 :param (str,str) options["region_name"]: an optional name to \
190 use for this PSyData area, provided as a 2-tuple containing a \
191 location name followed by a local name. The pair of strings \
192 should uniquely identify a region unless aggregate information \
193 is required (and is supported by the runtime library).
194
195 :raises TransformationError: if the supplied list of nodes is empty.
196 :raises TransformationError: if the PSyData node is inserted \
197 between an OpenMP/ACC directive and the loop(s) to which it \
198 applies.
199 :raises TransformationError: if the 'prefix' or 'region_name' options \
200 are not valid.
201 :raises TransformationError: if there will be a name clash between \
202 any existing symbols and those that must be imported from the \
203 appropriate PSyData library.
204
205 '''
206 # pylint: disable=too-many-branches
207 node_list = self.get_node_list(nodes)
208
209 if not node_list:
210 raise TransformationError("Cannot apply transformation to an "
211 "empty list of nodes.")
212
213 node_parent = node_list[0].parent
214 if isinstance(node_parent, Schedule) and \
215 isinstance(node_parent.parent, (OMPDoDirective, ACCLoopDirective)):
216 raise TransformationError("A PSyData node cannot be inserted "
217 "between an OpenMP/ACC directive and "
218 "the loop(s) to which it applies!")
219
220 if node_list[0].ancestor(ACCDirective):
221 raise TransformationError("A PSyData node cannot be inserted "
222 "inside an OpenACC region.")
223
224 if options is None:
225 options = {}
226 if "region_name" in options:
227 name = options["region_name"]
228 # pylint: disable=too-many-boolean-expressions
229 if not isinstance(name, tuple) or not len(name) == 2 or \
230 not name[0] or not isinstance(name[0], str) or \
231 not name[1] or not isinstance(name[1], str):
232 raise TransformationError(
233 f"Error in {self.name}. User-supplied region name "
234 f"must be a tuple containing two non-empty strings.")
235 # pylint: enable=too-many-boolean-expressions
236 prefix = options.get("prefix", None)
237 if "prefix" in options:
238 prefix = options.get("prefix", None)
239 if prefix not in Config.get().valid_psy_data_prefixes:
240 raise TransformationError(
241 f"Error in 'prefix' parameter: found '{prefix}', while"
242 f" one of {Config.get().valid_psy_data_prefixes} was "
243 f"expected as defined in {Config.get().filename}")
244
245 # We have to create an instance of the node that will be inserted in
246 # order to find out what module name it will use.
247 pdata_node = self._node_class(options=options)
248 table = node_list[0].scope.symbol_table
249 for name in ([sym.name for sym in pdata_node.imported_symbols] +
250 [pdata_node.fortran_module]):
251 try:
252 _ = table.lookup_with_tag(name)
253 except KeyError as err:
254 # The tag doesn't exist which means that we haven't already
255 # added this symbol as part of a PSyData transformation. Check
256 # for any clashes with existing symbols.
257 try:
258 _ = table.lookup(name)
259 raise TransformationError(
260 f"Cannot add PSyData calls because there is already a "
261 f"symbol named '{name}' which clashes with one of "
262 f"those used by the PSyclone PSyData API. ") from err
263 except KeyError:
264 pass
265
266 super().validate(node_list, options)

269 def apply(self, nodes, options=None):
270 # pylint: disable=arguments-renamed
271 '''Apply this transformation to a subset of the nodes within a
272 schedule - i.e. enclose the specified Nodes in the
273 schedule within a single PSyData region.
274
275 :param nodes: can be a single node or a list of nodes.
276 :type nodes: :py:obj:`psyclone.psyir.nodes.Node` or list of \
277 :py:obj:`psyclone.psyir.nodes.Node`
278 :param options: a dictionary with options for transformations.
279 :type options: Optional[Dict[str, Any]]
280 :param str options["prefix"]: a prefix to use for the PSyData module \
281 name (``PREFIX_psy_data_mod``) and the PSyDataType \
282 (``PREFIX_PSYDATATYPE``) - a "_" will be added automatically. \
283 It defaults to "".
284 :param (str,str) options["region_name"]: an optional name to \
285 use for this PSyData area, provided as a 2-tuple containing a \
286 location name followed by a local name. The pair of strings \
287 should uniquely identify a region unless aggregate information \
288 is required (and is supported by the runtime library).
289
290 '''
291 node_list = self.get_node_list(nodes)
292
293 # Perform validation checks
294 self.validatevalidatevalidate(node_list, options)
295
296 # Get useful references
297 parent = node_list[0].parent
298 position = node_list[0].position
299
300 # We always use the Routine symbol table
301 table = node_list[0].ancestor(Routine).symbol_table
302
303 # Create an instance of the required class that implements
304 # the code extraction using the PSyData API, e.g. a
305 # ExtractNode. We pass the user-specified options to the
306 # create() method. An example use case for this is the
307 # 'create_driver' flag, where the calling program can control if
308 # a stand-alone driver program should be created or not (when
309 # performing kernel extraction).
310 for node in node_list:
311 node.detach()
312
313 psy_data_node = self._node_class.create(
314 node_list, symbol_table=table, options=options)
315 parent.addchild(psy_data_node, position)
