82def check_line_length(filename):
83 '''Check that the code contained within the filename file
84 conforms to the 132 line length limit.
85
86 :param str filename: The name of the file containing the code.
87
88 :raises InternalError: if the specified file can not be opened or read.
89 :raises ParseError: if one of more lines are longer than the 132
90 line length limit.
91 '''
92 fll = FortLineLength()
93 try:
94 with io.open(filename, "r", encoding='utf8') as myfile:
95 code_str = myfile.read()
96 except IOError as excinfo:
97 raise InternalError(
98 f"In utils.py:check_line_length: {excinfo}") from excinfo
99
100 if fll.long_lines(code_str):
101 raise ParseError(
102 f"File '{filename}' does not conform to the specified {fll.length}"
103 f" line-length limit. Either correct the file or change the "
104 f"'-l/--limit' setting on the PSyclone command line.")

107def parse_fp2(filename):
108 '''Parse a Fortran source file contained in the file 'filename' using
109 fparser2.
110
111 :param str filename: source file (including path) to read.
112 :returns: fparser2 AST for the source file.
113 :rtype: :py:class:`fparser.two.Fortran2003.Program`
114 :raises ParseError: if the file could not be parsed.
115
116 '''
117 # We get the directories to search for any Fortran include files from
118 # our configuration object.
119 config = Config.get()
120 try:
121 reader = FortranFileReader(filename, include_dirs=config.include_paths)
122 except IOError as error:
123 raise ParseError(
124 f"algorithm.py:parse_fp2: Failed to parse file '{filename}'. "
125 f"Error returned was ' {error} '.") from error
126 parser = ParserFactory().create(std=config.fortran_standard)
127 try:
128 parse_tree = parser(reader)
129 except FortranSyntaxError as msg:
130 raise ParseError(
131 f"algorithm.py:parse_fp2: Syntax error in file '{filename}':\n"
132 f"{msg}") from msg
133 return parse_tree
