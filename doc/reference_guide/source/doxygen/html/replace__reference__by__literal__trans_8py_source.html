60class ReplaceReferenceByLiteralTrans(Transformation):
61 '''
62 This transformation takes a psyir Routine and replace all Reference psyir
63 Nodes by Literal if the corresponding symbol from the symbol table is
64 constant. That is to say the symbol is a Fortran parameter.
65 NOTE: this transformation does not handle parent scopes recursively yet.
66 For example:
67
68
69 >>> from psyclone.psyir.backend.fortran import FortranWriter
70 >>> from psyclone.psyir.symbols import INTEGER_TYPE
71 >>> from psyclone.psyir.transformations import (
72 ReplaceReferenceByLiteralTrans)
73 >>> source = """program test
74 ... use mymod
75 ... type(my_type):: t1, t2, t3, t4
76 ... integer, parameter :: x=3, y=12, z=13
77 ... integer, parameter :: u1=1, u2=2, u3=3, u4=4
78 ... integer i, invariant, ic1, ic2, ic3
79 ... real, dimension(10) :: a
80 ... invariant = 1
81 ... do i = 1, 10
82 ... t1%a = z
83 ... a(ic1) = u1+(ic1+x)*ic1
84 ... a(ic2) = u2+(ic2+y)*ic2
85 ... a(ic3) = u3+(ic3+z)*ic3
86 ... a(t1%a) = u4+(t1%a+u4*z)*t1%a
87 ... end do
88 ... end program test"""
89 >>> fortran_writer = FortranWriter()
90 >>> fortran_reader = FortranReader()
91 >>> psyir = fortran_reader.psyir_from_source(source)
92 >>> routine = psyir.walk(Routine)[0]
93 >>> rrbl = ReplaceReferenceByLiteralTrans()
94 >>> rrbl.apply(routine)
95 >>> written_code = fortran_writer(routine)
96 >>> print(written_code)
97 program test
98 use mymod
99 integer, parameter :: x = 3
100 integer, parameter :: y = 12
101 integer, parameter :: z = 13
102 integer, parameter :: u1 = 1
103 integer, parameter :: u2 = 2
104 integer, parameter :: u3 = 3
105 integer, parameter :: u4 = 4
106 type(my_type) :: t1
107 type(my_type) :: t2
108 type(my_type) :: t3
109 type(my_type) :: t4
110 integer :: i
111 integer :: invariant
112 integer :: ic1
113 integer :: ic2
114 integer :: ic3
115 real, dimension(10) :: a
116 <BLANKLINE>
117 invariant = 1
118 do i = 1, 10, 1
119 t1%a = 13
120 a(ic1) = 1 + (ic1 + 3) * ic1
121 a(ic2) = 2 + (ic2 + 12) * ic2
122 a(ic3) = 3 + (ic3 + 13) * ic3
123 a(t1%a) = 4 + (t1%a + 4 * 13) * t1%a
124 enddo
125 <BLANKLINE>
126 end program test
127 <BLANKLINE>
128
129 '''
130 def __init__(self) -> None:
131 super().__init__()
132 # Dictionary with Literal values of the corresponding symbol
133 # from symbol_table (based on symbol name as a string).
134 self._param_table: Dict[str, Literal] = {}
135
136 def _update_param_table(
137 self,
138 param_table: Dict[str, Literal],
139 symbol_table: SymbolTable,
140 ) -> Dict[str, Literal]:
141 """This methods takes a param_table as entry, updates this dictionary
142 and then returns the same dictionary updated.
143
144 * Goes through all datasymbols in the symbol_table.
145 * if symbol already in param_table or initial_value is not Literal:
146 * annotate code with warning.
147 * copy and detach the symbol.initial_value (Literal)
148 * update the param_table with this copy.
149 * Returns the updated param_table
150
151 :param param_table: To be updated
152 :param symbol_table: scope symbol table to look for the symbols.
153
154 :return: the updated param_table (same reference as the entry one)
155 """
156 for sym in symbol_table.datasymbols:
157 sym: DataSymbol
158 if sym.is_constant:
159 sym_name = sym.name
160 if param_table.get(sym_name):
161 message = (
162 f"{self.name}:"
163 + f" Symbol already found '{sym_name}'."
164 + " A conflict is possible."
165 + "To avoid replacing the symbol with the wrong value,"
166 + " we skip this symbol."
167 )
168 sym.preceding_comment += message
169 param_table.pop(sym_name)
170 continue
171 if not isinstance(sym.initial_value, Literal):
172
173 message = (
174 f"{self.name}: only "
175 + "supports symbols which have a Literal"
176 + " as their initial value but "
177 + f"'{sym_name}' is assigned "
178 + f"a {type(sym.initial_value)}"
179 )
180 sym.preceding_comment += message
181 continue
182 new_literal: Literal = sym.initial_value.copy().detach()
183 param_table[sym_name] = new_literal
184 else:
185 if isinstance(sym.datatype, UnsupportedFortranType):
186 if "PARAMETER" in sym.datatype.declaration:
187 message = (
188 f"{self.name}:"
189 + " only support constant (parameter) but "
190 + f"{sym.datatype} is not seen by "
191 + "Psyclone as a constant."
192 )
193 sym.preceding_comment += message
194 return param_table

196 def _replace_bounds(
197 self,
198 current_shape: List[Union[Literal, Reference]],
199 param_table: Dict[str, Literal],
200 ) -> List[Union[Literal, Reference]]:
201 """From the param_table and the current_shape of an array,
202 this method create a new_shape with the reference replaced by literal
203 when they are found in the param_table.
204
205 :param current_shape: shape before transformation
206 :param param_table: map of parameters to Literal values.
207
208 :return: the new shape with any references to constants replaced by
209 their Literal values.
210 """
211 new_shape = []
212 for dim in current_shape:
213 if isinstance(dim, ArrayType.ArrayBounds):
214 dim_upper = dim.upper.copy()
215 dim_lower = dim.lower.copy()
216 ref: DataNode = dim.upper
217 if isinstance(ref, Reference) and ref.name in param_table:
218 literal: Literal = param_table[ref.name]
219 dim_upper = literal.copy()
220 ref = dim.lower
221 if isinstance(ref, Reference) and ref.name in param_table:
222 literal: Literal = param_table[ref.name]
223 dim_lower = literal.copy()
224 new_bounds = ArrayType.ArrayBounds(dim_lower, dim_upper)
225 new_shape.append(new_bounds)
226 else:
227 # This dimension is specified with an ArrayType.Extent
228 # so no need to copy.
229 new_shape.append(dim)
230 return new_shape

233 def apply(self, node: Routine, options: Optional[Dict[str, Any]] = None):
234 """Applies the transformation to a Routine node:
235 * First update a dictionary (param_table) with the Literal of constant
236 (parameter) symbol from node.parent symbol_table, and from
237 node.symbol_table.
238 * Second, use this updated param_table to replace reference in node
239 psyir_tree with the corresponsing Literal.
240 * Third, use this updated param_table to replace reference in node
241 symbol_table DataSymbol array's dimensions with the corresponding
242 Literal.
243
244 Note: If an update cannot be performed for any reason then the
245 substitution is skipped and a comment starting with
246 'Psyclone(ReplaceReferenceByLiteral):' is added to the generated code.
247
248 :param node: node on which the transformation is applied
249 :param options: a dictionary with options for transformations.
250 """
251
252 self._param_table = {}
253 self.validatevalidate(node, options)
254
256 if node.parent is not None and isinstance(node.parent, Container):
257 self._param_table = self._update_param_table(
258 self._param_table, node.parent.symbol_table
259 )
260
261 self._param_table = self._update_param_table(
262 self._param_table, node.symbol_table
263 )
264
265 for ref in node.walk(Reference):
266 ref: Reference
267 if ref.name in self._param_table:
268 literal: Literal = self._param_table[ref.name]
269 ref.replace_with(literal.copy())
270
271 for sym in node.symbol_table.datasymbols:
272 sym: DataSymbol
273 if sym.is_array:
274 if not isinstance(sym.datatype, UnsupportedFortranType):
275 new_shape: List[Union[Literal, Reference]] = (
276 self._replace_bounds(sym.shape, self._param_table)
277 )
278 sym.datatype = ArrayType(sym.datatype.datatype, new_shape)
