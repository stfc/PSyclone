36'''Module containing the MetaMeshArgMetadata class which
37captures the argument values for the LFRic kernel
38META_MESH metadata.
39
40'''
41from psyclone.domain.lfric import LFRicConstants
42from psyclone.domain.lfric.kernel.common_arg_metadata import CommonArgMetadata
43
44
45class MetaMeshArgMetadata(CommonArgMetadata):
46 '''Class to capture the LFRic kernel metadata information for a
47 meta_mesh argument.
48
49 :param str mesh: the name of the mesh property.
50
51 '''
52 def __init__(self, mesh):
53 super().__init__()
54 self.meshmeshmesh = mesh
55
56 @staticmethod
57 def create_from_fparser2(fparser2_tree):
58 '''Create an instance of this class from an fparser2 tree.
59
60 :param fparser2_tree: fparser2 tree containing the metadata \
61 for a meta_mesh argument.
62 :type fparser2_tree: :py:class:`fparser.two.Fortran2003.Part_Ref`
63
64 :returns: an instance of this class.
65 :rtype: :py:class:`psyclone.domain.lfric.kernel.MetaMeshArgMetadata`
66
67 '''
68 MetaMeshArgMetadata.check_fparser2_arg(
69 fparser2_tree, type_name="mesh_data_type")
70 MetaMeshArgMetadata.check_nargs(fparser2_tree, 1)
71 mesh = MetaMeshArgMetadata.get_arg(fparser2_tree, 0)
72 return MetaMeshArgMetadata(mesh)
73
74 def fortran_string(self):
75 '''
76 :returns: the metadata represented by this class as Fortran.
77 :rtype: str
78 '''
79 return f"mesh_data_type({self.mesh})"
80
81 @property
82 def mesh(self):
83 '''
84 :returns: the mesh property for this meta_mesh argument.
85 :rtype: str
86 '''
87 return self._mesh
88
89 @mesh.setter
90 def mesh(self, value):
91 '''
92 :param str value: set the mesh property to the specified value.
93 '''
94 const = LFRicConstants()
95 self.validate_scalar_value(
96 value, const.VALID_MESH_NAMES, "mesh property")
97 self._mesh = value.lower()
