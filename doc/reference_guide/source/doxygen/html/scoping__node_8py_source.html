46class ScopingNode(Node):
47 ''' Abstract node that has an associated Symbol Table to keep track of
48 symbols declared in its scope (symbols that can be accessed by this node
49 and any of its descendants). If a pre-existing symbol table is provided,
50 it will be attached to the node (raising an error if the symbol table
51 is already attached to another scope), otherwise a new empty Symbol Table
52 will be created.
53
54 :param children: the PSyIR nodes that are children of this node.
55 :type children: List[:py:class:`psyclone.psyir.nodes.Node`]
56 :param parent: the parent node of this node in the PSyIR.
57 :type parent: Optional[:py:class:`psyclone.psyir.nodes.Node`]
58 :param symbol_table: attach the given symbol table to the new node.
59 :type symbol_table:
60 Optional[:py:class:`psyclone.psyir.symbols.SymbolTable`]
61
62 '''
63 # Polymorphic parameter to initialize the Symbol Table of the ScopingNode
64 _symbol_table_class = SymbolTable
65
66 def __init__(self, children=None, parent=None, symbol_table=None):
67 self._symbol_table = None
68
69 # As Routines (a ScopingNode subclass) add their own symbol
70 # into their parent's symbol table its necessary to ensure
71 # that the _symbol_table exists before we add the children
72 # to the Node (which happens in the super constructor).
73 if symbol_table is not None:
74 # Attach the provided symbol table to this scope
75 symbol_table.attach(self)
76 else:
77 # Create a new symbol table attached to this scope
78 self._symbol_table = self._symbol_table_class(self)
79
80 super().__init__(children=children, parent=parent)

82 def __eq__(self, other):
83 '''
84 Checks whether two nodes are equal. Scoping nodes are equal if their
85 symbol tables are equal.
86
87 :param object other: the object to check equality to.
88
89 :returns: whether other is equal to self.
90 :rtype: bool
91 '''
92 is_eq = super().__eq__(other)
93 is_eq = is_eq and self.symbol_table == other.symbol_table
94 return is_eq

96 def _refine_copy(self, other):
97 ''' Refine the object attributes when a shallow copy is not the most
98 appropriate operation during a call to the copy() method.
99
100 This method creates a deep copy of the SymbolTable associated with
101 the `other` scoping node and then calls `replace_symbols_using` to
102 update all Symbols referenced in the tree below this node.
103
104 .. warning::
105
106 Since `replace_symbols_using` only uses symbol *names*, this won't
107 get the correct symbol if the PSyIR has symbols shadowed in nested
108 scopes, e.g.:
109
110 .. code-block::
111
112 subroutine test
113 integer :: a
114 integer :: b = 1
115 if condition then
116 ! PSyIR declares a shadowed, locally-scoped a'
117 a' = 1
118 if condition2 then
119 ! PSyIR declares a shadowed, locally-scoped b'
120 b' = 2
121 a = a' + b'
122
123 Here, the final assignment will end up being `a' = a' + b'` and
124 thus the semantics of the code are changed. TODO #2666.
125
126 :param other: object we are copying from.
127 :type other: :py:class:`psyclone.psyir.node.Node`
128
129 '''
130 # Reorganise the symbol table construction to occur before we add
131 # the children.
132 self._symbol_table = other.symbol_table.deep_copy()
133 self._symbol_table._node = self # Associate to self
134
135 # Remove symbols corresponding to Routines that are contained in this
136 # ScopingNode. These symbols will be added automatically by the Routine
137 # objects when we add them to our child list in the super refine_copy
138 # call.
139 # We have to import Routine here to avoid a circular dependency.
140 # pylint: disable=import-outside-toplevel
141 from psyclone.psyir.nodes.routine import Routine
142 removed_tags = {}
143 for node in other.walk(Routine):
144 try:
145 if isinstance(self._symbol_table.lookup(node.name),
146 RoutineSymbol):
147 # We have to force the removal of the symbol.
148 symbol = self._symbol_table.lookup(node.name)
149 norm_name = self._symbol_table._normalize(symbol.name)
150 self._symbol_table._symbols.pop(norm_name)
151 # If the symbol had a tag, it should be disassociated
152 for tag, tagged_symbol in list(
153 self._symbol_table._tags.items()):
154 if symbol is tagged_symbol:
155 removed_tags[tag] = tagged_symbol
156 del self._symbol_table._tags[tag]
157 except KeyError:
158 pass
159
160 super(ScopingNode, self)._refine_copy(other)
161
162 # Add any routine tags back
163 for tag in removed_tags.keys():
164 # pylint: disable-next=protected-access
165 self._symbol_table._tags[tag] = self._symbol_table.lookup(
166 removed_tags[tag].name)
167
168 # Now we've updated the symbol table, walk back down the tree and
169 # update any Symbols.
170 # Since this will get called from every ScopingNode in the tree, there
171 # could be a lot of repeated walks back down the tree. If this becomes
172 # a performance issue we could keep track of the depth of the recursive
173 # call to _refine_copy and only do this call when that depth is zero.
174 self.replace_symbols_usingreplace_symbols_using(self._symbol_table)

176 def reference_accesses(self, access_info: VariablesAccessInfo):
177 '''
178 Get all variable access information. This specialisation is required
179 to query the SymbolTable associated with a Scoping node and ensure
180 that any Symbols appearing in precision specifications, array shapes or
181 initialisation expressions are captured.
182
183 :param var_accesses: VariablesAccessInfo instance that stores the
184 information about variable accesses.
185
186 '''
187 def _get_accesses(dtype: DataType, info: VariablesAccessInfo):
188 '''
189 Store information on any symbols referenced within the supplied
190 datatype.
191
192 :param dtype: the datatype to query.
193 :param info: the VariablesAccessInfo instance in which to store
194 information.
195 '''
196 if (hasattr(dtype, "precision") and isinstance(dtype.precision,
197 Symbol)):
198 # The use of a Symbol to specify precision does not constitute
199 # a read (since it is resolved at compile time).
200 access_info.add_access(
201 Signature(dtype.precision.name),
202 AccessType.TYPE_INFO, self)
203
204 if isinstance(dtype, DataTypeSymbol):
205 # The use of a DataTypeSymbol in a declaration is a compile-
206 # time access.
207 info.add_access(Signature(dtype.name),
208 AccessType.TYPE_INFO, self)
209 elif isinstance(dtype, StructureType):
210 for cmpt in sym.datatype.components.values():
211 # Recurse for members of a StructureType
212 _get_accesses(cmpt.datatype, info)
213 if cmpt.initial_value:
214 cmpt.initial_value.reference_accesses(info)
215 elif isinstance(dtype, ArrayType):
216 for dim in dtype.shape:
217 if isinstance(dim, ArrayType.ArrayBounds):
218 dim.lower.reference_accesses(access_info)
219 dim.upper.reference_accesses(access_info)
220 elif (isinstance(dtype, UnsupportedFortranType) and
221 dtype.partial_datatype):
222 # Recurse to examine partial datatype information.
223 _get_accesses(dtype.partial_datatype, info)
224
225 # Examine the datatypes and initial values of all DataSymbols.
226 for sym in self._symbol_table.datasymbols:
227 _get_accesses(sym.datatype, access_info)
228
229 if sym.initial_value:
230 sym.initial_value.reference_accesses(access_info)
231
232 # Examine the definition of each DataTypeSymbol.
233 for sym in self._symbol_table.datatypesymbols:
234 _get_accesses(sym.datatype, access_info)
235
236 super().reference_accesses(access_info)

238 def replace_symbols_using(self, table):
239 '''
240 Update any Symbols referenced by this Node (and its descendants) with
241 those in the supplied table with matching names. If there is no match
242 for a given Symbol then it is left unchanged.
243
244 Since this is a ScopingNode, it is associated with a symbol table.
245 Therefore, if the supplied table is the one for the scope containing
246 this node (if any), the one passed to the child nodes is updated to be
247 the one associated with this node.
248
249 :param table: the symbol table in which to look up replacement symbols.
250 :type table: :py:class:`psyclone.psyir.symbols.SymbolTable`
251
252 '''
253 next_table = table
254 if self.parentparentparent:
255 try:
256 # If this node is not within a scope we get a SymbolError
257 if table is self.parentparentparent.scope.symbol_table:
258 next_table = self.symbol_table
259 except SymbolError:
260 pass
261
262 for child in self.childrenchildrenchildrenchildren:
263 child.replace_symbols_using(next_table)
