98def load_script(
99 script_name: str, function_name: str = "trans",
100 is_optional: bool = False
101) -> Tuple[Callable, List[str], Union[bool, List[str]]]:
102 ''' Loads the specified script containing a psyclone recipe. We also
103 prepend the script path to the sys.path, so that the script itself and
104 any imports that it has from the same directory can be found.
105
106 :param script_name: name of the script to load.
107 :param function_name: the name of the function to call in the script.
108 :param is_optional: whether the function is optional or not. Defaults to
109 False.
110
111 :returns: callable recipe, list of files to skip, whether to resolve
112 modules (or which ones).
113
114 :raises IOError: if the file is not found.
115 :raises GenerationError: if the file does not have .py extension.
116 :raises GenerationError: if the named function does not exist or can not
117 be called.
118
119 '''
120 filepath, filename = os.path.split(script_name)
121 module_name, fileext = os.path.splitext(filename)
122 # the file must either be:
123 # a) at the given path or, given no path, in the current directory; or
124 # b) given no path, in the system path
125 if not (os.path.isfile(script_name) or
126 not filepath and any(os.path.isfile(os.path.join(p, filename))
127 for p in sys.path)):
128 raise GenerationError(
129 f"generator: script file '{script_name}' not found")
130 # the file must have the .py extension
131 if fileext != '.py':
132 raise GenerationError(
133 f"generator: expected the script file '{filename}' to have "
134 f"the '.py' extension")
135 # prepend file path - if none, the empty string equates to the current
136 # working directory - to the system path to guarantee we find the user
137 # provided module instead of a similarly named module that might
138 # already exist elsewhere in the system path
139 sys.path.insert(0, filepath)
140 recipe_module = importlib.import_module(module_name)
141
142 if hasattr(recipe_module, "FILES_TO_SKIP"):
143 files_to_skip = recipe_module.FILES_TO_SKIP
144 else:
145 files_to_skip = []
146
147 if hasattr(recipe_module, "RESOLVE_IMPORTS"):
148 imports_to_resolve = recipe_module.RESOLVE_IMPORTS
149 else:
150 imports_to_resolve = []
151
152 if hasattr(recipe_module, function_name):
153 transformation_recipe = getattr(recipe_module, function_name)
154 if callable(transformation_recipe):
155 # Everything is good, return recipe and files_to_skip
156 return transformation_recipe, files_to_skip, imports_to_resolve
157 elif is_optional:
158 return None, files_to_skip, imports_to_resolve
159 raise GenerationError(
160 f"generator: attempted to use specified PSyclone "
161 f"transformation module '{module_name}' but it does not "
162 f"contain a callable '{function_name}' function")
163

165def generate(filename, api="", kernel_paths=None, script_name=None,
166 line_length=False,
167 distributed_memory=None,
168 kern_out_path="",
169 kern_naming="multiple"):
170 # pylint: disable=too-many-arguments, too-many-statements
171 # pylint: disable=too-many-branches, too-many-locals
172 '''Takes a PSyclone algorithm specification as input and outputs the
173 associated generated algorithm and psy-layer codes suitable for
174 compiling with the specified kernel(s) and support
175 infrastructure.
176
177 :param str filename: the file containing the algorithm specification.
178 :param str api: the name of the API to use. Defaults to empty string.
179 :param kernel_paths: the directories from which to recursively
180 search for the files containing the kernel source (if
181 different from the location of the algorithm specification).
182 Defaults to None.
183 :type kernel_paths: Optional[List[str]]
184 :param str script_name: a script file that can apply optimisations
185 to the PSy layer (can be a path to a file or a filename that
186 relies on the PYTHONPATH to find the module). Defaults to None.
187 :param bool line_length: a logical flag specifying whether we care
188 about line lengths being longer than 132 characters. If so,
189 the input (algorithm and kernel) code is checked to make sure
190 that it conforms. The default is False.
191 :param bool distributed_memory: a logical flag specifying whether
192 to generate distributed memory code. The default is set in the
193 'config.py' file.
194 :param str kern_out_path: directory to which to write transformed
195 kernel code. Defaults to empty string.
196 :param bool kern_naming: the scheme to use when re-naming transformed
197 kernels. Defaults to "multiple".
198 :return: 2-tuple containing the fparser1 AST for the algorithm code and
199 the fparser1 AST or a string (for NEMO) of the psy code.
200 :rtype: Tuple[:py:class:`fparser.one.block_statements.BeginSource`,
201 :py:class:`fparser.one.block_statements.Module`] |
202 Tuple[:py:class:`fparser.one.block_statements.BeginSource`, str]
203
204 :raises GenerationError: if an invalid API is specified.
205 :raises GenerationError: if an invalid kernel-renaming scheme is specified.
206 :raises GenerationError: if there is an error raising the PSyIR to
207 domain-specific PSyIR.
208 :raises GenerationError: if a kernel functor is not named in a use
209 statement.
210 :raises IOError: if the filename or search path do not exist.
211 :raises NoInvokesError: if no invokes are found in the algorithm file.
212
213 For example:
214
215 >>> from psyclone.generator import generate
216 >>> alg, psy = generate("algspec.f90")
217 >>> alg, psy = generate("algspec.f90", kernel_paths=["src/kernels"])
218 >>> alg, psy = generate("algspec.f90", script_name="optimise.py")
219 >>> alg, psy = generate("algspec.f90", line_length=True)
220 >>> alg, psy = generate("algspec.f90", distributed_memory=False)
221
222 '''
223 if kernel_paths is None:
224 kernel_paths = []
225
226 if distributed_memory is None:
227 distributed_memory = Config.get().distributed_memory
228
229 if api not in Config.get().supported_apis:
230 raise GenerationError(
231 f"generate: Unsupported API '{api}' specified. Supported "
232 f"types are {Config.get().curated_api_list}.")
233
234 # Store Kernel-output options in our Configuration object
235 Config.get().kernel_output_dir = kern_out_path
236 try:
237 Config.get().kernel_naming = kern_naming
238 except ValueError as verr:
239 raise GenerationError(
240 f"Invalid kernel-renaming scheme supplied: {str(verr)}") from verr
241
242 if not os.path.isfile(filename):
243 raise IOError(f"File '{filename}' not found")
244 for kernel_path in kernel_paths:
245 if not os.access(kernel_path, os.R_OK):
246 raise IOError(
247 f"Kernel search path '{kernel_path}' not found")
248
249 # TODO #2011: investigate if kernel search path and module manager
250 # can be combined.
251 ModuleManager.get().add_search_path(kernel_paths)
252
253 ast, invoke_info = parse(filename, api=api, invoke_name="invoke",
254 kernel_paths=kernel_paths,
255 line_length=line_length)
256
257 if api in LFRIC_API_NAMES and not LFRIC_TESTING:
258 psy = PSyFactory(api, distributed_memory=distributed_memory)\
259 .create(invoke_info)
260 if script_name is not None:
261 # Apply provided recipe to PSyIR
262 recipe, _, _ = load_script(script_name)
263 recipe(psy.container.root)
264 alg_gen = None
265
266 elif api in GOCEAN_API_NAMES or (api in LFRIC_API_NAMES and LFRIC_TESTING):
267 # Create language-level PSyIR from the Algorithm file
268 reader = FortranReader()
269 if api in LFRIC_API_NAMES:
270 # avoid undeclared builtin errors in PSyIR by adding a
271 # wildcard use statement.
272 fp2_tree = parse_fp2(filename)
273 # Choose a module name that is invalid Fortran so that it
274 # does not clash with any existing names in the algorithm
275 # layer.
276 builtins_module_name = "_psyclone_builtins"
277 add_builtins_use(fp2_tree, builtins_module_name)
278 psyir = Fparser2Reader().generate_psyir(fp2_tree)
279 # Check that there is only one module/program per file.
280 check_psyir(psyir, filename)
281 else:
282 psyir = reader.psyir_from_file(filename)
283
284 # Raise to Algorithm PSyIR
285 if api in GOCEAN_API_NAMES:
286 alg_trans = AlgTrans()
287 else: # api in LFRIC_API_NAMES
288 alg_trans = LFRicAlgTrans()
289 try:
290 alg_trans.apply(psyir)
291 except TransformationError as info:
292 raise GenerationError(
293 f"In algorithm file '{filename}':\n{info.value}") from info
294
295 if not psyir.walk(AlgorithmInvokeCall):
296 raise NoInvokesError(
297 "Algorithm file contains no invoke() calls: refusing to "
298 "generate empty PSy code")
299
300 if script_name is not None:
301 # Call the optimisation script for algorithm optimisations
302 recipe, _, _ = load_script(script_name, "trans_alg",
303 is_optional=True)
304 if recipe:
305 recipe(psyir)
306
307 # For each kernel called from the algorithm layer
308 kernels = {}
309 for invoke in psyir.walk(AlgorithmInvokeCall):
310 kernels[id(invoke)] = {}
311 for kern in invoke.walk(KernelFunctor):
312 if isinstance(kern, LFRicBuiltinFunctor):
313 # Skip builtins
314 continue
315 if isinstance(kern.symbol.interface, UnresolvedInterface):
316 # This kernel functor is not specified in a use statement.
317 # Find all container symbols that are in scope.
318 st_ref = kern.scope.symbol_table
319 container_symbols = [
320 symbol.name for symbol in st_ref.containersymbols]
321 while st_ref.parent_symbol_table():
322 st_ref = st_ref.parent_symbol_table()
323 container_symbols += [
324 symbol.name for symbol in st_ref.containersymbols]
325 message = (
326 f"Kernel functor '{kern.name}' in routine "
327 f"'{kern.ancestor(Routine).name}' from algorithm file "
328 f"'{filename}' must be named in a use statement "
329 f"(found {container_symbols})")
330 if api in LFRIC_API_NAMES:
331 message += (
332 f" or be a recognised built-in (one of "
333 f"{list(BUILTIN_MAP.keys())})")
334 message += "."
335 raise GenerationError(message)
336 container_symbol = kern.symbol.interface.container_symbol
337
338 # Find the kernel file containing the container
339 filepath = get_kernel_filepath(
340 container_symbol.name, kernel_paths, filename)
341
342 try:
343 # Create language-level PSyIR from the kernel file
344 kernel_psyir = reader.psyir_from_file(filepath)
345 except InternalError as info:
346 print(f"In kernel file '{filepath}':\n{str(info.value)}",
347 file=sys.stderr)
348 sys.exit(1)
349
350 # Raise to Kernel PSyIR
351 if api in GOCEAN_API_NAMES:
352 kern_trans = RaisePSyIR2GOceanKernTrans(kern.symbol.name)
353 kern_trans.apply(kernel_psyir)
354 else: # api in LFRIC_API_NAMES
355 kern_trans = RaisePSyIR2LFRicKernTrans()
356 kern_trans.apply(
357 kernel_psyir,
358 options={"metadata_name": kern.symbol.name})
359
360 kernels[id(invoke)][id(kern)] = kernel_psyir
361
362 # Transform 'invoke' calls into calls to PSy-layer subroutines
363 if api in GOCEAN_API_NAMES:
364 invoke_trans = GOceanAlgInvoke2PSyCallTrans()
365 else: # api in LFRIC_API_NAMES
366 invoke_trans = LFRicAlgInvoke2PSyCallTrans()
367 for invoke in psyir.walk(AlgorithmInvokeCall):
368 invoke_trans.apply(
369 invoke, options={"kernels": kernels[id(invoke)]})
370 if api in LFRIC_API_NAMES:
371 # Remove any use statements that were temporarily added to
372 # avoid the PSyIR complaining about undeclared builtin
373 # names.
374 for node in psyir.walk((Routine, Container)):
375 symbol_table = node.symbol_table
376 if builtins_module_name in symbol_table:
377 symbol = symbol_table.lookup(builtins_module_name)
378 symbol_table.remove(symbol)
379
380 # Create Fortran from Algorithm PSyIR
381 writer = FortranWriter()
382 alg_gen = writer(psyir)
383
384 # Create the PSy-layer
385 # TODO: issue #1629 replace invoke_info with alg and kern psyir
386 psy = PSyFactory(api, distributed_memory=distributed_memory)\
387 .create(invoke_info)
388
389 if script_name is not None:
390 # Call the optimisation script for psy-layer optimisations
391 recipe, _, _ = load_script(script_name)
392 recipe(psy.container.root)
393
394 # TODO issue #1618 remove Alg class and tests from PSyclone
395 if api in LFRIC_API_NAMES and not LFRIC_TESTING:
396 alg_gen = Alg(ast, psy).gen
397
398 # Add profiling nodes to schedule if automatic profiling has
399 # been requested.
400 for invoke in psy.invokes.invoke_list:
401 Profiler.add_profile_nodes(invoke.schedule, Loop)
402
403 return alg_gen, psy.gen

406def main(arguments):
407 '''
408 Parses and checks the command line arguments, calls the generate
409 function if all is well, catches any errors and outputs the
410 results.
411
412 :param arguments: the list of command-line arguments that PSyclone has
413 been invoked with.
414 :type arguments: List[str]
415
416 '''
417 # pylint: disable=too-many-statements,too-many-branches
418
419 # Make sure we have the supported APIs defined in the Config singleton,
420 # but postpone loading the config file till the command line was parsed
421 # in case that the user specifies a different config file.
422 Config.get(do_not_load_file=True)
423
424 parser = argparse.ArgumentParser(
425 description='Transform a file using the PSyclone source-to-source '
426 'Fortran compiler')
427 # Common options
428 parser.add_argument('filename', help='input source code')
429 parser.add_argument(
430 '-v', '--version', action='version',
431 version=f'PSyclone version: {__VERSION__}',
432 help='display version information')
433 parser.add_argument('-c', '--config', help='config file with '
434 'PSyclone specific options')
435 parser.add_argument('-s', '--script', help='filename of a PSyclone'
436 ' optimisation recipe')
437 parser.add_argument(
438 '-I', '--include', default=[], action="append",
439 help='path to Fortran INCLUDE or module files')
440 parser.add_argument(
441 '-l', '--limit', dest='limit', default='off',
442 choices=['off', 'all', 'output'],
443 help="limit the Fortran line length to 132 characters (default "
444 "'%(default)s'). Use 'all' to apply limit to both input and "
445 "output Fortran. Use 'output' to apply line-length limit to output "
446 "Fortran only.")
447 parser.add_argument(
448 '-p', '--profile', action="append", choices=Profiler.SUPPORTED_OPTIONS,
449 help="add profiling hooks for 'kernels', 'invokes' or 'routines'")
450 parser.add_argument(
451 '--backend', dest='backend',
452 choices=['enable-validation', 'disable-validation'],
453 help=("options to control the PSyIR backend used for code generation. "
454 "Use 'disable-validation' to disable the validation checks that "
455 "are performed by default."))
456
457 # Code-transformation mode flags
458 parser.add_argument('-o', metavar='OUTPUT_FILE',
459 help='(code-transformation mode) output file')
460
461 # PSyKAl mode flags
462 parser.add_argument('-api', '--psykal-dsl', metavar='DSL',
463 help=f'whether to use a PSyKAl DSL (one of '
464 f'{Config.get().curated_api_list})')
465 parser.add_argument('-oalg', metavar='OUTPUT_ALGORITHM_FILE',
466 help='(psykal mode) filename of transformed '
467 'algorithm code')
468 parser.add_argument('-opsy', metavar='OUTPUT_PSY_FILE',
469 help='(psykal mode) filename of generated '
470 'PSy-layer code')
471 parser.add_argument('-okern', metavar='OUTPUT_KERNEL_PATH',
472 help='(psykal mode) directory in which to put '
473 'transformed kernels, default is the current working'
474 ' directory')
475 parser.add_argument(
476 '-d', '--directory', default=[], action="append", help='(psykal mode) '
477 'path to a root directory structure containing kernel source code. '
478 'Multiple roots can be specified by using multiple -d arguments.')
479 parser.add_argument(
480 '-dm', '--dist_mem', dest='dist_mem', action='store_true',
481 help='(psykal mode) generate distributed memory code')
482 parser.add_argument(
483 '-nodm', '--no_dist_mem', dest='dist_mem', action='store_false',
484 help='(psykal mode) do not generate distributed memory code')
485 parser.add_argument(
486 '--kernel-renaming', default="multiple",
487 choices=VALID_KERNEL_NAMING_SCHEMES,
488 help='(psykal mode) naming scheme to use when re-naming transformed'
489 ' kernels')
490 parser.set_defaults(dist_mem=Config.get().distributed_memory)
491
492 args = parser.parse_args(arguments)
493
494 # Validate that the given arguments are for the right operation mode
495 if not args.psykal_dsl:
496 if (args.oalg or args.opsy or args.okern or args.directory):
497 print(f"When using the code-transformation mode (with no -api or "
498 f"--psykal-dsl flags), the psykal-mode arguments must not be"
499 f" present in the command, but found {arguments}")
500 sys.exit(1)
501 else:
502 if args.o:
503 print("The '-o' flag is not valid when using the psykal mode "
504 "(-api/--psykal-dsl flag), use the -oalg, -opsy, -okern to "
505 "specify the output destination of each psykal layer.")
506 sys.exit(1)
507
508 # If no config file name is specified, args.config is none
509 # and config will load the default config file.
510 Config.get().load(args.config)
511
512 # Check whether a PSyKAl API has been specified.
513 if args.psykal_dsl is None:
514 api = ""
515 elif args.psykal_dsl not in Config.get().supported_apis:
516 print(f"Unsupported PSyKAL DSL / API '{args.psykal_dsl}' specified. "
517 f"Supported DSLs are {Config.get().curated_api_list}.",
518 file=sys.stderr)
519 sys.exit(1)
520 else:
521 # There is a valid API specified on the command line. Set it
522 # as API in the config object as well.
523 api = args.psykal_dsl
524 Config.get().api = api
525
526 # Record any profiling options.
527 if args.profile:
528 try:
529 Profiler.set_options(args.profile, api)
530 except ValueError as err:
531 print(f"Invalid profiling option: {err}", file=sys.stderr)
532 sys.exit(1)
533 if args.backend:
534 # A command-line flag overrides the setting in the Config file (if
535 # any).
536 Config.get().backend_checks_enabled = (
537 str(args.backend) == "enable-validation")
538
539 # The Configuration manager checks that the supplied path(s) is/are
540 # valid so protect with a try
541 try:
542 if args.include:
543 Config.get().include_paths = args.include
544 else:
545 # Default is to instruct fparser2 to look in the directory
546 # containing the file being parsed
547 Config.get().include_paths = ["./"]
548 except ConfigurationError as err:
549 print(str(err), file=sys.stderr)
550 sys.exit(1)
551
552 if not args.psykal_dsl:
553 code_transformation_mode(input_file=args.filename,
554 recipe_file=args.script,
555 output_file=args.o,
556 line_length=args.limit)
557 else:
558 # PSyKAl-DSL mode
559
560 # If an output directory has been specified for transformed kernels
561 # then check that it is valid
562 if args.okern:
563 if not os.path.exists(args.okern):
564 print(f"Specified kernel output directory ({args.okern}) does "
565 f"not exist.", file=sys.stderr)
566 sys.exit(1)
567 if not os.access(args.okern, os.W_OK):
568 print(f"Cannot write to specified kernel output directory "
569 f"({args.okern}).", file=sys.stderr)
570 sys.exit(1)
571 kern_out_path = args.okern
572 else:
573 # We write any transformed kernels to the current working directory
574 kern_out_path = os.getcwd()
575
576 try:
577 alg, psy = generate(args.filename, api=api,
578 kernel_paths=args.directory,
579 script_name=args.script,
580 line_length=(args.limit == 'all'),
581 distributed_memory=args.dist_mem,
582 kern_out_path=kern_out_path,
583 kern_naming=args.kernel_renaming)
584 except NoInvokesError:
585 _, exc_value, _ = sys.exc_info()
586 print(f"Warning: {exc_value}")
587 # no invoke calls were found in the algorithm file so we do
588 # not need to process it, or generate any psy layer code, so
589 # output the original algorithm file and set the psy file to
590 # be empty
591 with open(args.filename, encoding="utf8") as alg_file:
592 alg = alg_file.read()
593 psy = ""
594 except (OSError, IOError, ParseError, GenerationError,
595 RuntimeError):
596 _, exc_value, _ = sys.exc_info()
597 print(exc_value, file=sys.stderr)
598 sys.exit(1)
599 except Exception: # pylint: disable=broad-except
600 print("Error, unexpected exception, please report to the authors:",
601 file=sys.stderr)
602 traceback.print_exception(*sys.exc_info(), file=sys.stderr)
603 sys.exit(1)
604 if args.limit != 'off':
605 # Limit the line length of the output Fortran to ensure it conforms
606 # to the 132 characters mandated by the standard.
607 fll = FortLineLength()
608 psy_str = fll.process(str(psy))
609 alg_str = fll.process(str(alg))
610 else:
611 psy_str = str(psy)
612 alg_str = str(alg)
613 if args.oalg is not None:
614 with open(args.oalg, mode='w', encoding="utf8") as alg_file:
615 alg_file.write(alg_str)
616 else:
617 print(f"Transformed algorithm code:\n{alg_str}")
618
619 if not psy_str:
620 # empty file so do not output anything
621 pass
622 elif args.opsy is not None:
623 with open(args.opsy, mode='w', encoding="utf8") as psy_file:

692def code_transformation_mode(input_file, recipe_file, output_file,
693 line_length="off"):
694 ''' Process the input_file with the recipe_file instructions and
695 store it in the output_file.
696
697 Note: there is some duplicated logic in the PSyKAl path, we could attempt
698 to merge them when adopting the LFRIC_TESTING PATH and removing the
699 previous way.
700
701 :param input_file: the given input file.
702 :type input_file: str | os.PathLike
703 :param recipe_file: the given transformation recipe file.
704 :type input_file: Optional[str | os.PathLike]
705 :param output_file: the output file where to store the resulting code.
706 :type output_file: Optional[str | os.PathLike]
707 :param str line_length: set to "output" to break the output into lines
708 of 123 chars, and to "all", to additionally check the input code.
709
710 '''
711 # Load recipe file
712 if recipe_file:
713 trans_recipe, files_to_skip, resolve_mods = load_script(recipe_file)
714 else:
715 trans_recipe, files_to_skip, resolve_mods = (None, [], False)
716
717 _, filename = os.path.split(input_file)
718 if filename not in files_to_skip:
719 # If line_length "all" is provided, check the input source
720 fll = FortLineLength()
721 if line_length == "all":
722 with open(input_file, "r", encoding='utf8') as myfile:
723 code_str = myfile.read()
724 if fll.long_lines(code_str):
725 print(f"'{filename}' does not conform to the specified "
726 f"{fll.length} line-length limit. Either fix the "
727 f"file or change the '-l/--limit' argument on the "
728 f"PSyclone command line.", file=sys.stderr)
729 sys.exit(1)
730
731 # Parse file
732 psyir = FortranReader(resolve_modules=resolve_mods)\
733 .psyir_from_file(input_file)
734
735 # Modify file
736 if trans_recipe:
737 trans_recipe(psyir)
738
739 # Add profiling if automatic profiling has been requested
740 for routine in psyir.walk(Routine):
741 Profiler.add_profile_nodes(routine, Loop)
742
743 # Generate Fortran (We can disable the backend copy because at this
744 # point we also drop the PSyIR and we don't need to guarantee that
745 # is left unmodified)
746 output = FortranWriter(disable_copy=True)(psyir)
747 # Fix line_length if requested
748 if line_length in ("output", "all"):
749 output = fll.process(output)
750
751 if output_file:
752 with open(output_file, mode='w', encoding="utf8") as ofile:
753 ofile.write(output)
754 else:
755 print(output, file=sys.stdout)
756 else:
757 # Skip parsing and transformation and copy contents of file directly
758 if output_file:
759 shutil.copyfile(input_file, output_file)
760 else:
761 print(f"File '{input_file}' skipped because it is listed in "
762 "FILES_TO_SKIP.", file=sys.stdout)
