54# pylint: disable=too-many-public-methods, no-member
55class KernelInterface(ArgOrdering):
56 '''Create the kernel arguments for the supplied kernel as specified by
57 the associated kernel metadata and the kernel ordering rules
58 encoded in the ArgOrdering base class as method callbacks.
59
60 A PSyIR symbol table is created containing appropriate LFRic PSyIR
61 symbols to specify the arguments. If an argument is an array with
62 one or more dimension sizes specified by another argument, then
63 the associated array symbol is created so that it references the
64 appropriate symbol.
65
66 Related arguments - e.g. a field has an associated dofmap - are
67 not directly connected, they must be inferred from the function
68 space names. It is not yet clear whether this would be useful or
69 not.
70
71 TODO #928: This class should replace the current kernel stub generation
72 code when all of its methods are implemented.
73
74 :param kern: the kernel for which to create arguments.
75 :type kern: :py:class:`psyclone.domain.lfric.LFRicKern`
76
77 '''
78 #: Mapping from a generic PSyIR datatype to the equivalent
79 #: LFRic-specific field datasymbol.
80 field_mapping = {
81 "integer": "IntegerFieldDataSymbol",
82 "real": "RealFieldDataSymbol",
83 "logical": "LogicalFieldDataSymbol"}
84 #: Mapping from a generic PSyIR datatype to the equivalent
85 #: LFRic-specific vector field datasymbol.
86 vector_field_mapping = {
87 "integer": "IntegerVectorFieldDataSymbol",
88 "real": "RealVectorFieldDataSymbol",
89 "logical": "LogicalVectorFieldDataSymbol"}
90 #: Mapping from the LFRic metadata description of quadrature to the
91 #: associated LFRic-specific basis function datasymbol.
92 basis_mapping = {
93 "gh_quadrature_xyoz": "BasisFunctionQrXyozDataSymbol",
94 "gh_quadrature_face": "BasisFunctionQrFaceDataSymbol",
95 "gh_quadrature_edge": "BasisFunctionQrEdgeDataSymbol"}
96 #: Mapping from the LFRic metadata description of quadrature to the
97 #: associated LFRic-specific differential basis function datasymbol.
98 diff_basis_mapping = {
99 "gh_quadrature_xyoz": "DiffBasisFunctionQrXyozDataSymbol",
100 "gh_quadrature_face": "DiffBasisFunctionQrFaceDataSymbol",
101 "gh_quadrature_edge": "DiffBasisFunctionQrEdgeDataSymbol"}
102 _read_access = ArgumentInterface(ArgumentInterface.Access.READ)
103
104 def __init__(self, kern):
105 super().__init__(kern)
106 # We need a brand new symbol table, not a reference to the one for
107 # the Schedule containing this kernel call (which is what the
108 # ArgOrdering constructor defaults to). This is so that we can
109 # specify the correct interface for those symbols passed
110 # as arguments.
111 self._forced_symtab_forced_symtab = LFRicSymbolTable()

113 def generate(self, var_accesses=None):
114 '''Call the generate base class then add the argument list as it can't
115 be appended as we go along.
116
117 :param var_accesses: an unused optional argument that stores \
118 information about variable accesses.
119 :type var_accesses: :\
120 py:class:`psyclone.core.VariablesAccessInfo`
121
122 '''
123 super().generate(var_accesses=var_accesses)
124 # Set the argument list for the symbol table. This is done at
125 # the end after incrementally adding symbols to the _args
126 # list, as it is not possible to incrementally add symbols to
127 # the symbol table argument list.
128 self._symtab.specify_argument_list(self._arglist_arglist)
129 # While data dependence analysis does not use the symbol
130 # table, see #845, we have to provide variable information
131 # separately. This is done by using the base class append()
132 # method. However, this method also adds the variable names to the
133 # internal _arglist list which we do not want as we have
134 # already added our symbols there. Therefore we need to remove
135 # them afterwards.
136 # Map from symbol table accesses to dependence analysis accesses.
137 mapping = {ArgumentInterface.Access.READ: AccessType.READ,
138 ArgumentInterface.Access.READWRITE: AccessType.READWRITE,
139 ArgumentInterface.Access.WRITE: AccessType.WRITE}
140 len_arglist = len(self._arglist_arglist)
141 for symbol in self._symtab.symbols:
142 self.append(symbol.name, var_accesses,
143 mode=mapping[symbol.interface.access])
144 self._arglist_arglist = self._arglist_arglist[:len_arglist]

216 def field_vector(self, argvect, var_accesses=None):
217 '''Create LFRic field vector arguments and add them to the symbol
218 table and argument list. Also declare the associated "undf"
219 symbol if it has not already been declared so that it can be
220 used to dimension the field vector arguments.
221
222 :param argvect: the field vector to add.
223 :type argvect: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
224 :param var_accesses: an unused optional argument that stores \
225 information about variable accesses.
226 :type var_accesses: :\
227 py:class:`psyclone.core.VariablesAccessInfo`
228
229 :raises NotImplementedError: if the datatype of the vector \
230 field is not supported.
231
232 '''
233 fs_name = argvect.function_space.orig_name
234 undf_symbol = self._symtab.find_or_create_tag(
235 f"undf_{fs_name}", fs=fs_name,
236 symbol_type=LFRicTypes("NumberOfUniqueDofsDataSymbol"),
237 interface=self._read_access)
238
239 interface = ArgumentInterface(INTENT_MAPPING[argvect.intent])
240 try:
241 type_name = self.vector_field_mapping[argvect.intrinsic_type]
242 field_class = LFRicTypes(type_name)
243 except KeyError as info:
244 message = (f"kernel interface does not support a vector field of "
245 f"type '{argvect.intrinsic_type}'.")
246 raise NotImplementedError(message) from info
247 for idx in range(argvect.vector_size):
248 tag = f"{argvect.name}_v{idx}"
249 field_data_symbol = self._symtab.find_or_create_tag(
250 tag, symbol_type=field_class, dims=[Reference(undf_symbol)],
251 fs=fs_name, interface=interface)
252
253 self._arg_index_to_metadata_index[len(self._arglist_arglist)] = (
254 argvect.metadata_index)
255 self._arglist_arglist.append(field_data_symbol)

257 def field(self, arg, var_accesses=None):
258 '''Create an LFRic field argument and add it to the symbol table and
259 argument list. Also declare the associated "undf" symbol if it
260 has not already been declared so that it can be used to
261 dimension the field argument.
262
263 :param arg: the field to add.
264 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
265 :param var_accesses: an unused optional argument that stores \
266 information about variable accesses.
267 :type var_accesses: :\
268 py:class:`psyclone.core.VariablesAccessInfo`
269
270 :raises NotImplementedError: if the datatype of the field is \
271 not supported.
272
273 '''
274 fs_name = arg.function_space.orig_name
275
276 undf_symbol = self._symtab.find_or_create_tag(
277 f"undf_{fs_name}",
278 symbol_type=LFRicTypes("NumberOfUniqueDofsDataSymbol"),
279 fs=fs_name, interface=self._read_access)
280
281 try:
282 type_name = self.field_mapping[arg.intrinsic_type]
283 field_class = LFRicTypes(type_name)
284 except KeyError as info:
285 message = (f"kernel interface does not support a field of type "
286 f"'{arg.intrinsic_type}'.")
287 raise NotImplementedError(message) from info
288 field_data_symbol = self._symtab.find_or_create_tag(
289 arg.name, interface=ArgumentInterface(INTENT_MAPPING[arg.intent]),
290 symbol_type=field_class, dims=[Reference(undf_symbol)], fs=fs_name)
291
292 self._arg_index_to_metadata_index[len(self._arglist_arglist)] = (
293 arg.metadata_index)
294 self._arglist_arglist.append(field_data_symbol)

343 def operator(self, arg, var_accesses=None):
344 '''Create an LFRic operator argument and an ncells argument and add
345 them to the symbol table and argument list. Also declare the
346 associated 'fs_from', 'fs_to' symbols if they have not already
347 been declared so that they can be used to dimension the
348 operator symbol (as well as ncells).
349
350 :param arg: the operator to add.
351 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
352 :param var_accesses: an unused optional argument that stores
353 information about variable accesses.
354 :type var_accesses: :py:class:`psyclone.core.VariablesAccessInfo`
355
356 :raises NotImplementedError: if the datatype of the field is
357 not supported.
358
359 '''
360 fs_from_name = arg.function_space_from.orig_name
361
362 ndf_symbol_from = self._symtab.find_or_create_tag(
363 f"ndf_{fs_from_name}", fs=fs_from_name,
364 symbol_type=LFRicTypes("NumberOfDofsDataSymbol"),
365 interface=self._read_access)
366 fs_to_name = arg.function_space_to.orig_name
367 ndf_symbol_to = self._symtab.find_or_create_tag(
368 f"ndf_{fs_to_name}", fs=fs_to_name,
369 symbol_type=LFRicTypes("NumberOfDofsDataSymbol"),
370 interface=self._read_access)
371
372 # We may already have a symbol for this argument as we add it for each
373 # operator (TODO #2074).
374 ncells = self._symtab.find_or_create(
375 "ncell_3d",
376 symbol_type=LFRicTypes("NumberOfCellsDataSymbol"),
377 interface=self._read_access)
378 self._arglist_arglist.append(ncells)
379
380 op_arg_symbol = self._symtab.find_or_create_tag(
381 arg.name, symbol_type=LFRicTypes("OperatorDataSymbol"),
382 dims=[Reference(ndf_symbol_from), Reference(ndf_symbol_to),
383 Reference(ncells)],
384 fs_from=fs_from_name, fs_to=fs_to_name,
385 interface=ArgumentInterface(INTENT_MAPPING[arg.intent]))
386
387 self._arg_index_to_metadata_index[len(self._arglist_arglist)] = (
388 arg.metadata_index)
389 self._arglist_arglist.append(op_arg_symbol)

586 def field_bcs_kernel(self, function_space, var_accesses=None):
587 '''Create the boundary-dofs mask argument required for the
588 enforce_bc_code kernel. Adds it to the symbol table and the argument
589 list.
590
591 :param function_space: the function space for this boundary condition.
592 :type function_space: :py:class:`psyclone.domain.lfric.FunctionSpace`
593 :param var_accesses: an unused optional argument that stores
594 information about variable accesses.
595 :type var_accesses: :py:class:`psyclone.core.VariablesAccessInfo`
596
597 :raises InternalError: if the kernel does not have a single field as
598 argument.
599 :raises InternalError: if the field argument is not on the
600 'ANY_SPACE_1' function space.
601
602 '''
603 # Sanity check - expect the enforce_bc_code to have a single argument
604 # that is a field.
605 if (len(self._kern.arguments.args) != 1 or
606 not self._kern.arguments.args[0].is_field):
607 const = LFRicConstants()
608 raise InternalError(
609 f"Kernel '{self._kern.name}' applies boundary conditions to a "
610 f"field and therefore should have a single, field argument "
611 f"(one of {const.VALID_FIELD_NAMES}) but got "

674 def quad_rule(self, var_accesses=None):
675 '''Create LFRic arguments associated with the required quadrature, if
676 they do not already exist, and add them to the symbol table
677 and argument list. The arguments depend on the type of
678 quadrature requested.
679
680 :param var_accesses: an unused optional argument that stores \
681 information about variable accesses.
682 :type var_accesses: :\

684
685 :raises InternalError: if an unsupported quadrature shape is \
686 found.
687
688 '''
689 # The kernel captures all the required quadrature shapes
690 for shape in self._kern.qr_rules:

692 nqp_xy = self._symtab.find_or_create_tag(

704 weights_z = self._symtab.find_or_create_tag(
705 "weights_z",
706 symbol_type=LFRicTypes("QrWeightsInZDataSymbol"),
707 dims=[Reference(nqp_z)], interface=self._read_access)
708 self._arglist.extend([nqp_xy, nqp_z, weights_xy, weights_z])
709 elif shape == "gh_quadrature_face":
710 nfaces = self._symtab.find_or_create_tag(
711 "nfaces",
712 symbol_type=LFRicTypes("NumberOfFacesDataSymbol"),
713 interface=self._read_access)
714 nqp = self._symtab.find_or_create_tag(
715 "nqp_faces",
716 symbol_type=LFRicTypes(
717 "NumberOfQrPointsInFacesDataSymbol"),
718 interface=self._read_access)
719 weights = self._symtab.find_or_create_tag(
720 "weights_faces",
721 symbol_type=LFRicTypes("QrWeightsInFacesDataSymbol"),
722 dims=[Reference(nqp)], interface=self._read_access)
723 self._arglist.extend([nfaces, nqp, weights])
724 elif shape == "gh_quadrature_edge":
725 nedges = self._symtab.find_or_create_tag(
726 "nedges",
727 symbol_type=LFRicTypes("NumberOfEdgesDataSymbol"),
728 interface=self._read_access)
729 nqp = self._symtab.find_or_create_tag(
730 "nqp_edges",
731 symbol_type=LFRicTypes(
732 "NumberOfQrPointsInEdgesDataSymbol"),
733 interface=self._read_access)
734 weights = self._symtab.find_or_create_tag(
735 "weights_edges",
736 symbol_type=LFRicTypes("QrWeightsInEdgesDataSymbol"),
737 dims=[Reference(nqp)], interface=self._read_access)
738 self._arglist.extend([nedges, nqp, weights])

740 raise InternalError(f"Unsupported quadrature shape '{shape}' "
741 f"found in kernel_interface.")

743 def _create_basis(self, function_space, mapping, basis_name_func,
744 first_dim_value_func):
745 '''Internal utility to create an LFRic basis or differential basis
746 function argument specific to the particular quadrature that
747 is being used and add it to the symbol table and argument
748 list. Also declare the associated "ndf" symbol and any
749 quadrature-specific symbols if they have not already been
750 declared so that they can be used to dimension the basis or
751 differential basis symbol.
752
753 This utility function is used to avoid code replication as the
754 structure of a basis function is very similar to the structure
755 of a differential basis function.
756
757 :param function_space: the function space that this basis or \
758 differential basis function is on.
759 :type function_space: :py:class:`psyclone.domain.lfric.FunctionSpace`
760 :param dict mapping: a mapping from quadrature type to basis \
761 or differential basis class name.
762 :param method basis_name_func: a method that returns the name \
763 of the basis or differential basis function for the \
764 current function space.
765 :param function first_dim_value_func: a function that returns \
766 the size of the first dimension of the basis or \
767 differential basis function for the current function \
768 space.
769
770 :raises NotImplementedError: if an evaluator shape is found \
771 that is not a quadrature shape (currently just \
772 'gh_evaluator').
773 :raises InternalError: if the supplied evaluator shape is not \
774 recognised.
775
776 '''
777 # pylint: disable=too-many-locals
778 const = LFRicConstants()
779 for shape in self._kern.eval_shapes:
780 fs_name = function_space.orig_name
781 ndf_symbol = self._symtab.find_or_create_tag(
782 f"ndf_{fs_name}",
783 symbol_type=LFRicTypes("NumberOfDofsDataSymbol"),
784 fs=fs_name, interface=self._read_access)
785
786 # Create the qr tag by appending the last part of the shape
787 # name to "qr_".
788 quad_name = shape.split("_")[-1]
789 basis_tag = basis_name_func(qr_var="qr_"+quad_name)
790 if shape == "gh_quadrature_xyoz":
791 nqp_xy = self._symtab.find_or_create_tag(
792 "nqp_xy",
793 symbol_type=LFRicTypes("NumberOfQrPointsInXyDataSymbol"),
794 interface=self._read_access)
795 nqp_z = self._symtab.find_or_create_tag(
796 "nqp_z",
797 symbol_type=LFRicTypes("NumberOfQrPointsInZDataSymbol"),
798 interface=self._read_access)
799 type_name = mapping["gh_quadrature_xyoz"]
800 arg = LFRicTypes(type_name)(
801 basis_tag, [int(first_dim_value_func(function_space)),
802 Reference(ndf_symbol), Reference(nqp_xy),
803 Reference(nqp_z)],
804 fs_name, interface=self._read_access)
805 elif shape == "gh_quadrature_face":
806 nfaces = self._symtab.find_or_create_tag(
807 "nfaces",
808 symbol_type=LFRicTypes("NumberOfFacesDataSymbol"),
809 interface=self._read_access)
810 nqp = self._symtab.find_or_create_tag(
811 "nqp_faces",
812 symbol_type=LFRicTypes(
813 "NumberOfQrPointsInFacesDataSymbol"),
814 interface=self._read_access)
815 type_name = mapping["gh_quadrature_face"]
816 arg = LFRicTypes(type_name)(
817 basis_tag, [int(first_dim_value_func(function_space)),
818 Reference(ndf_symbol), Reference(nqp),
819 Reference(nfaces)],
820 fs_name, interface=self._read_access)
821 elif shape == "gh_quadrature_edge":
822 nedges = self._symtab.find_or_create_tag(
823 "nedges",
824 symbol_type=LFRicTypes("NumberOfEdgesDataSymbol"),
825 interface=self._read_access)
826 nqp = self._symtab.find_or_create_tag(
827 "nqp_edges",
828 symbol_type=LFRicTypes(
829 "NumberOfQrPointsInEdgesDataSymbol"),
830 interface=self._read_access)
831 type_name = mapping["gh_quadrature_edge"]
832 arg = LFRicTypes(type_name)(
833 basis_tag, [int(first_dim_value_func(function_space)),
834 Reference(ndf_symbol), Reference(nqp),
835 Reference(nedges)],
836 fs_name, interface=self._read_access)
837 elif shape in const.VALID_EVALUATOR_SHAPES:
838 # Need a (diff) basis array for each target space upon
839 # which the basis functions have been
840 # evaluated. _kern.eval_targets is a dict where the
841 # values are 2-tuples of (FunctionSpace, argument).
842 for _, _ in self._kern.eval_targets.items():
843 raise NotImplementedError(
844 "TODO #928: Evaluator shapes not implemented in "
845 "kernel_interface class.")
846 else:
847 raise InternalError(
848 f"Unrecognised quadrature or evaluator shape '{shape}'. "
849 f"Expected one of: {const.VALID_EVALUATOR_SHAPES}.")
850 self._symtab.add(arg)
851 self._arglist_arglist.append(arg)
