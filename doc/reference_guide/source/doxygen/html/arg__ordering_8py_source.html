57class ArgOrdering:
58 # pylint: disable=too-many-public-methods
59 # TODO: #845 Check that all implicit variables have the right type.
60 '''Base class capturing the arguments, type and ordering of data in
61 a Kernel call. This base class implements some functionality of a list
62 (extend and append functions), but not using list as a base class.
63 Reason is that many typical functions of a list make only sense to
64 be used after ``generate`` is called, which would then require a large
65 number of functions to be re-implemented. So instead the property
66 ``arglist`` checks that ``generate`` has been called and then provides
67 a list.
68
69 :param kern: the kernel call object to use.
70 :type kern: :py:class:`psyclone.domain.lfric.LFRicKern`
71
72 '''
73 def __init__(self, kern):
74 # TODO #2503: This reference will not survive some tree modifications
75 self._kern = kern
76 self._generate_called = False
77
78 # TODO #2503: This reference will not survive some tree modifications
79 self._forced_symtab = None
80
81 # TODO #1934 Completely remove the usage of strings, instead
82 # use the PSyIR representation.
83 self._arglist = []
84
85 # This stores the PSyIR representation of the arguments
86 self._psyir_arglist = []
87 self._arg_index_to_metadata_index = {}

90 def _symtab(self):
91 ''' Provide a reference to the associate Invoke SymbolTable, usually
92 following the `self._kernel.ancestor(InvokeSchedule)._symbol_table`
93 path unless a _forced_symtab has been provided.
94
95 If no symbol table is available it creates a temporary symbol table
96 for the operation to suceed but it will not be preserved.
97
98 Note: This could be improved by TODO #2503
99
100 :returns: the associate invoke symbol table.
101 :rtype: :py:class:`psyclone.psyir.symbols.SymbolTable`
102 '''
103 if self._forced_symtab:
104 return self._forced_symtab
105 elif self._kern and self._kern.ancestor(psyGen.InvokeSchedule):
106 return self._kern.ancestor(psyGen.InvokeSchedule).symbol_table
107 else:
108 return LFRicSymbolTable()

119 def append(self, var_name, var_accesses=None, var_access_name=None,
120 mode=AccessType.READ, metadata_posn=None):
121 # pylint: disable=too-many-arguments
122 '''Appends the specified variable name to the list of all arguments and
123 stores the mapping between the position of this actual argument and
124 the corresponding metadata entry. If var_accesses is given, it will
125 also record the access to the variable. The name of the variable
126 accessed can be overwritten by specifying var_access_name. By default
127 it is assumed that access mode is READ (which can be set with
128 ``mode``).
129
130 :param str var_name: the name of the variable.
131 :param var_accesses: optional class to store variable access \
132 information.
133 :type var_accesses: \
134 :py:class:`psyclone.core.VariablesAccessInfo`
135 :param str var_access_name: optional name of the variable for \
136 which access information is stored (used e.g. when the \
137 actual argument is field_proxy, but the access is to be \
138 recorded for field).
139 :param mode: optional access mode (defaults to READ).
140 :type mode: :py:class:`psyclone.core.access_type.AccessType`
141 :param int metadata_posn: the location of the corresponding entry in \
142 the list of arguments in the kernel metadata (if any).
143
144 '''
145 # Keep track of which metadata argument this actual argument
146 # corresponds to.
147 self._arg_index_to_metadata_index[len(self._arglist)] = metadata_posn
148
149 self._arglist.append(var_name)
150
151 if var_accesses is not None:
152 if var_access_name:
153 var_accesses.add_access(Signature(var_access_name), mode,
154 self._kern)
155 else:
156 var_accesses.add_access(Signature(var_name), mode,
157 self._kern)

159 def extend(self, list_var_name, var_accesses=None,
160 mode=AccessType.READ, list_metadata_posn=None):
161 '''Appends all variable names in the argument list to the list of
162 all arguments. If var_accesses is given, it will also record the
163 access to the variables. By default any access will be recorded as a
164 read-only access, but this can be changed (for all variables
165 included) using mode.
166
167 :param list_var_name: the list with name of the variables to append.
168 :type list_var_name: list of str.
169 :param var_accesses: optional class to store variable access \
170 information.
171 :type var_accesses: \
172 :py:class:`psyclone.core.VariablesAccessInfo`
173 :param mode: optional access mode (defaults to READ).
174 :type mode: Optional[:py:class:`psyclone.core.access_type.AccessType`]
175 :param Optional[List[int]] list_metadata_posn: list of metadata \
176 argument positions.
177
178 '''
179 for idx, var in enumerate(list_var_name):
180 if list_metadata_posn:
181 self.append(var, var_accesses=var_accesses, mode=mode,
182 metadata_posn=list_metadata_posn[idx])
183 else:
184 self.append(var, mode=mode, var_accesses=var_accesses)

186 def append_integer_reference(self, name, tag=None):
187 '''This function adds a reference to an integer variable to the list
188 of PSyIR nodes. If the symbol does not exist, it will be added to the
189 symbol table. If no tag is specified, is uses the name as tag. It also
190 returns the symbol.
191
192 :param str name: name of the integer variable to declare.
193 :param tag: optional tag of the integer variable to declare.
194 :type tag: Optional[str]
195
196 :returns: the symbol to which a reference was added.
197 :rtype: :py:class:`psyclone.psyir.symbols.Symbol`
198
199 '''
200 if tag is None:
201 tag = name
202 sym = self._symtab.find_or_create_integer_symbol(name, tag)
203 self.psyir_append(Reference(sym))
204 return sym

206 def get_array_reference(self, array_name, indices, intrinsic_type,
207 tag=None, symbol=None):
208 # pylint: disable=too-many-arguments
209 '''This function creates an array reference. If there is no symbol
210 with the given tag, a new array symbol will be defined using the given
211 intrinsic_type. If a symbol already exists but has no type, it will
212 be replaced.
213
214 :param str array_name: the name and tag of the array.
215 :param indices: the indices to be used in the PSyIR reference. It \
216 must either be ":", or a PSyIR node.
217 :type indices: List[Union[str, py:class:`psyclone.psyir.nodes.Node`]]
218 :param intrinsic_type: the intrinsic type of the array.
219 :type intrinsic_type: \
220 :py:class:`psyclone.psyir.symbols.datatypes.ScalarType.Intrinsic`
221 :param tag: optional tag for the symbol.
222 :type tag: Optional[str]
223 :param symbol: optional the symbol to use.
224 :type: Optional[:py:class:`psyclone.psyir.symbols.Symbol`]
225
226 :returns: a reference to the symbol used.
227 :rtype: :py:class:`psyclone.psyir.nodes.Reference`
228
229 '''
230 if not tag:
231 tag = array_name
232 if not symbol:
233 symbol = self._symtab.find_or_create_array(array_name,
234 len(indices),
235 intrinsic_type,
236 tag)
237 else:
238 if symbol.name != array_name:
239 raise InternalError(f"Specified symbol '{symbol.name}' has a "
240 f"different name than the specified array "
241 f"name '{array_name}'.")
242
243 # If all indices are specified as ":", just use the name itself
244 # to reproduce the current look of the code.
245 if indices == [":"]*len(indices):
246 ref = Reference(symbol)
247 else:
248 ref = ArrayReference.create(symbol, indices)
249 return ref

251 def append_array_reference(self, array_name, indices, intrinsic_type,
252 tag=None, symbol=None):
253 # pylint: disable=too-many-arguments
254 '''This function adds an array reference. If there is no symbol with
255 the given tag, a new array symbol will be defined using the given
256 intrinsic_type. If a symbol already exists but has no type, it will
257 be replaced. The created reference is added to the list of PSyIR
258 expressions, and the symbol is returned to the user.
259
260 :param str array_name: the name and tag of the array.
261 :param indices: the indices to be used in the PSyIR reference. It \
262 must either be ":", or a PSyIR node.
263 :type indices: List[Union[str, py:class:`psyclone.psyir.nodes.Node`]]
264 :param intrinsic_type: the intrinsic type of the array.
265 :type intrinsic_type: \
266 :py:class:`psyclone.psyir.symbols.datatypes.ScalarType.Intrinsic`
267 :param tag: optional tag for the symbol.
268 :type tag: Optional[str]
269 :param symbol: optional the symbol to use.
270 :type symbol: Optional[:py:class:`psyclone.psyir.symbols.Symbol`]
271
272 :returns: the symbol used in the added reference.
273 :rtype: :py:class:`psyclone.psyir.symbols.Symbol`
274
275 '''
276
277 ref = self.get_array_reference(array_name, indices, intrinsic_type,
278 tag=tag, symbol=symbol)
279 self.psyir_append(ref)
280 return ref.symbol

338 def generate(self, var_accesses=None):
339 # pylint: disable=too-many-statements, too-many-branches
340 '''
341 Specifies which arguments appear in an argument list, their type
342 and their ordering. Calls methods for each type of argument
343 that can be specialised by a child class for its particular need.
344 If the optional argument var_accesses is supplied, this function
345 will also add variable access information for each implicit argument
346 (i.e. that is not explicitly listed in kernel metadata) that is
347 added. These accesses will be marked as read.
348
349 :param var_accesses: optional VariablesAccessInfo instance that \
350 stores the information about variable accesses.
351 :type var_accesses: \
352 :py:class:`psyclone.core.VariablesAccessInfo`
353
354 :raises GenerationError: if the kernel arguments break the \
355 rules for the LFRic API.
356
357 '''
358 # Setting this first is important, since quite a few derived classes
359 # will access self.arglist during generate() (e.g. to test if an
360 # argument is already contained in the argument list).
361 self._generate_called = True
362 if self._kern.arguments.has_operator():
363 # All operator types require the cell index to be provided
364 self.cell_position(var_accesses=var_accesses)
365 # Pass the number of layers in the mesh unless this kernel is
366 # applying a CMA operator or doing a CMA matrix-matrix calculation
367 if self._kern.cma_operation not in ["apply", "matrix-matrix"]:
368 self.mesh_height(var_accesses=var_accesses)
369 # Pass the number of cells in the mesh if this kernel has a
370 # LMA operator argument
371 # TODO this code should replace the code that currently includes
372 # this quantity for *every* operator it encounters.
373 # if self._kern.arguments.has_operator(op_type="gh_operator"):
374 # self.mesh_ncell3d()
375
376 # If this kernel iterates into the halos then include a halo-depth
377 # argument.
378 if "halo" in self._kern.iterates_over:
379 self.halo_depth(var_accesses=var_accesses)
380
381 # Pass the number of columns in the mesh if this kernel operates on
382 # the 'domain' or has a CMA operator argument. For the former we
383 # exclude halo columns.
384 if self._kern.iterates_over == "domain":
385 self._mesh_ncell2d_no_halos(var_accesses=var_accesses)
386
387 if self._kern.arguments.has_operator(op_type="gh_columnwise_operator"):
388 self._mesh_ncell2d(var_accesses=var_accesses)
389
390 if self._kern.is_intergrid:
391 # Inter-grid kernels require special arguments
392 # The cell-map for the current column providing the mapping from
393 # the coarse to the fine mesh.
394 self.cell_map(var_accesses=var_accesses)
395
396 # For each argument in the order they are specified in the
397 # kernel metadata, call particular methods depending on what
398 # type of argument we find (field, field vector, operator or
399 # scalar). If the argument is a field or field vector and also
400 # has a stencil access then also call appropriate stencil
401 # methods.
402 const = LFRicConstants()
403 for arg in self._kern.arguments.args:
404 if arg.is_field:
405 if arg.vector_size > 1:
406 self.field_vector(arg, var_accesses=var_accesses)
407 else:
408 self.field(arg, var_accesses=var_accesses)
409 if arg.descriptor.stencil:
410 if not arg.descriptor.stencil['extent']:
411 if arg.descriptor.stencil['type'] == "cross2d":
412 # stencil extent is not provided in the
413 # metadata so must be passed from the Algorithm
414 # layer.
415 self.stencil_2d_unknown_extent(
416 arg, var_accesses=var_accesses)
417 # Due to the nature of the stencil extent array
418 # the max size of a stencil branch must be passed
419 # from the Algorithm layer.
420 self.stencil_2d_max_extent(
421 arg, var_accesses=var_accesses)
422 else:
423 # stencil extent is not provided in the
424 # metadata so must be passed from the Algorithm
425 # layer.
426 self.stencil_unknown_extent(
427 arg, var_accesses=var_accesses)
428 if arg.descriptor.stencil['type'] == "xory1d":
429 # if "xory1d is specified then the actual
430 # direction must be passed from the Algorithm layer.
431 self.stencil_unknown_direction(arg, var_accesses)
432 # stencil information that is always passed from the
433 # Algorithm layer.
434 if arg.descriptor.stencil['type'] == "cross2d":
435 self.stencil_2d(arg, var_accesses=var_accesses)
436 else:
437 self.stencil(arg, var_accesses=var_accesses)
438 elif arg.argument_type == "gh_operator":
439 self.operator(arg, var_accesses=var_accesses)
440 elif arg.argument_type == "gh_columnwise_operator":
441 self.cma_operator(arg, var_accesses=var_accesses)
442 elif arg.is_scalar:
443 self.scalar(arg, var_accesses=var_accesses)
444 else:
445 raise GenerationError(
446 f"ArgOrdering.generate(): Unexpected argument type found. "
447 f"Expected one of '{const.VALID_ARG_TYPE_NAMES}' but "
448 f"found '{arg.argument_type}'")
449 # For each function space (in the order they appear in the
450 # metadata arguments)
451 for unique_fs in self._kern.arguments.unique_fss:
452 # Provide arguments common to LMA operators and fields on
453 # a space *unless* this is an inter-grid or CMA
454 # matrix-matrix kernel
455 if self._kern.cma_operation not in ["matrix-matrix"] and \
456 not self._kern.is_intergrid:
457 self.fs_common(unique_fs, var_accesses=var_accesses)
458 # Provide additional arguments if there is a
459 # field on this space
460 if unique_fs.field_on_space(self._kern.arguments):
461 if self._kern.is_intergrid:
462 self.fs_intergrid(unique_fs, var_accesses=var_accesses)
463 else:
464 self.fs_compulsory_field(unique_fs,
465 var_accesses=var_accesses)
466 cma_op = unique_fs.cma_on_space(self._kern.arguments)
467 if cma_op:
468 if self._kern.cma_operation == "assembly":
469 # CMA-assembly requires banded dofmaps
470 self.banded_dofmap(unique_fs, var_accesses=var_accesses)
471 elif self._kern.cma_operation == "apply":
472 # Applying a CMA operator requires indirection dofmaps
473 self.indirection_dofmap(unique_fs, operator=cma_op,
474 var_accesses=var_accesses)
475
476 # Provide any optional arguments. These arguments are
477 # associated with the keyword arguments (basis function
478 # and differential basis function) for a function space.
479 if self._kern.fs_descriptors.exists(unique_fs):
480 descriptors = self._kern.fs_descriptors
481 descriptor = descriptors.get_descriptor(unique_fs)
482 if descriptor.requires_basis:
483 self.basis(unique_fs, var_accesses=var_accesses)
484 if descriptor.requires_diff_basis:
485 self.diff_basis(unique_fs, var_accesses=var_accesses)
486 # Fix for boundary_dofs array to the boundary condition
487 # kernel (enforce_bc_kernel) arguments
488 if (MetadataToArgumentsRules.bc_kern_regex.match(self._kern.name)
489 and unique_fs.orig_name.lower() == "any_space_1"):
490 self.field_bcs_kernel(unique_fs, var_accesses=var_accesses)
491
492 # Add boundary dofs array to the operator boundary condition
493 # kernel (enforce_operator_bc_kernel) arguments
494 if self._kern.name.lower() == "enforce_operator_bc_code":
495 # Sanity checks - this kernel should only have a single LMA
496 # operator as argument
497 if len(self._kern.arguments.args) > 1:
498 raise GenerationError(
499 f"Kernel {self._kern.name} has "
500 f"{len(self._kern.arguments.args)} arguments when it "
501 f"should only have 1 (an LMA operator)")
502 op_arg = self._kern.arguments.args[0]
503 if op_arg.argument_type != "gh_operator":
504 raise GenerationError(
505 f"Expected an LMA operator from which to look-up boundary "
506 f"dofs but kernel {self._kern.name} has argument "
507 f"{op_arg.argument_type}.")
508 if op_arg.access != AccessType.READWRITE:
509 raise GenerationError(
510 f"Kernel {self._kern.name} is recognised as a kernel which"
511 f" applies boundary conditions to an operator. However its"
512 f" operator argument has access "
513 f"{op_arg.access.api_specific_name()} rather than "
514 f"gh_readwrite.")
515 self.operator_bcs_kernel(op_arg.function_space_to,
516 var_accesses=var_accesses)
517
518 # Reference-element properties
519 if self._kern.reference_element:
520 self.ref_element_properties(var_accesses=var_accesses)
521
522 # Mesh properties
523 if self._kern.mesh:
524 self.mesh_properties(var_accesses=var_accesses)
525
526 # Provide qr arguments if required
527 if self._kern.qr_required:
528 self.quad_rule(var_accesses=var_accesses)
