53class OMPTaskDirective(OMPRegionDirective):
54 """
55 Class representing an OpenMP TASK directive in the PSyIR after lowering.
56 This node should not be created by any transformation, it is solely used
57 to represent TASK directives after lowering a DynamicOMPTaskDirective.
58
59 :param list children: list of Nodes that are children of this Node.
60 :type children: List[:py:class:`psyclone.psyir.nodes.Node`]
61 :param parent: the Node in the AST that has this directive as a child
62 :type parent: :py:class:`psyclone.psyir.nodes.Node`
63 :param clauses: optional list of clauses to be added to this Node.
64 :type clauses: Optional[List[:py:class:`psyclone.psyir.nodes.Clause`]]
65 """

67 # the clause computation - this should not currently cause much issues but

83 @staticmethod
84 def _validate_child(position, child):
85 """
86 Decides whether a given child and position are valid for this node.
87 The rules are:
88 1. Child 0 must always be a Schedule.
89 2. Child 1 must always be an OMPPrivateClause
90 3. Child 2 must always be an OMPFirstprivateClause
91 4. Child 3 must always be an OMPSharedClause
92 5. Child 4 and 5 must always be OMPDependClauses
93
94 :param int position: the position to be validated.
95 :param child: a child to be validated.
96 :type child: :py:class:`psyclone.psyir.nodes.Node`
97
98 :return: whether the given child and position are valid for this node.
99 :rtype: bool
100
101 """
102 if position == 0:
103 return isinstance(child, Schedule)
104 if position == 1:
105 return isinstance(child, OMPPrivateClause)
106 if position == 2:
107 return isinstance(child, OMPFirstprivateClause)
108 if position == 3:
109 return isinstance(child, OMPSharedClause)
110 if position in (4, 5):
111 return isinstance(child, OMPDependClause)
112 return False

155 def validate_global_constraints(self):
156 """
157 Perform validation checks that can only be done at code-generation
158 time.
159
160 :raises GenerationError: if this OMPTaskDirective is not \
161 enclosed within an OpenMP serial region.
162 """
163 # It is only at the point of code generation that we can check for
164 # correctness (given that we don't mandate the order that a user
165 # can apply transformations to the code). A taskloop
166 # directive, we must have an OMPSerialDirective as an
167 # ancestor back up the tree.
168 if not self.ancestor(OMPSingleDirective):
169 raise GenerationError(
170 "OMPTaskDirective must be inside an OMP Single region "
171 "but could not find an ancestor node."
172 )
173
174 # If there is a nowait clause on the ancestor OMPSingleDirective,
175 # it is possible that we could make other tasks we expect to be
176 # dependent that wouldn't be, as dependencies are only counted
177 # in OpenMP if spawned by the same thread.
178 if self.ancestor(OMPSingleDirective).nowait:
179 raise GenerationError(
180 "OMPTaskDirective found inside an OMP Single region "
181 "with nowait attached. This means we can't guarantee "
182 "correctness with other potential Single regions so is "
183 "forbidden with PSyclone.")
