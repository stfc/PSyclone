45class Range(Node):
46 '''The ``Range`` node is used to capture a range of integers via
47 ``start``, ``stop`` and ``step`` expressions. For example,
48 ``start=2``, ``stop=6`` and ``step=2`` indicates the values ``2``,
49 ``4`` and ``6``.
50
51 At the moment the only valid use of ``Range`` in the PSyIR is to
52 describe a set of accesses to an Array dimension (so-called array
53 notation in Fortran). Therefore, the parent of a ``Range`` node should
54 only be an ``Array`` node.
55
56 The ``Range`` node has three children nodes, the first child captures
57 the start of the range, the second child captures the end of the range
58 and the third captures the step within the range.
59
60 The nodes for each of the children must return an integer. Potentially
61 valid nodes are therefore ``Literal``, ``Reference``, ``Operation``
62 and ``CodeBlock``.
63
64 A common use case is to want to specify all the elements of a given
65 array dimension without knowing the extent of that dimension. In the
66 PSyIR this is achieved by using the ``LBOUND``, and ``UBOUND``
67 intrinsics::
68
69 >>> one = Literal("1", INTEGER_TYPE)
70 >>> # Declare a 1D real array called 'a' with 10 elements
71 >>> symbol = DataSymbol("a", ArrayType(REAL_TYPE, [10]))
72 >>> # Return the lower bound of the first dimension of array 'a'
73 >>> lbound = IntrinsicCall.create(
74 IntrinsicCall.Intrinsic.LBOUND,
75 [Reference(symbol), one.copy()])
76 >>> # Return the upper bound of the first dimension of array 'a'
77 >>> ubound = IntrinsicCall.create(
78 IntrinsicCall.Intrinsic.UBOUND,
79 [Reference(symbol), one.copy()])
80 >>> # Step defaults to 1 so no need to include it when creating range
81 >>> my_range = Range.create(lbound, ubound)
82 >>> # Create an access to all elements in the 1st dimension of array 'a'
83 >>> array_access = ArrayReference.create(symbol, [my_range])
84
85 In Fortran the above access ``array_access`` can be represented by
86 ``a(:)``. The Fortran front-ends and back-ends are aware of array
87 notation. Therefore the Fortran frontend is able to convert array
88 notation to PSyIR and the Fortran backend is able to convert PSyIR
89 back to array notation.
90 '''
91 # Textual description of the node.
92 _children_valid_format = "DataNode, DataNode, DataNode"
93 _text_name = "Range"
94 _colour = "white"

109 @staticmethod
110 def create(start, stop, step=None):
111 '''
112 Create an internally-consistent Range object. If no step
113 is provided then it defaults to an integer Literal with value 1.
114
115 :param start: the PSyIR for the start value.
116 :type start: :py:class:`psyclone.psyir.nodes.Node`
117 :param stop: the PSyIR for the stop value.
118 :type stop: :py:class:`psyclone.psyir.nodes.Node`
119 :param step: the PSyIR for the increment/step or None.
120 :type step: :py:class:`psyclone.psyir.nodes.Node` or NoneType
121 :param parent: the parent node of this Range in the PSyIR.
122 :type parent: :py:class:`psyclone.psyir.nodes.Node` or NoneType
123
124 :returns: a fully-populated Range object.
125 :rtype: :py:class:`psyclone.psyir.nodes.ranges.Range`
126
127 '''
128 erange = Range()
129 erange.start = start
130 erange.stop = stop
131 if step:
132 erange.step = step
133 else:
134 # No step supplied so default to a value of 1
135 erange.step = Literal("1", INTEGER_TYPE)
136 return erange

140 '''
141 Perform checks that the supplied value is valid as a child of a
142 Range node.
143
144 :param object value: entity to check.
145 :param str name: the name of the quantity for which this value has \
146 been supplied.
147

158 raise TypeError(
159 f"If the {name} value of a Range is a Literal then it "
160 f"must be of type INTEGER but got {value.datatype}")
