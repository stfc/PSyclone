60class InlineTrans(Transformation):
61 '''
62 This transformation takes a Call (which may have a return value)
63 and replaces it with the body of the target routine. It is used as
64 follows:
65
66 >>> from psyclone.psyir.backend.fortran import FortranWriter
67 >>> from psyclone.psyir.frontend.fortran import FortranReader
68 >>> from psyclone.psyir.nodes import Call, Routine
69 >>> from psyclone.psyir.transformations import InlineTrans
70 >>> code = """
71 ... module test_mod
72 ... contains
73 ... subroutine run_it()
74 ... integer :: i
75 ... real :: a(10)
76 ... do i=1,10
77 ... a(i) = 1.0
78 ... call sub(a(i))
79 ... end do
80 ... end subroutine run_it
81 ... subroutine sub(x)
82 ... real, intent(inout) :: x
83 ... x = 2.0*x
84 ... end subroutine sub
85 ... end module test_mod"""
86 >>> psyir = FortranReader().psyir_from_source(code)
87 >>> call = psyir.walk(Call)[0]
88 >>> inline_trans = InlineTrans()
89 >>> inline_trans.apply(call)
90 >>> # Uncomment the following line to see a text view of the schedule
91 >>> # print(psyir.walk(Routine)[0].view())
92 >>> print(FortranWriter()(psyir.walk(Routine)[0]))
93 subroutine run_it()
94 integer :: i
95 real, dimension(10) :: a
96 <BLANKLINE>
97 do i = 1, 10, 1
98 a(i) = 1.0
99 a(i) = 2.0 * a(i)
100 enddo
101 <BLANKLINE>
102 end subroutine run_it
103 <BLANKLINE>
104
105 .. warning::
106 Routines/calls with any of the following characteristics are not
107 supported and will result in a TransformationError:
108
109 * the routine is not in the same file as the call;
110 * the routine contains an early Return statement;
111 * the routine contains a variable with UnknownInterface;
112 * the routine contains a variable with StaticInterface;
113 * the routine contains an UnsupportedType variable with
114 ArgumentInterface;
115 * the routine has a named argument;
116 * the shape of any array arguments as declared inside the routine does
117 not match the shape of the arrays being passed as arguments;
118 * the routine accesses an un-resolved symbol;
119 * the routine accesses a symbol declared in the Container to which it
120 belongs.
121
122 Some of these restrictions will be lifted by #924.
123
124 '''
125 def apply(self, node, options=None):
126 '''
127 Takes the body of the routine that is the target of the supplied
128 call and replaces the call with it.
129
130 :param node: target PSyIR node.
131 :type node: :py:class:`psyclone.psyir.nodes.Routine`
132 :param options: a dictionary with options for transformations.
133 :type options: Optional[Dict[str, Any]]
134 :param bool options["force"]: whether or not to permit the inlining
135 of Routines containing CodeBlocks. Default is False.
136 '''
137 self.validatevalidate(node, options)
138 # The table associated with the scoping region holding the Call.
139 table = node.scope.symbol_table
140 # Find the routine to be inlined.
141 orig_routine = node.get_callees()[0]
142
143 if not orig_routine.children or isinstance(orig_routine.children[0],
144 Return):
145 # Called routine is empty so just remove the call.
146 node.detach()
147 return
148
149 # Ensure we don't modify the original Routine by working with a
150 # copy of it.
151 routine = orig_routine.copy()
152 routine_table = routine.symbol_table
153
154 # Construct lists of the nodes that will be inserted and all of the
155 # References that they contain.
156 new_stmts = []
157 refs = []
158 for child in routine.children:
159 new_stmts.append(child.copy())
160 refs.extend(new_stmts[-1].walk(Reference))
161
162 # Shallow copy the symbols from the routine into the table at the
163 # call site.
164 table.merge(routine_table,
165 symbols_to_skip=routine_table.argument_list[:])
166
167 # When constructing new references to replace references to formal
168 # args, we need to know whether any of the actual arguments are array
169 # accesses. If they use 'array notation' (i.e. represent a whole array)
170 # then they won't have index expressions and will have been captured
171 # as a Reference.
172 ref2arraytrans = Reference2ArrayRangeTrans()
173
174 for child in node.arguments:
175 try:
176 # TODO #1858, this won't yet work for arrays inside structures.
177 ref2arraytrans.apply(child)
178 except (TransformationError, ValueError):
179 pass
180
181 # Replace any references to formal arguments with copies of the
182 # actual arguments.
183 formal_args = routine_table.argument_list
184 for ref in refs[:]:
185 self._replace_formal_arg(ref, node, formal_args)
186
187 # Store the Routine level symbol table and node's current scope
188 # so we can merge symbol tables later if required.
189 ancestor_table = node.ancestor(Routine).scope.symbol_table
190 scope = node.scope
191
192 # Copy the nodes from the Routine into the call site.
193 # TODO #924 - while doing this we should ensure that any References
194 # to common/shared Symbols in the inlined code are updated to point
195 # to the ones at the call site.
196 if isinstance(new_stmts[-1], Return):
197 # If the final statement of the routine is a return then
198 # remove it from the list.
199 del new_stmts[-1]
200
201 if routine.return_symbol:
202 # This is a function
203 assignment = node.ancestor(Statement)
204 parent = assignment.parent
205 idx = assignment.position-1
206 for child in new_stmts:
207 idx += 1
208 parent.addchild(child, idx)
209 table = parent.scope.symbol_table
210 # Avoid a potential name clash with the original function
211 table.rename_symbol(
212 routine.return_symbol, table.next_available_name(
213 f"inlined_{routine.return_symbol.name}"))
214 node.replace_with(Reference(routine.return_symbol))
215 else:
216 # This is a call
217 parent = node.parent
218 idx = node.position
219 node.replace_with(new_stmts[0])
220 for child in new_stmts[1:]:
221 idx += 1
222 parent.addchild(child, idx)
223
224 # If the scope we merged the inlined function's symbol table into
225 # is not a Routine scope then we now merge that symbol table into
226 # the ancestor Routine. This avoids issues like #2424 when
227 # applying ParallelLoopTrans to loops containing inlined calls.
228 if ancestor_table is not scope.symbol_table:
229 ancestor_table.merge(scope.symbol_table)
230 replacement = type(scope.symbol_table)()
231 scope.symbol_table.detach()
232 replacement.attach(scope)

234 def _replace_formal_arg(self, ref, call_node, formal_args):
235 '''
236 Recursively combines any References to formal arguments in the supplied
237 PSyIR expression with the corresponding Reference (actual argument)
238 from the call site to make a new Reference for use in the inlined code.
239 If the supplied node is not a Reference to a formal argument then it is
240 just returned (after we have recursed to any children).
241
242 :param ref: the expression to update.
243 :type ref: :py:class:`psyclone.psyir.nodes.Node`
244 :param call_node: the call site.
245 :type call_node: :py:class:`psyclone.psyir.nodes.Call`
246 :param formal_args: the formal arguments of the called routine.
247 :type formal_args: List[:py:class:`psyclone.psyir.symbols.DataSymbol`]
248
249 :returns: the replacement reference.
250 :rtype: :py:class:`psyclone.psyir.nodes.Reference`
251
252 '''
253 if not isinstance(ref, Reference):
254 # Recurse down in case this is e.g. an Operation or Range.
255 for child in ref.children[:]:
256 self._replace_formal_arg(child, call_node, formal_args)
257 return ref
258
259 if ref.symbol not in formal_args:
260 # The supplied reference is not to a formal argument.
261 return ref
262
263 # Lookup the actual argument that corresponds to this formal argument.
264 actual_arg = call_node.arguments[formal_args.index(ref.symbol)]
265
266 # If the local reference is a simple Reference then we can just
267 # replace it with a copy of the actual argument, e.g.
268 #
269 # call my_sub(my_struc%data(i,j))
270 #
271 # subroutine my_sub(var)
272 # ...
273 # var = 0.0
274 #
275 # pylint: disable=unidiomatic-typecheck
276 if type(ref) is Reference:
277 arg_copy = actual_arg.copy()
278 # If the local reference we are replacing has a parent then we
279 # must ensure the parent's child list is updated. (It may not
280 # have a parent if we are in the process of constructing a brand
281 # new reference.)
282 if ref.parent:
283 ref.replace_with(arg_copy)
284 return arg_copy
285
286 # Local reference is not simple but the actual argument is, e.g.:
287 #
288 # call my_sub(my_struc)
289 #
290 # subroutine my_sub(var)
291 # ...
292 # var%data(i,j) = 0.0
293 #
294 if type(actual_arg) is Reference:
295 ref.symbol = actual_arg.symbol
296 return ref
297
298 # Neither the actual or local references are simple, i.e. they
299 # include array accesses and/or structure accesses.
300 new_ref = self._replace_formal_struc_arg(actual_arg, ref, call_node,
301 formal_args)
302 # If the local reference we are replacing has a parent then we must
303 # ensure the parent's child list is updated. (It may not have a parent
304 # if we are in the process of constructing a brand new reference.)
305 if ref.parent:
306 ref.replace_with(new_ref)
307 return new_ref

309 def _create_inlined_idx(self, call_node, formal_args,
310 local_idx, decln_start, actual_start):
311 '''
312 Utility that creates the PSyIR for an inlined array-index access
313 expression. This is not trivial since a formal argument may be
314 declared with bounds that are shifted relative to those of an
315 actual argument.
316
317 If local_idx is the index of the access in the routine;
318 local_decln_start is the starting index of the dimension as
319 declared in the routine;
320 actual_start is the starting index of the slice at the callsite
321 (whether from the array declaration or a slice);
322
323 then the index of the inlined access will be::
324
325 inlined_idx = local_idx - local_decln_start + 1 + actual_start - 1
326 = local_idx - local_decln_start + actual_start
327
328 :param call_node: the Call that we are inlining.
329 :type call_node: :py:class:`psyclone.psyir.nodes.Call`
330 :param formal_args: the formal arguments of the routine being called.
331 :type formal_args: List[:py:class:`psyclone.psyir.symbols.DataSymbol`]
332 :param local_idx: a local array-index expression (i.e. appearing \
333 within the routine being inlined).
334 :type local_idx: :py:class:`psyclone.psyir.nodes.Node`
335 :param decln_start: the lower bound of the corresponding array \
336 dimension, as declared inside the routine being inlined.
337 :type decln_start: :py:class:`psyclone.psyir.nodes.Node`
338 :param actual_start: the lower bound of the corresponding array \
339 dimension, as defined at the call site.
340 :type actual_start: :py:class:`psyclone.psyir.nodes.Node`
341
342 :returns: PSyIR for the corresponding inlined array index.
343 :rtype: :py:class:`psyclone.psyir.nodes.Node`
344
345 '''
346 if isinstance(local_idx, Range):
347 lower = self._create_inlined_idx(call_node, formal_args,
348 local_idx.start, decln_start,
349 actual_start)
350 upper = self._create_inlined_idx(call_node, formal_args,
351 local_idx.stop, decln_start,
352 actual_start)
353 step = self._replace_formal_arg(local_idx.step, call_node,
354 formal_args)
355 return Range.create(lower.copy(), upper.copy(), step.copy())
356
357 uidx = self._replace_formal_arg(local_idx, call_node, formal_args)
358 if decln_start == actual_start:
359 # If the starting indices in the actual and formal arguments are
360 # the same then we don't need to shift the index.
361 return uidx
362
363 ustart = self._replace_formal_arg(decln_start, call_node, formal_args)
364 start_sub = BinaryOperation.create(BinaryOperation.Operator.SUB,
365 uidx.copy(), ustart.copy())
366 return BinaryOperation.create(BinaryOperation.Operator.ADD,
367 start_sub, actual_start.copy())

369 def _update_actual_indices(self, actual_arg, local_ref,
370 call_node, formal_args):
371 '''
372 Create a new list of indices for the supplied actual argument
373 (ArrayMixin) by replacing any Ranges with the appropriate expressions
374 from the local access in the called routine. If there are no Ranges
375 then the returned list of indices just contains copies of the inputs.
376
377 :param actual_arg: (part of) the actual argument to the routine.
378 :type actual_arg: :py:class:`psyclone.psyir.nodes.ArrayMixin`
379 :param local_ref: the corresponding Reference in the called routine.
380 :param call_node: the call site.
381 :type call_node: :py:class:`psyclone.psyir.nodes.Call`
382 :param formal_args: the formal arguments of the called routine.
383 :type formal_args: List[:py:class:`psyclone.psyir.symbols.DataSymbol`]
384
385 :returns: new indices for the actual argument.
386 :rtype: List[:py:class:`psyclone.psyir.nodes.Node`]
387
388 '''
389 if isinstance(local_ref, ArrayMixin):
390 local_indices = [idx.copy() for idx in local_ref.indices]
391 # Get the locally-declared shape of the formal argument in case its
392 # bounds are shifted relative to the caller.
393 if isinstance(local_ref.symbol.datatype, ArrayType):
394 local_decln_shape = local_ref.symbol.datatype.shape
395 else:
396 local_decln_shape = []
397
398 new_indices = [idx.copy() for idx in actual_arg.indices]
399 local_idx_posn = 0
400 for pos, idx in enumerate(new_indices[:]):
401
402 if not isinstance(idx, Range):
403 continue
404
405 # Starting index of slice of actual argument.
406 if actual_arg.is_lower_bound(pos):
407 # Range starts at lower bound of argument so that's what
408 # we store.
409 actual_start = actual_arg.get_lbound_expression(pos)
410 else:
411 actual_start = idx.start
412
413 local_decln_start = None
414 if local_decln_shape:
415 if isinstance(local_decln_shape[local_idx_posn],
416 ArrayType.ArrayBounds):
417 # The formal argument declaration has a shape.
418 local_shape = local_decln_shape[local_idx_posn]
419 local_decln_start = local_shape.lower
420 if isinstance(local_decln_start, Node):
421 # Ensure any references to formal arguments within
422 # the declared array lower bound are updated.
423 local_decln_start = self._replace_formal_arg(
424 local_decln_start, call_node, formal_args)
425 elif (local_decln_shape[local_idx_posn] ==
426 ArrayType.Extent.DEFERRED):
427 # The formal argument is declared to be allocatable and
428 # therefore has the same bounds as the actual argument.
429 local_shape = None
430 local_decln_start = actual_start
431
432 if not local_decln_start:
433 local_shape = None
434 local_decln_start = _ONE
435
436 if local_ref.is_full_range(local_idx_posn):
437 # If the local Range is for the full extent of the formal
438 # argument then the actual Range is defined by that of the
439 # actual argument and no change is required unless the formal
440 # argument is declared as having a Range with an extent that is
441 # less than that supplied. In general we're not going to know
442 # that so we have to be conservative.
443 if local_shape:
444 new = Range.create(local_shape.lower.copy(),
445 local_shape.upper.copy())
446 new_indices[pos] = self._create_inlined_idx(
447 call_node, formal_args,
448 new, local_decln_start, actual_start)
449 else:
450 # Otherwise, the local index expression replaces the Range.
451 new_indices[pos] = self._create_inlined_idx(
452 call_node, formal_args,
453 local_indices[local_idx_posn],
454 local_decln_start, actual_start)
455 # Each Range corresponds to one dimension of the formal argument.
456 local_idx_posn += 1
457 return new_indices

459 def _replace_formal_struc_arg(self, actual_arg, ref, call_node,
460 formal_args):
461 '''
462 Called by _replace_formal_arg() whenever a formal or actual argument
463 involves an array or structure access that can't be handled with a
464 simple substitution, e.g.
465
466 .. code-block:: fortran
467
468 call my_sub(my_struc%grid(:,2,:), 10)
469
470 subroutine my_sub(grid, ngrids)
471 ...
472 do igrid = 1, ngrids
473 do jgrid = ...
474 do i = 1, 10
475 do j = 1, 10
476 grid(igrid, jgrid)%data(i,j) = 0.0
477
478 The assignment in the inlined code should become
479
480 .. code-block:: fortran
481
482 my_struc%grid(igrid,2,jgrid)%data(i,j) = 0.0
483
484 This routine therefore recursively combines any References to formal
485 arguments in the supplied Reference (including any array-index
486 expressions) with the corresponding Reference
487 from the call site to make a new Reference for use in the inlined code.
488
489 :param actual_arg: an actual argument to the routine being inlined.
490 :type actual_arg: :py:class:`psyclone.psyir.nodes.Reference`
491 :param ref: the corresponding reference to a formal argument.
492 :type ref: :py:class:`psyclone.psyir.nodes.Reference`
493 :param call_node: the call site.
494 :type call_node: :py:class:`psyclone.psyir.nodes.Call`
495 :param formal_args: the formal arguments of the called routine.
496 :type formal_args: List[:py:class:`psyclone.psyir.symbols.DataSymbol`]
497
498 :returns: the replacement reference.
499 :rtype: :py:class:`psyclone.psyir.nodes.Reference`
500
501 '''
502 # The final stage of this method creates a brand new
503 # [ArrayOf]Structure[s]Reference so we have to collect the indices and
504 # members as we walk down both the actual and local references.
505 local_indices = None
506 members = []
507
508 # Actual arg could be var, var(:)%index, var(i,j)%grid(:) or
509 # var(j)%data(i) etc. Any Ranges must correspond to dimensions of the
510 # formal argument. The validate() method has already ensured that we
511 # do not have any indirect accesses or non-unit strides.
512
513 if isinstance(ref, ArrayMixin):
514 local_indices = [idx.copy() for idx in ref.indices]
515
516 # Since a Range can occur at any level of a Structure access in the
517 # actual argument, we walk down it and check each Member. Any Ranges
518 # are updated according to how that dimension is accessed by the
519 # reference inside the routine.
520 cursor = actual_arg
521 while True:
522 if isinstance(cursor, ArrayMixin):
523 new_indices = self._update_actual_indices(
524 cursor, ref, call_node, formal_args)
525 members.append((cursor.name, new_indices))
526 else:
527 members.append(cursor.name)
528
529 if not isinstance(cursor, (StructureMember, StructureReference)):
530 break
531 cursor = cursor.member
532
533 # TODO #1858 - once we support converting structure accesses into
534 # explicit array accesses, we can put back the testing in
535 # inline_trans_test.py that covers this code and remove the pragma:
536 if not actual_arg.walk(Range) and local_indices: # pragma: no cover
537 # There are no Ranges in the actual argument but the local
538 # reference is an array access.
539 # Create updated index expressions for that access.
540 new_indices = []
541 for idx in local_indices:
542 new_indices.append(
543 self._replace_formal_arg(
544 idx.copy(), call_node, formal_args))
545 # Replace the last entry in the `members` list with a new array
546 # access.
547 members[-1] = (cursor.name, new_indices)
548
549 # We now walk down the *local* access, skipping its head (as that is
550 # replaced by the actual arg). We don't need to worry about updating
551 # index expressions in the actual argument as they are independent of
552 # any array accesses within a structure passed as a formal argument.
553 cursor = ref
554 while isinstance(cursor, (StructureReference, StructureMember)):
555 cursor = cursor.member
556 if isinstance(cursor, ArrayMixin):
557 new_indices = []
558 for idx in cursor.indices:
559 # Update each index expression in case it refers to
560 # formal arguments.
561 new_indices.append(
562 self._replace_formal_arg(
563 idx.copy(), call_node, formal_args))
564 members.append((cursor.name, new_indices))
565 else:
566 members.append(cursor.name)
567
568 # Finally, construct the new Reference using the information we've
569 # collected from both the actual argument and local access.
570 if len(members) > 1:
571 # We have some form of Structure reference.
572 if isinstance(members[0], tuple):
573 # Root of access is an array access.
574 return ArrayOfStructuresReference.create(actual_arg.symbol,
575 members[0][1],
576 members[1:])
577 return StructureReference.create(actual_arg.symbol, members[1:])
578
579 # Just an array reference.
580 return ArrayReference.create(actual_arg.symbol, members[0][1])

582 def validate(self, node, options=None):
583 '''
584 Checks that the supplied node is a valid target for inlining.
585
586 :param node: target PSyIR node.
587 :type node: subclass of :py:class:`psyclone.psyir.nodes.Call`
588 :param options: a dictionary with options for transformations.
589 :type options: Optional[Dict[str, Any]]
590 :param bool options["force"]: whether or not to ignore any CodeBlocks
591 in the candidate routine. Default is False.
592
593 :raises TransformationError: if the supplied node is not a Call or is
594 an IntrinsicCall or call to a PSyclone-generated routine.
595 :raises TransformationError: if the routine has a return value.
596 :raises TransformationError: if the routine body contains a Return
597 that is not the first or last statement.
598 :raises TransformationError: if the routine body contains a CodeBlock
599 and the 'force' option is not True.
600 :raises TransformationError: if the called routine has a named
601 argument.
602 :raises TransformationError: if any of the variables declared within
603 the called routine are of UnknownInterface.
604 :raises TransformationError: if any of the variables declared within
605 the called routine have a StaticInterface.
606 :raises TransformationError: if any of the subroutine arguments is of
607 UnsupportedType.
608 :raises TransformationError: if a symbol of a given name is imported
609 from different containers at the call site and within the routine.
610 :raises TransformationError: if the routine accesses an un-resolved
611 symbol.
612 :raises TransformationError: if the number of arguments in the call
613 does not match the number of formal arguments of the routine.
614 :raises TransformationError: if a symbol declared in the parent
615 container is accessed in the target routine.
616 :raises TransformationError: if the shape of an array formal argument
617 does not match that of the corresponding actual argument.
618
619 '''
620 super().validate(node, options=options)
621
622 options = {} if options is None else options
623 forced = options.get("force", False)
624
625 # The node should be a Call.
626 if not isinstance(node, Call):
627 raise TransformationError(
628 f"The target of the InlineTrans transformation "
629 f"should be a Call but found '{type(node).__name__}'.")
630
631 if isinstance(node, IntrinsicCall):
632 raise TransformationError(
633 f"Cannot inline an IntrinsicCall ('{node.routine.name}')")
634 name = node.routine.name
635
636 # Check that we can find the source of the routine being inlined.
637 # TODO #924 allow for multiple routines (interfaces).
638 try:
639 routine = node.get_callees()[0]
640 except (NotImplementedError, FileNotFoundError, SymbolError) as err:
641 raise TransformationError(
642 f"Cannot inline routine '{name}' because its source cannot be "
643 f"found: {err}") from err
644
645 if not routine.children or isinstance(routine.children[0], Return):
646 # An empty routine is fine.
647 return
648
649 return_stmts = routine.walk(Return)
650 if return_stmts:
651 if len(return_stmts) > 1 or not isinstance(routine.children[-1],
652 Return):
653 # Either there is more than one Return statement or there is
654 # just one but it isn't the last statement of the Routine.
655 raise TransformationError(
656 f"Routine '{name}' contains one or more "
657 f"Return statements and therefore cannot be inlined.")
658
659 if routine.walk(CodeBlock) and not forced:
660 # N.B. we permit the user to specify the "force" option to allow
661 # CodeBlocks to be included.
662 raise TransformationError(
663 f"Routine '{name}' contains one or more CodeBlocks and "
664 "therefore cannot be inlined. (If you are confident that "
665 "the code may safely be inlined despite this then use "
666 "`options={'force': True}` to override.)")
667
668 # Support for routines with named arguments is not yet implemented.
669 # TODO #924.
670 for arg in node.argument_names:
671 if arg:
672 raise TransformationError(
673 f"Routine '{routine.name}' cannot be inlined because it "
674 f"has a named argument '{arg}' (TODO #924).")
675
676 table = node.scope.symbol_table
677 routine_table = routine.symbol_table
678
679 for sym in routine_table.datasymbols:
680 # We don't inline symbols that have an UnsupportedType and are
681 # arguments since we don't know if a simple assignment if
682 # enough (e.g. pointers)
683 if isinstance(sym.interface, ArgumentInterface):
684 if isinstance(sym.datatype, UnsupportedType):
685 raise TransformationError(
686 f"Routine '{routine.name}' cannot be inlined because "
687 f"it contains a Symbol '{sym.name}' which is an "
688 f"Argument of UnsupportedType: "
689 f"'{sym.datatype.declaration}'")
690 # We don't inline symbols that have an UnknownInterface, as we
691 # don't know how they are brought into this scope.
692 if isinstance(sym.interface, UnknownInterface):
693 raise TransformationError(
694 f"Routine '{routine.name}' cannot be inlined because it "
695 f"contains a Symbol '{sym.name}' with an UnknownInterface:"
696 f" '{sym.datatype.declaration}'")
697 # Check that there are no static variables in the routine (because
698 # we don't know whether the routine is called from other places).
699 if (isinstance(sym.interface, StaticInterface) and
700 not sym.is_constant):
701 raise TransformationError(
702 f"Routine '{routine.name}' cannot be inlined because it "
703 f"has a static (Fortran SAVE) interface for Symbol "
704 f"'{sym.name}'.")
705
706 # We can't handle a clash between (apparently) different symbols that
707 # share a name but are imported from different containers.
708 try:
709 table.check_for_clashes(
710 routine_table,
711 symbols_to_skip=routine_table.argument_list[:])
712 except SymbolError as err:
713 raise TransformationError(
714 f"One or more symbols from routine '{routine.name}' cannot be "
715 f"added to the table at the call site.") from err
716
717 # Check for unresolved symbols or for any accessed from the Container
718 # containing the target routine.
719 # TODO #1792 - kind parameters will not be found by simply doing
720 # `walk(Reference)`. Although SymbolTable has the
721 # `precision_datasymbols` property, this only returns those Symbols
722 # that are used to define the precision of other Symbols in the same
723 # table. If a precision symbol is only used within Statements then we
724 # don't currently capture the fact that it is a precision symbol.
725 ref_or_lits = routine.walk((Reference, Literal))
726 # Check for symbols in any initial-value expressions
727 # (including Fortran parameters) or array dimensions.
728 for sym in routine_table.datasymbols:
729 if sym.initial_value:
730 ref_or_lits.extend(
731 sym.initial_value.walk((Reference, Literal)))
732 if isinstance(sym.datatype, ArrayType):
733 for dim in sym.shape:
734 if isinstance(dim, ArrayType.ArrayBounds):
735 if isinstance(dim.lower, Node):
736 ref_or_lits.extend(dim.lower.walk(Reference,
737 Literal))
738 if isinstance(dim.upper, Node):
739 ref_or_lits.extend(dim.upper.walk(Reference,
740 Literal))
741 # Keep a reference to each Symbol that we check so that we can avoid
742 # repeatedly checking the same Symbol.
743 _symbol_cache = set()
744 for lnode in ref_or_lits:
745 if isinstance(lnode, Literal):
746 if not isinstance(lnode.datatype.precision, DataSymbol):
747 continue
748 sym = lnode.datatype.precision
749 else:
750 sym = lnode.symbol
751 # If we've already seen this Symbol then we can skip it.
752 if sym in _symbol_cache:
753 continue
754 _symbol_cache.add(sym)
755 if isinstance(sym, IntrinsicSymbol):
756 continue
757 # We haven't seen this Symbol before.
758 if sym.is_unresolved:
759 try:
760 routine_table.resolve_imports(symbol_target=sym)
761 except KeyError:
762 # The symbol is not (directly) imported into the symbol
763 # table local to the routine.
764 # pylint: disable=raise-missing-from
765 raise TransformationError(
766 f"Routine '{routine.name}' cannot be inlined "
767 f"because it accesses variable '{sym.name}' and this "
768 f"cannot be found in any of the containers directly "
769 f"imported into its symbol table.")
770 else:
771 if sym.name not in routine_table:
772 raise TransformationError(
773 f"Routine '{routine.name}' cannot be inlined because "
774 f"it accesses variable '{sym.name}' from its "
775 f"parent container.")
776
777 # Check that the shapes of any formal array arguments are the same as
778 # those at the call site.
779 if len(routine_table.argument_list) != len(node.arguments):
780 raise TransformationError(LazyString(
781 lambda: f"Cannot inline '{node.debug_string().strip()}' "
782 f"because the number of arguments supplied to the call "
783 f"({len(node.arguments)}) does not match the number of "
784 f"arguments the routine is declared to have "
785 f"({len(routine_table.argument_list)})."))
786
787 for formal_arg, actual_arg in zip(routine_table.argument_list,
788 node.arguments):
789 # If the formal argument is an array with non-default bounds then
790 # we also need to know the bounds of that array at the call site.
791 if not isinstance(formal_arg.datatype, ArrayType):
792 # Formal argument is not an array so we don't need to do any
793 # further checks.
794 continue
795
796 if not isinstance(actual_arg, (Reference, Literal)):
797 # TODO #1799 this really needs the `datatype` method to be
798 # extended to support all nodes. For now we have to abort
799 # if we encounter an argument that is not a scalar (according
800 # to the corresponding formal argument) but is not a
801 # Reference or a Literal as we don't know whether the result
802 # of any general expression is or is not an array.
803 # pylint: disable=cell-var-from-loop
804 raise TransformationError(LazyString(
805 lambda: f"The call '{node.debug_string()}' cannot be "
806 f"inlined because actual argument "
807 f"'{actual_arg.debug_string()}' corresponds to a "
808 f"formal argument with array type but is not a "
809 f"Reference or a Literal."))
810
811 # We have an array argument. We are only able to check that the
812 # argument is not re-shaped in the called routine if we have full
813 # type information on the actual argument.
814 # TODO #924. It would be useful if the `datatype` property was
815 # a method that took an optional 'resolve' argument to indicate
816 # that it should attempt to resolve any UnresolvedTypes.
817 if (isinstance(actual_arg.datatype,
818 (UnresolvedType, UnsupportedType)) or
819 (isinstance(actual_arg.datatype, ArrayType) and
820 isinstance(actual_arg.datatype.intrinsic,
821 (UnresolvedType, UnsupportedType)))):
822 raise TransformationError(
823 f"Routine '{routine.name}' cannot be inlined because "
824 f"the type of the actual argument "
825 f"'{actual_arg.symbol.name}' corresponding to an array"
826 f" formal argument ('{formal_arg.name}') is unknown.")
827
828 formal_rank = 0
829 actual_rank = 0
830 if isinstance(formal_arg.datatype, ArrayType):
831 formal_rank = len(formal_arg.datatype.shape)
832 if isinstance(actual_arg.datatype, ArrayType):
833 actual_rank = len(actual_arg.datatype.shape)
834 if formal_rank != actual_rank:
835 # It's OK to use the loop variable in the lambda definition
836 # because if we get to this point then we're going to quit
837 # the loop.
838 # pylint: disable=cell-var-from-loop
839 raise TransformationError(LazyString(
840 lambda: f"Cannot inline routine '{routine.name}' "
841 f"because it reshapes an argument: actual argument "
842 f"'{actual_arg.debug_string()}' has rank {actual_rank}"
843 f" but the corresponding formal argument, "
844 f"'{formal_arg.name}', has rank {formal_rank}"))
845 if actual_rank:
846 ranges = actual_arg.walk(Range)
847 for rge in ranges:
848 ancestor_ref = rge.ancestor(Reference)
849 if ancestor_ref is not actual_arg:
850 # Have a range in an indirect access.
851 # pylint: disable=cell-var-from-loop
852 raise TransformationError(LazyString(
853 lambda: f"Cannot inline routine '{routine.name}' "
854 f"because argument '{actual_arg.debug_string()}' "
855 f"has an array range in an indirect access (TODO "
856 f"#924)."))
857 if rge.step != _ONE:
858 # TODO #1646. We could resolve this problem by making
859 # a new array and copying the necessary values into it.
860 # pylint: disable=cell-var-from-loop
861 raise TransformationError(LazyString(
862 lambda: f"Cannot inline routine '{routine.name}' "
863 f"because one of its arguments is an array slice "
864 f"with a non-unit stride: "
865 f"'{actual_arg.debug_string()}' (TODO #1646)"))
