51class ModuleManager:
52 '''This class implements a singleton that manages module
53 dependencies.
54
55 :param use_caching: Whether to use (`True`) or
56 disable (`False`) caching
57 '''
58
59 # Class variable to store the singleton instance
60 _instance = None
61
62 # How similar a file name must be to a module name to trigger reading
63 # of the file.
64 _threshold_similarity = 0.7
65
66 # ------------------------------------------------------------------------
67 @staticmethod
68 def get(use_caching: bool = None):
69 '''Static function that if necessary creates and returns the singleton
70 ModuleManager instance.
71
72 :param use_caching: If `True`, a file-based caching of the fparser
73 tree will be used. This can significantly accelerate obtaining
74 a PSyIR from a source file.
75 For parallel builds, parallel race conditions to the cache file
76 can happen, but this shouldn't lead to wrong results. However,
77 that's untested so far.
78
79 '''
80 if not ModuleManager._instance:
81 ModuleManager._instance = ModuleManager(use_caching)
82
83 return ModuleManager._instance

85 # ------------------------------------------------------------------------
86 def __init__(
87 self,
88 use_caching: bool = None
89 ):
90
91 if ModuleManager._instance is not None:
92 raise InternalError("You need to use 'ModuleManager.get()' "
93 "to get the singleton instance.")
94
95 # Disable caching by default
96 self._use_caching = use_caching if use_caching is not None else False
97
98 self._modules = {}
99 self._visited_files = {}
100
101 # The list of all search paths which have not yet all their files
102 # checked. It is stored as an ordered dict to make it easier to avoid
103 # duplicating entries.
104 self._remaining_search_paths = OrderedDict()
105 self._original_search_paths = []
106
107 self._ignore_modules_ignore_modules = set()
108
109 # Setup the regex used to find Fortran modules. Have to be careful not
110 # to match e.g. "module procedure :: some_sub".
111 self._module_pattern = re.compile(r"^\s*module\s+([a-z]\S*)\s*$",
112 flags=(re.IGNORECASE | re.MULTILINE))
113
114 # ------------------------------------------------------------------------
115 def add_search_path(self, directories, recursive=True):
116 '''If the directory is not already contained in the search path,
117 add it. Directory can either be a string, in which case it is a single
118 directory, or a list of directories, each one a string.
119
120 :param directories: the directory/directories to add.
121 :type directories: str | list[str]
122
123 :param bool recursive: whether recursively all subdirectories should
124 be added to the search path.
125
126 '''
127 if isinstance(directories, str):
128 # Make sure we always have a list
129 directories = [directories]
130
131 for directory in directories:
132 if not os.access(directory, os.R_OK):
133 raise IOError(f"Directory '{directory}' does not exist or "
134 f"cannot be read.")
135 self._remaining_search_paths[directory] = 1
136 self._original_search_paths.append(directory)
137 if recursive:
138 for root, dirs, _ in os.walk(directory):
139 for current_dir in dirs:
140 new_dir = os.path.join(root, current_dir)
141 self._remaining_search_paths[new_dir] = 1
142 self._original_search_paths.append(new_dir)
143
144 # ------------------------------------------------------------------------
145 def _add_all_files_from_dir(self, directory):
146 '''This function creates (and caches) FileInfo objects for all files
147 with an extension of (F/f/X/x)90 in the given directory that have
148 not previously been visited. The new FileInfo objects are returned.
149
150 :param str directory: the directory containing Fortran files
151 to analyse.
152
153 :returns: the FileInfo objects for any files that we have not
154 previously visited.
155 :rtype: list[:py:class:`psyclone.parse.FileInfo` | None]
156
157 '''
158 new_files = []
159 with os.scandir(directory) as all_entries:
160 for entry in all_entries:
161 _, ext = os.path.splitext(entry.name)
162 if (not entry.is_file() or
163 ext not in [".F90", ".f90", ".X90", ".x90"]):
164 continue
165 full_path = os.path.join(directory, entry.name)
166 if full_path in self._visited_files:
167 continue
168 self._visited_files[full_path] = \
169 FileInfo(full_path, use_caching=self._use_caching)
170 new_files.append(self._visited_files[full_path])
171 return new_files

173 # ------------------------------------------------------------------------
174 def _find_module_in_files(self, name, file_list):
175 '''
176 Searches the files represented by the supplied list of FileInfo objects
177 to find the one defining the named Fortran module.
178
179 :param str name: the name of the module to locate.
180 :param file_list: the files to search.
181 :type file_list: list[:py:class:`psyclone.parse.FileInfo`]
182
183 :returns: information on the file that contains the module or None if
184 it wasn't found.
185 :rtype: :py:class:`psyclone.parse.FileInfo` | None
186
187 '''
188 mod_info = None
189 for finfo in file_list:
190 finfo: FileInfo
191 # We only proceed to read a file to check for a module if its
192 # name is sufficiently similar to that of the module.
193 score = SequenceMatcher(None,
194 finfo.basename, name).ratio()
195 if score > self._threshold_similarity:
196 mod_names = self.get_modules_in_file(finfo)
197 if name in mod_names:
198 # We've found the module we want. Create a ModuleInfo
199 # object for it and cache it.
200 mod_info = ModuleInfo(name, finfo)
201 self._modules[name] = mod_info
202 # A file that has been (or does not require)
203 # preprocessing always takes precendence.
204 if finfo.filename.endswith(".f90"):
205 return mod_info
206 return mod_info

225 # ------------------------------------------------------------------------
226 def get_module_info(self, module_name):
227 '''This function returns the ModuleInformation for the specified
228 module.
229
230 :param str module_name: name of the module.
231
232 :returns: object describing the requested module or None if the
233 manager has been configured to ignore this module.
234 :rtype: :py:class:`psyclone.parse.ModuleInfo` | None
235
236 :raises FileNotFoundError: if the module_name is not found in
237 either the cached data nor in the search path.
238
239 '''
240 mod_lower = module_name.lower()
241
242 if mod_lower in self._ignore_modules_ignore_modules:
243 return None
244
245 # First check if we have already seen this module. We only end the
246 # search early if the file we've found does not require pre-processing
247 # (i.e. has a .f90 suffix).
248 mod_info: ModuleInfo = self._modules.get(mod_lower, None)
249 if mod_info and mod_info.filename.endswith(".f90"):
250 return mod_info
251 old_mod_info = mod_info
252 # Are any of the files that we've already seen a good match?
253 mod_info = self._find_module_in_files(mod_lower,
254 self._visited_files.values())
255 if mod_info and mod_info.filename.endswith(".f90"):
256 return mod_info
257 old_mod_info = mod_info
258
259 # If not, check the search paths. To avoid frequent accesses to
260 # the directories, we search directories one at a time, and
261 # add the list of all files in that directory to our cache
262 # `_visited_files`.
263 while self._remaining_search_paths:
264 # Get the first element from the search path list:
265 directory, _ = self._remaining_search_paths.popitem(last=False)
266 new_files = self._add_all_files_from_dir(directory)
267 mod_info = self._find_module_in_files(mod_lower, new_files)
268 if mod_info:
269 return mod_info
270
271 if old_mod_info:
272 return old_mod_info
273
274 raise FileNotFoundError(f"Could not find source file for module "
275 f"'{module_name}' in any of the directories "
276 f"'{', '.join(self._original_search_paths)}'. "
277 f"You can add search paths using the '-d' "
278 f"command line option.")
279
280 # ------------------------------------------------------------------------
281 def get_modules_in_file(self, finfo: FileInfo):
282 '''
283 Uses a regex search to find all modules defined in the file with the
284 supplied name.
285
286 :param finfo: object holding information on the file to examine.
287 :type finfo: :py:class:`psyclone.parse.FileInfo`
288
289 :returns: the names of any modules present in the supplied file.
290 :rtype: list[str]
291
292 '''
293 # TODO #2597: perhaps use the fparser FortranReader here as this regex
294 # could be defeated by e.g.
295 # module &
296 # my_mod
297 # `finfo.get_source_code()` will read the file if it hasn't already
298 # been cached.
299 mod_names = self._module_pattern.findall(finfo.get_source_code())
300
301 return [name.lower() for name in mod_names]

304 def get_all_dependencies_recursively(self, all_mods):
305 '''This function collects recursively all module dependencies
306 for any of the modules in the ``all_mods`` set. I.e. it will
307 add all modules used by any module listed in ``all_mods``,
308 and any modules used by the just added modules etc. In the end,
309 it will return a dictionary that for each module lists which
310 modules it depends on. This dictionary will be complete,
311 i.e. all modules that are required for the original set of modules
312 (and that could be found) will be a key in the dictionary. It will
313 include the original set of modules as well.
314
315 If a module cannot be found (e.g. its path was not given to the
316 ModuleManager, or it might be a system module for which the sources
317 are not available, a message will be printed, and this module will
318 be ignored (i.e. not listed in any dependencies).
319 # TODO 2120: allow a choice to abort or ignore.
320
321 :param set[str] all_mods: the set of all modules for which to collect
322 module dependencies.
323
324 :returns: a dictionary with all modules that are required (directly
325 or indirectly) for the modules in ``all_mods``.
326 :rtype: dict[str, set[str]]
327
328 '''
329 # This contains the mapping from each module name to the
330 # list of the dependencies and is returned as result:
331 module_dependencies = {}
332
333 # Work on a copy to avoid modifying the caller's set:
334 todo = all_mods.copy()
335
336 # This set contains module that could not be found (to avoid
337 # adding them to the todo list again
338 not_found = set()
339
340 while todo:
341 # Pick one (random) module to handle (convert to lowercase
342 # in case that the code use inconsistent capitalisation)
343 module = todo.pop().lower()
344
345 # Ignore any modules that we were asked to ignore
346 if module in self.ignores():
347 continue
348 try:
349 mod_deps = self.get_module_info(module).get_used_modules()
350 # Convert to set since we continue with a set
351 mod_deps = set(mod_deps)
352 except (FileNotFoundError, ModuleInfoError):
353 if module not in not_found:
354 # We don't have any information about this module,
355 # ignore it.
356 # TODO 2120: allow a choice to abort or ignore.
357 print(f"Could not find module '{module}'.")
358 not_found.add(module)
359 # Remove this module as dependencies from any other
360 # module in our todo list, so the final result will
361 # only contain known modules
362 for dep in module_dependencies.values():
363 if module in dep:
364 dep.remove(module)
365 continue
366
367 # Remove all dependencies which we don't know anything about:
368 mod_deps = mod_deps.difference(not_found)
369
370 # Add the dependencies of `module` to the result dictionary:
371 module_dependencies[module] = mod_deps
372
373 # Remove all dependencies from the list of new dependencies
374 # of `module` that have already been handled:
375 new_deps = mod_deps.difference(module_dependencies.keys())
376
377 # Then add these really new modules to the list of modules
378 # that still need to be handled
379 todo |= new_deps
380
381 return module_dependencies

384 def sort_modules(self, module_dependencies):
385 '''This function sorts the given dependencies so that all
386 dependencies of a module are before any module that
387 needs it. Input is a dictionary that contains all modules to
388 be sorted as keys, and the value for each module is the set
389 of dependencies that the module depends on.
390
391 :param module_dependencies: the list of modules required as keys, \
392 with all their dependencies as value.
393 :type module_dependencies: dict[str, set[str]]
394
395 :returns: the sorted list of modules.
396 :rtype: list[str]
397
398 '''
399 result = []
400
401 # Create a copy to avoid modifying the callers data structure:
402 todo = copy.deepcopy(module_dependencies)
403
404 # Consistency check: test that all dependencies listed are also
405 # a key in the list, otherwise there will be a dependency that
406 # breaks sorting. If an unknown dependency is detected, print
407 # a warning, and remove it (otherwise no sort order could be
408 # determined).
409 for module, dependencies in todo.items():
410 # Take a copy so we can modify the original set of dependencies:
411 dependencies_copy = dependencies.copy()
412 for dep in dependencies_copy:
413 if dep in todo:
414 continue
415 # Print a warning if this module is not supposed to be ignored
416 if dep not in self.ignores():
417 # TODO 2120: allow a choice to abort or ignore.
418 print(f"Cannot find module `{dep}` which is used by "
419 f"module '{module}'.")
420 dependencies.remove(dep)
421
422 while todo:
423 # Find one module that has no dependencies, which is the
424 # next module to be added to the results.
425 for mod, dep in todo.items():
426 if not dep:
427 break
428 else:
429 # If there is no module without a dependency, there
430 # is a circular dependency
431 print(f"Circular dependency - cannot sort "
432 f"module dependencies: {todo}")
433 # TODO 2120: allow a choice to abort or ignore.
434 # In this case pick a module with the least number of
435 # dependencies, the best we can do in this case - and
436 # it's better to provide all modules (even if they cannot)
437 # be sorted, than missing some.
438 all_mods_sorted = sorted((mod for mod in todo.keys()),
439 key=lambda x: len(todo[x]))
440 mod = all_mods_sorted[0]
441
442 # Add the current module to the result and remove it from
443 # the todo list.
444 result.append(mod)
445 del todo[mod]
446
447 # Then remove this module from the dependencies of all other
448 # modules:
449 for dep in todo.values():
450 if mod in dep:
451 dep.remove(mod)
452
453 return result
