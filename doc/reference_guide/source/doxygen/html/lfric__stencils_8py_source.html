51class LFRicStencils(LFRicCollection):
52 '''
53 Stencil information and code generation associated with a PSy-layer
54 routine or Kernel stub.
55
56 :param node: the Invoke or Kernel stub for which to provide stencil info.
57 :type node: :py:class:`psyclone.dynamo0p3.LFRicInvoke` or
58 :py:class:`psyclone.domain.lfric.LFRicKern`
59
60 :raises GenerationError: if a literal has been supplied for a stencil
61 direction.
62
63 '''
64 def __init__(self, node):
65 # pylint: disable=too-many-branches
66 super().__init__(node)
67
68 # List of arguments which have an extent value passed to this
69 # invoke routine from the algorithm layer. Duplicate argument
70 # names are removed.
71 self._unique_extent_args = []
72 extent_names = []
73 # pylint: disable=too-many-nested-blocks
74 for call in self._calls_calls:
75 for arg in call.arguments.args:
76 if arg.stencil:
77 # Check for the existence of arg.extent here as in
78 # the future we plan to support kernels which
79 # specify the value of extent in metadata. If this
80 # is the case then an extent argument is not
81 # required.
82 # TODO #963
83 if not arg.stencil.extent:
84 if not arg.stencil.extent_arg.is_literal():
85 if arg.stencil.extent_arg.text not in extent_names:
86 extent_names.append(
87 arg.stencil.extent_arg.text)
88 self._unique_extent_args.append(arg)
89
90 # A list of arguments that have a direction variable passed in
91 # to this invoke routine from the algorithm layer. Duplicate
92 # argument names are removed.
93 self._unique_direction_args = []
94 direction_names = []
95 for call in self._calls_calls:
96 for idx, arg in enumerate(call.arguments.args):
97 if arg.stencil and arg.stencil.direction_arg:
98 if arg.stencil.direction_arg.is_literal():
99 raise GenerationError(
100 f"Kernel {call.name}, metadata arg {idx}, a "
101 f"literal is not a valid value for a stencil "
102 f"direction.")
103 if arg.stencil.direction_arg.text.lower() not in \
104 ["x_direction", "y_direction"]:
105 if arg.stencil.direction_arg.text not in \
106 direction_names:
107 direction_names.append(
108 arg.stencil.direction_arg.text)
109 self._unique_direction_args.append(arg)
110
111 # List of stencil args with an extent variable passed in. The same
112 # field name may occur more than once here from different kernels.
113 self._kern_args_kern_args = []
114 for call in self._calls_calls:
115 for arg in call.arguments.args:
116 if arg.stencil:
117 if not arg.stencil.extent:
118 self._kern_args_kern_args.append(arg)

120 @staticmethod
121 def extent_value(arg):
122 '''
123 Returns the content of the stencil extent which may be a literal
124 value (a number) or a variable name. This function simplifies this
125 problem by returning a string in either case.
126
127 :param arg: the argument with which the stencil is associated.
128 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
129
130 :returns: the content of the stencil extent.
131 :rtype: str
132
133 '''
134 if arg.stencil.extent_arg.is_literal():
135 return arg.stencil.extent_arg.text
136 return arg.stencil.extent_arg.varname

138 @staticmethod
139 def stencil_unique_str(arg, context):
140 '''
141 Creates a unique identifier for a stencil. As a stencil
142 differs due to the function space it operates on, type of
143 stencil and extent of stencil, we concatenate these things together
144 to create a unique string.
145
146 :param arg: kernel argument with which stencil is associated.
147 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
148 :param str context: a context for this stencil (e.g. "size" or
149 "direction").
150
151 :returns: unique string identifying the stencil for this argument.
152 :rtype: str
153
154 :raises GenerationError: if an explicit stencil extent is found in
155 the metadata for the kernel argument.
156
157 '''
158 unique = context
159 unique += arg.function_space.mangled_name
160 unique += arg.descriptor.stencil['type']
161 if arg.descriptor.stencil['extent']:
162 raise GenerationError(
163 "Found a stencil with an extent specified in the metadata. "
164 "This is not coded for.")
165 unique += arg.stencil.extent_arg.text.lower()
166 if arg.descriptor.stencil['type'] == 'xory1d':
167 unique += arg.stencil.direction_arg.text.lower()
168 return unique

170 def map_name(self, arg):
171 '''
172 Creates and registers a name for the stencil map associated with the
173 supplied kernel argument.
174
175 :param arg: kernel argument with which the stencil is associated.
176 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
177
178 :returns: a valid unique map name for a stencil in the PSy layer.
179 :rtype: str
180
181 '''
182 root_name = arg.name + "_stencil_map"
183 unique = LFRicStencils.stencil_unique_str(arg, "map")
184 return self._symbol_table.find_or_create_tag(unique, root_name).name

186 @staticmethod
187 def dofmap_symbol(symtab, arg):
188 '''
189 Creates and registers a symbol for the stencil dofmap associated with
190 the supplied kernel argument.
191
192 :param symtab: symbol table that will contain (or already contains)
193 the symbol with this name.
194 :type symtab: :py:class:`psyclone.psyir.symbols.SymbolTable`
195 :param arg: kernel argument with which the stencil is associated.
196 :type arg: :py:class:`psyclone.dynamo0p3.DynKernelArgument`
197
198 :returns: a dofmap symbol for a stencil in the PSy layer.
199 :rtype: :py:class:`psyclone.psyir.symbols.Symbol`
200
201 '''
202 root_name = arg.name + "_stencil_dofmap"
203 unique = LFRicStencils.stencil_unique_str(arg, "dofmap")
204 if arg.descriptor.stencil['type'] == "cross2d":
205 num_dimensions = 4
206 else:
207 num_dimensions = 3
208 return symtab.find_or_create_array(root_name, num_dimensions,
209 ScalarType.Intrinsic.INTEGER,
210 tag=unique)

444 def initialise(self, parent):
445 '''
446 Adds in the code to initialise stencil dofmaps to the PSy layer.
447
448 :param parent: the node in the f2pygen AST to which to add the
449 initialisations.
450 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
451
452 :raises GenerationError: if an unsupported stencil type is encountered.
453
454 '''
455 if not self._kern_args_kern_args:
456 return
457
458 parent.add(CommentGen(parent, ""))
459 parent.add(CommentGen(parent, " Initialise stencil dofmaps"))
460 parent.add(CommentGen(parent, ""))
461 api_config = Config.get().api_conf("lfric")
462 stencil_map_names = []
463 const = LFRicConstants()
464 for arg in self._kern_args_kern_args:
465 map_name = self.map_name(arg)
466 if map_name not in stencil_map_names:
467 # Only initialise maps once.
468 stencil_map_names.append(map_name)
469 stencil_type = arg.descriptor.stencil['type']
470 symtab = self._symbol_table
471 if stencil_type == "xory1d":
472 direction_name = arg.stencil.direction_arg.varname
473 for direction in ["x", "y"]:
474 if_then = IfThenGen(parent, direction_name +
475 " .eq. " + direction +
476 "_direction")
477 if_then.add(
478 AssignGen(
479 if_then, pointer=True, lhs=map_name,
480 rhs=arg.proxy_name_indexed +
481 "%vspace%get_stencil_dofmap("
482 "STENCIL_1D" + direction.upper() +
483 ","+self.extent_value(arg)+")"))
484 parent.add(if_then)
485 elif stencil_type == "cross2d":
486 parent.add(
487 AssignGen(parent, pointer=True, lhs=map_name,
488 rhs=arg.proxy_name_indexed +
489 "%vspace%get_stencil_2D_dofmap(" +
490 "STENCIL_2D_CROSS" + "," +
491 self.extent_value(arg) + ")"))
492 # Max branch length in the CROSS2D stencil is used when
493 # defining the stencil_dofmap dimensions at declaration of
494 # the dummy argument in the kernel. This value is 1
495 # greater than the stencil extent as the central cell
496 # is included as part of the stencil_dofmap.
497 parent.add(
498 AssignGen(parent,
499 lhs=self.max_branch_length_name(symtab,
500 arg),
501 rhs=self.extent_value(arg) + " + 1_" +
502 api_config.default_kind["integer"]))
503 else:
504 try:
505 stencil_name = const.STENCIL_MAPPING[stencil_type]
506 except KeyError as err:
507 raise GenerationError(
508 f"Unsupported stencil type "
509 f"'{arg.descriptor.stencil['type']}' supplied. "
510 f"Supported mappings are "
511 f"{str(const.STENCIL_MAPPING)}") from err
512 parent.add(
513 AssignGen(parent, pointer=True, lhs=map_name,
514 rhs=arg.proxy_name_indexed +
515 "%vspace%get_stencil_dofmap(" +
516 stencil_name + "," +
517 self.extent_value(arg) + ")"))
518
519 parent.add(AssignGen(parent, pointer=True,
520 lhs=self.dofmap_symbol(symtab, arg).name,
521 rhs=map_name + "%get_whole_dofmap()"))
522
523 # Add declaration and look-up of stencil size
524 dofmap_size_name = self.dofmap_size_symbol(symtab, arg).name
525 parent.add(AssignGen(parent, pointer=True,
526 lhs=dofmap_size_name,
527 rhs=map_name + "%get_stencil_sizes()"))

529 def _declare_maps_invoke(self, parent):
530 '''
531 Declare all stencil maps in the PSy layer.
532
533 :param parent: the node in the f2pygen AST to which to add
534 declarations.
535 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
536
537 :raises GenerationError: if an unsupported stencil type is encountered.
538
539 '''
540 api_config = Config.get().api_conf("lfric")
541
542 if not self._kern_args_kern_args:
543 return
544
545 symtab = self._symbol_table
546 stencil_map_names = []
547 const = LFRicConstants()
548
549 for arg in self._kern_args_kern_args:
550 map_name = self.map_name(arg)
551
552 if map_name in stencil_map_names:
553 continue
554
555 stencil_map_names.append(map_name)
556 stencil_type = arg.descriptor.stencil['type']
557 if stencil_type == "cross2d":
558 smap_type = const.STENCIL_TYPE_MAP["stencil_2D_dofmap"]["type"]
559 smap_mod = const.STENCIL_TYPE_MAP[
560 "stencil_2D_dofmap"]["module"]
561 parent.add(UseGen(parent, name=smap_mod, only=True,
562 funcnames=[smap_type, "STENCIL_2D_CROSS"]))
563 parent.add(TypeDeclGen(parent, pointer=True,
564 datatype=smap_type,
565 entity_decls=[map_name +
566 " => null()"]))
567 parent.add(DeclGen(parent, datatype="integer",
568 kind=api_config.default_kind["integer"],
569 pointer=True,
570 entity_decls=[self.dofmap_symbol(symtab,
571 arg).name +
572 "(:,:,:,:) => null()"]))
573 dofmap_size_name = self.dofmap_size_symbol(symtab, arg).name
574 parent.add(DeclGen(parent, datatype="integer",
575 kind=api_config.default_kind["integer"],
576 pointer=True,
577 entity_decls=[f"{dofmap_size_name}(:,:) "
578 f"=> null()"]))
579 parent.add(DeclGen(parent, datatype="integer",
580 kind=api_config.default_kind["integer"],
581 entity_decls=[self.max_branch_length_name(
582 symtab, arg)]))
583 else:
584 smap_type = const.STENCIL_TYPE_MAP["stencil_dofmap"]["type"]
585 smap_mod = const.STENCIL_TYPE_MAP["stencil_dofmap"]["module"]
586 parent.add(UseGen(parent, name=smap_mod,
587 only=True, funcnames=[smap_type]))
588 if stencil_type == 'xory1d':
589 drct_mod = const.STENCIL_TYPE_MAP["direction"]["module"]
590 parent.add(UseGen(parent, name=drct_mod,
591 only=True, funcnames=["x_direction",
592 "y_direction"]))
593 parent.add(UseGen(parent, name=smap_mod,
594 only=True, funcnames=["STENCIL_1DX",
595 "STENCIL_1DY"]))
596 else:
597 try:
598 stencil_name = const.STENCIL_MAPPING[stencil_type]
599 except KeyError as err:
600 raise GenerationError(
601 f"Unsupported stencil type "
602 f"'{arg.descriptor.stencil['type']}' supplied. "
603 f"Supported mappings are "
604 f"{const.STENCIL_MAPPING}") from err
605 parent.add(UseGen(parent, name=smap_mod,
606 only=True, funcnames=[stencil_name]))
607
608 parent.add(TypeDeclGen(parent, pointer=True,
609 datatype=smap_type,
610 entity_decls=[map_name+" => null()"]))
611 parent.add(DeclGen(parent, datatype="integer",
612 kind=api_config.default_kind["integer"],
613 pointer=True,
614 entity_decls=[self.dofmap_symbol(symtab,
615 arg).name +
616 "(:,:,:) => null()"]))
617 dofmap_size_name = self.dofmap_size_symbol(symtab, arg).name
618 parent.add(DeclGen(parent, datatype="integer",
619 kind=api_config.default_kind["integer"],
620 pointer=True,
621 entity_decls=[f"{dofmap_size_name}(:) "
622 f"=> null()"]))

624 def _declare_maps_stub(self, parent):
625 '''
626 Add declarations for all stencil maps to a kernel stub.
627
628 :param parent: the node in the f2pygen AST representing the kernel
629 stub routine.
630 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
631
632 '''
633 api_config = Config.get().api_conf("lfric")
634
635 symtab = self._symbol_table
636 for arg in self._kern_args_kern_args:
637 if arg.descriptor.stencil['type'] == "cross2d":
638 parent.add(DeclGen(
639 parent, datatype="integer",
640 kind=api_config.default_kind["integer"], intent="in",
641 dimension=",".join([arg.function_space.ndf_name,
642 self.max_branch_length_name(
643 symtab, arg), "4"]),
644 entity_decls=[self.dofmap_symbol(symtab, arg).name]))
645 else:
