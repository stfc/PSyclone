64class ACCDirective(metaclass=abc.ABCMeta):
65 # pylint: disable=too-few-public-methods
66 '''
67 Base mixin class for all OpenACC directive statements.
68
69 This class is useful to provide a unique common ancestor to all the
70 OpenACC directives, for instance when traversing the tree with
71 `node.walk(ACCDirective)`
72
73 Note that classes inheriting from it must place the ACCDirective in
74 front of the other Directive node sub-class, so that the Python
75 MRO gives preference to this class's attributes.
76 '''
77 _PREFIX = "ACC"

217class ACCEnterDataDirective(ACCStandaloneDirective):
218 '''
219 Class representing a "!$ACC enter data" OpenACC directive in
220 an InvokeSchedule. Must be sub-classed for a particular API because the way
221 in which fields are marked as being on the remote device is API-dependent.
222
223 :param children: list of nodes which the directive should have as children.
224 :type children: List[:py:class:`psyclone.psyir.nodes.Node`]
225 :param parent: the node in the InvokeSchedule to which to add this \
226 directive as a child.
227 :type parent: :py:class:`psyclone.psyir.nodes.Node`
228 '''
229 def __init__(self, children=None, parent=None):
230 super().__init__(children=children, parent=parent)
231 self._acc_dirs = None # List of parallel directives
232
233 self._sig_set = set()
234
235 def gen_code(self, parent):
236 '''Generate the elements of the f2pygen AST for this Node in the
237 Schedule.
238
239 :param parent: node in the f2pygen AST to which to add node(s).
240 :type parent: :py:class:`psyclone.f2pygen.BaseGen`
241
242 :raises GenerationError: if no data is found to copy in.
243
244 '''
245 self.validate_global_constraints()
246 self.lower_to_language_levellower_to_language_level()
247 # Leverage begin_string() to raise an exception if there are no
248 # variables to copyin but discard the generated string since it is
249 # incompatible with class DirectiveGen() we are using below.
250 self.begin_string()
251
252 # Add the enter data directive.
253 sym_list = _sig_set_to_string(self._sig_set)
254 copy_in_str = f"copyin({sym_list})"
255 parent.add(DirectiveGen(parent, "acc", "begin", "enter data",
256 copy_in_str))
257 # Call an API-specific subclass of this class in case
258 # additional declarations are required.
259 self.data_on_device(parent)
260 parent.add(CommentGen(parent, ""))

262 def lower_to_language_level(self):
263 '''
264 In-place replacement of this directive concept into language level
265 PSyIR constructs.
266
267 :returns: the lowered version of this node.
268 :rtype: :py:class:`psyclone.psyir.node.Node`
269
270 '''
271 # We must generate a list of all of the fields accessed within OpenACC
272 # compute constructs (i.e. OpenACC parallel and kernels directives)
273 # 1. Find all parallel and kernels directives. We store this list
274 # for later use in any sub-class.
275 self._acc_dirs = self.ancestor(Routine).walk(
276 (ACCParallelDirective, ACCKernelsDirective))
277 # 2. For each directive, add the fields used by the kernels it
278 # contains (as given by signatures) and add it to our set.
279 # TODO GOcean grid properties are duplicated in this set under
280 # different names (the OpenACC deep copy support should spot this).
281 for pdir in self._acc_dirs:
282 self._sig_set.update(*pdir.signatures)
283
284 return super().lower_to_language_level()

287 '''Returns the beginning statement of this directive. The visitor is
288 responsible for adding the correct directive beginning (e.g. "!$").
289
290 :returns: the opening statement of this directive.
291 :rtype: str
292
293 :raises GenerationError: if there are no variables to copy to
294 the device.
295 '''
296 if not self._sig_set:
297 # There should be at least one variable to copyin.
298 # TODO #1872: this directive needs reimplementing using the Clause
299 # class and proper lowering. When this is fixed it may be possible
300 # to change RegionTrans.validate() so that it always uses
301 # Node.debug_string() rather than only for CodeBlocks.
302 raise GenerationError(
303 "ACCEnterData directive did not find any data to copyin. "
304 "Perhaps there are no ACCParallel or ACCKernels directives "
305 "within the region?")
306
307 sym_list = _sig_set_to_string(self._sig_set)
308
309 # Variables need lexicographic sorting since sets guarantee no ordering
310 # and members of composite variables must appear later in deep copies.
311 return f"acc enter data copyin({sym_list})"

426class ACCLoopDirective(ACCRegionDirective):
427 '''
428 Class managing the creation of a '!$acc loop' OpenACC directive.
429
430 :param int collapse: Number of nested loops to collapse into a single
431 iteration space or None.
432 :param bool independent: Whether or not to add the `independent` clause
433 to the loop directive.
434 :param bool sequential: whether or not to add the `seq` clause to the
435 loop directive.
436 :param bool gang: whether or not to add the `gang` clause to the
437 loop directive.
438 :param bool vector: whether or not to add the `vector` clause to the
439 loop directive.
440 :param kwargs: additional keyword arguments provided to the super class.
441 :type kwargs: unwrapped dict.
442 '''
443 def __init__(self, collapse=None, independent=True, sequential=False,
444 gang=False, vector=False, **kwargs):
445 self.collapsecollapsecollapsecollapse = collapse
446 self._independent = independent
447 self._sequential = sequential
448 self._gang = gang
449 self._vector = vector
450 self._check_clauses_consistent()
451 super().__init__(**kwargs)
452
453 def __eq__(self, other):
454 '''
455 Checks whether two nodes are equal. Two ACCLoopDirective nodes are
456 equal if their collapse, independent, sequential, gang, and vector
457 members are equal.
458
459 :param object other: the object to check equality to.
460
461 :returns: whether other is equal to self.
462 :rtype: bool
463 '''
464 is_eq = super().__eq__(other)
465 is_eq = is_eq and self.collapsecollapsecollapsecollapse == other.collapse
466 is_eq = is_eq and self.independentindependent == other.independent
467 is_eq = is_eq and self.sequentialsequential == other.sequential
468 is_eq = is_eq and self.ganggang == other.gang
469 is_eq = is_eq and self.vectorvectorvector == other.vector
470
471 return is_eq
472
473 def _check_clauses_consistent(self):
474 '''
475 Check that the clauses applied to the loop directive make sense.
476
477 :raises ValueError: if sequential is used in conjunction with gang
478 and/or vector
479 '''
480 if self.sequentialsequential and (self.ganggang or self.vectorvectorvector):
481 raise ValueError(
482 "The OpenACC seq clause cannot be used in conjunction with the"
483 " gang or vector clauses."
484 )

1006 '''
1007 OpenACC directive to represent that the memory accesses in the associated
1008 assignment must be performed atomically.
1009 Note that the standard supports blocks with 2 assignments but this is
1010 currently unsupported in the PSyIR.
1011
1012 '''
1013 def begin_string(self):

1015 :returns: the opening string statement of this directive.
1016 :rtype: str

1019 return "acc atomic"

1023 :returns: the ending string statement of this directive.
1024 :rtype: str
1025

1029 @staticmethod
1030 def is_valid_atomic_statement(stmt):
1031 ''' Check if a given statement is a valid OpenACC atomic expression.
1032
1033 :param stmt: a node to be validated.
1034 :type stmt: :py:class:`psyclone.psyir.nodes.Node`
1035
1036 :returns: whether a given statement is compliant with the OpenACC
1037 atomic expression.
1038 :rtype: bool
1039
1040 '''
1041 if not isinstance(stmt, Assignment):
1042 return False
1043
1044 # Not all rules are checked, just that:
1045 # - operands are of a scalar intrinsic type
1046 if not isinstance(stmt.lhs.datatype, ScalarType):
1047 return False
1048
1049 # - the top-level operator is one of: +, *, -, /, AND, OR, EQV, NEQV
1050 if isinstance(stmt.rhs, BinaryOperation):
1051 if stmt.rhs.operator not in (BinaryOperation.Operator.ADD,
1052 BinaryOperation.Operator.SUB,
1053 BinaryOperation.Operator.MUL,
1054 BinaryOperation.Operator.DIV,
1055 BinaryOperation.Operator.AND,
1056 BinaryOperation.Operator.OR,
1057 BinaryOperation.Operator.EQV,
1058 BinaryOperation.Operator.NEQV):
1059 return False
1060 # - or intrinsics: MAX, MIN, IAND, IOR, or IEOR
1061 if isinstance(stmt.rhs, IntrinsicCall):
1062 if stmt.rhs.intrinsic not in (IntrinsicCall.Intrinsic.MAX,
1063 IntrinsicCall.Intrinsic.MIN,
1064 IntrinsicCall.Intrinsic.IAND,
1065 IntrinsicCall.Intrinsic.IOR,
1066 IntrinsicCall.Intrinsic.IEOR):
1067 return False
1068
1069 # - one of the operands should be the same as the lhs
1070 if stmt.lhs not in stmt.rhs.children:
1071 return False
1072
1073 return True
