52class Loop(Statement):
53 # pylint: disable=too-many-instance-attributes
54 '''Node representing a loop within the PSyIR. It has 4 mandatory children:
55 the first one represents the loop lower bound, the second one represents
56 the loop upper bound, the third one represents the step value and the
57 fourth one is always a PSyIR Schedule node containing the statements inside
58 the loop body.
59
60 (Note: Loop only represents the equivalent to Fortran counted do loops.
61 This means the loop is bounded by start/stop/step expressions evaluated
62 before the loop starts. See WhileLoop for while loops, including the
63 Fortran do while and do loop with no condition.)
64
65 :param variable: optional reference to the loop iterator \
66 variable. Defaults to None.
67 :type variable: Optional[:py:class:`psyclone.psyir.symbols.DataSymbol`]
68 :param annotations: One or more labels that provide additional information\
69 about the node (primarily relating to the input code that it was \
70 created from).
71 :type annotations: Optional[List[str]]
72 :param kwargs: additional keyword arguments provided to the PSyIR node.
73 :type kwargs: unwrapped dict.
74
75 :raises InternalError: if the 'was_single_stmt' annotation is supplied \
76 without the 'was_where' annotation.
77
78 '''
79 valid_annotations = ('was_where', 'was_single_stmt', 'chunked')
80 # Textual description of the node.
81 _children_valid_format = "DataNode, DataNode, DataNode, Schedule"
82 _text_name = "Loop"
83 _colour = "red"
84
85 # Set of rules that give a loop a certain loop_type by inspecting
86 # its variable name
87 _loop_type_inference_rules = {}

89 def __init__(self, variable=None, annotations=None, **kwargs):
90 # Although the base class checks on the annotations individually, we
91 # need to do further checks here
92 if annotations:
93 if 'was_single_stmt' in annotations and \
94 'was_where' not in annotations:
95 raise InternalError(
96 f"A Loop with the 'was_single_stmt' annotation "
97 f"must also have the 'was_where' annotation but"
98 f" got: {annotations}")
99
100 super().__init__(self, annotations=annotations, **kwargs)
101 # Call the variable setter for error checking
102 self._variable_variable = None
103 if variable is not None:
104 self.variablevariablevariable = variable
105 # Hold the set of symbols that will be private/local to the interation
106 # if this loop is run concurrently. Alternatively this could be
107 # implemented by moving the symbols to the loop_body symbol table.
108 self._explicitly_private_symbols = set()

110 def __eq__(self, other):
111 '''
112 Checks whether two nodes are equal. Two Loop nodes are equal
113 if they have the same iteration variable and their children are
114 equal.
115
116 :param object other: the object to check equality to.
117
118 :returns: whether other is equal to self.
119 :rtype: bool
120 '''
121 is_eq = super().__eq__(other)
122 # Similar to Reference equality, it is enough to compare the name
123 # since if the same-named symbols represent the same is already
124 # done in their respective scope symbol_table equality check.
125 is_eq = is_eq and self.variablevariablevariable.name == other.variable.name
126
127 return is_eq

148 @classmethod
149 def set_loop_type_inference_rules(cls, rules):
150 '''
151 Specify the rules that define a loop type by inspecting its variable,
152 name. This affects all instances of the Loop class. For example:
153
154 .. code-block::
155
156 rules = {
157 "lon": {"variable": "ji"},
158 "lat": {"variable": "jj"}
159 }
160
161 :param rules: new set of rules for inferring loop_types.
162 :type rules: dict[str, dict[str, str]]
163 '''
164 if rules is None:
165 cls._loop_type_inference_rules = {}
166 return
167
168 # Check that the provided rules have the right format
169 if not isinstance(rules, dict):
170 raise TypeError(f"The rules argument must be of type 'dict' but "
171 f"found '{type(rules)}'.")
172 for key, rule in rules.items():
173 if not isinstance(key, str):
174 raise TypeError(f"The rules keys must be of type 'str' but "
175 f"found '{type(key)}'.")
176 if not isinstance(rule, dict):
177 raise TypeError(f"The rules values must be of type 'dict' but "
178 f"found '{type(rule)}'.")
179 for rkey, value in rule.items():
180 if not isinstance(rkey, str) or not isinstance(value, str):
181 raise TypeError(
182 f"All the values of the rule definition must be "
183 f"of type 'str' but found '{rule}'.")
184 if rkey != "variable":
185 raise TypeError(f"Currently only the 'variable' rule key"
186 f" is accepted, but found: '{rkey}'.")
187 if "variable" not in rule:
188 raise TypeError(f"A rule must at least have a 'variable' field"
189 f" to specify the loop variable name that "
190 f"defines this loop_type, but the rule for "
191 f"'{key}' does not have it.")
192
193 # Convert the rules to a dictionary with variable as a key
194 inference_rules = {}
195 for key, rule in rules.items():
196 inference_rules[rule["variable"]] = key
197 cls._loop_type_inference_rules = inference_rules

244 def create(cls, variable, start, stop, step, children):
245 # pylint: disable=too-many-arguments
246 '''Create a Loop instance given valid instances of a variable,
247 start, stop and step nodes, and a list of child nodes for the
248 loop body.
249
250 :param variable: the PSyIR node containing the variable \
251 of the loop iterator.
252 :type variable: :py:class:`psyclone.psyir.symbols.DataSymbol`
253 :param start: the PSyIR node determining the value for the \
254 start of the loop.
255 :type start: :py:class:`psyclone.psyir.nodes.Node`
256 :param end: the PSyIR node determining the value for the end \
257 of the loop.
258 :type end: :py:class:`psyclone.psyir.nodes.Node`
259 :param step: the PSyIR node determining the value for the loop \
260 step.
261 :type step: :py:class:`psyclone.psyir.nodes.Node`
262 :param children: a list of PSyIR nodes contained in the \
263 loop.
264 :type children: list of :py:class:`psyclone.psyir.nodes.Node`
265
266 :returns: a Loop instance.
267 :rtype: :py:class:`psyclone.psyir.nodes.Loop`
268
269 :raises GenerationError: if the arguments to the create method \
270 are not of the expected type.
271
272 '''
273 cls._check_variable(variable)
274
275 if not isinstance(children, list):
276 raise GenerationError(
277 f"children argument in create method of Loop class "
278 f"should be a list but found '{type(children).__name__}'.")
279
280 loop = cls(variable=variable)
281 schedule = Schedule(parent=loop, children=children)
282 loop.children = [start, stop, step, schedule]
283 return loop

430 def replace_symbols_using(self, table):
431 '''
432 Replace the Symbol referred to by this object's `variable` and
433 `explicit_local_symbols` properties with those in the supplied
434 SymbolTable with a matching name. If there is no matches then they
435 are left unchanged.
436
437 :param table: symbol table in which to look up the replacement symbol.
438 :type table: :py:class:`psyclone.psyir.symbols.SymbolTable`
439
440 '''
441 if self._variable_variable:
442 try:
443 new_sym = table.lookup(self._variable_variable.name)
444 self.variablevariablevariable = new_sym
445 except KeyError:
446 pass
447
448 for symbol in list(self._explicitly_private_symbols):
449 try:
450 new_sym = table.lookup(symbol.name)
451 self._explicitly_private_symbols.remove(symbol)
452 self._explicitly_private_symbols.add(new_sym)
453 except KeyError:
454 pass
455 super().replace_symbols_using(table)

473 def reference_accesses(self, var_accesses):
474 '''Get all variable access information. It combines the data from
475 the loop bounds (start, stop and step), as well as the loop body.
476 The loop variable is marked as 'READ+WRITE' and references in start,
477 stop and step are marked as 'READ'.
478
479 :param var_accesses: VariablesAccessInfo instance that stores the \
480 information about variable accesses.
481 :type var_accesses: \
482 :py:class:`psyclone.core.VariablesAccessInfo`
483 '''
484
485 # Only add the loop variable and start/stop/step values if this is
486 # not an LFRic domain loop. We need to access the variable directly
487 # to avoid a crash in the getter if the loop variable is not defined.
488 if self._variable_variable:
489 # It is important to first add the WRITE access, since this way
490 # the dependency analysis for declaring openmp private variables
491 # will automatically declare the loop variables to be private
492 # (write access before read)
493 var_accesses.add_access(Signature(self.variablevariablevariable.name),
494 AccessType.WRITE, self)
495 var_accesses.add_access(Signature(self.variablevariablevariable.name),
496 AccessType.READ, self)
497
498 # Accesses of the start/stop/step expressions
499 self.start_exprstart_expr.reference_accesses(var_accesses)
500 self.stop_exprstop_expr.reference_accesses(var_accesses)
501 self.step_exprstep_expr.reference_accesses(var_accesses)
502 var_accesses.next_location()
503
504 for child in self.loop_body.children:
505 child.reference_accesses(var_accesses)
506 var_accesses.next_location()

508 def independent_iterations(self,
509 test_all_variables=False,
510 signatures_to_ignore=None,
511 dep_tools=None):
512 '''This function analyses a loop in the PSyIR to see whether

514

516 accesses are independent, otherwise it will stop after the first
517 variable access is found that isn't.
518 :param signatures_to_ignore: list of signatures for which to skip
519 the access checks.
520 :type signatures_to_ignore: Optional[
521 List[:py:class:`psyclone.core.Signature`]]
522 :param dep_tools: an optional instance of DependencyTools so that the
523 caller can access any diagnostic messages detailing why the loop
524 iterations are not independent.
525 :type dep_tools: Optional[
526 :py:class:`psyclone.psyir.tools.DependencyTools`]
527
528 :returns: True if the loop iterations are independent, False otherwise.
529 :rtype: bool
530
531 '''
532 if not dep_tools:
533 # pylint: disable=import-outside-toplevel
534 from psyclone.psyir.tools import DependencyTools
535 dtools = DependencyTools()
536 else:
537 dtools = dep_tools
538 return dtools.can_loop_be_parallelised(
539 self, test_all_variables=test_all_variables,
540 signatures_to_ignore=signatures_to_ignore)

543 '''
544 Generate the Fortran Loop and any associated code.
545
546 :param parent: the node in the f2pygen AST to which to add content.
547 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
548

551 # pylint: disable=import-outside-toplevel

559 # that the lowering changes are not reflected in the provided node.

561 # to ancestors will be lost here because the ancestors use gen_code
562 # instead of lowering+backend.

564 # table to be updated.
565 from psyclone.psyGen import CodedKern

568 if kernel.modified:
569 kernel.rename_and_write()

577 # to ancestors will be lost here because the ancestors use gen_code
578 # instead of lowering+backend.
579 # Therefore we need to replicate the lowering ancestor changes
580 # manually here (all this can be removed when the invoke schedule also
581 # uses the lowering+backend), these are:

595 entity_decls=[name])

599 for kernel in self.walk(CodedKern):
600 if not kernel.module_inline:
601 parent.add(UseGen(parent, name=kernel.module_name, only=True,
602 funcnames=[kernel.name]))
