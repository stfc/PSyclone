38'''Module providing a transformation from a PSyIR MATMUL operator to
39PSyIR code. This could be useful if the MATMUL operator is not
40supported by the back-end or if the performance in the inline code is
41better than the intrinsic. MATMUL supports both matrix multiply and
42matrix vector multiply. This transformation supports both with the
43restriction that the first matrix must be of at least rank 2.
44
45'''

50 DataSymbol, INTEGER_TYPE, REAL_TYPE, ArrayType, UnsupportedType)
51from psyclone.psyir.transformations.intrinsics.intrinsic2code_trans import (
52 Intrinsic2CodeTrans)

55def _create_array_ref(array_symbol, loop_idx_symbols, other_dims,
56 loop_idx_order, other_dims_order):
57 '''
58 Utility function to create a reference to an array element being
59 accessed using one or more loop indices followed by zero or more
60 other index expressions.
61
62 :param array_symbol: the symbol representing the array.
63 :type array_symbol: :py:class:`psyclone.psyir.symbols.DataSymbol`
64 :param loop_idx_symbols: loop indices which will index into the array.
65 :type loop_idx_symbols: List[:py:class:`psyclone.psyir.symbols.DataSymbol`]
66 :param other_dims: index expressions for any other dimensions that are
67 not being looped over. (If there are none then this
68 must be an empty list.)
69 :type other_dims: List[:py:class:`psyclone.psyir.nodes.ExpressionNode`]
70 :param List[int] loop_idx_order: list of indices in the original array
71 where the loop_idx_symbols are.
72 :param List[int] other_dims_order: list of indices in the original array
73 where the other_dims are.
74
75 :returns: the new reference to an array element.
76 :rtype: :py:class:`psyclone.psyir.nodes.ArrayReference`
77
78 '''
79 # If there are no other dims, then we simply create
80 # an array with the loop_idx_symbols in order.
81 if len(other_dims) == 0:
82 indices = [Reference(sym) for sym in loop_idx_symbols]
83 else:
84 n_indices = len(loop_idx_symbols) + len(other_dims)
85 indices = [-1]*n_indices
86 for it, order_idx in enumerate(loop_idx_order):
87 indices[order_idx] = Reference(loop_idx_symbols[it])
88 # Fill the remaining indices with the other dims
89 for it, order_idx in enumerate(other_dims_order):
90 indices[order_idx] = other_dims[it].copy()
91 return ArrayReference.create(array_symbol, indices)

94def _get_array_bound(array, index):
95 '''A utility function that returns the appropriate loop bounds (lower,
96 upper and step) for a particular index of an array reference. At
97 the moment all entries for the index are assumed to be accessed.
98 If the array symbol is declared with known bounds (an integer or a
99 symbol) then these bound values are used. If the size is unknown
100 (a deferred or attribute type) then the LBOUND and UBOUND PSyIR
101 nodes are used.
102
103 :param array: the reference that we are interested in.
104 :type array: :py:class:`psyir.nodes.Reference`
105 :param int index: the (array) reference index that we are
106 interested in.
107 :returns: the loop bounds for this array index.
108 :rtype: (Literal, Literal, Literal) or
109 (BinaryOperation, BinaryOperation, Literal)
110
111 :raises TransformationError: if the shape of the array's symbol is
112 not supported.
113
114 '''
115 # The 'shape' getter performs validation checks.
116 try:
117 my_dim = array.symbol.shape[index]
118 except TypeError as err:
119 # Added import here to avoid circular dependencies.
120 # pylint: disable=import-outside-toplevel
121 from psyclone.psyir.transformations import TransformationError
122 raise TransformationError(
123 f"Unsupported index type found for array '{array.name}': "
124 f"{err}") from err
125
126 dim_index = index + 1 # The Fortran dim argument is 1-indexed
127 if isinstance(my_dim, ArrayType.ArrayBounds):
128 # Use .copy() to ensure we return new nodes.
129 lower_bound = my_dim.lower.copy()
130 if my_dim.upper == ArrayType.Extent.ATTRIBUTE:
131 # Assumed-shape array.
132 upper_bound = IntrinsicCall.create(
133 IntrinsicCall.Intrinsic.UBOUND,
134 [Reference(array.symbol),
135 ("dim", Literal(str(dim_index), INTEGER_TYPE))])
136 else:
137 upper_bound = my_dim.upper.copy()
138 else:
139 lower_bound = IntrinsicCall.create(
140 IntrinsicCall.Intrinsic.LBOUND,
141 [Reference(array.symbol),
142 ("dim", Literal(str(dim_index), INTEGER_TYPE))])
143 upper_bound = IntrinsicCall.create(
144 IntrinsicCall.Intrinsic.UBOUND,
145 [Reference(array.symbol),
146 ("dim", Literal(str(dim_index), INTEGER_TYPE))])
147
148 step = Literal("1", INTEGER_TYPE)
149 return (lower_bound, upper_bound, step)

152class Matmul2CodeTrans(Intrinsic2CodeTrans):
153 '''Provides a transformation from a PSyIR MATMUL Operator node to
154 equivalent code in a PSyIR tree. Validity checks are also
155 performed.
156
157 For a matrix-vector multiplication, if the dimensions of ``R``, ``A``,
158 and ``B`` are ``R(N)``, ``A(N,M)``, ``B(M)``, the transformation replaces:
159
160 .. code-block:: fortran
161
162 R=MATMUL(A,B)
163
164 with the following code:
165
166 .. code-block:: fortran
167
168 do i=1,N
169 R(i) = 0.0
170 do j=1,M
171 R(i) = R(i) + A(i,j) * B(j)
172
173 For a matrix-matrix multiplication, if the dimensions of ``R``, ``A``,
174 and ``B`` are ``R(P,M)``, ``A(P,N)``, ``B(N,M)``, the MATMUL is replaced
175 with the following code:
176
177 .. code-block:: fortran
178
179 do j=1,M
180 do i=1,P
181 R(i,j) = 0.0
182 do ii=1,N
183 R(i,j) = R(i,j) + A(i,ii) * B(ii,j)
184
185 Note that this transformation does *not* support the case where ``A`` is
186 a rank-1 array.
187
188 '''
189 def __init__(self):
190 super().__init__()
191 self._intrinsic_intrinsic = IntrinsicCall.Intrinsic.MATMUL

193 def _get_full_range_split(self, array):
194 '''A utility function that returns the number of full ranges
195 and the list of indices which are not full ranges.
196
197 :param array: the reference that we are interested in.
198 :type array: :py:class:`psyir.nodes.Reference`
199 :returns: tuple with list of full range indices,
200 the list of non full range indices, and
201 list of non full range nodes for this array.
202 :rtype: (List[int], List[int], List[psyclone.psyir.nodes.DataNode])
203
204 :raises TransformationError: if any Range nodes are not full, or
205 if there are more than two full range nodes.
206
207 '''
208 from psyclone.psyir.transformations import TransformationError
209 non_full_ranges = []
210 full_range_order = []
211 non_full_range_order = []
212 for it, idx in enumerate(array.children):
213 if isinstance(idx, Range):
214 if array.is_full_range(it):
215 full_range_order.append(it)
216 else:
217 raise TransformationError(
218 f"To use {self.name} on matmul, each Range index "
219 f"of the array '{array.debug_string()}' "
220 f"must be a full range but found "
221 f"non full range at position {it}.")
222 else:
223 non_full_ranges.append(idx)
224 non_full_range_order.append(it)
225
226 # Error raising if we go above 2 full ranges
227 n_full_ranges = len(full_range_order)
228 if n_full_ranges > 2:
229 raise TransformationError(
230 f"To use {self.name} on matmul, no more than "
231 f"two indices of the array '{array.debug_string()}' "
232 f"must be full ranges but found {n_full_ranges}.")
233 return (full_range_order, non_full_range_order, non_full_ranges)

235 def _validate_array_full_ranges(self, array, permitted_full_ranges):
236 '''A utility function that checks if the number of full ranges
237 of an array is allowed.
238
239 :param array: the reference that we are interested in.
240 :type array: :py:class:`psyir.nodes.Reference`
241 :param List[int] permitted_full_ranges: the permitted number
242 of full ranges for array
243
244 :raises TransformationError: if the number of full ranges
245 is not permitted.
246
247 '''
248 from psyclone.psyir.transformations import TransformationError
249 # Make sure the array has as many full ranges as needed
250 if (len(array.symbol.shape) in permitted_full_ranges
251 and not array.children):
252 # If the array only has 1 or 2 dimensions and all of its
253 # data is used in the matrix multiply then the reference does
254 # not need to supply any dimension information.
255 pass
256 else:
257 # There should be one index per dimension. This is enforced
258 # by the array create method so is not tested here.
259 full_range_order, _, _ = self._get_full_range_split(array)
260 n_full_ranges = len(full_range_order)
261 if n_full_ranges not in permitted_full_ranges:
262 # Formatting error message nicely.
263 if len(permitted_full_ranges) == 1:
264 permit_str = f"{permitted_full_ranges[0]}"
265 else:
266 permit_str = f"either {permitted_full_ranges[0]} " \
267 f"or {permitted_full_ranges[1]}"
268 raise TransformationError(
269 f"To use {self.name} on matmul, {permit_str} "
270 f"indices of the array '{array.debug_string()}' "
271 f"must be full ranges but found {n_full_ranges}.")

273 def validate(self, node, options=None):
274 '''Perform checks to ensure that it is valid to apply the
275 Matmul2CodeTran transformation to the supplied node.
276
277 :param node: the node that is being checked.
278 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
279 :param options: options for the transformation.
280 :type options: Optional[Dict[str, Any]]
281
282 :raises TransformationError: if the node argument is not the
283 expected type.
284 :raises TransformationError: if the parent of the MATMUL
285 operation is not an assignment.
286 :raises TransformationError: if the matmul arguments are not in
287 the required form.
288 :raises TransformationError: if sub-sections of an array are present
289 in the arguments.
290
291 '''
292 # pylint: disable=too-many-branches
293
294 # Import here to avoid circular dependencies.
295 # pylint: disable=import-outside-toplevel
296 from psyclone.psyir.transformations import TransformationError
297
298 super().validate(node, options)
299
300 # Check the matmul is the only code on the rhs of an assignment
301 # i.e. ... = matmul(a,b)
302 if not isinstance(node.parent, Assignment):
303 raise TransformationError(
304 "Matmul2CodeTrans only supports the transformation of a "
305 "MATMUL operation when it is the sole operation on the rhs "
306 "of an assignment.")
307
308 matrix1 = node.arguments[0]
309 matrix2 = node.arguments[1]
310 result = node.parent.lhs
311
312 # The arguments of matvec should be References
313 if not (isinstance(matrix1, Reference) and
314 isinstance(matrix2, Reference) and
315 isinstance(result, Reference)):
316 raise TransformationError(
317 f"Expected result and operands of MATMUL IntrinsicCall to "
318 f"be references, but found: '{node.parent.debug_string()}'.")
319
320 # The arguments of matvec should be References to arrays
321 if any(isinstance(var.symbol.datatype, UnsupportedType) for var
322 in [matrix1, matrix2, result]):
323 raise TransformationError(
324 f"Must have full type information for result and operands of "
325 f"MATMUL IntrinsicCall but found '{result.symbol}', "
326 f"'{matrix1.symbol}' and '{matrix2.symbol}'.")
327 if (len(matrix1.symbol.shape) == 0 or len(matrix2.symbol.shape) == 0 or
328 len(result.symbol.shape) == 0):
329 raise TransformationError(
330 f"Expected result and operands of MATMUL IntrinsicCall to "
331 f"be references to arrays but found '{result.symbol}', "
332 f"'{matrix1.symbol}' and '{matrix2.symbol}'.")
333
334 # The first child (matrix1) should be declared as an array
335 # with at least 2 dimensions.
336 if len(matrix1.symbol.shape) < 2:
337 raise TransformationError(
338 f"Expected 1st child of a MATMUL IntrinsicCall to be a "
339 f"matrix with at least 2 dimensions, but found "
340 f"'{len(matrix1.symbol.shape)}'.")
341
342 if len(matrix1.symbol.shape) > 2 and not matrix1.children:
343 # If matrix1 has no children then it is a reference. If
344 # it is a reference then the number of dimensions must be 2.
345 raise TransformationError(
346 f"Expected 1st child of a MATMUL IntrinsicCall to have 2 "
347 f"dimensions, but found '{len(matrix1.symbol.shape)}'.")
348 self._validate_array_full_ranges(matrix1, [2])
349
350 if len(matrix2.symbol.shape) > 2 and not matrix2.children:
351 # If matrix2 has no children then it is a reference. If
352 # it is a reference then the number of dimensions must be
353 # 1 or 2.
354 raise TransformationError(
355 f"Expected 2nd child of a MATMUL IntrinsicCall to have 1 "
356 f"or 2 dimensions, but found '{len(matrix2.symbol.shape)}'.")
357 self._validate_array_full_ranges(matrix2, [1, 2])
358
359 # Make sure the result has as many full ranges as needed
360 self._validate_array_full_ranges(result, [1, 2])
361
362 # Make sure the result is not one of the MATMUL operands
363 if result.symbol in (matrix1.symbol, matrix2.symbol):
364 raise TransformationError(
365 f"'{result.symbol.name}' is the result location and one of "
366 f"the MATMUL operators. This is not supported.")

368 def apply(self, node, options=None):
369 '''Apply the MATMUL intrinsic conversion transformation to the
370 specified node. This node must be a MATMUL IntrinsicCall. The first
371 argument must currently have two dimensions while the second must have
372 either one or two dimensions. Each argument is permitted to have
373 additional dimensions (i.e. more than 2) but in each case it is only
374 the first one or two which may be ranges. Further, the ranges must
375 currently be for the full index space for that dimension (i.e. array
376 subsections are not supported). If the transformation is
377 successful then an assignment which includes a MATMUL
378 IntrinsicCall node is converted to equivalent inline code.
379
380 :param node: a MATMUL IntrinsicCall node.
381 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
382 :param options: options for the transformation.
383 :type options: Optional[Dict[str, Any]]
384
385 '''
386 self.validatevalidatevalidate(node, options)
387
388 arg2 = node.arguments[1]
389 full_range_order, _, _ = self._get_full_range_split(arg2)
390 n_full_ranges = len(full_range_order)
391 if (n_full_ranges > 1 or
392 not arg2.children and len(arg2.symbol.shape) == 2):
393 self._apply_matrix_matrix(node)
394 else:
395 self._apply_matrix_vector(node)

397 def _apply_matrix_vector(self, node):
398 '''
399 Apply the transformation for the case of a matrix-vector
400 multiplication.
401
402 :param node: a MATMUL IntrinsicCall node.
403 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
404
405 '''
406 # pylint: disable=too-many-locals
407 assignment = node.parent
408 matrix = node.arguments[0]
409 vector = node.arguments[1]
410 result = node.parent.lhs
411
412 # Create new i and j loop iterators.
413 symbol_table = node.scope.symbol_table
414 i_loop_sym = symbol_table.new_symbol("i", symbol_type=DataSymbol,
415 datatype=INTEGER_TYPE)
416 j_loop_sym = symbol_table.new_symbol("j", symbol_type=DataSymbol,
417 datatype=INTEGER_TYPE)
418
419 # Create "result(i)"
420 r_fr_order, r_nfr_order, r_nfr = self._get_full_range_split(result)
421 result_ref = _create_array_ref(result.symbol,
422 [i_loop_sym],
423 r_nfr,
424 r_fr_order,
425 r_nfr_order)
426 # Create "vector(j)"
427 v_fr_order, v_nfr_order, v_nfr = self._get_full_range_split(vector)
428 vector_array_reference = _create_array_ref(vector.symbol,
429 [j_loop_sym],
430 v_nfr,
431 v_fr_order,
432 v_nfr_order)
433 m_fr_order, m_nfr_order, m_nfr = self._get_full_range_split(matrix)
434 # Create "matrix(i,j)"
435 matrix_array_reference = _create_array_ref(matrix.symbol,
436 [i_loop_sym, j_loop_sym],
437 m_nfr,
438 m_fr_order,
439 m_nfr_order)
440 # Create "matrix(i,j) * vector(j)"
441 multiply = BinaryOperation.create(
442 BinaryOperation.Operator.MUL, matrix_array_reference,
443 vector_array_reference)
444 # Create "result(i) + matrix(i,j) * vector(j)"
445 rhs = BinaryOperation.create(
446 BinaryOperation.Operator.ADD, result_ref, multiply)
447 # Create "result(i) = result(i) + matrix(i,j) * vector(j)"
448 assign = Assignment.create(result_ref.copy(), rhs)
449 # Create j loop and add the above code as a child
450 # Work out the bounds
451 if len(v_fr_order) == 0:
452 # If no full ranges, then the vector was specified
453 # without them i.e.: only has one dimension.
454 first_pos = 0
455 else:
456 first_pos = v_fr_order[0]
457 lower_bound, upper_bound, step = _get_array_bound(vector, first_pos)
458 jloop = Loop.create(j_loop_sym, lower_bound, upper_bound, step,
459 [assign])
460 # Create "result(i) = 0.0"
461 assign = Assignment.create(result_ref.copy(),
462 Literal("0.0", REAL_TYPE))
463 # Create i loop and add assignment and j loop as children
464 if len(m_fr_order) == 0:
465 # If no full ranges, then matrix was specified
466 # without them i.e.: only has two dimensions.
467 first_pos = 0
468 else:
469 first_pos = m_fr_order[0]
470 lower_bound, upper_bound, step = _get_array_bound(matrix, first_pos)
471 iloop = Loop.create(i_loop_sym, lower_bound, upper_bound, step,
472 [assign, jloop])
473 # Replace the existing assignment with the new loop.
474 assignment.replace_with(iloop)

476 def _apply_matrix_matrix(self, node):
477 '''
478 Apply the transformation for the case of a matrix-matrix
479 multiplication.
480
481 :param node: a MATMUL IntrinsicCall.
482 :type node: :py:class:`psyclone.psyir.nodes.IntrinsicCall`
483
484 '''
485 # pylint: disable=too-many-locals
486 assignment = node.parent
487 matrix1 = node.arguments[0]
488 matrix2 = node.arguments[1]
489 result = node.parent.lhs
490
491 # Create new i, j and ii loop iterators.
492 symbol_table = node.scope.symbol_table
493 i_loop_sym = symbol_table.new_symbol("i", symbol_type=DataSymbol,
494 datatype=INTEGER_TYPE)
495 j_loop_sym = symbol_table.new_symbol("j", symbol_type=DataSymbol,
496 datatype=INTEGER_TYPE)
497 ii_loop_sym = symbol_table.new_symbol("ii", symbol_type=DataSymbol,
498 datatype=INTEGER_TYPE)
499 # Create "result(i,j)"
500 r_fr_order, r_nfr_order, r_nfr = self._get_full_range_split(result)
501 result_ref = _create_array_ref(result.symbol,
502 [i_loop_sym, j_loop_sym],
503 r_nfr,
504 r_fr_order,
505 r_nfr_order)
506 # Create "matrix2(ii,j)"
507 m2_fr_order, m2_nfr_order, m2_nfr = self._get_full_range_split(matrix2)
508 m2_array_reference = _create_array_ref(matrix2.symbol,
509 [ii_loop_sym, j_loop_sym],
510 m2_nfr,
511 m2_fr_order,
512 m2_nfr_order)
513 # Create "matrix1(i,ii)"
514 m1_fr_order, m1_nfr_order, m1_nfr = self._get_full_range_split(matrix1)
515 m1_array_reference = _create_array_ref(matrix1.symbol,
516 [i_loop_sym, ii_loop_sym],
517 m1_nfr,
518 m1_fr_order,
519 m1_nfr_order)
520 # Create "matrix1(i,ii) * matrix2(ii,j)"
521 multiply = BinaryOperation.create(
522 BinaryOperation.Operator.MUL, m1_array_reference,
523 m2_array_reference)
524 # Create "result(i,j) + matrix1(i,ii) * matrix2(ii,j)"
525 rhs = BinaryOperation.create(
526 BinaryOperation.Operator.ADD, result_ref, multiply)
527 # Create "result(i,j) = result(i,j) + matrix1(i,ii) * matrix2(ii,j)"
528 assign = Assignment.create(result_ref.copy(), rhs)
529 # Create ii loop and add the above code as a child
530 # Work out the bounds
531 if len(m1_fr_order) == 0:
532 # If no full ranges, then matrix was specified
533 # without them i.e.: only has two dimensions.
534 pos_last = 1
535 else:
536 pos_last = m1_fr_order[-1]
537 lower_bound, upper_bound, step = _get_array_bound(matrix1, pos_last)
538 iiloop = Loop.create(ii_loop_sym, lower_bound, upper_bound, step,
539 [assign])
540 # Create "result(i,j) = 0.0"
541 assign = Assignment.create(result_ref.copy(),
542 Literal("0.0", REAL_TYPE))
543 # Create i loop and add assignment and ii loop as children.
544 if len(m1_fr_order) == 0:
545 # If no full ranges, then matrix was specified
546 # without them i.e.: only has two dimensions.
547 pos_first = 0
548 else:
549 pos_first = m1_fr_order[0]
550 lower_bound, upper_bound, step = _get_array_bound(matrix1, pos_first)
551 iloop = Loop.create(i_loop_sym, lower_bound, upper_bound, step,
552 [assign, iiloop])
553 # Create j loop and add i loop as child.
554 if len(m2_fr_order) == 0:
555 # If no full ranges, then matrix was specified
556 # without them i.e.: only has two dimensions.
557 pos_last = 1
558 else:
559 pos_last = m2_fr_order[-1]
560 lower_bound, upper_bound, step = _get_array_bound(matrix2, pos_last)
561 jloop = Loop.create(j_loop_sym, lower_bound, upper_bound, step,
562 [iloop])
563 # Replace the original assignment with the new loop.
564 assignment.replace_with(jloop)
