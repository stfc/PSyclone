52class ReadOnlyVerifyTrans(PSyDataTrans):
53 '''This transformation inserts a ReadOnlyVerifyNode or a node derived
54 from ReadOnlyVerifyNode into the PSyIR of a schedule. At code creation
55 time this node will use the PSyData API to create code that will
56 verify that read-only quantities are not modified.
57
58 After applying the transformation the Nodes marked for verification are
59 children of the ReadOnlyVerifyNode.
60 Nodes to verify can be individual constructs within an Invoke (e.g.
61 Loops containing a Kernel or BuiltIn call) or entire Invokes.
62
63 :param node_class: The class of Node which will be inserted \
64 into the tree (defaults to ReadOnlyVerifyNode), but can be any \
65 derived class.
66 :type node_class: :py:class:`psyclone.psyir.nodes.ReadOnlyVerifyNode` or \
67 derived class
68
69 '''
70 # The types of node that this transformation can enclose
71 valid_node_types = (Loop, Kern, BuiltIn, Directive, Literal, Reference)
72
73 def __init__(self, node_class=ReadOnlyVerifyNode):
74 super().__init__(node_class=node_class)
75
76 # -------------------------------------------------------------------------
77 def validate(self, node_list, options=None):
78 # pylint: disable=arguments-renamed
79 '''Performs validation checks specific to read-only-based
80 transformations.
81
82 :param node_list: the list of Node(s) we are checking.
83 :type node_list: list of :py:class:`psyclone.psyir.nodes.Node`
84 :param options: a dictionary with options for transformations.
85 :type options: Optional[Dict[str, Any]]
86
87 :raises TransformationError: if transformation is applied to a \
88 Kernel or a BuiltIn call without its \
89 parent Loop.
90 :raises TransformationError: if transformation is applied to a Loop \
91 without its parent Directive when \
92 optimisations are applied.
93 :raises TransformationError: if transformation is applied to an \
94 orphaned Directive without its parent \
95 Directive.
96
97 '''
98 # Check ReadOnlyVerifyTrans specific constraints.
99 # Check constraints not covered by valid_node_types for
100 # individual Nodes in node_list.
101 for node in node_list:
102
103 # Check that a ReadOnlyVerifyNode is not inserted between a Loop
104 # and its parent Directive when optimisations are applied, as
105 # this may result in including the end Directive for verification
106 # but not the beginning.
107 if isinstance(node, Loop) and isinstance(node.parent, Schedule) \
108 and isinstance(node.parent.parent, Directive):
109 raise TransformationError(
110 f"Error in {self.name}: Application to a Loop without its "
111 f"parent Directive is not allowed.")
112
113 # Check that the ReadOnlyVerifyNode is not inserted within a
114 # thread parallel region when optimisations are applied.
115 if node.ancestor((OMPParallelDirective, ACCParallelDirective)):
116 raise TransformationError(
117 f"Error in {self.name}: Application to Nodes enclosed "
118 f"within a thread-parallel region is not allowed.")
119
120 # Performs validation checks specific to PSyData-based
121 # transformations.
122 super().validate(node_list, options)
