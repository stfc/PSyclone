57class RaisePSyIR2AlgTrans(Transformation):
58 '''Transform a generic PSyIR representation of an Algorithm-layer
59 invoke call to a PSyclone version with specialised domain-specific
60 nodes.
61
62 '''
63 def __init__(self):
64 super().__init__()
65 self._call_name_call_name = None

67 @staticmethod
68 def _parse_args(code_block, fp2_node):
69 '''Return the arguments from a Structure Constructor stored as a
70 CodeBlock containing an fparser2 ast.
71
72 :param code_block: the CodeBlock containing a StructureConstructor.
73 :type code_block: :py:class:`psyclone.psyir.nodes.CodeBlock`
74 :param fp2_node: the fparser2 Structure Constructor node.
75 :type fp2_node: \
76 :py:class:`fparser.two.Fortran2003.Structure_Constructor`
77
78 :returns: a list of PSyIR nodes containing the \
79 StructureConstructor arguments.
80 :rtype: list of :py:class:`psyclone.psyir.nodes.Node`
81
82 '''
83 dummy_call = Call(parent=code_block.parent)
84 dummy_call.addchild(Reference(RoutineSymbol("dummy")))
85 fparser2 = Fparser2Reader()
86 for arg in fp2_node.children[1].children:
87 fparser2.process_nodes(dummy_call, [arg])
88 # Return the list of detached arguments
89 return dummy_call.pop_all_children()[1:]
90
91 @staticmethod
92 def _get_symbol(call, fp2_node):
93 '''Return the name of a Structure Constructor stored as a CodeBlock
94 containing an fparser2 ast.
95
96 :param code_block: the CodeBlock containing a StructureConstructor.
97 :type code_block: :py:class:`psyclone.psyir.nodes.CodeBlock`
98 :param fp2_node: the fparser2 Structure Constructor node.
99 :type fp2_node: \
100 :py:class:`fparser.two.Fortran2003.Structure_Constructor`
101
102 :returns: the symbol capturing the name and type of the \
103 StructureConstructor.
104 :rtype: :py:class:`psyclone.psyir.symbols.Symbol`
105
106 '''
107 name = fp2_node.children[0].string
108 symbol_table = call.scope.symbol_table
109 try:
110 type_symbol = symbol_table.lookup(name)
111 except KeyError:
112 type_symbol = DataTypeSymbol(name, StructureType())
113 symbol_table.add(type_symbol)
114 return type_symbol
115
116 @staticmethod
117 def _specialise_symbol(symbol):
118 '''If the symbol argument is a Symbol then change it into a
119 DataTypeSymbol.
120
121 :param symbol: a symbol that will be modified to a DataTypeSymbol \
122 if it is a Symbol.
123 :type symbol: :py:class:`psyclone.psyir.symbols.Symbol`
124
125 '''
126 # pylint: disable=unidiomatic-typecheck
127 if type(symbol) is Symbol:
128 symbol.specialise(DataTypeSymbol)
129 symbol.datatype = StructureType()

131 def _validate_fp2_node(self, fp2_node):
132 '''Validation routine for an fparser2 node within a code block.
133
134 :param fp2_node: an fparser2 Structure Constructor.
135 :type fp2_node: \
136 :py:class:`fparser.two.Fortran2003.Structure_Constructor`
137
138 :raises TransformationError: if the fparser2 node is not the \
139 expected type.
140
141 '''
142 if not isinstance(fp2_node, Structure_Constructor):
143 raise TransformationError(
144 f"Error in {self.name} transformation. Expecting an algorithm "
145 f"invoke codeblock to contain a Structure-Constructor, but "
146 f"found '{type(fp2_node).__name__}'.")

148 def validate(self, node, options=None):
149 '''Validate the node argument.
150
151 :param node: a PSyIR call node capturing an invoke call in
152 generic PSyIR.
153 :type node: :py:class:`psyclone.psyir.nodes.Call`
154 :param options: a dictionary with options for transformations.
155 :type options: Optional[Dict[str, Any]]
156
157 :raises TransformationError: if the supplied call argument is
158 not a PSyIR Call node.
159 :raises TransformationError: if the supplied call argument
160 does not have the expected name which would identify it as an
161 invoke call.
162 :raises TransformationError: if there is more than one named argument.
163 :raises TransformationError: if the named argument does not
164 conform to the name=str format.
165 :raises TransformationError: if the name of the invoke is invalid.
166 :raises TransformationError: if the invoke arguments are not a
167 PSyIR ArrayReference or CodeBlock.
168
169 '''
170 self._call_name_call_name = None
171
172 if not isinstance(node, Call):
173 raise TransformationError(
174 f"Error in {self.name} transformation. The supplied call "
175 f"argument should be a `Call` node but found "
176 f"'{type(node).__name__}'.")
177 if not node.routine.name.lower() == "invoke":
178 raise TransformationError(
179 f"Error in {self.name} transformation. The supplied call "
180 f"argument should be a `Call` node with name 'invoke' but "
181 f"found '{node.routine.name}'.")
182 names = [name for name in node.argument_names if name]
183 if len(names) > 1:
184 raise TransformationError(
185 f"Error in {self.name} transformation. There should be at "
186 f"most one named argument in an invoke, but there are "
187 f"{len(names)} in '{node.debug_string()}'.")
188 for idx, arg in enumerate(node.arguments):
189 if node.argument_names[idx]:
190 if (not node.argument_names[idx].lower() == "name"
191 or not (isinstance(arg, Literal) and
192 isinstance(arg.datatype, ScalarType) and
193 arg.datatype.intrinsic ==
194 ScalarType.Intrinsic.CHARACTER)):
195 raise TransformationError(
196 f"Error in {self.name} transformation. If there "
197 f"is a named argument, it must take the form name"
198 f"='str', but found '{node.debug_string()}'.")
199 try:
200 FortranReader.validate_name(arg.value)
201 except (TypeError, ValueError) as err:
202 raise TransformationError(
203 f"Problem with invoke name: {err}") from err
204 if node.argument_names[idx]:
205 pass
206 elif isinstance(arg, ArrayReference):
207 pass
208 elif isinstance(arg, CodeBlock):
209 # pylint: disable=protected-access
210 for fp2_node in arg._fp2_nodes:
211 self._validate_fp2_node(fp2_node)
212 else:
213 if isinstance(arg, Call):
214 info = (
215 f"The invoke call argument '{arg.routine.name}' has "
216 f"been used as a routine name. This is not allowed.")
217 else:
218 info = (
219 f"The arguments to this invoke call are expected to "
220 f"be kernel calls which are represented in generic "
221 f"PSyIR as CodeBlocks or ArrayReferences, but "
222 f"'{arg.debug_string()}' is of type "
223 f"'{type(arg).__name__}'.")
224 raise TransformationError(
225 f"Error in {self.name} transformation. {info}")

227 def apply(self, call, index, options=None):
228 ''' Apply the transformation to the supplied node.
229
230 :param call: a PSyIR call node capturing an invoke call in \
231 generic PSyIR.
232 :type call: :py:class:`psyclone.psyir.nodes.Call`
233 :param int index: the position of this invoke call relative to \
234 other invokes in the algorithm layer.
235 :param options: a dictionary with options for transformations.
236 :type options: Optional[Dict[str, Any]]
237
238 '''
239 self.validatevalidate(call, options=options)
240
241 call_name = None
242 calls = []
243 for idx, call_arg in enumerate(call.arguments):
244
245 # pylint: disable=protected-access
246 arg_info = []
247 if call.argument_names[idx]:
248 call_name = f"{call_arg.value}"
249 continue
250 elif isinstance(call_arg, ArrayReference):
251 # kernel misrepresented as ArrayReference
252 args = call_arg.pop_all_children()
253 type_symbol = call_arg.symbol
254 arg_info.append((type_symbol, args))
255 else:
256 # The validates check that this can only be a Codeblock with
257 # a StructureConstructor fparser2 node inside
258 for fp2_node in call_arg.get_ast_nodes:
259 # This child is a kernel
260 type_symbol = self._get_symbol(call, fp2_node)
261 args = self._parse_args(call_arg, fp2_node)
262 arg_info.append((type_symbol, args))
263
264 for (type_symbol, args) in arg_info:
265 self._specialise_symbol(type_symbol)
266 calls.append(KernelFunctor.create(type_symbol, args))
267
268 invoke_call = AlgorithmInvokeCall.create(
269 call.routine.symbol, calls, index, name=call_name)
270 call.replace_with(invoke_call)
