39'''
40This module contains the LFRicDofmaps class which holds all the
41information and methods for the dofmaps required by an invoke such as:
42generating the calls to the LFRic infrastructure that look-up the necessary
43dofmaps; declaring all unique function space dofmaps in the PSy layer as
44pointers to integer arrays; and adding dofmap-related declarations to a
45Kernel stub.
46
47LFRicDofmaps is used in the LFRicInvoke module.
48'''
49

53from psyclone.configuration import Config
54from psyclone.domain.lfric import LFRicCollection
55from psyclone.errors import GenerationError, InternalError
56from psyclone.f2pygen import AssignGen, CommentGen, DeclGen

59class LFRicDofmaps(LFRicCollection):
60 '''
61 Holds all information on the dofmaps (including column-banded and
62 indirection) required by an invoke.
63
64 :param node: Kernel or Invoke for which to manage dofmaps.
65 :type node: :py:class:`psyclone.domain.lfric.LFRicKern` or \
66 :py:class:`psyclone.dynamo0p3.LFRicInvoke`
67
68 '''
69 def __init__(self, node):
70 # pylint: disable=too-many-branches
71 super().__init__(node)
72
73 # Look at every kernel call in this invoke and generate a list
74 # of the unique function spaces involved.
75 # We create a dictionary whose keys are the map names and entries
76 # are the corresponding field objects.
77 self._unique_fs_maps_unique_fs_maps = OrderedDict()
78 # We also create a dictionary of column-banded dofmaps. Entries
79 # in this one are themselves dictionaries containing two entries:
80 # "argument" - the object holding information on the CMA kernel
81 # argument
82 # "direction" - whether the dofmap is required for the "to" or
83 # "from" function space of the operator.
84 self._unique_cbanded_maps_unique_cbanded_maps_unique_cbanded_maps = OrderedDict()
85 # A dictionary of required CMA indirection dofmaps. As with the
86 # column-banded dofmaps, each entry is itself a dictionary with
87 # "argument" and "direction" entries.
88 self._unique_indirection_maps_unique_indirection_maps_unique_indirection_maps = OrderedDict()
89
90 for call in self._calls_calls:
91 # We only need a dofmap if the kernel operates on cells
92 # rather than dofs.
93 if call.iterates_over != "dof":
94 for unique_fs in call.arguments.unique_fss:
95 # We only need a dofmap if there is a *field* on this
96 # function space. If there is then we use it to look
97 # up the dofmap.
98 fld_arg = unique_fs.field_on_space(call.arguments)
99 if fld_arg:
100 map_name = unique_fs.map_name
101 if map_name not in self._unique_fs_maps_unique_fs_maps:
102 self._unique_fs_maps_unique_fs_maps[map_name] = fld_arg
103 if call.cma_operation == "assembly":
104 # A kernel that assembles a CMA operator requires
105 # column-banded dofmaps for its 'to' and 'from'
106 # function spaces
107 cma_args = psyGen.args_filter(
108 call.arguments.args,
109 arg_types=["gh_columnwise_operator"])
110
111 # Sanity check - we expect only one CMA argument
112 if len(cma_args) != 1:
113 raise GenerationError(
114 f"Internal error: there should only be one CMA "
115 f"operator argument for a CMA assembly kernel but "
116 f"found {len(cma_args)}")
117
118 map_name = \
119 cma_args[0].function_space_to.cbanded_map_name
120 if map_name not in self._unique_cbanded_maps_unique_cbanded_maps_unique_cbanded_maps:
121 self._unique_cbanded_maps_unique_cbanded_maps_unique_cbanded_maps[map_name] = {
122 "argument": cma_args[0],
123 "direction": "to"}
124 map_name = \
125 cma_args[0].function_space_from.cbanded_map_name
126 if map_name not in self._unique_cbanded_maps_unique_cbanded_maps_unique_cbanded_maps:
127 self._unique_cbanded_maps_unique_cbanded_maps_unique_cbanded_maps[map_name] = {
128 "argument": cma_args[0],
129 "direction": "from"}
130 elif call.cma_operation == "apply":
131 # A kernel that applies (or applies the inverse of) a
132 # CMA operator requires the indirection dofmaps for the
133 # to- and from-spaces of the operator.
134 cma_args = psyGen.args_filter(
135 call.arguments.args,
136 arg_types=["gh_columnwise_operator"])
137
138 # Sanity check - we expect only one CMA argument
139 if len(cma_args) != 1:
140 raise GenerationError(
141 f"Internal error: there should only be one CMA "
142 f"operator argument for a kernel that applies a "
143 f"CMA operator but found {len(cma_args)}")
144
145 map_name = cma_args[0].function_space_to\
146 .cma_indirection_map_name
147 if map_name not in self._unique_indirection_maps_unique_indirection_maps_unique_indirection_maps:
148 self._unique_indirection_maps_unique_indirection_maps_unique_indirection_maps[map_name] = {
149 "argument": cma_args[0],
150 "direction": "to"}
151 map_name = cma_args[0].function_space_from\
152 .cma_indirection_map_name
153 if map_name not in self._unique_indirection_maps_unique_indirection_maps_unique_indirection_maps:
154 self._unique_indirection_maps_unique_indirection_maps_unique_indirection_maps[map_name] = {
155 "argument": cma_args[0],
156 "direction": "from"}

158 def initialise(self, parent):
159 ''' Generates the calls to the LFRic infrastructure that
160 look-up the necessary dofmaps. Adds these calls as children
161 of the supplied parent node. This must be an appropriate
162 f2pygen object. '''
163
164 # If we've got no dofmaps then we do nothing
165 if self._unique_fs_maps_unique_fs_maps:
166 parent.add(CommentGen(parent, ""))
167 parent.add(CommentGen(parent,
168 " Look-up dofmaps for each function space"))
169 parent.add(CommentGen(parent, ""))
170
171 for dmap, field in self._unique_fs_maps_unique_fs_maps.items():
172 parent.add(AssignGen(parent, pointer=True, lhs=dmap,
173 rhs=field.proxy_name_indexed +
174 "%" + field.ref_name() +
175 "%get_whole_dofmap()"))
176 if self._unique_cbanded_maps_unique_cbanded_maps_unique_cbanded_maps:
177 parent.add(CommentGen(parent, ""))
178 parent.add(CommentGen(parent,
179 " Look-up required column-banded dofmaps"))
180 parent.add(CommentGen(parent, ""))
181
182 for dmap, cma in self._unique_cbanded_maps_unique_cbanded_maps_unique_cbanded_maps.items():
183 parent.add(AssignGen(parent, pointer=True, lhs=dmap,
184 rhs=cma["argument"].proxy_name_indexed +
185 "%column_banded_dofmap_" +
186 cma["direction"]))
187
188 if self._unique_indirection_maps_unique_indirection_maps_unique_indirection_maps:
189 parent.add(CommentGen(parent, ""))
190 parent.add(CommentGen(parent,
191 " Look-up required CMA indirection dofmaps"))
192 parent.add(CommentGen(parent, ""))
193
194 for dmap, cma in self._unique_indirection_maps_unique_indirection_maps_unique_indirection_maps.items():
195 parent.add(AssignGen(parent, pointer=True, lhs=dmap,
196 rhs=cma["argument"].proxy_name_indexed +
197 "%indirection_dofmap_"+cma["direction"]))

199 def _invoke_declarations(self, parent):
200 '''
201 Declare all unique function space dofmaps in the PSy layer as pointers
202 to integer arrays of rank 2.
203
204 :param parent: the f2pygen node to which to add the declarations.
205 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
206
207 '''
208 api_config = Config.get().api_conf("lfric")
209
210 # Function space dofmaps
211 decl_map_names = \
212 [dmap+"(:,:) => null()" for dmap in sorted(self._unique_fs_maps_unique_fs_maps)]
213
214 if decl_map_names:
215 parent.add(DeclGen(parent, datatype="integer",
216 kind=api_config.default_kind["integer"],
217 pointer=True, entity_decls=decl_map_names))
218
219 # Column-banded dofmaps
220 decl_bmap_names = \
221 [dmap+"(:,:) => null()" for dmap in self._unique_cbanded_maps_unique_cbanded_maps_unique_cbanded_maps]
222 if decl_bmap_names:
223 parent.add(DeclGen(parent, datatype="integer",
224 kind=api_config.default_kind["integer"],
225 pointer=True, entity_decls=decl_bmap_names))
226
227 # CMA operator indirection dofmaps
228 decl_ind_map_names = \
229 [dmap+"(:) => null()" for dmap in self._unique_indirection_maps_unique_indirection_maps_unique_indirection_maps]
230 if decl_ind_map_names:
231 parent.add(DeclGen(parent, datatype="integer",
232 kind=api_config.default_kind["integer"],
233 pointer=True, entity_decls=decl_ind_map_names))

235 def _stub_declarations(self, parent):
236 '''
237 Add dofmap-related declarations to a Kernel stub.
238
239 :param parent: node in the f2pygen AST representing the Kernel stub.
240 :type parent: :py:class:`psyclone.f2pygen.SubroutineGen`
241
242 '''
243 api_config = Config.get().api_conf("lfric")
244
245 # Function space dofmaps
246 for dmap in sorted(self._unique_fs_maps_unique_fs_maps):
247 # We declare ndf first as some compilers require this
248 ndf_name = \
249 self._unique_fs_maps_unique_fs_maps[dmap].function_space.ndf_name
250 parent.add(DeclGen(parent, datatype="integer",
251 kind=api_config.default_kind["integer"],
252 intent="in", entity_decls=[ndf_name]))
253 parent.add(DeclGen(parent, datatype="integer",
254 kind=api_config.default_kind["integer"],
255 intent="in", dimension=ndf_name,
256 entity_decls=[dmap]))
257 # Column-banded dofmaps
258 for dmap, cma in self._unique_cbanded_maps_unique_cbanded_maps_unique_cbanded_maps.items():
259 if cma["direction"] == "to":
260 ndf_name = cma["argument"].function_space_to.ndf_name
261 elif cma["direction"] == "from":
262 ndf_name = cma["argument"].function_space_from.ndf_name
263 else:
264 raise InternalError(
265 f"Invalid direction ('{cma['''direction''']}') found for "
266 f"CMA operator when collecting column-banded dofmaps. "
267 f"Should be either 'to' or 'from'.")
268 parent.add(DeclGen(parent, datatype="integer",
269 kind=api_config.default_kind["integer"],
270 intent="in", entity_decls=[ndf_name]))
271 parent.add(DeclGen(parent, datatype="integer",
272 kind=api_config.default_kind["integer"],
273 intent="in",
274 dimension=",".join([ndf_name, "nlayers"]),
275 entity_decls=[dmap]))
276 # CMA operator indirection dofmaps
277 for dmap, cma in self._unique_indirection_maps_unique_indirection_maps_unique_indirection_maps.items():
278 if cma["direction"] == "to":
279 dim_name = cma["argument"].name + "_nrow"
280 elif cma["direction"] == "from":
281 dim_name = cma["argument"].name + "_ncol"
282 else:
283 raise InternalError(
284 f"Invalid direction ('{cma['''direction''']}') found for "
285 f"CMA operator when collecting indirection dofmaps. "
286 f"Should be either 'to' or 'from'.")
287 parent.add(DeclGen(parent, datatype="integer",
288 kind=api_config.default_kind["integer"],
289 intent="in", entity_decls=[dim_name]))
290 parent.add(DeclGen(parent, datatype="integer",
291 kind=api_config.default_kind["integer"],
292 intent="in", dimension=dim_name,
293 entity_decls=[dmap]))
