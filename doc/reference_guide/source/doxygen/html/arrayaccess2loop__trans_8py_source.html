56class ArrayAccess2LoopTrans(Transformation):
57 '''Provides a transformation to transform a constant index access to
58 an array (i.e. one that does not contain a loop iterator) to a
59 single trip loop. For example:
60
61 >>> from psyclone.psyir.transformations import ArrayAccess2LoopTrans
62 >>> from psyclone.psyir.backend.fortran import FortranWriter
63 >>> from psyclone.psyir.frontend.fortran import FortranReader
64 >>> from psyclone.psyir.nodes import Assignment
65 >>> code = ("program example\\n"
66 ... " real a(10)\\n"
67 ... " a(1) = 0.0\\n"
68 ... "end program example\\n")
69 >>> psyir = FortranReader().psyir_from_source(code)
70 >>> assignment = psyir.walk(Assignment)[0]
71 >>> ArrayAccess2LoopTrans().apply(assignment.lhs.children[0])
72 >>> print(FortranWriter()(psyir))
73 program example
74 real, dimension(10) :: a
75 integer :: ji
76 <BLANKLINE>
77 do ji = 1, 1, 1
78 a(ji) = 0.0
79 enddo
80 <BLANKLINE>
81 end program example
82 <BLANKLINE>
83
84 '''
85 def apply(self, node, options=None):
86 '''Apply the ArrayAccess2Loop transformation if the supplied node
87 is an access to an array index within an Array Reference that
88 is on the left-hand-side of an Assignment node. The access
89 must be a scalar (i.e. not a range) and must not include a
90 loop variable (as we are transforming a single access to a
91 loop).
92
93 If the constraints are satisfied then the array access is replaced
94 with a loop iterator and a single trip loop. The new loop will be
95 placed immediately around the assignment.
96
97 :param node: an array index.
98 :type node: :py:class:`psyclone.psyir.nodes.Node`
99 :param options: a dictionary with options for transformations.
100 This is an optional argument that defaults to None.
101 :type options: Optional[Dict[str, Any]]
102
103 '''
104 self.validatevalidate(node, options)
105
106 array_index = node.position
107 array_reference = node.parent
108 assignment = array_reference.parent
109 symbol_table = node.scope.symbol_table
110
111 node_copy = node.copy()
112 loop_variable_name = symbol_table.next_available_name("idx")
113
114 # Look up the loop variable in the symbol table. If it does
115 # not exist then create it.
116 loop_variable_symbol = symbol_table.find_or_create(
117 loop_variable_name, symbol_type=DataSymbol,
118 datatype=INTEGER_TYPE)
119
120 # Replace current access with loop variable.
121 for array in assignment.walk(ArrayReference):
122 if not array.ancestor(ArrayReference):
123 # This is not a nested access e.g. a(b(n)).
124 array.indices[array_index] = Reference(loop_variable_symbol)
125
126 loop_body = assignment
127 loc_parent = loop_body.parent
128 loc_index = loop_body.position
129
130 # Create the new single-trip loop and add its children.
131 step = Literal("1", INTEGER_TYPE)
132 loop = Loop.create(loop_variable_symbol, node_copy,
133 node_copy.copy(), step, [loop_body.detach()])
134
135 # Replace the original assignment with a loop containing the
136 # modified assignment.
137 loc_parent.children.insert(loc_index, loop)

139 def validate(self, node, options=None):
140 '''Perform various checks to ensure that it is valid to apply the
141 ArrayAccess2LoopTrans transformation to the supplied PSyIR Node.
142
143 :param node: the node that is being checked.
144 :type node: :py:class:`psyclone.psyir.nodes.Node`
145 :param options: a dictionary with options for transformations.
146 This is an optional argument that defaults to None.
147 :type options: Optional[Dict[str, Any]]
148
149 '''
150 # Not a PSyIR node
151 if not isinstance(node, Node):
152 raise TransformationError(
153 f"Error in ArrayAccess2LoopTrans transformation. The "
154 f"supplied node argument should be a PSyIR Node, but found "
155 f"'{type(node).__name__}'.")
156 # Not within an array reference
157 if not node.parent or not isinstance(node.parent, ArrayReference):
158 raise TransformationError(
159 f"Error in ArrayAccess2LoopTrans transformation. The "
160 f"supplied node argument should be within an ArrayReference "
161 f"node, but found '{type(node.parent).__name__}'.")
162 array_ref = node.parent
163 # Array reference not within an assignment
164 if not array_ref.parent or not isinstance(array_ref.parent,
165 Assignment):
166 raise TransformationError(
167 f"Error in ArrayAccess2LoopTrans transformation. The "
168 f"supplied node argument should be within an ArrayReference "
169 f"node that is within an Assignment node, but found "
170 f"'{type(array_ref.parent).__name__}' instead of an "
171 f"Assignment.")
172 assignment = array_ref.parent
173 # Array reference not on lhs of the assignment
174 if assignment.lhs is not array_ref:
175 raise TransformationError(
176 f"Error in ArrayAccess2LoopTrans transformation. The "
177 f"supplied node argument should be within an ArrayReference "
178 f"node that is within the left-hand-side of an Assignment "
179 f"node, but '{array_ref.debug_string()}' is on the "
180 f"right-hand-side of '{assignment.debug_string()}'.")
181
182 # Contains a range node
183 if node.walk(Range):
184 raise TransformationError(
185 f"Error in ArrayAccess2LoopTrans transformation. The "
186 f"supplied node should not be or contain a Range node "
187 f"(array notation) as it should be single valued, but found "
188 f"'{node.debug_string()}'.")
189
190 # Capture loop iterator symbols in order
191 iterator_symbols = []
192 location = node.parent.parent
193 while (isinstance(location.parent, Schedule) and
194 isinstance(location.parent.parent, Loop)):
195 location = location.parent.parent
196 if isinstance(location, Loop):
197 iterator_symbols.append(location.variable)
198
199 # Index contains a loop iterator
200 for reference in node.walk(Reference):
201 if reference.symbol in iterator_symbols:
202 raise TransformationError(
203 "Error in ArrayAccess2LoopTrans transformation. The "
204 "supplied node should not be or contain a loop iterator, "
205 "it should be single valued.")
206
207 # Indices on lhs and rhs array accesses are not the same
208 index_pos = node.position
209 assignment = node.parent.parent
210 sym_maths = SymbolicMaths.get()
211 for array_reference in assignment.rhs.walk(ArrayReference):
212 if array_reference.ancestor(ArrayReference):
213 # skip validation as this is an array reference within
214 # an array reference.
215 continue
216 if not sym_maths.equal(array_reference.children[index_pos], node):
217 raise TransformationError(
218 f"Expected index '{index_pos}' for rhs array "
219 f"'{array_reference.symbol.name}' to be the same "
220 f"as that for the lhs array '{node.parent.name}', but "
221 f"they differ in '{assignment.debug_string()}'.")
